276function Manager:updateActIconVisible()
277    if not self:isEnabled() then return end
278    local hasNetWork = (_G.amfBean("NetworkStatusManager"):getNetworkStatus() ~= _G.NetworkConst.NetworkStatusDes.NONETWORK)
279    local isFMJPlayer = _G.amfBean("UserStateManager"):isFMJ()
280    self:log("isIconVisible : isEnabled", self:isEnabled(), "hasNetWork", hasNetWork, "isFMJPlayer", isFMJPlayer, "fullLevelButNeedShow", self.fullLevelButNeedShow)
281    local newVisiblity = (self:isEnabled() and hasNetWork and (isFMJPlayer or self.fullLevelButNeedShow)) and true or false
