55function Manager:onActActInfoUpdate(event)
56    local data = event:getData()
57    local actInfos = data.actInfos or {}
58    local fromServer = data.fromServer
59    local actInfo = actInfos[tostring(self:getConst().actId)]
60    -- not 服务器发来且有actInfo，就返回
61    if not (fromServer and actInfo) then return end
62
63	local actInfo = _G.amfBean("ActivityMgr"):getActInfo(self:getConst().actId)
64	if actInfo == nil then
65		self:clearData()
66	end

69---强弹
70function Manager:checkSh01ActivityPop(dontDispatch)
71    -- 能拿到actInfo，活动开了，发事件
72    local actInfo = _G.amfBean("ActivityMgr"):getActInfo(self:getConst().actId)
73    if not actInfo then
74        return false
75    end
76
77    --回流不弹
78    if actInfo.extra and actInfo.extra ~= "" then
79        local data = table.deserialize(actInfo.extra) or {}
80        if data.livenessTag and tonumber(data.livenessTag) == 2 then
81            return false
82        end
83    end
84
85    local bGetSucess, key = self:getPopMainPanelKey()
86    -- 本地数据没准备好不弹
87    if not bGetSucess then return end
88    local bHadPop = self:getModel():getValueFromLocalByKey(key, tostring(self:getModel():getMainBeginTime()))
89    --弹过不弹
90    if bHadPop then return end
91
92    if not dontDispatch then
93        AutoPopEventCenter:dispatch(AutoPopEvent.ACTIVITY, {
94            popupCheckPoint = PopupCheckPoint.SH01_MAIN_POP
95        })
96    end
97
98    return true
99end

105function Manager:handleSh01ActivityPop( endCall )

177	local function hideBoard()

179			return
180		end

183		Tween.to(fruitTreePlugUI,AppearTime,{
184	        fAlpha = 0
185	    })

187		local moveBy = CCEaseBackIn:create(CCMoveBy:create(AppearTime, ccp(0, AppearDistanceY)))

189			if self.isDisposed then return end

197		fruitTreePlugUI:runAction(action)
198	end
199
200	childList.bg:setTouchHandler(function()
201		hideBoard()
202	end, nil, function() return true end)
