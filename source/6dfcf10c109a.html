86        :next(function()

94        self._isResReady = true
95        return deferred.success()
96    end
97    local d = deferred.new()
98    self._isDownloadingRes = true
99    local onLoadSuccess = function()
100        self._isDownloadingRes = false
101        self._isResReady = true
102        d:resolve()
103        printx(33, "CollectCardManager:DownLoadRes success")
104    end
105    local onLoadFail = function()
106        if showLoading then
107            CommonTip:showTip("活动加载失败，请重试")
108        end
109        self._isDownloadingRes = false
110        d:reject()
111        printx(33, "CollectCardManager:DownLoadRes failed")
112    end
113    local onCancel = function()
114        self._isDownloadingRes = false
115        d:reject()
116        printx(33, "CollectCardManager:DownLoadRes canceled")
117    end
118    AssetsUtil:loadDynamicModule('meta/CollectCardRes.xml', onLoadSuccess, onLoadFail, onCancel, showLoading)
119    return d

256function Manager:applySwapOperateSuccess(data, params, context)
257    local model = self:getModel()
258    local Const = model:getConst()
259    local gainCardId
260    local receiveData
261    if context.type == Const.SwapCreateType.GIFT
262        and context.direction == Const.SwapMessageDirection.RECEIVED
263        and params.op == Const.SwapOperateType.ACCEPT
264    then
265        gainCardId = context.cardId
266        receiveData = {type = context.type, profile = context.profile}
267    elseif context.type == Const.SwapCreateType.EXCHANGE then
268        if context.direction == Const.SwapMessageDirection.RECEIVED
269            and context.state == Const.SwapMessageState.WAIT_CREATOR_CONFIRM
270            and params.op == Const.SwapOperateType.CANCEL
271        then
272            gainCardId = context.selectedCardId or params.selectedCardId
273        elseif context.direction == Const.SwapMessageDirection.SENT
274            and context.state == Const.SwapMessageState.WAIT_RECEIVER
275            and params.op == Const.SwapOperateType.CANCEL
276        then
277            gainCardId = context.cardId
278        elseif context.direction == Const.SwapMessageDirection.SENT
279            and context.state == Const.SwapMessageState.WAIT_CREATOR_CONFIRM
280        then
281            if params.op == Const.SwapOperateType.ACCEPT then
282                gainCardId = context.selectedCardId or params.selectedCardId
283                receiveData = {type = context.type, profile = context.profile}
284            elseif params.op == Const.SwapOperateType.REJECT then
285                gainCardId = context.cardId
286            end
287        end
288    end
289    if gainCardId then
290        if receiveData then
291            local cardData = model:getCardDataById(gainCardId)
292            local wasNew = cardData:getGotNum() <= 0
293            self:onGetCard(gainCardId, 1)

297            }}, {
298                dcFeature = "collect_card",

300            })

304                string.format(
305                    "%s_%s_%s",
306                    cardData:getCardId(),
307                    cardData:getStarLevel(),
308                    cardData:getAlbumId()
309                ),
310                receiveData.profile.xxlId,
311                receiveData.type == Const.SwapCreateType.EXCHANGE and 2 or 1,
312                wasNew and 1 or 2,
313                repeatStarNum
314            )
315        else
316            self:onGetCard(gainCardId, 1)
317        end

513function Manager:startCollectCardFlow(data)
514    if self._isCreaingNewCardPanel or self._isNewCardPanelOpen then
515        if EnvUtil:isDevelopMode() then
516            CommonTip:showTip("新卡大礼包与开包流程同时触发")
517        end
518        table.insert(self._pendingCollectCardFlows, data)
519        return
520    end
521
522    local flowType = data.flowType or _G.COLLECT_CARD_FLOW_TYPE.OPEN_AND_SETTLE
523
524    if flowType == _G.COLLECT_CARD_FLOW_TYPE.OPEN_CARD then
525        return self:_executeOpenCardFlow(data)
526            :next(function(context)
527                CollectCardLog:log("OpenCardPackFlow complete")
528                if data.onComplete then data.onComplete(context) end
529            end, function(error)
530                CollectCardLog:log("OpenCardPackFlow error", error)
531                if data.onError then data.onError(error) end
532                return deferred.error(error)
533            end)
534
535    elseif flowType == _G.COLLECT_CARD_FLOW_TYPE.SETTLE then
536        return self:_executeSettleCardRewardFlow(data)
537            :next(function(context)
538                CollectCardLog:log("SettleCardRewardFlow complete")
539                if data.onComplete then data.onComplete(context) end
540            end, function(error)
541                CollectCardLog:log("SettleCardRewardFlow error", error)
542                if data.onError then data.onError(error) end
543                return deferred.error(error)
544            end)
545
546    elseif flowType == _G.COLLECT_CARD_FLOW_TYPE.OPEN_AND_SETTLE then
547        return self:_executeOpenCardFlow(data)
548            :next(function(context)
549                data.cardPackRewards = context.cardPackRewards
550                return self:_executeSettleCardRewardFlow(data)
551            end, function(error)
552                data.cardPackRewards = {}
553                return self:_executeSettleCardRewardFlow(data)
554            end)
555            :next(function(context)
556                CollectCardLog:log("OpenAndSettleFlow complete")
557                if data.onComplete then data.onComplete(context) end
558            end, function(error)
559                CollectCardLog:log("OpenAndSettleFlow error", error)
560                if data.onError then data.onError(error) end
561                return deferred.error(error)
562            end)
563    else
564        if data.onError then data.onError() end
565        return deferred.error()
566    end
567end

601            return deferred.error("download resource failed")

1629--- 找出满足 >= target 的最小组合
1630---@param target number 目标值
1631---@return number 实际消耗总值
1632---@return table 实际消耗对应的卡牌组合
1633function Manager:pickBestCardCombo(target)
1634    if not target or target <= 0 then return 0, {} end
1635
1636    local cards = self:getModel():getCards()
1637    local starGroups = {} -- 按星级分组的卡牌
1638    for i = 1, 5 do starGroups[i] = {} end
1639
1640    for i, cardData in ipairs(cards) do
1641        if cardData:getGotNum() > 1 then
1642            local starLevel = cardData:getStarLevel()
1643            table.insert(starGroups[starLevel], cardData)
1644        end
1645    end
1646
1647    for star = 1, 5 do
1648        table.sort(starGroups[star], function(left, right)
1649            local leftAvailable = left:getGotNum() - 1
1650            local rightAvailable = right:getGotNum() - 1
1651            if leftAvailable ~= rightAvailable then
1652                return leftAvailable > rightAvailable
1653            end
1654            return tonumber(left:getCardId()) < tonumber(right:getCardId())
1655        end)
1656    end
1657
1658    local availableCounts = {}
1659    for star = 1, 5 do
1660        availableCounts[star] = 0
1661        for i, cardData in ipairs(starGroups[star]) do
1662            -- 需要保留一张卡片
1663            availableCounts[star] = availableCounts[star] + cardData:getGotNum() - 1
1664        end
1665    end
1666
1667    local priorityFiveCount = math.min(math.floor(target / 5), availableCounts[5])
1668    availableCounts[5] = availableCounts[5] - priorityFiveCount
1669    local remainingTarget = target - priorityFiveCount * 5
1670
1671    local maxCapacity = remainingTarget + 4
1672    local dp = {}
1673    local choice = {}
1674    for sum = 0, maxCapacity do dp[sum] = false end
1675    dp[0] = true
1676
1677    for star = 1, 5 do
1678        local count = availableCounts[star]
1679        if count > 0 then
1680            local usedCount = {}
1681            for sum = 0, maxCapacity do usedCount[sum] = 0 end
1682            for sum = star, maxCapacity do
1683                if not dp[sum] and dp[sum - star] and usedCount[sum - star] < count then
1684                    dp[sum] = true
1685                    usedCount[sum] = usedCount[sum - star] + 1
1686                    choice[sum] = star
1687                end
1688            end
1689        end
1690    end
1691
1692    local best_sum
1693    for sum = remainingTarget, maxCapacity do
1694        if dp[sum] then
1695            best_sum = sum
1696            break
1697        end
1698    end
1699    if not best_sum then return 0, {} end
1700
1701    local needCounts = { [1] = 0, [2] = 0, [3] = 0, [4] = 0, [5] = priorityFiveCount }
1702    local currentSum = best_sum
1703    while currentSum > 0 do
1704        local usedStar = choice[currentSum]
1705        needCounts[usedStar] = needCounts[usedStar] + 1
1706        currentSum = currentSum - usedStar
1707    end
1708
1709    -- 提取卡牌实体
1710    local combo = {}
1711    for star = 1, 5 do
1712        local need = needCounts[star]

1714            local group = starGroups[star]
1715            -- 从该星级池中取出 need 张卡
1716            for _, cardData in ipairs(group) do
1717                if need <= 0 then break end
1718                local costNum = math.min(cardData:getGotNum() - 1, need)
1719                -- 记录使用了哪些卡和数量
1720                table.insert(combo, {
1721                    cardId = cardData:getCardId(),

1723                })
