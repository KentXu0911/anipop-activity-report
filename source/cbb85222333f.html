85-- 新字展示时间只跟随 actInfo；没有完整时间说明活动数据未就绪，不展示新字。
86function LuckyBagPartyIcon:isInNewFlagTime()
87    local extra = self:getActInfoExtra()
88    local beginTs = extra and tonumber(extra.beginTs)
89    local endTs = extra and tonumber(extra.endTs)
90    if beginTs and endTs then
91        local now = TimeKit:timeInSec()
92        return now >= math.floor(beginTs / 1000) and now <= math.floor(endTs / 1000)
93    end
94    return false
95end

97-- 读取当前入口展示结束时间，优先使用 actInfo，其次兜底本地活动数据。
98function LuckyBagPartyIcon:getActDisplayEndTime()
99    local extra = self:getActInfoExtra()
100    local endTs = extra and tonumber(extra.endTs)
101    if endTs and endTs > 0 then
102        return math.floor(endTs / 1000)

104
105    local mgr = self:getMgr()

107    if localDataServer and localDataServer.getActMainEndTime then

123-- 注册活动结束刷新 timer；到期后入口需要主动触发右下角池重算并删除自己。
124function LuckyBagPartyIcon:refreshActEndTimer()
125    self:stopActEndTimer()
126    if self:isDevForceShowOnly() then return end
127
128    local endTime = self:getActDisplayEndTime()
129    local now = TimeKit:timeInSec()
130    if not endTime or endTime <= now then return end
131
132    -- isActInMain 使用 now <= endTime，延后一秒刷新，确保边界秒已经跨出活动期。
133    self.actEndTimerId = TimerUtil.addAlarm(function()
134        if self.isDisposed then return end
135        self.actEndTimerId = nil
136        self:onActDisplayExpired()
137    end, math.max(endTime - now + 1, 0.1))
138end

161-- HeGuide 会临时把 icon 换父节点；引导期间新增 icon 先保活延迟布局，避免福袋原布局被中途改写。

169-- 释放右下角引导布局锁；delay 用于等待 HeGuide 恢复父节点，dispose 时可立即兜底释放。
170function LuckyBagPartyIcon:releaseRightCornerGuideLayoutLock(delay)
171    local rightCornerPool = self:getRightCornerPool()
172    local owner = self
173    local function release()
174        if rightCornerPool and rightCornerPool.endGuideLayoutLock then
175            rightCornerPool:endGuideLayoutLock(owner)
176        end
177    end
178
179    if delay and delay > 0 then
180        setTimeOut(release, delay)
181    else
182        release()
183    end
184end

244-- 监听活动打开事件，活动框架拉起成功后清新字。
245function LuckyBagPartyIcon:registerEvents()
246    self.onActivityOpened = function(event)
247        local data = event:getData()
248        local mgr = self:getMgr()
249        if mgr and tonumber(data.actId) == tonumber(mgr.ACT_ID) then
250            self:cancelNewActState()
251        end
252    end
253    ActivityEventCenter:register(ActivityEvent.ACTIVITY_OPENED, self.onActivityOpened)
254
255    self.onActInfosUpdate = function()
256        if self.isDisposed then return end
257        setTimeOut(function()
258            if self.isDisposed then return end
259            self:refreshActEndTimer()
260            if not self:canShow() then
261                self:onActDisplayExpired()
262            end
263        end, 0)
264    end
265    ActivityEventCenter:register(ActivityEvent.ACT_INFOS_UPDATE, self.onActInfosUpdate)
266end

311-- 加入 condition 强弹队列，延迟两帧等待 icon 层完成创建。
312function LuckyBagPartyIcon:tryAddMoveGuideAutoPop()
313    if not self:canShowMoveGuide() then return end
314
315    local function creator()
316        setTimeOut(function()
317            local icon = _G.amfBean("IconManager"):getIconByIconKey(IconKey.kLuckyBagParty)
318            if icon and icon.showMoveGuide then
319                icon:showMoveGuide()
320            else
321                AutoPopEventCenter:dispatch(AutoPopEvent.SINGLE_POP_OVER, PopupCheckPoint.LUCKY_BAG_PARTY_MOVE_GUIDE)
322            end
323        end, 2 / 30)
324    end
325    _G.amfBean("AutoPopupMgr"):addPopupConfig(PopupCheckPoint.LUCKY_BAG_PARTY_MOVE_GUIDE, creator)
326end

328-- 播放福袋换位引导；使用 HeGuide 指向右下角池内的福袋 icon。
329function LuckyBagPartyIcon:showMoveGuide()
330    local function onHandleOver()
331        AutoPopEventCenter:dispatch(AutoPopEvent.SINGLE_POP_OVER, PopupCheckPoint.LUCKY_BAG_PARTY_MOVE_GUIDE)
332    end
333
334    if not self:canShowMoveGuide() then
335        safeRunCall(onHandleOver)
336        return
337    end
338
339    IconEventCenter:dispatch(IconEvent.OPEN_ICON_POOL, IconMeta.PoolType.kRightCorner)
340
341    setTimeOut(function()
342        if self.isDisposed or not self:canShowMoveGuide() then
343            safeRunCall(onHandleOver)
344            return
345        end
346
347        self:beginRightCornerGuideLayoutLock()
348        local HeGuide = require "amf.middleware.ui.kit.component.HeGuide"
349        local heGuide = HeGuide:create(self, true, nil, nil, true)
350        heGuide:addAccurateGuide(self, {self})
351        heGuide:setMinMaxTime(nil, 5)
352        heGuide:setEndCall(function()
353            self.heGuide = nil
354            self:endRightCornerGuideLayoutLock()
355            safeRunCall(onHandleOver)
356        end)
357        local contentList = {
358            {ttf = "福袋派对", color = "00A5C8FF"},
359            {ttf = "已开启，以后请到右下角“+”内寻找福袋派对哦~"},
360        }
361        heGuide:showNpc2(contentList, 60)
362        self.heGuide = heGuide
363
364        if self._serverCacheManager then
365            self._serverCacheManager:setGuideFlag(GuideFlagConst.kLuckyBagPartyMoveGuide)
366        end
367    end, 0.2)
368end

370-- 点击入口拉起福袋活动。
371function LuckyBagPartyIcon:onTouch()
372    BaseIcon.onTouch(self)
373    if self.isDisposed then return end
374    if self.heGuide and not self.heGuide.isDisposed then
375        self.heGuide:removeGuide()

377        self:endRightCornerGuideLayoutLock()
378    end

383    end
384end
