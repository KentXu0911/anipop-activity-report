32        -- 加推送

72function Manager:onReceiveTransaction(evt)
73    if not self:isEnabled() then return end
74    if evt:getName() ~= TransactionEvent.LOCALPUSH then return end
75    local data = evt:getData()
76    if data.action == "add" then
77        -- 闹钟时间不对
78        if not LocalPushUtil:checkAddAlarmTimeValid(data.timeStamp) then return end
79        -- 没有配置
80        if not self:getNotificationMeta(data.type) then return end
81        -- 加闹钟
82        self:addNotification(data)
83    elseif data.action == "remove" then
84        if EnvUtil:isDevelopMode() then
85            -- 公用的活动push 类型
86            if tonumber(data.type) and table.exist(LocalPushConst.ACT_TYPES, tonumber(data.type)) then
87                CommonTip:showTip("删除闹钟有问题，活动用公用类型，删除不要传type，需要传入groupKey")
88                ALog.error("删除闹钟有问题，活动用公用类型，删除不要传type，需要传入groupKey")
89                return
90            end
91        end
92        -- 删除闹钟
93        self:removeNotification(data)
94    elseif data.action == "removeAll" then
95        -- 删除所有闹钟
96        self:removePushedAlarms()
97    else
98        assert(false, "LocalPushManager:onReceiveTransaction: action is not add or remove")
99    end
100end

102-- 获取推送配置
103function Manager:getNotificationMeta(type)
104    local allNotificationMeta = self:getAllNotificationMeta()
105    for k, v in pairs(allNotificationMeta) do
106        if v.type == type then
107            return _G.READ_ONLY(v)
108        end
109    end
110    return nil
111end

113-- 获取所有推送配置
114function Manager:getAllNotificationMeta()
115    if table.isEmpty(self._localPushMetas) then
116        local LocalPushMetaRef = require "amf.middleware.notification.localpush.model.LocalPushMetaRef"
117        self._localPushMetas = self._metaManager:parseOrderedList("notification_v2", LocalPushMetaRef)
118        -- 删除不可用的和not active的
119        for id, v in pairs(self._localPushMetas) do
120            if not v.activate or v.pusher ~= LocalPushConst.PushType.CLIENT then
121                self._localPushMetas[id] = nil
122            end
123        end
124    end
125    return self._localPushMetas
126end

128-- 添加闹钟
129function Manager:addNotification(data, fromRevert)
130    local metaCfg = self:getNotificationMeta(data.type)
131    local structAddAlarm = StructAddAlarm.new()
132        :setType(data.type)
133        :setTimeStamp(data.timeStamp)
134        :setVoiceId(metaCfg.channel)
135        :setLabelType(metaCfg.textId)
136        :setGroupKey(data.groupKey)
137    -- 修改文案，title，message ...
138    structAddAlarm = LocalPushUtil:makeStructRandomMessage(structAddAlarm, metaCfg, data)
139    --每天22:00~次日7:00，不会给玩家发送任何推送；如果此时触发时效类推送，改为次日7:00后发
140    structAddAlarm = LocalPushUtil:makeStructChangePushTime(structAddAlarm, metaCfg)
141    -- 修改推送时间 这里不能修改时间，因为不能确定是要推上午还是下午，有优先级问题
142    local function ifHas(list, data)
143        local tmpList = list or {}
144        for i, v in ipairs(tmpList) do
145            if v.type == data.type and v.timeStamp == data.timeStamp then
146                return true, i
147            end
148        end
149        return false
150    end
151    local function ifHasByLimit(list, data)
152        local tmpList = list or {}
153        for i, v in ipairs(tmpList) do
154            local dayIdx = TimeKit:getDayStartTimeByTS(TimeUtil:convertSec(data.timeStamp))
155            local vDayIdx = TimeKit:getDayStartTimeByTS(TimeUtil:convertSec(v.timeStamp))
156            if v.type == data.type and dayIdx == vDayIdx then
157                return true, i
158            end
159        end
160        return false
161    end
162    -- 加到对应列表中
163    if metaCfg.groupId == LocalPushConst.AlarmType.NO_LIMIT then
164        local has, index = ifHas(self._noLimitPushList, data)
165        if has then
166            if not fromRevert then -- 如果是还原操作就不替换原本的，新添进来的和原本的相同就替换
167                self._noLimitPushList[index] = structAddAlarm
168            -- else
169            --     self:log("fromRevert nolimit -- ", tts(data))
170            end
171        else
172            table.insert(self._noLimitPushList, structAddAlarm)
173        end
174    else
175        local has, index = ifHasByLimit(self._limitPushList, data)
176        if has then
177            if not fromRevert then -- 如果是还原操作就不替换原本的，新添进来的和原本的相同就替换
178                self._limitPushList[index] = structAddAlarm
179            -- else
180            --     self:log("fromRevert limit -- ", tts(data))
181            end
182        else
183            table.insert(self._limitPushList, structAddAlarm)
184        end
185    end
186    -- print("addNotification  ----", tts(self._limitPushList), tts(self._noLimitPushList))
187end

189-- 删除闹钟
190function Manager:removeNotification(data)
191    local type = data.type
192    local groupKey = data.groupKey
193    local timeStamp = data.timeStamp
194    -- 删除即时闹钟
195    for i = #self._noLimitPushList, 1, -1 do
196        if ((type and self._noLimitPushList[i].type == type) 
197            and (not timeStamp or (timeStamp and self._noLimitPushList[i].timeStamp == timeStamp)))
198            or (groupKey and self._noLimitPushList[i].groupKey == groupKey) then
199            table.remove(self._noLimitPushList, i)
200        end
201    end
202    -- 删除限制闹钟
203    for i = #self._limitPushList, 1, -1 do
204        if (type and self._limitPushList[i].type == type) or (groupKey and self._limitPushList[i].groupKey == groupKey) then
205            table.remove(self._limitPushList, i)
206        end
207    end
208    -- print("removeNotification ----", tts(self._limitPushList), tts(self._noLimitPushList))
209end

242-- 到后台事件之后
243function Manager:onSystemEnterBackground()
244    if not self:isEnabled() then return end
245    local list = self:getPushList()
246
247    -- 先删除所有闹钟，再去添加，因为有不知名问题会导致alarm数量到500上限
248    self._localPushSDK:removeAllAlarm(list)
249
250    -- 先存
251    self._model:setValue("localPushList", list, true)
252    -- 添加闹钟
253    for _, v in ipairs(list) do
254        self._localPushSDK:addAlarm(v)
255    end
256    if EnvUtil:isDevelopMode() then
257        self:log("LocalPush: 添加闹钟列表:结果:", tts(list))
258    end
259end

261function Manager:getLimitPushList()
262    local dayMap = {}
263    -- 限制推送的
264    for i, v in ipairs(self._limitPushList) do
265        local day = tostring(TimeKit:getDayStartTimeByTS(TimeUtil:convertSec(v.timeStamp)))
266        if dayMap[day] == nil then
267            dayMap[day] = {}
268        end
269        table.insert(dayMap[day], v)
270    end
271    for k, vList in pairs(dayMap) do
272        table.sort(vList, function (a, b)
273            local metaCfgA = self:getNotificationMeta(a.type)
274            local metaCfgB = self:getNotificationMeta(b.type)
275            return metaCfgA.priority < metaCfgB.priority
276        end)
277    end
278    -- 限制推送列表
279    local limitList = {}
280    for dayStr, vList in pairs(dayMap) do
281        local day = tonumber(dayStr)
282        for index, v in ipairs(vList) do
283            local metaCfg = self:getNotificationMeta(v.type)
284            -- 找出上午优先级最高的要推送的
285            if metaCfg.groupLimit == LocalPushConst.GroupLimitTime.LIMIT1 or metaCfg.groupLimit == LocalPushConst.GroupLimitTime.LIMIT2 then
286                local timeStamp = day + 3600 * 8
287                if LocalPushUtil:checkAddAlarmTimeValid(timeStamp) and not LocalPushUtil:isNearOneHour(timeStamp) then
288                    v:setTimeStamp(timeStamp)
289                    local context = table.remove(vList, index)
290                    table.insert(limitList, context)
291                    break
292                end
293            end
294        end
295        for index, v in ipairs(vList) do
296            local metaCfg = self:getNotificationMeta(v.type)
297            -- 找出下午优先级最高的要推送的
298            if metaCfg.groupLimit == LocalPushConst.GroupLimitTime.LIMIT1 or metaCfg.groupLimit == LocalPushConst.GroupLimitTime.LIMIT3 then
299                --换算到指定时间
300                local timeStamp = day + 3600*20.5
301
302                if LocalPushUtil:checkAddAlarmTimeValid(timeStamp) and not LocalPushUtil:isNearOneHour(timeStamp) then
303                    v:setTimeStamp(timeStamp)
304                    local context = table.remove( vList, index )
305                    table.insert( limitList, context )
306                end
307                break
308            end
309        end
310    end
311    return limitList
312end

314function Manager:getNolimitPushList()
315    -- 时效性推送的 规则
316    local dayMap = {}
317    for i, v in ipairs(self._noLimitPushList) do
318        local day = tostring(TimeKit:getDayStartTimeByTS(TimeUtil:convertSec(v.timeStamp)))
319        if dayMap[day] == nil then
320            dayMap[day] = {}
321        end
322        if LocalPushUtil:checkAddAlarmTimeValid(v.timeStamp) then
323            table.insert(dayMap[day], v)
324        end
325    end
326    for k, vList in pairs(dayMap) do
327        table.sort(vList, function (a, b)
328            local metaCfgA = self:getNotificationMeta(a.type)
329            local metaCfgB = self:getNotificationMeta(b.type)
330            if metaCfgA.priority ~= metaCfgB.priority then
331                return metaCfgA.priority < metaCfgB.priority
332            else
333                return a.timeStamp < b.timeStamp
334            end
335        end)
336    end
337    local noLimitList = {}
338    local cacheAddMap = {}
339    local notAddMap = {}
340    for dayStr, vList in pairs(dayMap) do
341        for i, v in ipairs(vList) do
342            if not cacheAddMap[dayStr] then
343                cacheAddMap[dayStr] = {}
344            end
345            if not notAddMap[dayStr] then
346                notAddMap[dayStr] = {}
347            end
348            if v.groupKey then
349                if LocalPushConst.NO_LIMIT_RULE.GROUPKEY[tostring(v.groupKey)] then
350                    table.insert(noLimitList, v)
351                else
352                    if not cacheAddMap[dayStr][v.groupKey] then
353                        table.insert(noLimitList, v)
354                        cacheAddMap[dayStr][v.groupKey] = true
355                    else
356                        table.insert(notAddMap[dayStr], v)
357                    end
358                end
359            else
360                if LocalPushConst.NO_LIMIT_RULE.TYPE[tostring(v.type)] then
361                    table.insert(noLimitList, v)
362                else
363                    if not cacheAddMap[dayStr][tostring(v.type)] then
364                        table.insert(noLimitList, v)
365                        cacheAddMap[dayStr][tostring(v.type)] = true
366                    else
367                        table.insert(notAddMap[dayStr], v)
368                    end
369                end
370            end
371        end
372    end
373    if EnvUtil:isDevelopMode() then
374        self:log("闹钟notAddMap =-====", tts(notAddMap))
375    end
376    return noLimitList
377end

379-- 获得要推送的列表
380function Manager:getPushList()
381    local limitList = self:getLimitPushList()
382    local noLimitList = self:getNolimitPushList()
383    -- self:log("getPushList : noLimitList ----", tts(noLimitList), " limitList = ", tts(limitList))
384    -- 合并要推送的列表
385    local list = table.union(noLimitList, limitList)
386    return list
387end
