646function WeekTaskModel:taskAddNum( inTaskType, inAddNum  )
647	local addNum = inAddNum or 1
648
649	local completeTaskInfos = {}
650
651	local perDayMap = self:getPerDayMap()
652	for i,v in ipairs(perDayMap) do
653		if addNum == 0 then
654			break
655		end
656
657		for j,k in ipairs(v.tasks) do
658			if not k.finished then
659				local weekDay = i
660				local taskId = k.taskId
661				local curValue = k.curValue
662                local taskType = self:getConst().TaskType.INVALID
663
664				local bGetSucss, taskConfig = self:getTaskConfig( weekDay, taskId )
665				if bGetSucss then
666					taskType = taskConfig.taskType
667				end
668
669				--任务类型相同，并且还未增加过值
670				if taskType == inTaskType then
671					
672					if bGetSucss then
673						local target = taskConfig.target
674						if k.curValue < target then
675							local allNeedNum = target - k.curValue
676							local canAddNum = 0
677							if addNum < allNeedNum then
678								canAddNum = addNum
679							else
680								canAddNum = allNeedNum
681							end
682							k.curValue = k.curValue + canAddNum
683
684							--变更值的任务加入记录
685							if k.curValue == target then
686								if self:getCurDay() >= weekDay then
687									local info = { weekDay = weekDay, taskId=taskId, taskType = taskType, oldValue = curValue, newValue = k.curValue}
688									table.insert( completeTaskInfos, info )
689								end
690							end
691
692							if not self:checkTaskCanAddAll(taskType) then
693								--不能叠加的，要扣除已用的值
694								addNum = addNum - canAddNum
695								if addNum == 0 then
696									break
697								end
698							end
699						end
700					end
701				end
702			end
703		end
704	end
705
706	self:setDataByKey('perDays',perDayMap)
707
708	return completeTaskInfos
709end
