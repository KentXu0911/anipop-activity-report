2local PolicyHandler = class("PolicyHandler")
3
4function PolicyHandler:create(autoPopupMgr)
5	local handler = PolicyHandler.new()
6	handler:init(autoPopupMgr)
7	return handler
8end
9
10function PolicyHandler:init(autoPopupMgr)
11	self._autoPopupMgr = autoPopupMgr
12	AutoPopEventCenter:register(AutoPopEvent.POLICY, self.onHandlePopup, self)
13end
14
15function PolicyHandler:onHandlePopup(evt)
16	local data = evt:getData()
17	local popupCheckPoint = data.popupCheckPoint
18	local params = data.params
19
20	lua_switch(popupCheckPoint){
21		[PopupCheckPoint.FORBID_PLAYING] = function ()
22			self:handleForbidPlaying(params)
23		end,
24		[PopupCheckPoint.ANTI_ADDICTION] = function ()
25			self:handleAntiAddiction(params)
26		end,
27		[PopupCheckPoint.RECALL_SHIMING] = function ()
28			self:handleRecallShiMing(params)
29		end,
30		[PopupCheckPoint.HW_SHIMING_SYNC_ALERT] = function ()
31			self:handleHWShiMingSyncAlert(params)
32		end,
33		default = function ()
34			assert(false, "PolicyHandler invalid popupCheckPoint : " .. tostring(popupCheckPoint))
35		end
36	}
37end
38
39function PolicyHandler:handleForbidPlaying(popType)
40	local function onHandleOver()
41		AutoPopEventCenter:dispatch(AutoPopEvent.SINGLE_POP_OVER, PopupCheckPoint.FORBID_PLAYING)
42	end
43
44	local function creator()
45		local ForbidLoginLogic = require "amf.feature.login.stage.login.ForbidLoginLogic"
46		ForbidLoginLogic:doAutoPopup(popType, onHandleOver)
47	end
48
49	self._autoPopupMgr:addPopupConfig(PopupCheckPoint.FORBID_PLAYING, creator)
50end
51
52function PolicyHandler:handleAntiAddiction(alertType)
53	if not alertType then return end
54	if alertType ~= ShiMingConst.AntiAddictionType.JUVENILE_CUMULATIVE and
55	   alertType ~= ShiMingConst.AntiAddictionType.JUVENILE_PERIOD then
56		-- assert(false, "PolicyHandler invalid alertType : " .. tostring(alertType))
57		return
58	end
59
60	local function onHandleOver()
61		AutoPopEventCenter:dispatch(AutoPopEvent.SINGLE_POP_OVER, PopupCheckPoint.ANTI_ADDICTION)
62	end
63
64	local function creator()
65		if alertType == ShiMingConst.AntiAddictionType.JUVENILE_CUMULATIVE then
66			_G.amfBean("ShiMingManager"):popPanelJuvenileTimeCumulativeLimit()
67		elseif alertType == ShiMingConst.AntiAddictionType.JUVENILE_PERIOD then
68			_G.amfBean("ShiMingManager"):popPanelJuvenileTimePeriodLimit()
69		end
70	end
71
72	self._autoPopupMgr:addPopupConfig(PopupCheckPoint.ANTI_ADDICTION, creator)
73end
74
75function PolicyHandler:handleRecallShiMing()
76	local function onHandleOver()
77		AutoPopEventCenter:dispatch(AutoPopEvent.SINGLE_POP_OVER, PopupCheckPoint.RECALL_SHIMING)
78	end
79
80	local function creator()
81		_G.amfBean("ShiMingManager"):handleAutoPopupMainPanel(onHandleOver)
82	end
83
84	self._autoPopupMgr:addPopupConfig(PopupCheckPoint.RECALL_SHIMING, creator)
85end
86
87function PolicyHandler:handleHWShiMingSyncAlert(params)
88	local regType = params.regType
89	local alertPanelType = params.alertPanelType
90	local function onHandleOver()
91		AutoPopEventCenter:dispatch(AutoPopEvent.SINGLE_POP_OVER, PopupCheckPoint.HW_SHIMING_SYNC_ALERT)
92	end
93
94	local function creator()
95		if alertPanelType == ShiMingConst.PanelId.HW_JUVENILE_ALERT then
96			_G.amfBean("ShiMingManager"):popPanelHWJuvenileAlert(regType, onHandleOver)
97		elseif alertPanelType == ShiMingConst.PanelId.HW_UNKNOWN_ALERT then
98			_G.amfBean("ShiMingManager"):popPanelHWUnknownAlert(regType, onHandleOver)
99		else
100			assert(false, "PolicyHandler invalid alertPanelType : " .. tostring(alertPanelType))
101		end
102	end
103
104	self._autoPopupMgr:addPopupConfig(PopupCheckPoint.HW_SHIMING_SYNC_ALERT, creator)
105end
106
107return PolicyHandler
