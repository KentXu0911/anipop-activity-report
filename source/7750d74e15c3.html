13        local nextLevelType = _G.RecommendLevelConst.NextLevelType.FARM_SCORE_LEVEL
14        if userState > _G.UserStateConst.UserSegmentState.FMC then
15            if curLevelId + 1 <= _G.getTopLevelId() then
16                nextLevelId = curLevelId + 1
17            end
18        end
19        return nextLevelId, nextLevelType, nextLevelScene
20    end
21
22    -- 本次是补星
23    if params.levelFarmingType == _G.UBFLevelFarmingType.FARM_STAR then
24        local unlockManager = _G.amfBean("UnlockManager")
25        -- 打前15关且能解锁
26        if _G.getTopLevelId() - curLevelId < 15 and unlockManager:canUnlockArea() then
27            return _G.getTopPassedLevelId() + 1, _G.RecommendLevelConst.NextLevelType.TOP_LEVEL, nextLevelScene
28        else
29            local nextLevelType = _G.RecommendLevelConst.NextLevelType.FARM_STAR_LEVEL
30            local nextLevelId = RecommendLevelUtil:getMainNotFullYellowLevel(curLevelId + 1)
31            if not nextLevelId then nextLevelId = RecommendLevelUtil:getMainNotFullYellowLevel(1, curLevelId - 1) end
32            if not nextLevelId then nextLevelId = RecommendLevelUtil:getMainNotFullColorfulLevel(curLevelId + 1) end
33            if not nextLevelId then nextLevelId = RecommendLevelUtil:getMainNotFullColorfulLevel(1, curLevelId - 1) end
34            return nextLevelId, nextLevelType, nextLevelScene
35        end
36    end
37
38    -- 正常打关
39    local nextLevelId = _G.getTopLevelId()
40    local nextLevelType = _G.RecommendLevelConst.NextLevelType.TOP_LEVEL
41    if nextLevelId == curLevelId then
42        nextLevelId = nextLevelId + 1
43    end
44    return nextLevelId, nextLevelType, nextLevelScene
45end
