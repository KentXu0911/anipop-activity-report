5function PopupDoPopState:create(fsm)

11function PopupDoPopState:init(fsm)
12	self.fsm = fsm
13	self._autoPopupMgr = fsm:getAutoPopupMgr()
14end
15
16function PopupDoPopState:onEnter(context)
17	assert(context and context.checkPoint and context.popupEntry,
18		"invalid popup dispatch context")
19	local popupCheckPoint = context.checkPoint
20	local popupEntry = self._autoPopupMgr:popPopupEntry(popupCheckPoint, context.popupEntry)
21	if popupEntry then
22		local popupCreator = popupEntry.creator
23		AutoPopupDebugLog:log("PopupDoPopState:doPop ==>", popupCheckPoint)
24		AutoPopupDebugLog:setLastPopupCheckPoint(popupCheckPoint)
25		self._autoPopupMgr:logCurrentPopup()
26		self._autoPopupMgr:clearCheckPointLimitedStages(popupCheckPoint)
27		self._autoPopupMgr:increaseCheckPointPopNum(popupCheckPoint)
28
29		-- popupCheckPoint 同时是身份、规则和槽位；限流归属其 PopType。
30		local popType = self._autoPopupMgr:getPopType(popupCheckPoint)
31		self._autoPopupMgr:increaseTmpPopTypePopNum(popType)
32		self._autoPopupMgr:increaseDailyPopTypePopNum(popType)
33
34		self.fsm:changeState(self.fsm.checkCloseState, {
35			popupEntry = popupEntry,
36			checkPoint = popupCheckPoint,
37		})
38
39		local dcPopStage = AutoPopupDC:getDcPopStage()		--放到popupCreator调用之前，因为popupCreator可能会停掉强弹，导致打点时的场景为空
40		AutoPopupDC:sendPop(popupCheckPoint, dcPopStage)
41		-- 通知业务侧“强弹即将展示”；具体业务打点由监听方决定，不能阻断强弹流程。
42		AutoPopEventCenter:dispatch(AutoPopEvent.POPUP_WILL_SHOW, {
43			popupEntry = popupEntry,
44			checkPoint = popupCheckPoint,
45			popStage = dcPopStage,
46		})
47
48		-- 放在最后，不然可能会有入栈出栈顺序带来的隐患
49		AutoPopupDebugLog:recordPopupStarted(popupEntry, popupCheckPoint)
50		popupCreator()
51	else
52		self.fsm:changeState(self.fsm.checkPopState)
53	end
54end
55
