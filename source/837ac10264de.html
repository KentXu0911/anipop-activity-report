373function PersonalCenterManager:handleProfileOperations(operations, customProfile)
374    local cardInfos = self._model:getNameCardInfoStack()
375    local info = table.find(cardInfos, function(info)
376        return self:isMe(info)
377    end)
378    local profile = info and info.getProfile and info:getProfile()
379
380    if customProfile then
381        self._profileManager:setExtendValue("customProfile", customProfile)
382        if profile then
383            profile.customProfile = true
384        end
385    end
386
387    for i,v in ipairs(operations) do
388        local key = v.key
389        local op = v.op
390        local params = v.params
391
392        if key == "name" and op == "update" then
393            self._profileManager:setProfileValue("name", HeDisplayUtil:urlEncode(params))
394            if profile then
395                profile.name = params
396            end
397        elseif key == "signature" and op == "update" then
398            -- self._profileManager:setProfileValue("signature", params)
399            if profile then
400                profile.signature = params
401            end
402        elseif key == "location" and op == "update" then
403            local data = table.deserialize(params)
404            self._profileManager:setProfileValue("location", data.loc)
405            self._profileManager:setProfileValue("locationId", data.locId)
406            self._profileManager:setProfileValue("adcode", data.adcode)
407
408            if profile then
409                profile.location = data.loc
410                profile.locationId = data.locId
411                profile.adcode = data.adcode
412            end
413
414            self:updateLocationUpdateTime()
415        elseif key == "birthDate" and op == "update" then
416            self._profileManager:setBirthDate(params)
417            if profile then
418                profile.birthDate = params
419            end
420        elseif key == "gender" and op == "update" then
421            self._profileManager:setGender(params)
422            if profile then
423                profile.gender = params
424            end
425        -- elseif key == "fileIds" and op == "update" then 目前缓存的已经改过了，profilemanager又不关心，所以无视
426        elseif key == "headUrl" and op == "update" then
427            self._profileManager:setProfileValue("headUrl", params)
428            if profile then
429                profile.headUrl = params
430            end
431        elseif key == "snsDetail" and op == "delSnsHead" then
432            local snsPlatform = table.deserialize(params).snsPlatformEnum
433            self._accountManager:updateSnsInfo(snsPlatform, nil, nil, "", "")
434            if profile then
435                profile.snsMap = self._accountManager:getSnsMap(false)
436            end
437        elseif key == "snsDetail" and op == "update" then
438            local snsInfo = table.deserialize(params)
439            self._accountManager:updateSnsInfo(snsInfo.snsPlatformEnum, snsInfo.snsName, snsInfo.name, snsInfo.headUrl, nil, true)
440
441            if profile then
442                profile.snsMap = self._accountManager:getSnsMap(false)
443            end
444        elseif key == "secret" and op == "update" then
445            local secretV2 = self._profileManager:getProfileValue("secretV2")
446
447            local curSecretV2 = table.deserialize(params)
448            if curSecretV2 then
449                for k,v in pairs(curSecretV2) do
450                    secretV2[k] = v
451                end
452            end
453            self._profileManager:setProfileValue("secretV2", secretV2)
454            if profile then
455                profile.secretV2 = secretV2
456            end
457        end
458    end
459    if info and profile then
460        info:setProfile(profile)
461    end
462end
