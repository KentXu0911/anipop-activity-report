412function UBFUEPlpBottomWinRepeatRank:checkHasRankRewards(rank)
413    local rewardInfo = self.rankRewardMap[rank] or self.rankRewardMap[tostring(rank)]
414    local rewards = rewardInfo and rewardInfo.rewards or {}
415    local extraRewards = rewardInfo and rewardInfo.extraRewards
416    local mergeRewards = table.union(rewards, extraRewards)

484function UBFUEPlpBottomWinRepeatRank:startAnim()
485    local myNode = self.myNode
486    if not isSafeCocosObj(myNode) then
487        self:finishAnim()
488        return deferred.success()
489    end
490
491    local refs = self:getCellRefs(myNode)
492    if not refs or not isSafeCocosObj(refs.root) then
493        self:finishAnim()
494        return deferred.success()
495    end
496
497    local flow = deferred.success()
498    local function addTask(task)
499        flow = flow:next(function()
500            if not self:isViewSafe() then return deferred.success() end
501            return task()
502        end)
503    end
504
505    if self.hasScoreChange then
506        local label = refs.collectPartScore
507        addTask(function()
508            return self:deferAction(label, {
509                CCScaleTo:create(3 * FRAMETIME, 1.5, 1.5),
510            })
511        end)
512        addTask(function()
513            return self:deferCall(function(done)
514                if not self:isViewSafe() or not isSafeCocosObj(label) then
515                    done()
516                    return
517                end
518                ActionUtil:rollingNumber({
519                    label = label,
520                    startNumber = self.lastScore,
521                    endNumber = self.myScore,
522                    callback = done,
523                })
524            end)
525        end)
526        addTask(function()
527            return self:deferAction(label, {
528                CCDelayTime:create(10 * FRAMETIME),
529                CCCallFunc:create(function()
530                    if not self:isViewSafe() or not isSafeCocosObj(label) then return end
531                    label:setString(self:getShowScore(self.myScore))
532                end),
533                CCScaleTo:create(3 * FRAMETIME, 1, 1),
534            })
535        end)
536    end
537
538    if self.hasRankChange and self.hasRankUp then
539        addTask(function()
540            return self:deferCall(function(done)
541                if not self:isViewSafe() then
542                    done()
543                    return
544                end
545                local offset = self:getScrollLabelOffsetByRank(self.myRank, false)
546                local curOffset = self.scrollView:getXOffset()
547                if offset <= curOffset then
548                    done()
549                    return
550                end
551                self.scrollView:gotoPositionX(offset, RANK_SCROLL_MOVE_TIME)
552                done()
553            end)
554        end)
555        addTask(function()
556            return self:deferCall(function(done)
557                if not self:isViewSafe() then
558                    done()
559                    return
560                end
561                for i = self.lastRank - 1, self.myRank, -1 do
562                    local node = self.nodeList[i]
563                    local effectHolderViewName = self:getReferencedCellViewName(node and node.cellName, node, "effectHolder")
564                    if isSafeCocosObj(node) and effectHolderViewName then
565                        node:runAction(UBFViewHelper:sequence({
566                            CCDelayTime:create(2 * FRAMETIME * (self.lastRank - i)),
567                            CCCallFunc:create(function()
568                                if not self:isViewSafe() then return end
569                                self:switchNodeRank(node, i + 1, i)
570                            end),
571                            CCMoveBy:create(OTHER_CELL_MOVE_TIME, ccp(CELL_WIDTH, 0)),
572                            CCCallFunc:create(function()
573                                if not self:isViewSafe() then return end
574                                self:playUpDownFx(effectHolderViewName, "animCell_" .. i, "down")
575                            end),
576                        }))
577                    end
578                end
579
580                refs.root:runAction(UBFViewHelper:sequence({
581                    CCScaleTo:create(2 * FRAMETIME, 0.99, 0.99),
582                    CCScaleTo:create(3 * FRAMETIME, 1.15, 1.15),
583                    CCScaleTo:create(((self.lastRank - self.myRank) >= 4 and 22 or 15) * FRAMETIME, 1.2, 1.2),
584                    CCScaleTo:create(2 * FRAMETIME, 0.96, 0.96),
585                    CCScaleTo:create(3 * FRAMETIME, 1, 1),
586                    CCCallFunc:create(function()
587                        if self:isViewSafe() then
588                            local effectHolderViewName = self:getReferencedCellViewName(myNode.cellName, myNode, "effectHolder")
589                            self:playUpDownFx(effectHolderViewName, "_animCell_up", "up")
590                        end
591                        done()
592                    end),
593                }))
594
595                myNode:runAction(UBFViewHelper:sequence({
596                    CCDelayTime:create(5 * FRAMETIME),
597                    CCCallFunc:create(function()
598                        if not self:isViewSafe() then return end
599                        self:switchNodeRank(myNode, self.myRank, self.lastRank)
600                    end),
601                    CCMoveBy:create(((self.lastRank - self.myRank) >= 4 and MY_CELL_LONG_MOVE_TIME or MY_CELL_FAST_MOVE_TIME), ccp(CELL_WIDTH * (self.myRank - self.lastRank), 0)),
602                }))
603            end)
604        end)
605    end

608        addTask(function()

611        addTask(function()

613                if self:isViewSafe() then
614                    local effectHolderViewName = self:getReferencedCellViewName(myNode.cellName, myNode, "effectHolder")
615                    self:playUpDownFx(effectHolderViewName, "myNodeanim", "down")
616                end
617                done()
618            end)
619        end)
620    end

624    end)

636function UBFUEPlpBottomWinRepeatRank:switchNodeRank(node, rank, lastRank)
637    local refs = self:getCellRefs(node)
638    if not refs or not isSafeCocosObj(refs.rankPart) then return end
639
640    local info = node.info
641    info.rank = rank
642
643    local disapearUseTime = 2 * FRAMETIME
644    local showUseTime = 3 * FRAMETIME
645    local allUseTime = disapearUseTime + showUseTime
646    if lastRank > 3 and rank > 3 then
647        refs.topOther:runAction(UBFViewHelper:sequence({
648            ActionUtil:fadeToU3D(refs.topOther, disapearUseTime, 100),
649            CCCallFunc:create(function()
650                if not self:isViewSafe() then return end
651                refs.topOtherLabel:setString("" .. rank)
652            end),
653            ActionUtil:fadeToU3D(refs.topOther, showUseTime, 255),
654        }))
655    else
656        if lastRank > 3 then
657            refs.topOther:runAction(UBFViewHelper:sequence({
658                ActionUtil:fadeOutU3D(refs.topOther, allUseTime),
659            }))
660        else
661            local oldTop = refs["top" .. lastRank]
662            if isSafeCocosObj(oldTop) then
663                oldTop:runAction(CCFadeOut:create(allUseTime))
664            end
665        end
666
667        if rank > 3 then
668            refs.topOtherLabel:setString("" .. rank)
669            refs.topOther:runAction(CCFadeIn:create(allUseTime))
670            refs.topOther:setVisible(true)
671        else
672            local newTop = refs["top" .. rank]
673            if isSafeCocosObj(newTop) then
674                newTop:runAction(CCFadeIn:create(allUseTime))
675                newTop:setVisible(true)
676            end
677        end
678    end
679
680    local giftBox = refs.giftBox
681    local bCurHasBoxShow = giftBox:isVisible()
682    local bHasReward, rewards, extraRewards, mergeRewards = self:checkHasRankRewards(rank)
683    if bCurHasBoxShow then
684        giftBox:runAction(UBFViewHelper:sequence({
685            ActionUtil:fadeToU3D(giftBox, disapearUseTime, 100),
686            CCCallFunc:create(function()
687                if not self:isViewSafe() then return end
688                if bHasReward then
689                    self:updateGiftBoxItemImage(node, rank, mergeRewards)
690                    giftBox:setVisible(true)
691                else
692                    giftBox:setVisible(false)
693                end
694            end),
695            ActionUtil:fadeToU3D(giftBox, showUseTime, 255),
696        }))
697    elseif bHasReward then
698        giftBox:setVisible(true)
699        giftBox:setOpacity(0)
700        self:updateGiftBoxItemImage(node, rank, mergeRewards)
701        giftBox:runAction(UBFViewHelper:sequence({
702            CCFadeIn:create(allUseTime),
703        }))
704    end
705end
