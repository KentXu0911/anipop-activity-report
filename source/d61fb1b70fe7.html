986function WaterfallPanel:doFlyCollectionAction(srcPos, targetPos, done)
987    local function playEndVfx(endPos)
988		if self.isDisposed then return end
989        local anim = UICreateUtil:createSpine("amf/activity/waterfallprogress/spine/reward_vfx.skel")
990        self.ui:addChild(anim)
991        anim:setPosition(endPos)
992        anim:setScale(0.6)
993        anim:playOnceAndDispose("end")
994    end
995
996    local flyNum = 10
997    local flyDownCount = 0
998    local function callBack(endPos)
999        flyDownCount = flyDownCount + 1
1000
1001        if flyDownCount == 1 or flyDownCount == 5 or flyDownCount == 9 then
1002            playEndVfx(endPos)
1003        end
1004
1005        if flyDownCount >= flyNum then
1006            safeRunCall(done)
1007        end
1008    end
1009
1010    local function createBezierObject(startPos, endPos, time)
1011        local p1 = ccp((0.2+0.05*math.random(-2,2))*(endPos.x - startPos.x), (-0.2+0.05*math.random(-2,2))*(endPos.y - startPos.y))
1012        local p2 = ccp((0.8+0.05*math.random(-2,2))*(endPos.x - startPos.x), (-0.2+0.05*math.random(-2,2))*(endPos.y - startPos.y))
1013
1014        local bezierConfig = ccBezierConfig:new()
1015        bezierConfig.controlPoint_1 = ccp(startPos.x +  p1.x, startPos.y +  p1.y)
1016        bezierConfig.controlPoint_2 = ccp(startPos.x +  p2.x, startPos.y +  p2.y)
1017        bezierConfig.endPosition = ccp(endPos.x,endPos.y)
1018        local object = CCEaseSineIn:create(CCBezierTo:create(time, bezierConfig))
1019        return object
1020    end
1021
1022    local function flyCash(index)
