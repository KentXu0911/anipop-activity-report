7local function isMiniGame()
8    return EnvUtil
9        and type(EnvUtil.isMiniGame) == "function"
10        and EnvUtil:isMiniGame() == true
11end
12
13local function createIconAsset()
14    if isMiniGame() then
15        return {
16            type = "spriteFile",
17            path = ICON_V2_MINIGAME,
18        }
19    end
20    return {
21        type = "spine",
22        path = ICON_V2_SPINE,
23        animations = { idle = "idle1", reward = "idle2" },
24    }
25end
26
27local function createSurface()
28    return {
29        size = { width = 136, height = 136 },
30        layers = {
31            {
32                id = "giftBg",
33                zOrder = 0,
34                asset = {
35                    type = "spriteFile",
36                    path = IconV2VisualAssets.GiftPurpleBackground,
37                },
38            },
39        },
40    }
41end
42
43function IconV2Config.getPreloadFile()

45end

47function IconV2Config.getPreloadFiles(path)

55function IconV2Config.create(manager, iconEventCenter)
56    local IconConfig = require "amf.feature.mainscene.module.icon.config.IconConfig"
57    local IconData = require "amf.feature.localactivity.collectcard.icon.CollectCardGiftIconData"
58    local CollectCardGiftPackageIcon = require
59        "amf.feature.localactivity.collectcard.icon.CollectCardGiftPackageIcon"
60    local ownerIconKey = IconKey.kCollectCardGiftIcon
61    local info = assert(IconConfig:getInfoByKey(ownerIconKey), "COLLECT_CARD_GIFT_ICON_INFO_REQUIRED")
62    local provider = IconData:create(manager)
63
64    return {
65        ownerIconKey = ownerIconKey,
66        ownerIdentity = assert(manager.iconModule, "COLLECT_CARD_GIFT_ICON_OWNER_REQUIRED"),
67        iconEventCenter = iconEventCenter,
68        v2Class = CollectCardGiftPackageIcon,
69        legacyFactory = function()
70            return manager.iconModule:createLegacyGiftIcon()
71        end,
72        metadata = {
73            actId = manager.ACT_ID,
74            iconKey = ownerIconKey,
75            iconInfo = info,
76            poolType = info:getPoolType(),
77            basePriority = info:getPriority(),
78            label = "新卡礼包",
79            ambientIdle2Eligible = true,
80            asset = createIconAsset(),
81            surface = createSurface(),
82            newActivityType = "B",
83        },
84        provider = provider,
85        newActivityRecordStore = manager:createCollectCardGiftIconNewActivityRecordStore(),
86        newActivityCycleKeyResolver = function()
87            return provider:resolveCycleKey()
88        end,
89        countdownReachedCommand = function()
90            local iconModule = manager.iconModule
91            if not iconModule or type(iconModule.clearGiftIcon) ~= "function" then
92                return false, "COLLECT_CARD_GIFT_ICON_OWNER_CLEANUP_REQUIRED"
93            end
94            if iconModule.giftIconAdapter == nil and iconModule.giftIcon == nil then
95                return true
96            end
97            return iconModule:clearGiftIcon()
98        end,
99        isNewActivityTimePermitted = function()
100            return manager.iconModule:needShowGiftIcon()
101        end,
102        touchCommand = function()
103            if not manager.iconModule:needShowGiftIcon() then
104                return false, "COLLECT_CARD_GIFT_ICON_NOT_AVAILABLE", false
105            end
106            manager:onGiftIconTapped()
107            return true, nil, true
108        end,
109    }, provider
110end

112return IconV2Config
