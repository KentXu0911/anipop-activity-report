44function MergeGiftModule:createBuyButton(buttonData)
45    if (not buttonData) or type(buttonData) ~= 'table' then return end
46
47    local goodsId = buttonData.goodsId
48    local price,isRMB,discount,originPrice = self:getPrice(goodsId)
49    if discount and discount >= 1 then discount = nil end
50    if buttonData.buyNum then
51        price = price * buttonData.buyNum
52    end
53
54    local suffixStr = buttonData.suffixStr or " 购买"
55    if isRMB then
56        buttonData.str = "￥"..price
57        if buttonData.rmbNeedSuffix then
58            buttonData.str = buttonData.str .. suffixStr
59        end
60    else
61        buttonData.str = {"gold",tostring(price),suffixStr}
62    end
63
64    if buttonData.needOriginPrice then
65        buttonData.originPrice = {text = originPrice,isBelow = true}
66    end
67
68    --控制某些参数，剩下自己发挥
69    buttonData.widthMode = UIButtonConfig.Widths.kShort
70    buttonData.scaleMode = buttonData.scaleMode or UIButtonConfig.BtnScales.k90
71    buttonData.colorMode = UIButtonConfig.ColorMode.blue
72    buttonData.discount = discount and discount * 10
73    if buttonData.noDiscountTip then buttonData.discount = nil end
74
75    local button = UIButton:createButton(buttonData)
76    if buttonData.touchFunc then
77        button:setCallBack(buttonData.touchFunc)
78    end
79    button.isRMB = isRMB
80    return button
81end

84function MergeGiftModule:getPrice(goodsId)
85    if not goodsId or goodsId == 0 then return 0 end
86    local goodsObj = GoodsManager:getGoods(goodsId)
87    if not goodsObj then return 0 end
88    local moneyType = goodsObj:getMoneyType()
89    return goodsObj:getPrice(), moneyType == MoneyType.kRmb, goodsObj:getDiscount() / 10, goodsObj:getOriginPrice()
90end

92function MergeGiftModule:checkTimeToday(str)
93    local actInfo = self.mgr:getActInfo()
94    if not actInfo then return false end
95    if not actInfo.extra then return false end
96    local extra = table.deserialize(actInfo.extra)
97    if not extra then return false end
98    local buyTime = tonumber(extra[str] or 0)
99    return TimeUtil:isSameDay(buyTime, TimeKit:timeInSec())
100end
