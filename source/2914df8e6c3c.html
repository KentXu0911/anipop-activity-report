201-- 初始化卡包动画层（创建卡包节点）
202function OpenCardAnimLayer:initCardAnimLayer()
203    local cardPackLayer = Layer:create()
204    self:addChild(cardPackLayer)
205    local count = 0
206    local nodeInfos = {}
207    local rewards = self.model:getRewards()
208    for i, reward in ipairs(rewards) do
209        if count >= 8 then break end
210        local itemId = reward.itemId
211        if self.model:getManager():isCardPack(itemId) then
212            local level = self.model:getManager():getCardPackLevel(itemId)
213            local sp = self.uiBuilder:createCardPackSpine(level)
214            cardPackLayer:addChildAt(sp, 0)
215            count = count + 1
216            table.insert(nodeInfos, {node = sp, width = 200})
217        end
218    end
219    if not table.isEmpty(nodeInfos) then
220        LayoutUtil.sortNodes(nodeInfos, nil, -280+self.uiBuilder:getCardPackSpace(count))
221    end
222    cardPackLayer:setScale(0)
223    self.cardPackNodes = nodeInfos
224    self.cardPackLayer = cardPackLayer
225end

253--[[
254    OPENING 状态：正在开包
255    流程：更新UI（显示当前进度）-> 等待开包完成
256]]
257function OpenCardAnimLayer:onStateOpening(delegate)
258    self._openingEndCallback = delegate.endCallback
259    self.model:setOpeningIdx(delegate.curIdx)
260    -- OPENING状态下，关闭按钮触发小跳（与全局点击一致）
261    self._openingBackKeyFunc = function(context, evt)
262        if self.isDisposed then return end
263        if not self:isMyOpenCardFlowEvent(evt) then return end
264        evt:setReceived()
265        local stateMachine = self:getStateMachine()
266        if stateMachine and stateMachine:getState() == Config.AnimState.OPENING and not self.animController._cardShowFinish then
267            self.animController:skipOpenCardAnim()
268        end
269    end
270    CollectCardEventCenter:register(CollectCardEvent.CARD_PACK_FLOW_BACK_CLICKED, self._openingBackKeyFunc, self)
271    self:dispatchAnimEvent(CollectCardEvent.CARD_PACK_ANIM_OPENING, {
272        openingIdx = self.model:getOpeningIdx(),
273    })
274end

276function OpenCardAnimLayer:onStateFinishOpening(nextState)
277    if self._openingBackKeyFunc then
278        CollectCardEventCenter:unregister(CollectCardEvent.CARD_PACK_FLOW_BACK_CLICKED, self._openingBackKeyFunc)
279        self._openingBackKeyFunc = nil
280    end
281    -- 如果是跳过，不执行后续逻辑
282    if nextState == Config.AnimState.SKIP then
283        return
284    end
285
286    -- 如果是最后一个卡包，直接结束
287    local isLastPack = self.model:getOpeningIdx() and self.model:getOpeningIdx() >= #self.model:getRewards()
288    if isLastPack then
289        safeRunCall(self._openingEndCallback)
290        return
291    end
292
293    -- 结算重复卡 -> 切换下一个卡包
294    self.model:markPackSettled(self.model:getOpeningIdx())
295    deferred.success()
296    :next(function()
297        if self.isDisposed then return end
298        if self.animController.skipAllCardAnim then return end
299        return self.animController:settleCards(self.model:getShownCardList(), self.scrollShowLayer)
300    end)
301    :next(function()
302        if self.isDisposed then return end
303        if self.animController.skipAllCardAnim then return end
304        return self.animController:switchToNextCardPack()
305    end)
306    :next(function()
307        if self.isDisposed then return end
308        if self.animController.skipAllCardAnim then return end
309        safeRunCall(self._openingEndCallback)
310    end)
311end

367--[[

371function OpenCardAnimLayer:onStateSkip()
372    local totalNum = #self.model:getRewards()
373    local settledNum = self.model:getSettledPackCount()
374    local skippedNum = math.max(totalNum - settledNum, 0)
375    self:dispatchAnimEvent(CollectCardEvent.CARD_PACK_ANIM_SKIP, {
376        skippedNum = skippedNum,
377    })
378
379    local getCardList = self.model:getGetCardList()
380    local allNum = 0
381    for i,v in pairs(getCardList) do
382        allNum = allNum + v
383    end
384    WeekTaskEventCenter:dispatch(_G.WeekTaskEvent.ACT_TASK_COMPLETE, { type=_G.WeekTaskType.CARD_ACT_GET_NUM, num=allNum})
385
386    local showBtnDelay = 25 * Config.FRAME_TIME
387    self:hideClickSkipNode(showBtnDelay)
388    self:hideClickContinueNode()
389    self:removeGlobalClick()
390
391    self.animController:skipAllAnimShowFinal()
392    self._isShowClickBtn2 = false
393    self:showClickBtn("收下", function()
394        if self.isDisposed then return end
395        if self._isShowClickBtn2 then return end
396        self._isShowClickBtn2 = true
397        self:hideClickBtn()
398        self:setFlowCloseBtnVisible(false)
399        deferred.success()
400        :next(function()
401            return self.animController:settleAllCards(self.model:getFinalShownCardList(), self.scrollShowLayer)
402        end)
403        :next(function()
404            if self.isDisposed then return end
405            self.stateMachine:enterState(Config.AnimState.FINISH)
406        end)
407    end, showBtnDelay)
408end
