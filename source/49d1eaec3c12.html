126-- 解析活动标记 repairFlag：0 没有 / 1 可能有 / 2 已经有；解析失败按 0
127function PromoteDailyMgr:parseRepairFlag(extra)
128    if not extra or extra == "" then return 0 end
129    -- actInfo.extra 是服务端下发的 JSON 字符串（联调 mock 需用 table.serialize 生成）
130    local tb = table.deserialize(extra)
131    if type(tb) ~= "table" then
132        self:log("parseRepairFlag: extra is not a table:", tostring(extra))
133        return 0
134    end
135    local flag = tonumber(tb.repairFlag)
136    if flag == nil or flag < 0 or flag > 2 then
137        self:log("parseRepairFlag: invalid flag:", tostring(tb.repairFlag))
138        return 0
139    end
140    return flag
141end

143-- 活动信息更新：只用于「提示补拉主信息」，活动是否可用以 mainInfo.repairTaskInfo 为准
144--   1 / 2：本地没有任务数据时补拉一次（1 是「可能有」，见下面的冷却）
145--   1 / 2 -> 0：任务结束或完成，新旧七日任务数据可能已经变化，补拉一次
146function PromoteDailyMgr:onActInfoUpdate(evt)
147    local data = evt and evt:getData()
148    if not data or not data.fromServer then return end
149    local actInfos = data.actInfos or {}
150    -- 没有这条 actInfo（活动没配 / 已下线）不处理
151    local actInfo = actInfos[tostring(PromoteDailyConst.REPAIR_ACT_ID)]
152    if not actInfo then return end
153    local flag = self:parseRepairFlag(actInfo.extra)
154    -- 缓存最近一次 repairFlag（getRepairFlag 读取；supportRepairTask 现在固定 true）
155    self._repairFlagCache = flag
156    local lastFlag = self._lastActInfoFlag
157    self._lastActInfoFlag = flag
158    -- 标记变了就清掉「flag = 1 且没有任务」的冷却：换了状态说明后端任务数据已经变化
159    if lastFlag ~= nil and lastFlag ~= flag then
160        self._repairFlag1NoTaskCdEndTime = nil
161    end
162
163    local needRequest
164    if flag == 0 then
165        -- 1/2 -> 0：任务结束或完成，七日任务（新旧）数据可能已经变化，补拉一次
166        needRequest = lastFlag == 1 or lastFlag == 2
167    else
168        -- 1/2：本地已经有任务数据就不请求
169        needRequest = not self:isRepairActValid()
170        -- flag = 1 后端「可能有也可能没有任务」：上次请求确认没有任务后冷却期内不再请求
171        if needRequest and flag == 1 then
172            local cdEndTime = self._repairFlag1NoTaskCdEndTime
173            needRequest = cdEndTime == nil or TimeKit:timeInSec() >= cdEndTime
174        end
175        -- 同一个标记连续触发时，请求已经在飞就不重复发
176        if needRequest and self._repairMainInfoRequesting then
177            needRequest = false
178        end
179    end
180    if not needRequest then return end
181    -- 标记翻转（1/2 -> 0）要补发的那次不等上一次返回，直接发
182    self:requestRepairMainInfo()
183end

218-- 重燃活动数据唯一来源：mainInfo 下的 repairTaskInfo 字段（后端 RepairTaskInfoBean），直接读取，不做额外缓存
219-- 后端字段 → 客户端字段：repairPeriodId/startTimeSecond/endTimeSecond/targetProgress/progress
220--                          → repairPeriodId/beginTime/endTime/target/progress
221-- 字段缺失/非法返回 nil（非法记日志）
222function PromoteDailyMgr:getRepairActData()
223    local task = self:getDataValue(PromoteDailyConst.REPAIR_ACT_FIELD)
224    if task == nil then return nil end
225    if type(task) ~= "table" then
226        UError("PromoteDailyMgr: mainInfo." .. tostring(PromoteDailyConst.REPAIR_ACT_FIELD) .. " is not a table")
227        return nil
228    end
229    local beginTime = tonumber(task.startTimeSecond) or 0
230    local endTime = tonumber(task.endTimeSecond) or 0
231    local target = tonumber(task.targetProgress) or 0
232    if beginTime <= 0 or endTime <= beginTime or target <= 0 then
233        UError("PromoteDailyMgr: repairTaskInfo invalid, startTimeSecond/endTimeSecond/targetProgress =",
234            beginTime, endTime, target)
235        return nil
236    end
237    -- 后端 progress 为 byte（0~255），非法值（负数 / 超过目标）按 [0, target] 收敛并记日志，
238    -- 避免负进度被 isRepairActInProgress 误判为「进行中」、或超过目标后影响进度展示
239    local progress = tonumber(task.progress) or 0
240    if progress < 0 or progress > target then
241        UError("PromoteDailyMgr: repairTaskInfo.progress out of range:",
242            progress, "target =", target)
243        if progress < 0 then
244            progress = 0
245        else
246            progress = target
247        end
248    end
249    return {
250        repairPeriodId = tonumber(task.repairPeriodId) or 0,
251        beginTime = beginTime,
252        endTime = endTime,
253        target = target,
254        progress = progress,
255    }
256end

280function PromoteDailyMgr:settleExpiredRepairAct()
281    local act = self:getRepairActData()
282    if not act or act.progress >= act.target or TimeKit:timeInSec() < act.endTime then
283        return false
284    end
285    local changed = false
286    if not self._expiredRepairActSettled then
287        self._expiredRepairActSettled = true
288        self:setDataValue(PromoteDailyConst.CONTINUOUS_PASS_FIELD, 0)
289        changed = true
290    end
291    local year, month, day = self:getYear(), self:getMonth(), self:getDay()
292    if self:getSignState(year, month, day) == PromoteDailyConst.SignState.REPAIR then
293        self:setSignState(year, month, day, PromoteDailyConst.SignState.NONE)
294        self:resetContinuseInfo()
295        changed = true
296    end
297    return changed
298end

300-- 活动到期定时器：到点在本地把界面刷回非活动态。
301-- 只有「进行中」的活动需要排：已达标（结束由过关结算链路负责刷新，见 PromoteDailyLevelLogic:doActPlayLevelEnd）、
302-- 未开始、无活动数据都不排；活动数据每次刷新（主信息返回 / 跨天 / 本地加进度）都重排一次，避免拿到旧 endTime。
303function PromoteDailyMgr:_scheduleRepairActEndRefresh()
304    if self._repairActEndAlarmId then
305        TimerUtil.removeAlarm(self._repairActEndAlarmId)
306        self._repairActEndAlarmId = nil
307    end
308    if not self:isRepairActInProgress() then return end
309    self._expiredRepairActSettled = false
310    local delay = self:getRepairActEndTime() - TimeKit:timeInSec()
311    -- delay <= 0 说明已经到点，交给下一次刷新处理，不排一个「延迟 1 秒」（TimerUtil 会把 <=0 收敛成 1 秒）
312    if delay <= 0 then return end
313    self._repairActEndAlarmId = TimerUtil.addAlarm(function()
314        self._repairActEndAlarmId = nil
315        self:settleExpiredRepairAct()
316        self:refreshRepairActUI()
317    end, delay, 1)
318end

359-- 打点用：断签的日期区间 yyyyMMdd_yyyyMMdd（例 20260101_20260102）
360-- 口径（与连过天数的算法 PromoteDailyModelLogic:resetContinuseInfo / calculateMaxContiNum 同源）：
361--   ① 从今天前一天往前找「上一段连过」的开始 / 结束（状态 SIGN / SHIELD 才算连过），
362--      往前最多回溯到功能开启那天（PromoteDailyConst.ACT_OPEN_DAYNUM，与 calculateMaxContiNum 同一个下限）；
363--   ② 再从连过结束那天开始往今天遍历，取这段里未签到（status = 0）的第一个 / 最后一个日子，
364--      返回「区间第一天_今天前的最后未签到那天」；只有一天时只返回那一天（不带 "_"）。
365function PromoteDailyMgr:getRepairBreakDateRange()
366    local lastNum = self:getTodayNum() - 1
367    if lastNum <= 0 then return "" end
368    -- 功能开启那天：断签只可能发生在它之后，作为往前回溯的下限
369    local minNum = PromoteDailyConst.ACT_OPEN_DAYNUM
370    -- ① 上一段连过：功能开启以来都没有连过（新号 / 数据缺失）时，从功能开启那天开始算
371    local _, contiEndNum = self:getLastContinousPassRange(lastNum, minNum)
372    if contiEndNum <= 0 then
373        self:log("getRepairBreakDateRange: no continous pass range, fallback to ACT_OPEN_DAYNUM =", minNum)
374    end
375    local startNum = contiEndNum > 0 and contiEndNum or minNum
376    if startNum < minNum then startNum = minNum end
377    -- ② 从连过结束那天开始往今天遍历，取未签到的第一个 / 最后一个日子
378    local firstUnsignNum, lastUnsignNum
379    for dayNum = startNum, lastNum do
380        if self:_isRepairUnsignDay(dayNum) then
381            firstUnsignNum = firstUnsignNum or dayNum
382            lastUnsignNum = dayNum
383        end
384    end
385    if not firstUnsignNum then
386        self:log("getRepairBreakDateRange: no unsign day, startNum =", startNum, "lastNum =", lastNum)
387        return ""
388    end
389    -- 只有一天：不上报 "_"，直接给单个日期
390    if firstUnsignNum == lastUnsignNum then
391        return self:_getRepairDateStr(firstUnsignNum)
392    end
393    return self:_getRepairDateStr(firstUnsignNum) .. "_" .. self:_getRepairDateStr(lastUnsignNum)
394end

481-- 活动期间过关：本地累加活动进度（已完成关数）
482-- 说明：进度最终以服务端下发的 mainInfo.repairTaskInfo.progress 为准，这里先把本地表现推进，
483--       主信息刷新后会被服务端值覆盖；上限按 target 截断，避免本地超过目标关数。
484function PromoteDailyMgr:addRepairProgress(count)
485    count = tonumber(count) or 0
486    if count <= 0 then return self:getRepairProgress() end
487    local task = self:getDataValue(PromoteDailyConst.REPAIR_ACT_FIELD)
488    if type(task) ~= "table" then return self:getRepairProgress() end
489    local beginTime = tonumber(task.startTimeSecond) or 0
490    local endTime = tonumber(task.endTimeSecond) or 0
491    local target = tonumber(task.targetProgress) or 0
492    if beginTime <= 0 or endTime <= beginTime or target <= 0 then
493        return self:getRepairProgress()
494    end
495    local progress = tonumber(task.progress) or 0
496    progress = progress + count
497    if progress > target then progress = target end
498    if progress < 0 then progress = 0 end
499    task.progress = progress
500    self:setDataValue(PromoteDailyConst.REPAIR_ACT_FIELD, task)
501    -- 达标当关：先留住 Banner 播「光效飞向 icon → Banner 消失」，播完再摘
502    if progress >= target and self._repairBannerLogic then
503        self._repairBannerLogic:waitSignFlowForFinish()
504    end
505    -- 本地进度到顶即视为活动已达标，刷新 UI
506    PromoteDailyEventCenter:dispatch(PromoteDailyEvent.REPAIR_ACT_UPDATE)
507    -- 达标后活动不再是「进行中」，顺手把到期定时器撤掉（达标结束由过关结算链路刷新）
508    self:_scheduleRepairActEndRefresh()
509    return progress
510end

513-- 说明：连过天数最终以服务端下发的 mainInfo.continuousPassingDay 为准，这里先本地推进表现，

613-- 本地配置是否可用
614function PromoteDailyMgr:isLocalEnabled()
615    if not FeatureSwitch:isOpen(FeatureSwitch.Type.PROMOTE_DAILY) then return false, "FeatureSwitch is not open" end
616    if EnvUtil:isIOS() and MaintenanceManager:getInstance():isInReview() then return false, "提审期间不开放，iOS不开放" end
617    if not self._hasInited then return false, "disable 未初始化" end
618    local userVersion = self:getActValue("version2")
619    -- 前后端版本判断
620    if userVersion ~= PromoteDailyConst.VERSION then
621        return false, "disable 前后端版本不一致，不可用"
622    end
623    -- 版本判断
624    local ver = tonumber(string.split(_G.bundleVersion, ".")[2])
625    local cfgMinVer = tonumber(string.split(self:getActValue("minVersion"), ".")[2]) or 0
626    if ver < cfgMinVer then
627        return false, "disable 版本过低，不可用" .. ver .. "_" .. cfgMinVer
628    end
629    -- 开启等级判断
630    local openLevel = self:getActValue("activityOpenLevel")
631    if _G.getTopPassedLevelId() < openLevel then
632        return false, "disable 等级过低，不可用" .. openLevel
633    end
634    -- 是否在开启时间内
635    local beginTime = self:getActValue("beginTime") or 0
636    local endTime = self:getActValue("endTime") or 0
637    local curTime = TimeKit:timeInSec()
638    if curTime <= beginTime or curTime >= endTime then
639        return false, "disable 不在开启时间内"
640    end
641
642    return true, "ok"
643end
