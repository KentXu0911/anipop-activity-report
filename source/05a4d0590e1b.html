129--- 奖励强弹资格。
130--- 场景2每天最多强弹一次（needTime=true 的本地标记按自然日失效），场景4不做次数限制。
131--- @param stageParams table|nil 框架透传的场景参数，含 popStage
132function Manager:checkCanPopRewardMainPanel(stageParams)
133    if not self:isPlatformSupport() then return false end
134    if not self:isEnabled() then return false end
135    if not self:getModel():isInMainTime() then return false end
136
137    -- 奖励资格由服务端判断，Manager 只读取主协议结果。
138    if self:getDataByKey("canReceiveReward", false) ~= true then return false end
139
140    if not self:isRewardPopStageTwo(stageParams) then return true end
141
142    -- 场景2每天只弹一次：needTime=true 的本地标记按自然日失效，跨天自动重置。
143    local cacheKey = self:getConst().LocalCacheKey.RewardMainPanelPoppedToday
144    return not self:getFlagFromLocalByKey(cacheKey, true)
145end

153function Manager:popRewardMainPanel(onPopOver, stageParams)
154    if not self:checkCanPopRewardMainPanel(stageParams) then

157    end
158
159    -- 场景2弹过就落当日标记，避免同一天重复弹；场景4不落标记，不做限制。

162    end

166        endCallback = onPopOver,
167        failCallback = onPopOver,
168        extra = {
169            source = "auto_popup_reward",
170            autoPopType = "reward",
171        },
172    })
173    if opened == false then safeRunCall(onPopOver) end

409function Manager:_registerCrossPeriodCallbacks()
410    local PassDayTimer = require "amf.script.tool.timer.PassDayTimer"
411    local model = self:getModel()
412
413    self:_unregisterCrossPeriodCallbacks()
414
415    local mainBegin = model:getMainBeginTimeInSec()
416    local mainEnd = model:getMainEndTimeInSec()
417    local actEnd = model:getEndTimeInSec()
418
419    if mainBegin and mainBegin > 0 then
420        if mainEnd and mainEnd ~= mainBegin then
421            self._cbMainBegin = function() self:onActPassMainBegin() end
422            self._cbMainEnd = function() self:onActPassMainEnd() end
423            PassDayTimer:registerEndTimeCallback(mainBegin, self._cbMainBegin)
424            PassDayTimer:registerEndTimeCallback(mainEnd, self._cbMainEnd)
425        end
426        if actEnd and actEnd ~= mainEnd then
427            self._cbActEnd = function() self:onActPassActEnd() end
428            PassDayTimer:registerEndTimeCallback(actEnd, self._cbActEnd)
429        end
430    end
431end
432
433function Manager:_unregisterCrossPeriodCallbacks()
434    local PassDayTimer = require "amf.script.tool.timer.PassDayTimer"
435    local model = self:getModel()
436    if self._cbMainBegin then
437        PassDayTimer:unregisterEndTimeCallback(model:getMainBeginTimeInSec(), self._cbMainBegin)
438        self._cbMainBegin = nil
439    end
440    if self._cbMainEnd then
441        PassDayTimer:unregisterEndTimeCallback(model:getMainEndTimeInSec(), self._cbMainEnd)
442        self._cbMainEnd = nil
443    end
444    if self._cbActEnd then
445        PassDayTimer:unregisterEndTimeCallback(model:getEndTimeInSec(), self._cbActEnd)
446        self._cbActEnd = nil
447    end
448end
