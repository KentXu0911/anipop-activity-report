256function Manager:checkIconTip(config)
257    -- 抽条要加载资源，防止小游戏走加载资源逻辑
258    if EnvUtil:isMiniGame() then return end
259    if not self:isEnabled() then return end
260    local addProgress = tonumber(self.cacheAddProgress) or 0
261    if addProgress <= 0 then return end
262
263    local progressBeforeAdd = tonumber(self.cacheProgressBeforeAdd)
264    if not progressBeforeAdd then
265        progressBeforeAdd = math.max((tonumber(self:getProgress()) or 0) - addProgress, 0)
266    end
267
268    local stageProgressBeforeAdd = 0
269    local stageNeedProgress = 0
270    local completedProgress = 0
271    -- 动画档位由加进度前的位置决定，与奖励是否已领取无关。
272    for _, info in ipairs(self:getRewardStageProgressList()) do
273        local needProgress = tonumber(info.needProgress) or 0
274        if needProgress > 0 and progressBeforeAdd < completedProgress + needProgress then
275            stageProgressBeforeAdd = math.max(progressBeforeAdd - completedProgress, 0)
276            stageNeedProgress = needProgress
277            break
278        end
279        completedProgress = completedProgress + needProgress
280    end
281    if stageNeedProgress <= 0 then
282        self:clearIconTipProgressCache()
283        return
284    end
285
286    local targetProgress = stageProgressBeforeAdd + addProgress
287    local remain = stageNeedProgress - math.min(targetProgress, stageNeedProgress)
288    if remain > 20 then return end
289
290    self:addIconTip(stageProgressBeforeAdd, targetProgress, stageNeedProgress, config)
291end

815-- 此接口只读取当前翻倍状态，不消费翻倍次数；次数统一在 updatePassLevelData 中消费。
816---@param levelId number 本次结算的关卡 ID，预留给按关卡判断翻倍资格的规则。
817---@param levelScene string 本次结算的关卡场景，预留给按场景判断翻倍资格的规则。

820    if not self:isEnabled() then return false end
821    if not self:hasDailyDoubleRemain() then return false end
822
823    if LevelSceneFunc:isFullLevelLib(levelScene) then
824        return true
825    end

828    local scoreNode = levelAreaManager and levelAreaManager:getUserScore(levelId, levelScene) or {}
829    local star = tonumber(scoreNode.star) or 0
830
831    if LevelSceneFunc:isHideLevel(levelScene) then
832        return star ~= 3
833    end
834
835    if LevelSceneFunc:isMainLevel(levelScene) then
836        local topPassedLevelId = _G.getTopPassedLevelId()
837        if not topPassedLevelId then return false end
838
839        if self:getLocalLevelStatus() == 1 then
840            return tonumber(levelId) == topPassedLevelId + 1
841        else
842            return star ~= 4
843        end
844    end
845
846    return false
847end
