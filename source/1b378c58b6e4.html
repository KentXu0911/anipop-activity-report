79function ActivityCenterManager:hasReward()
80    for _, data in pairs(self:getVisibleDatas()) do
81        if data.actId then
82            local flag = self:getLocalCache("HasReward" .. tostring(data.actId))
83            local bHasReward = data.buttonText ~= "" and data.hasReward and not flag
84            if bHasReward then
85                return true
86            end
87        end
88    end
89
90    return false
91end

193function ActivityCenterManager:onActivityResolveOver(event)

203    if result then
204        local config = activityNode:getConfig()

266function ActivityCenterManager:getNewActInfo()
267    local latestData
268    for _, v in pairs(self:getVisibleDatas()) do
269        if not latestData or (v.getBeginTime() > latestData.getBeginTime()) then
270            latestData = v
271        end
272    end
273    local info = {
274        hasNew = latestData and true or false,
275        data = latestData,
276        newActKey = tostring(latestData and latestData.actId or "none")
277    }
