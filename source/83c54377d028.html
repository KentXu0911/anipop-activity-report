36local function first(state, paths, defaultValue)
37    for pathIndex = 1, #paths do
38        local value = state
39        local path = paths[pathIndex]
40        for index = 1, #path do
41            if type(value) ~= "table" then
42                value = nil
43                break
44            end
45            value = value[path[index]]
46        end
47        if value ~= nil then
48            return value
49        end
50    end
51    return defaultValue

54local function describePriorityOrder(order)
55    if type(order) ~= "table" then
56        return tostring(order)
57    end
58    local values = {}
59    for key, value in pairs(order) do
60        values[#values + 1] = tostring(key) .. "=" .. tostring(value)
61    end
62    table.sort(values)
63    return table.concat(values, ",")
64end

66local function validatePriorityOrder(order)
67    if order == nil then
68        return copyArray(DefaultPriorityOrder)
69    end
70    if type(order) ~= "table" then
71        return nil, "priority order must be a table"
72    end
73
74    local keyCount = 0
75    for key in pairs(order) do
76        keyCount = keyCount + 1
77        if type(key) ~= "number" or key % 1 ~= 0 or key < 1 or key > #DefaultPriorityOrder then
78            return nil, "priority order must be a dense array with at most five items"
79        end
80    end
81
82    local seen = {}
83    local copied = {}
84    for index = 1, keyCount do
85        local kind = order[index]
86        if kind == nil then
87            return nil, "priority order must be a dense array"
88        end
89        if not ValidReminderKinds[kind] then
90            return nil, "unsupported reminder kind " .. tostring(kind)
91        end
92        if seen[kind] then
93            return nil, "duplicate reminder kind " .. kind
94        end
95        seen[kind] = true
96        copied[index] = kind
97    end
98    for index = 1, #DefaultPriorityOrder do
99        local kind = DefaultPriorityOrder[index]
100        if not seen[kind] then
101            copied[#copied + 1] = kind
102        end
103    end
104    return copied
105end

107local function resolveReminderDynamic(reminder, kind)
108    -- Product policy: reward reminder animation is disabled.
109    if kind == "reward" then
110        return false
111    end
112    local dynamicKinds = type(reminder.dynamicKinds) == "table" and reminder.dynamicKinds or {}
113    if dynamicKinds[kind] ~= nil then
114        return dynamicKinds[kind] == true
115    end
116    return false
117end

119local function normalizeReminderIntent(intent)
120    intent = type(intent) == "table" and intent or {}
121    local priorityOrder, priorityError = validatePriorityOrder(intent.priorityOrder)
122    if not priorityOrder then
123        priorityOrder = copyArray(DefaultPriorityOrder)
124    end
125    local count = type(intent.count) == "number" and intent.count or 0
126    local dynamicKinds = {}
127    for kind in pairs(ValidReminderKinds) do
128        dynamicKinds[kind] = resolveReminderDynamic(intent, kind)
129    end
130    return {
131        visible = {
132            reward = intent.rewardVisible == true,
133            exclamation = intent.exclamationVisible == true,
134            new = intent.newVisible == true,
135            count = count > 0,
136            redDot = intent.redDotVisible == true or intent.hintVisible == true,
137        },
138        count = count,
139        dynamicKinds = dynamicKinds,
140        priorityOrder = priorityOrder,
141        priorityError = priorityError,
142        prioritySignature = describePriorityOrder(intent.priorityOrder),
143    }
144end

216local function selectReminder(reminder)

241local function inputFingerprint(reminder, discount)
242    local values = {
243        tostring(discount.visible),
244        tostring(discount.discount),
245        tostring(reminder.count),
246    }
247    for index = 1, #DefaultPriorityOrder do

312function RightTopSlotResolver:_recordPriorityDiagnostic(reminder)
313    if not reminder.priorityError then
314        self.lastPriorityDiagnostic = nil
315        return
316    end
317    local signature = reminder.priorityError .. ":" .. reminder.prioritySignature
318    if self.lastPriorityDiagnostic == signature then
319        return
320    end
321    self.lastPriorityDiagnostic = signature
322    local diagnostic = {
323        code = "ICON_V2_INVALID_REMINDER_PRIORITY_ORDER",
324        detail = reminder.priorityError,
325        configuredOrder = reminder.prioritySignature,
326        iconKey = self.icon and self.icon.iconKey,
327    }
328    self.diagnostics[#self.diagnostics + 1] = diagnostic
329    if ALog and ALog.error then
330        ALog.error(diagnostic.code .. ": iconKey=" .. tostring(diagnostic.iconKey))
331    end
332end

334function RightTopSlotResolver:_buildSnapshot(previousSnapshot)
335    self.revision = self.revision + 1
336    if self.discountIntent.visible and self.bubbleRenderer then
337        return {
338            slot = "rightTop",
339            kind = "discount",
340            discount = self.discountIntent.discount,
341            revision = self.revision,
342        }
343    end
344
345    local reminder = selectReminder(self.reminderIntent)
346    if not reminder then
347        return {
348            slot = "rightTop",
349            kind = "none",
350            revision = self.revision,
351        }
352    end
353    local recoveredFromDiscount = previousSnapshot and previousSnapshot.kind == "discount"
354    reminder.playAnimation = reminder.dynamic and not recoveredFromDiscount
355    return {
356        slot = "rightTop",
357        kind = "reminder",
358        reminder = reminder,
359        recoveredFromDiscount = recoveredFromDiscount == true,
360        revision = self.revision,
361    }
362end

364function RightTopSlotResolver:_applyReminderSnapshot(snapshot)

396function RightTopSlotResolver:_resolve(forceCommit)
397    local currentInputFingerprint = inputFingerprint(self.reminderIntent, self.discountIntent)
398    if forceCommit ~= true
399        and self.currentSnapshot
400        and self.lastInputFingerprint == currentInputFingerprint then
401        return {
402            changed = false,
403            selectionChanged = false,
404            snapshot = self.currentSnapshot,
405            recoveredFromDiscount = self.currentSnapshot.recoveredFromDiscount == true,
406        }
407    end
408    local previousSnapshot = self.currentSnapshot
409    local snapshot = self:_buildSnapshot(previousSnapshot)
410    local fingerprint = visualFingerprint(snapshot)
411    local changed = forceCommit == true or fingerprint ~= self.lastVisualFingerprint
412    self.currentSnapshot = snapshot
413    self.lastInputFingerprint = currentInputFingerprint
414    if changed then
415        self.lastVisualFingerprint = fingerprint
416        self:_commit(snapshot)
417    end
418    local previousReminderKind = previousSnapshot and previousSnapshot.reminder and previousSnapshot.reminder.kind
419    local currentReminderKind = snapshot.reminder and snapshot.reminder.kind
420    return {
421        changed = changed,
422        selectionChanged = previousSnapshot == nil
423            or previousSnapshot.kind ~= snapshot.kind
424            or previousReminderKind ~= currentReminderKind,
425        snapshot = snapshot,
426        recoveredFromDiscount = snapshot.recoveredFromDiscount == true,
427    }
428end
