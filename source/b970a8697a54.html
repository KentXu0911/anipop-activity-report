26local function tryEnableSwapAdvice()
27    if EnvUtil:isMiniGame() then return end
28    local inst = UBFGameSceneManager:getInstance()
29    if not inst then return end
30    local modModule = inst:getGameModule(UBFGameLogics.kGameMod)
31    if not modModule then return end
32
33    local guideMgr = modModule:getModMgrBySourceId(UBFModSourceIdEnum.kGuide)
34    if not guideMgr then return end
35    
36    local boardViewLogic = inst:getBoardViewLogic()
37    local advice = boardViewLogic:getExtendLogic(UBFBoardViewExtendLogics.kMatchAdvice)
38    if not advice then return end
39
40    guideMgr:updateAdviceSwap()
41    local swapList = advice:getLastSwapData()
42    if not swapList or #swapList == 0 then return end
43
44    local hasNice
45    for _, v in ipairs(swapList) do
46        local evaluate = v.evaluate or 0
47        if evaluate >= 8 then hasNice = true break end
48    end
49
50    if not hasNice then return end
51    guideMgr:rightNowShowSwapAdvice(true)
52end

1406    local state

1420    else
1421        state = UBFSuperMB2024SlpEntryState.NORMAL_CHARGING
1422    end

1426    nodeData.maxProgress = maxProgress
1427    nodeData.cardExpireAt = reviveCardState and reviveCardState.cardExpireAt or nodeData.cardExpireAt
1428    nodeData.refreshReason = refreshReason or "update"

1431        nodeData.iconIsActive = true
1432        nodeData.iconProgressNew = maxProgress
1433        nodeData.iconProgressOld = maxProgress
1434        nodeData.iconMaxProgress = maxProgress
