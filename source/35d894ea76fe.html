423function Model:hasBattlePassReward()
424    local value = self:getBattlePassValue()
425    local hasVip = self:getBattlePassVip()
426    local vipGot = self:getBattlePassRewardsId(true)
427    local freeGot = self:getBattlePassRewardsId(false)
428    local cfg = self:getConfigByKey("battlePass", {})
429
430    local canGetFree = {}
431    local canGetVip = {}
432
433    for id, rewardCfg in ipairs(cfg) do
434        local need = rewardCfg.need or 0
435        local meetNeed = value >= need
436
437        if meetNeed then
438            if not freeGot[id] then
439                table.insert(canGetFree, id)
440            end
441
442            if hasVip and not vipGot[id] then
443                table.insert(canGetVip, id)
444            end
445        end
446    end
447
448    local hasReward = (#canGetFree > 0) or (#canGetVip > 0)
449    return hasReward, canGetFree, canGetVip
450end

452function Model:clearBattlePassReward()

454
455    if not table.isEmpty(canGetFree) then

457        for _, v in ipairs(canGetFree) do
458            table.insert(ids, v)
459        end
460        self:setDataByKey("battlePass.freeAlreadyGotIds", ids)
461    end
462
463    if not table.isEmpty(canGetVip) then
464        local ids = self:getDataByKey("battlePass.vipAlreadyGotIds", {})
465        for _, v in ipairs(canGetVip) do
466            table.insert(ids, v)
467        end
468        self:setDataByKey("battlePass.vipAlreadyGotIds", ids)
469    end
470end
