248function Manager:hasRewards()
249    if not self.localDataServer:isBaseDataEnabled() then 
250        local actInfo = self:getActInfo()
251        return actInfo.reward
252    end 
253
254    local data = self:getData()
255    local nowCollectNum = tonumber(data:getDataByKey("nowCollectNum", 0)) or 0
256
257    if nowCollectNum > 0
258        and (self:hasGetAllMap() or data:isActInReward()) then
259        return true
260    end
261    local nowMapId = data:getNowMapId()
262    local nowMap = data:getMapBeanById(nowMapId)
263
264    if nowMap and not nowMap.progressReward then
265        local openNum, allNum = data:getMapGemNum(nowMapId)
266        if allNum > 0 and openNum >= allNum then
267            return true
268        end
269    end
270
271    return false
272end

321--解析活动数据
322function Manager:parseData(data)
323    if not data then return end
324
325    local localActPeriodId = self:getData():getDataByKey("periodId")
326    local serverActPeriodId = data.periodId
327    local hasLocalPeriod = localActPeriodId ~= nil and tostring(localActPeriodId) ~= "0"
328    local hasServerPeriod = serverActPeriodId ~= nil and tostring(serverActPeriodId) ~= "0"
329    local periodChanged = hasLocalPeriod and hasServerPeriod
330        and tostring(localActPeriodId) ~= tostring(serverActPeriodId)
331
332    if periodChanged then
333        local localConfig = self:getData():getDataByKey("localConfig")
334        self:getData():resetForNewPeriod()
335        if not data.extra and localConfig and not table.isEmpty(localConfig) then
336            data.extra = table.serialize(localConfig)
337        end
338    end
339
340    self:getData():fromServer(data,nil)
341    if serverActPeriodId ~= nil then
342        self:getData():setDataByKey("periodId", serverActPeriodId, true, true)
343    end
344    if not self.hasServerData then
345        self:updateTipView()
346        self.hasServerData = true
347    end
348end
