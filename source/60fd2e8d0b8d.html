7local function emitDiagnostic(logger, code, data)
8    if logger == nil then
9        return
10    end
11    if type(logger) == "function" then
12        pcall(logger, code, data)
13        return
14    end
15    if type(logger) ~= "table" then
16        return
17    end
18    local method = logger.warn or logger.log
19    if type(method) == "function" then
20        pcall(method, logger, code, data)
21    end
22end

24local function safeCallback(callback, result, logger)
25    if type(callback) ~= "function" then
26        return
27    end
28    local ok, err = pcall(callback, result)
29    if not ok then
30        emitDiagnostic(logger, "ICON_V2_NOTIFICATION_COMPLETION_CALLBACK_ERROR", {
31            error = tostring(err),
32            stage = result and result.stage,
33            token = result and result.token,
34        })
35    end
36end

38local function callRecordMethod(records, methodName, key)
39    if records == nil or key == nil then
40        return false
41    end
42    local method = records[methodName]
43    if type(method) ~= "function" then
44        return false
45    end
46    local ok, value = pcall(method, records, key)
47    if not ok then
48        return false
49    end
50    if methodName == "isConsumed" or methodName == "get" then
51        return value == true or (type(value) == "number" and value > 0)
52    end
53    return value ~= false
54end

56local function isConsumed(records, key)
57    if records == nil or key == nil then
58        return false
59    end
60    if type(records.isConsumed) == "function" then
61        return callRecordMethod(records, "isConsumed", key)
62    end
63    return callRecordMethod(records, "get", key)
64end
65
66local function consume(records, key)
67    if records == nil or key == nil then
68        return false
69    end
70    if type(records.consume) == "function" then
71        return callRecordMethod(records, "consume", key)
72    end
73    return callRecordMethod(records, "set", key)
74end

83local function settleRequest(request, result, logger)
84    if request == nil or request.settled then

115function NotificationCoordinator:beginFlow(options)
116    options = options or {}
117    self.flowId = self.flowId + 1
118    self.stage2Status = options.stage2Expected == true and "expected" or "unavailable"
119    local previousFallbacks = self.fallbackRequests
120    self.fallbackRequests = {}
121    for index = 1, #previousFallbacks do
122        settleRequest(previousFallbacks[index], {
123            displayed = false,

131function NotificationCoordinator:_restartBatchBeforeVisible(reason)
132    local batch = self.activeBatch
133    if batch == nil or batch.visible then
134        return false
135    end
136    batch.invalidated = true
137    self.activeBatch = nil
138    for index = 1, #batch.entries do

152function NotificationCoordinator:register(candidate)
153    if type(candidate) ~= "table" or type(candidate.cycleKey) ~= "string" or candidate.cycleKey == "" then
154        emitDiagnostic(self.logger, "ICON_V2_NOTIFICATION_CANDIDATE_INVALID", {
155            cycleKey = candidate and candidate.cycleKey,
156        })
157        return false, "ICON_V2_NOTIFICATION_CANDIDATE_INVALID"
158    end
159    if isConsumed(self.records, candidate.noticeShownRecordKey) then
160        self.entries[candidate.cycleKey] = {
161            candidate = candidate,
162            state = "consumed",
163        }
164        return false, "ICON_V2_NOTIFICATION_ALREADY_CONSUMED"
165    end
166    if isConsumed(self.records, candidate.newClearedRecordKey) then
167        self.entries[candidate.cycleKey] = nil
168        return false, "ICON_V2_NOTIFICATION_NEW_ALREADY_CLEARED"
169    end
170
171    local entry = self.entries[candidate.cycleKey]
172    if entry and entry.state == "record_failed" then
173        return false, "ICON_V2_NOTIFICATION_RECORD_WRITE_FAILED"
174    end
175    if entry and (entry.state == "reserved" or entry.state == "showing" or entry.state == "consumed") then
176        return false, "ICON_V2_NOTIFICATION_ALREADY_ACTIVE"
177    end
178    self.entries[candidate.cycleKey] = {
179        candidate = candidate,
180        state = "pending",
181    }
182    self:_restartBatchBeforeVisible("candidate_registered_before_visible")
183    return true
184end

200function NotificationCoordinator:_collectPending()
201    local pending = {}
202    for cycleKey, entry in pairs(self.entries) do
203        if entry.state == "pending" then
204            local candidate = entry.candidate
205            if isConsumed(self.records, candidate.noticeShownRecordKey) then
206                entry.state = "consumed"
207            elseif isConsumed(self.records, candidate.newClearedRecordKey) then
208                self.entries[cycleKey] = nil
209            else
210                table.insert(pending, entry)
211            end
212        end
213    end
214    table.sort(pending, function(left, right)
215        local leftOrder = left.candidate.poolIdx or math.huge
216        local rightOrder = right.candidate.poolIdx or math.huge
217        if leftOrder == rightOrder then
218            return left.candidate.cycleKey < right.candidate.cycleKey
219        end
220        return leftOrder < rightOrder
221    end)
222    if type(self.maxBatchSize) == "number" and self.maxBatchSize > 0 then
223        while #pending > self.maxBatchSize do
224            table.remove(pending)
225        end
226    end
227    return pending
228end

234function NotificationCoordinator:_settleBatch(batch, result)
235    if not self:_isCurrentBatch(batch) then
236        return false
237    end
238    batch.invalidated = true
239    self.activeBatch = nil
240    for index = 1, #batch.requests do
241        settleRequest(batch.requests[index], result, self.logger)
242    end
243    if batch.stage == PopStage.STAGE_TWO then
244        self.stage2Status = result.displayed and "completed" or "failed"
245        self:_drainFallbackRequests()
246    end
247    return true

250function NotificationCoordinator:_releaseBeforeVisible(batch, reason)
251    if not self:_isCurrentBatch(batch) then
252        return false
253    end
254    for index = 1, #batch.entries do
255        local entry = batch.entries[index]
256        if entry.state == "reserved" and entry.batchId == batch.id then
257            if isConsumed(self.records, entry.candidate.newClearedRecordKey) then
258                self.entries[entry.candidate.cycleKey] = nil
259            else
260                entry.state = "pending"
261                entry.batchId = nil
262            end
263        end
264    end
265    return self:_settleBatch(batch, {
266        displayed = false,
267        reason = reason or "before_visible_failure",
268        stage = batch.stage,
269        token = batch.token,

273function NotificationCoordinator:_markVisible(batch)
274    if not self:_isCurrentBatch(batch) or batch.visible then
275        return false
276    end
277    local visibleCount = 0
278    for index = 1, #batch.entries do
279        local entry = batch.entries[index]
280        local candidate = entry.candidate
281        if entry.state == "reserved" and entry.batchId == batch.id
282            and not isConsumed(self.records, candidate.newClearedRecordKey)
283        then
284            visibleCount = visibleCount + 1
285            local noticeWritten = consume(self.records, candidate.noticeShownRecordKey)
286            if not noticeWritten then
287                emitDiagnostic(self.logger, "ICON_V2_NOTICE_RECORD_WRITE_FAILED", {
288                    cycleKey = candidate.cycleKey,
289                    recordKey = candidate.noticeShownRecordKey,
290                })
291            end
292            local featureWritten = true
293            if candidate.temporaryC then
294                featureWritten = consume(self.records, candidate.firstCRecordKey)
295                if not featureWritten then
296                    emitDiagnostic(self.logger, "ICON_V2_FIRST_C_RECORD_WRITE_FAILED", {
297                        cycleKey = candidate.cycleKey,
298                        recordKey = candidate.firstCRecordKey,
299                    })
300                end
301            end
302            if noticeWritten then
303                entry.state = "showing"
304            else
305                entry.state = "record_failed"
306            end
307            batch.recordWriteFailed = batch.recordWriteFailed == true
308                or not noticeWritten or not featureWritten
309        elseif entry.state == "reserved" and entry.batchId == batch.id then
310            self.entries[candidate.cycleKey] = nil
311        end
312    end
313    if visibleCount == 0 then
314        return self:_releaseBeforeVisible(batch, "cancelled_before_visible")
315    end
316    batch.visible = true
317    return true
318end

320function NotificationCoordinator:_completeBatch(batch, reason)
321    if not self:_isCurrentBatch(batch) then
322        return false
323    end
324    if not batch.visible then
325        return self:_releaseBeforeVisible(batch, reason or "completed_before_visible")
326    end
327    for index = 1, #batch.entries do
328        local entry = batch.entries[index]
329        if entry.batchId == batch.id then
330            if entry.state == "showing" then
331                entry.state = "consumed"
332            end
333            entry.batchId = nil
334        end
335    end
336    local persistenceFailed = batch.recordWriteFailed == true
337    return self:_settleBatch(batch, {
338        displayed = true,
339        reason = reason or (persistenceFailed and "record_write_failed" or "completed"),
340        stage = batch.stage,
341        token = batch.token,
342        persistenceFailed = persistenceFailed,
343    })
344end

346function NotificationCoordinator:_startBatch(stage, requests)
347    local pending = self:_collectPending()
348    if #pending == 0 then
349        local result = {
350            displayed = false,
351            reason = "no_pending_candidate",
352            stage = stage,
353        }
354        for index = 1, #requests do
355            settleRequest(requests[index], result, self.logger)
356        end
357        if stage == PopStage.STAGE_TWO then
358            self.stage2Status = "completed"
359            self:_drainFallbackRequests()
360        end
361        return nil
362    end
363
364    self.nextBatchId = self.nextBatchId + 1
365    local batch = {
366        id = self.nextBatchId,
367        flowId = self.flowId,
368        stage = stage,
369        entries = pending,
370        requests = requests,
371        visible = false,
372    }
373    batch.token = table.concat({ tostring(batch.flowId), stage, tostring(batch.id) }, ":")
374    for index = 1, #pending do
375        pending[index].state = "reserved"
376        pending[index].batchId = batch.id
377    end
378    self.activeBatch = batch
379    if stage == PopStage.STAGE_TWO then
380        self.stage2Status = "running"
381    end
382
383    local hooks = {
384        token = batch.token,
385        onVisible = function()
386            return self:_markVisible(batch)
387        end,
388        onComplete = function(reason)
389            return self:_completeBatch(batch, reason)
390        end,
391        onFailure = function(reason)
392            if batch.visible then
393                return self:_completeBatch(batch, reason or "failure_after_visible")
394            end
395            return self:_releaseBeforeVisible(batch, reason)
396        end,
397    }
398
399    if type(self.presenter) ~= "function" then
400        self:_releaseBeforeVisible(batch, "presenter_missing")
401        return batch.token
402    end
403    local candidates = {}
404    for index = 1, #pending do
405        candidates[index] = pending[index].candidate
406    end
407    local ok, control = pcall(self.presenter, candidates, hooks)
408    if not ok then
409        emitDiagnostic(self.logger, "ICON_V2_NOTIFICATION_PRESENTER_ERROR", {
410            error = tostring(control),
411            token = batch.token,
412        })
413        self:_releaseBeforeVisible(batch, "presenter_error")
414    elseif type(control) == "table" then
415        batch.control = control
416    end
417    return batch.token
418end

420function NotificationCoordinator:_drainFallbackRequests()
421    if #self.fallbackRequests == 0 or self.activeBatch ~= nil then
422        return
423    end
424    local requests = self.fallbackRequests
425    self.fallbackRequests = {}
426    self:_startBatch(PopStage.STAGE_FOUR, requests)
427end

429function NotificationCoordinator:request(stage, callback)
430    assert(stage == PopStage.STAGE_TWO or stage == PopStage.STAGE_FOUR,
431        "ICON_V2_NOTIFICATION_STAGE_INVALID")
432    local request = newRequest(callback)
433    local active = self.activeBatch
434    if active ~= nil then
435        if active.stage == stage then
436            table.insert(active.requests, request)
437        elseif stage == PopStage.STAGE_FOUR and active.stage == PopStage.STAGE_TWO then
438            table.insert(self.fallbackRequests, request)
439        else
440            settleRequest(request, {
441                displayed = false,
442                reason = "another_batch_active",
443                stage = stage,
444            }, self.logger)
445        end
446        return active.token
447    end
448
449    if stage == PopStage.STAGE_TWO and self.stage2Status == "unavailable" then
450        settleRequest(request, {
451            displayed = false,
452            reason = "stage2_unavailable",
453            stage = stage,
454        }, self.logger)
455        return nil, "ICON_V2_NOTIFICATION_STAGE2_UNAVAILABLE"
456    end
457    if stage == PopStage.STAGE_FOUR
458        and (self.stage2Status == "expected" or self.stage2Status == "running") then
459        table.insert(self.fallbackRequests, request)
460        return nil, "ICON_V2_NOTIFICATION_STAGE4_WAITING_FOR_STAGE2"
461    end
462    return self:_startBatch(stage, { request })
463end

465function NotificationCoordinator:markStage2Unavailable(reason)

474function NotificationCoordinator:cancelBeforeVisible(cycleKey, reason)
475    local entry = self.entries[cycleKey]
476    if entry == nil then
477        return false
478    end
479    local batch = self.activeBatch
480    if entry.state == "pending" then
481        self.entries[cycleKey] = nil
482        return true
483    end
484    if entry.state == "reserved" and batch and entry.batchId == batch.id then
485        self.entries[cycleKey] = nil
486        entry.state = "cancelled"
487        entry.batchId = nil
488        local remaining = 0
489        for index = 1, #batch.entries do
490            local batchEntry = batch.entries[index]
491            if batchEntry.state == "reserved" and batchEntry.batchId == batch.id then
492                remaining = remaining + 1
493            end
494        end
495        if remaining == 0 then
496            if batch.control and type(batch.control.cancel) == "function" then
497                pcall(batch.control.cancel, batch.control, reason)
498            end
499            self:_releaseBeforeVisible(batch, reason or "cancelled_before_visible")
500        end
501        return true
502    end
503    if entry.state == "showing" or entry.state == "record_failed" then
504        if batch and batch.control and type(batch.control.skipToFinal) == "function" then
505            pcall(batch.control.skipToFinal, batch.control, reason)
506        end
507        return false, "ICON_V2_NOTIFICATION_ALREADY_VISIBLE"
508    end
509    return false
510end
