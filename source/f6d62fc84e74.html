157    -- icon:setScale(scale)
158    LayoutUtil:centerAdd(itemNode, icon)
159    local itemNum = itemNode.mbMap.itemNum
160    itemNum:removeFromParentAndCleanup(false)
161    itemNode:addChild(itemNum)
162    itemNode.mbMap.itemNum:setString(YH_REF_MAP.x .. reward.num)
163    if _G.amfBean("PropManager"):isMergableProp(reward.itemId) or reward.itemId == Meta.DoubleBuffId then
164        itemNode.mbMap.itemNum:setVisible(false)

168        UICreateUtil:move(itemNode.mbMap.itemNum, -5, 0)

223function MGift:onButton()
224    if self.isDisposed then return end
225    if self:isLocked() then return end
226
227    self.tracer:click()
228    local rewards = {}
229    local buyPowerNum = 0
230    for _, v in pairs(self.goodsObj:getItems()) do
231        if v.itemId == Meta.CollectionId then
232            buyPowerNum = v.num
233        end
234        local r = {itemId = v.itemId, num = v.num}
235        rewards[#rewards + 1] = r
236    end
237
238    local function successCallback(bBuy)
239        if bBuy then
240            self:lock()
241            local source = self.from
242            local goodsId = self.goodsId
243            local sucCall = function(...)
244                local mgr = _G.amfBean("MergeManager")
245                mgr:getData():addBuyGoodsCount(goodsId, 1)
246
247                if self.isDisposed then return end
248                self:unlock()
249                self.buyTime = self.buyTime + 1
250                self:refreshData()
251                self:refresh()
252                self:flyRewards()
253                safeRunCall(self.buySuccessFunc, self.goodsId)
254            end
255
256            local failCall = function(...)
257                if self.isDisposed then return end
258                self:unlock()
259            end
260
261            local cancelCall = function(...)
262                if self.isDisposed then return end
263                self:unlock()
264            end
265
266            self.buyLogic:requestBuyWithTracer(sucCall, failCall, cancelCall, self.tracer, self.targetId)
267        end
268    end
269
270    local totalPower = self.mgr:getData():getPowerTotal()
271    if buyPowerNum > 0 and totalPower + buyPowerNum > Meta.MaxPowerConfig then
272        self.mgr:openSecondConfirmPanel(successCallback, totalPower, buyPowerNum)
273    else
274        successCallback(true)
275    end
276end
