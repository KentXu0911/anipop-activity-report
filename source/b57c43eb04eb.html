95function NewLevelAtmosphereModule:getNextRequireTime()
96    local nowTime = TimeKit:timeInSec()
97    if not self.clockEndTime then return end
98    if not self.lastRandomTime then
99        self.lastRandomTime = 60 * randomCirleMinute
100    end
101    local newRandomTime = math.random(60 * randomCirleMinute)
102    local nextRequireTimeNeed = newRandomTime + ( randomCirleMinute * 60) - self.lastRandomTime
103    self.lastRandomTime = newRandomTime
104    return math.max(nextRequireTimeNeed + TimeKit:timeInSec(),self.clockEndTime - 60 * 60 * 2 - math.random(60 * randomCirleMinute) )--在线的话第一次同步放到16.50~17.00
105end

107function NewLevelAtmosphereModule:setTopNewLevelCountDownInfoClock()

109    local PassDayTimer = require"amf.script.tool.timer.PassDayTimer"

112        return
113    end

127        return

132        self:requireTopNewLevelCountDownInfo()

134        if nextRequireTime < newLevelOpenTime then
135            self.nextRequireTime = nextRequireTime
136            self.requireTimeFunc = getNewFunc
137            PassDayTimer:registerEndTimeCallback(self.nextRequireTime,self.requireTimeFunc)
138        end
139    end
