1-- @ref 集卡三选一礼包 → 强弹注册
2-- 集卡三选一礼包（4100）是开心卡册（9087）同期的礼包活动，活动包自身没有 Manager，
3-- 所以它的强弹注册与可弹判定都挂在本模块：由 CollectCardManager 的 AutoPop plug 统一注册与注销，
4-- 可弹判定（活动状态校验、场景强弹 meta、24 小时 CD、强弹条件）全部收在本模块，
5-- 活动包（collectcardthreeforone）只提供活动配置数据与面板打开。
6local ThreeForOneGiftModule = class("ThreeForOneGiftModule")
7
8-- 集卡三选一礼包活动 id
9ThreeForOneGiftModule.ACT_ID = 4100

11-- 活动标识，同时用于活动包路径与强弹本地状态 key
12local ACT_KEY = "collectcardthreeforone"
13
14-- 活动包 Config.lua 的运行期路径：活动 launcher 加载后与 ActivityNode 共用同一份缓存
15local ACT_CONFIG_PATH = "activity/" .. ACT_KEY .. "/Config.lua"
16
17-- 强弹条件类型：与活动包强弹配置下发的 conditionId 一致
18local AUTO_POP_TYPE = {
19    WINDCOIN = 1,
20    ENERGY = 2,
21    LOGIN = 3,
22    CONTINUEFAIL = 4,
23}
24
25-- 强弹 CD：弹出后 24 小时内不再弹
26local STRONG_PUSH_CD_SECONDS = 24 * 3600
27
28-- 上次强弹弹出时间 key：沿用活动包原来的 key 组成（活动标识 + actId + 本期起止时间），换期后自动失效
29local KEY_LAST_POPOUT_TIME = "StrongPushLastPopoutTime"
30
31-- 连续失败次数 key：由本模块自持，与活动包旧的 ContinueFailCount 区分，避免迁移期间两边同时计数
32local KEY_CONTINUE_FAIL_COUNT = "StrongPushContinueFailCount"
33
34-- 强弹场景：PopupRegistration 会在 check/pop 的入参上补 popStage，用它换算出活动包强弹 meta 的 sceneId
35-- （2 = 进入主场景，4 = 关后）。运行时取 _G.AutoPopupConst，避免依赖 require 顺序。
36local function _getSceneIdByPopStage(popStage)
37    if popStage == AutoPopupConst.PopStage.STAGE_TWO then
38        return 2
39    end
40    if popStage == AutoPopupConst.PopStage.STAGE_FOUR then
41        return 4
42    end
43    return nil
44end

61---强弹本地状态 key：活动标识 + actId + 本期起止时间 + key，换期后旧数据不再命中

73---上次强弹弹出时间（秒）

78---记录本次强弹弹出时间：弹出即写入，之后 24 小时内不再弹

83---强弹剩余 CD 秒数，小于等于 0 表示不在 CD 内

97---关卡结算：胜场清零、败场累加，供「连续失败 N 次」强弹条件使用
98function ThreeForOneGiftModule:onActPlayLevelEnd(event)
99    if not self:isActSupported() then
100        return
101    end
102
103    local data = event and event:getData() or {}
104    if data.isWin then
105        self:setContinueFailCount(0)
106    else
107        self:setContinueFailCount(self:getContinueFailCount() + 1)
108    end
109end

111---强弹条件：风车币少于阈值
112function ThreeForOneGiftModule:_checkRemainWindCoin(checkCount)
113    checkCount = checkCount or 30
114    local userManager = _G.amfBean("UserManager")
115    local coinNumber = userManager and userManager:getCash() or 0
116    return coinNumber < checkCount
117end

119---强弹条件：体力少于阈值（无限体力不弹）
120function ThreeForOneGiftModule:_checkRemainEnergy(checkCount)
121    checkCount = checkCount or 5
122    local energyManager = _G.amfBean("UserEnergyManager")
123    if not energyManager then
124        return false
125    end
126    if energyManager:getEnergyState() == UserEnergyState.INFINITE then
127        return false
128    end
129    return energyManager:getEnergy() < checkCount
130end

137---强弹条件：连续失败次数达到阈值

143---单条强弹 meta 是否满足条件，conditionId / conditionValue 由服务端强弹配置下发
144function ThreeForOneGiftModule:checkStrongPushCondition(meta)
145    if not meta then
146        return false
147    end
148
149    local conditionId = tonumber(meta.conditionId) or 0
150    local conditionValue = tonumber(meta.conditionValue) or 0
151    if conditionId == AUTO_POP_TYPE.WINDCOIN then
152        return self:_checkRemainWindCoin(conditionValue)
153    elseif conditionId == AUTO_POP_TYPE.ENERGY then
154        return self:_checkRemainEnergy(conditionValue)
155    elseif conditionId == AUTO_POP_TYPE.LOGIN then
156        return self:_checkOnEnterAutoPop()
157    elseif conditionId == AUTO_POP_TYPE.CONTINUEFAIL then
158        return self:_checkContinueFailCount(conditionValue)
159    end
160    return false
161end

163---该场景的强弹 meta 是否可弹：24 小时 CD + 命中任意一条强弹条件
164---不通过时把原因作为第二个返回值透出，框架会记进强弹场景日志
165function ThreeForOneGiftModule:checkStrongPushConditions(metaList)
166    if table.isEmpty(metaList or {}) then
167        return false, "strong push meta list empty"
168    end
169
170    local leftSeconds = self:getStrongPushCdLeftSeconds()
171    if leftSeconds > 0 then
172        return false, "strong push in cd, leftSeconds=" .. tostring(leftSeconds)
173    end
174
175    for _, meta in ipairs(metaList) do
176        if self:checkStrongPushCondition(meta) then
177            return true
178        end
179    end
180    return false, "all strong push conditions failed"
181end

183---场景强弹是否可弹：按当前强弹场景取该场景的强弹 meta，再走本模块的可弹判断
184function ThreeForOneGiftModule:checkStrongPush(stageParams)
185    if not self:isActSupported() then
186        return false
187    end
188
189    local sceneId = _getSceneIdByPopStage(stageParams and stageParams.popStage)
190    if not sceneId then
191        return false
192    end
193
194    return self:checkPopMainFunction(self:_getActConfig().getStrongPushMetaListByScene(sceneId))
195end

197---场景强弹是否可弹：活动强弹基础校验 + 24 小时 CD + 强弹条件
198---原实现在活动包（collectcardthreeforone）的 config.checkPopMainFunction 与 src/model/Model.lua，现都收在本模块
199function ThreeForOneGiftModule:checkPopMainFunction(strongPushMeta)
200    local supportFailReason = self:_getActConfig().getStrongPushSupportFailReason()
201    if supportFailReason then
202        return false, supportFailReason
203    end
204
205    return self:checkStrongPushConditions(strongPushMeta)
206end

208---场景强弹弹出：记录 24 小时 CD 后交给活动包开面板；不弹的每个分支都必须回调 onPopOver，否则强弹 FSM 会卡住
209function ThreeForOneGiftModule:popStrongPush(onPopOver)
210    if not self:isActSupported() then
211        safeRunCall(onPopOver)
212        return
213    end
214
215    -- 先记 CD 再开面板：面板打开失败或玩家中途退出也算弹过，与活动包原逻辑一致
216    self:setStrongPushLastPopoutTime()
217    self:_getActConfig().popMainFunction(onPopOver)
218end

220---注册式强弹声明，由 CollectCardManager:_getAutoPopConfigList 合并后统一注册
221function ThreeForOneGiftModule:getAutoPopConfigList()
222    -- 集卡三选一礼包主面板强弹：场景2/场景4 共用一个检测点，由 popStage 决定用哪个场景的强弹条件；
223    -- 次数由本模块的 24 小时 CD 控制，因此框架侧不再叠加每日次数限制。
224    return {
225        {
226            checkPoint = PopupCheckPoint.COLLECT_CARD_THREE_FOR_ONE_MAIN_POP,
227            stages = {AutoPopupConst.PopStage.STAGE_TWO, AutoPopupConst.PopStage.STAGE_FOUR},
228            actId = self.ACT_ID,
229            check = function(stageParams)
230                return self:checkStrongPush(stageParams)
231            end,
232            pop = function(_, onPopOver)
233                return function()
234                    self:popStrongPush(onPopOver)
235                end
236            end,
237        },
238    }
239end

241return ThreeForOneGiftModule
