6local function normalizeStageId(context)
7    if context and context.popStageId ~= nil then
8        local stageId = tonumber(context.popStageId)
9        if stageId then return stageId end
10    end
11
12    local popStage = context and context.popStage
13    for stageId, mappedStage in pairs(AutoPopupConst.PopStageKeyMapping) do
14        if mappedStage == popStage then
15            return tonumber(stageId)
16        end
17    end
18end

77function Manager:updatePredictionData(data)

99function Manager:_requestPredictionData(source)
100    -- printx(33, "AIPopupManager:_requestPredictionData", source, self._requestInFlight, self:_shouldFetchToday())
101    if not self:isMaintenanceEnabled() or self._requestInFlight or not self:_shouldFetchToday() then
102        return false
103    end
104    local model = self:getModel()
105    self._requestInFlight = true
106    local request = model:requestAiAutoPopInfo()
107    if not request then
108        self._requestInFlight = false
109        return false
110    end
111    request:next(function(response)
112        self._requestInFlight = false
113        local data = response
114        if response then
115            data = response:getData() or {}
116        end
117        if data == nil then
118            data = {}
119        end
120        if self:updatePredictionData(data) then
121            model:setLastFetchTime(TimeKit:timeInSec())
122        end
123    end, function()
124        self._requestInFlight = false
125    end)
126    return true
127end

165function Manager:_trackAIInterveneShow(popupEntry)
166    -- 被 AI 干预的凭据由调度写在 popupEntry 上：originalCheckPoint/originalActId
167    -- 只有真被挪过顺序位置的强弹才有值；位置没变的强弹即使有 AI 权重也不打点。
168    if not popupEntry or not popupEntry.originalCheckPoint or popupEntry.actId == nil then
169        return
170    end
171    -- printx(33, "AIPopupManager:_trackAIInterveneShow", tts(popupEntry),
172    --     tostring(popupEntry.originalCheckPoint), tostring(popupEntry.originalActId))
173
174    local targetDate = self:_getTargetDate()
175    local aiPopupRecInfo = self:getModel():getAiPopupRecInfoByDate(targetDate)
176    if not aiPopupRecInfo then return end
177
178    DcUtil:UserTrack({
179        category = "ai_recommend",
180        sub_category = "main_panel_show_intervene",
181        ai_popuprec_info = mime.b64(table.serialize(aiPopupRecInfo)),
182        t1 = popupEntry.actId,
183        t2 = popupEntry.originalActId,
184    })
185end

191--- 只重排带 actId 且有模型权重的候选项。没有 actId 或权重的强弹保留原顺序，
192-- 同一活动的多个强弹保持原有顺序。回写顺序由 PopupSchedule 统一完成。
193-- 传入的 candidates 已经是静态顺序，被挤掉位置的原候选由 PopupSchedule 记在
194-- popupEntry 上，这里只负责给出重排后的顺序。
195function Manager:resolveSchedule(candidates, context)
196    if not self:isEnabled() then return candidates end
197    if type(candidates) ~= "table" or #candidates < 2 then return candidates end
198
199    local stageId = normalizeStageId(context)
200    if stageId ~= 3 and stageId ~= 4 then return candidates end
201
202    local targetDate = self:_getTargetDate()
203    local ranked = {}
204    local rankedIndexes = {}
205    for index, candidate in ipairs(candidates) do
206        local hasScore, priority = self:_getActivityPriority(targetDate, candidate.actId)
207        if hasScore then
208            ranked[#ranked + 1] = {
209                candidate = candidate,
210                index = index,
211                priority = priority,
212            }
213            rankedIndexes[#rankedIndexes + 1] = index
214        end
215    end
216
217    if #ranked < 2 then return candidates end
218
219    table.sort(ranked, function(left, right)
220        if left.priority ~= right.priority then
221            return left.priority > right.priority
222        end
223        return left.index < right.index
224    end)
225
226    local result = {}
227    for index, candidate in ipairs(candidates) do
228        result[index] = candidate.checkPoint
229    end
230    for index, candidateIndex in ipairs(rankedIndexes) do
231        local candidate = ranked[index].candidate
232        result[candidateIndex] = candidate.checkPoint
233    end
234    return result
235end
