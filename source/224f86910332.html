341function FixBirdNestManager:onPassLevel(levelContext)
342	if not levelContext then return end
343	self.needRelife = false
344	self.pass_playId = levelContext.playId
345	-- 是否可以收集，关卡内发出来的
346	self.canShowBottom = levelContext.isPass
347	self.alreadyAllFinish = false
348	-- -- 所有任务完成就不处理
349	if self.model:isAllTaskFinish() then
350		self.alreadyAllFinish = true
351		return
352	end
353	local lastCurTask = self.model:getCurTaskInfo()
354	lastCurTask.showFinished = true
355	self.model:saveRecord("curTaskInLevel", lastCurTask)
356	self.levelPassed = true
357	local curTaskInfo = self.model:getRealCurTaskInfo()
358	local progress = curTaskInfo.progress
359	local targetProgress = progress + 1
360	local newTaskInfo = curTaskInfo
361	newTaskInfo.progress = targetProgress
362	if targetProgress >= curTaskInfo.target then
363		local curLife = self.model:getLifeNum()
364		if curLife < self.model:getMaxLife() then
365			self.needRelife = true
366		end
367		self.model:setLifeNum(self.model:getMaxLife())
368		self.model:resetCurTask(newTaskInfo)
369		self.model:addFinishedTasks(1)
370		self:tryGetMedal()
371		self.dc:dcTaskSuccess(curTaskInfo.taskId)
372	else
373		self.model:resetCurTask(newTaskInfo)
374	end
375	self:recordLastData(newTaskInfo)
376	self.model:saveRecord("lastProgress", progress)
377	self:setBeforeProgress()
378end

380function FixBirdNestManager:onFailLevel(levelContext)
381	if not levelContext then return end
382	self.canShowBottom = false
383	self.needRelife = false
384	-- 所有任务完成就不处理
385	if self.model:isAllTaskFinish() then
386		return
387	end
388	self.levelFailed = true
389	local curTaskInfo = self.model:getRealCurTaskInfo()
390	self:recordLastData(curTaskInfo)
391	if curTaskInfo.progress == 0 then
392		return
393	end
394	self.model:addLifeNum(-1)
395	self.model:addLostLives(1)
396	-- 如果没命了要重置
397	if self.model:getLifeNum() <= 0 then
398		self.model:setLifeNum(self.model:getMaxLife())
399		local newTaskInfo = curTaskInfo
400		newTaskInfo.progress = 0
401		self.model:resetCurTask(newTaskInfo)
402		self.failedTask = true
403	end
404	if levelContext.failReason == "retry" then
405		self:tryUpdateIcon()
406	end
407end
