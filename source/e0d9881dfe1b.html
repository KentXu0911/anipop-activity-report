176function Handler:_makeNoFlyRewards(rewards)
177    if table.isEmpty(rewards) then return rewards end
178
179    local noFlyRewards = {}
180    for index, reward in ipairs(rewards) do
181        if type(reward) == "table" then
182            noFlyRewards[index] = table.clone(reward) or {}
183            noFlyRewards[index].doNotFly = true
184        else
185            noFlyRewards[index] = reward
186        end
187    end
188    return noFlyRewards
189end
