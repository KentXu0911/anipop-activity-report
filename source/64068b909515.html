129function FullStarLevelLibManager:requestMainInfo(successCallback, failCallback, cancelCallback, noLoading)
130    if self.isRequesting then
131        return
132    end
133
134    local request = require("amf.feature.fullstarlevellib.http.RandomChallengeMainInfo")
135    self.isRequesting = true
136
137    local function onSuccess(event)
138        local data = event:getData()
139        self.isRequesting = false
140        self:updateData(data)
141        safeRunCall(successCallback)
142
143        if not self.firstUpdateMainInfo then
144            FullStarLevelLibEventCenter:dispatch(FullStarLevelLibEvent.FULL_STAR_LEVEL_LIB_FIRST_UPDATE)
145            self.firstUpdateMainInfo = true
146        end
147    end
148
149    local function onFail(event)
150        self.isRequesting = false
151        safeRunCall(failCallback)
152    end
153
154    local function onCancel(event)
155        self.isRequesting = false
156        safeRunCall(cancelCallback)
157    end
158
159    request:post({
160        noLoading = noLoading
161    }):next(onSuccess, function(response)
162        if response:isCancel() then
163            onCancel(response)
164        else
165            onFail(response)
166        end
167    end)
168end

552        return {}
553    end

556    local retLevelReward = {}
557    for k, v in ipairs(tbl) do
558        local data = string.split(v, ":")
