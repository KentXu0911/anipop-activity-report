92function Manager:onActInfoUpdate(evt)
93    local data = evt:getData()
94    local fromServer = data.fromServer
95    if not fromServer then return end
96    local actInfos = data.actInfos or {}
97    local actInfo = actInfos[tostring(self:getConst().actId)]
98    if not actInfo then
99        self:clearData()
100        return
101    end
102
103    self:registerGiftModule()
104
105    local extra = self:callActFunc("getActInfoExtraData")
106
107    if extra then
108        local beginTime = extra.beginTime
109        local mainEndTime = extra.mainEndTime
110        local endTime = extra.endTime
111        local time = {
112            beginTime     = beginTime,
113            mainBeginTime = beginTime,
114            mainEndTime   = mainEndTime,
115            endTime       = endTime,
116        }
117        local model = self:getModel()
118        local timeChanged = _toSec(beginTime) ~= model:getBeginTime()
119            or _toSec(mainEndTime) ~= model:getMainEndTime()
120            or _toSec(endTime) ~= model:getEndTime()
121        if timeChanged then
122            model:clearData()
123            model:updateTimeInfo(time)
124        end
125    end
126
127    if (self:isInMainTime() or self:isInRewardTime()) and not self:isResEnabled() then
128        self:callDownloadFunc("startDownload")
129    end
130end

292function Manager:getCurAvailableBuffAndEndTime(data)
293    local Const = self:getConst()
294    local now = TimeKit:timeInSec()
295    local buffRate = 0
296    local buffEndTime = 0
297
298    local score20EndTime = tonumber(self:getBuffExpireTsInSec(Const.BuffType.SCORE_20)) or 0
299    if score20EndTime > now then
300        buffRate = buffRate + 0.20
301        buffEndTime = score20EndTime
302    end
303
304    local score50EndTime = tonumber(self:getBuffExpireTsInSec(Const.BuffType.SCORE_50)) or 0
305    if score50EndTime > now then
306        buffRate = buffRate + 0.50
307        buffEndTime = buffEndTime > 0 and math.min(buffEndTime, score50EndTime) or score50EndTime
308    end
309
310    return buffRate, buffEndTime

438function Manager:getYinliuFinalRewards()
439    local cfg = self:getLocalConfig() or {}
440    local rewards = {}
441
442    local finalRewards = cfg.finalRewards or cfg.finalRewardList or cfg.finalRewardItemList
443    if not table.isEmpty(finalRewards) then
444        rewards = finalRewards
445    else
446        local stageRewards = cfg.stageRewardMetas or {}
447        local finalMeta
448        for _, meta in ipairs(stageRewards) do
449            if not finalMeta then
450                finalMeta = meta
451            else
452                local progress = meta.lastProgressId or meta.progressId or meta.id or 0
453                local finalProgress = finalMeta.lastProgressId or finalMeta.progressId or finalMeta.id or 0
454                if progress > finalProgress then
455                    finalMeta = meta
456                end
457            end
458        end
459        rewards = finalMeta and (finalMeta.rewards or finalMeta.rewardList or finalMeta.itemList) or {}
460    end
461
462    return rewards
463end

1051function Manager:_onCustomGain(event)
1052    -- PropEventCenter 派发的是 AMFEvent，必须用 getData() 取原始 reward table。
1053    local reward = event:getData()
1054    if not reward then return end
1055
1056    local Const = self:getConst()
1057    if reward.itemId == Const.ItemId.ENERGY then
1058        if self:isGraduated() then return end
1059        -- 活动体力直接累加到 ExploreMapGameModel.energy，不进入通用 bag。
1060        self:getModel():addEnergy(reward.num)
1061    elseif reward.itemId == Const.ItemId.BUFF_DOUBLE then
1062        -- 收集物翻倍 buff：reward.num 语义为分钟，Model:addBuffTime 内部换算为毫秒并叠加过期时间。
1063        self:getModel():addBuffTime(Const.BuffType.DOUBLE, reward.num)
1064    elseif reward.itemId == Const.ItemId.BUFF_SCORE_20 then
1065        -- 分数 +20% buff：scoreBuff 图标由 items_active_config.lua 的 effectType=14/effectValue=120 驱动。
1066        self:getModel():addBuffTime(Const.BuffType.SCORE_20, reward.num)
1067    elseif reward.itemId == Const.ItemId.BUFF_SCORE_50 then
1068        -- 分数 +50% buff：scoreBuff 图标由 items_active_config.lua 的 effectType=14/effectValue=150 驱动。
1069        self:getModel():addBuffTime(Const.BuffType.SCORE_50, reward.num)
1070    end
1071
1072    local isExploreReward = reward.itemId == Const.ItemId.ENERGY
1073        or reward.itemId == Const.ItemId.BUFF_DOUBLE
1074        or reward.itemId == Const.ItemId.BUFF_SCORE_20
1075        or reward.itemId == Const.ItemId.BUFF_SCORE_50
1076        or self:_isExploreMapGameProp(reward.itemId)
1077    if reward._fromLuckyExpress and isExploreReward then
1078        -- 直接快递发奖不经过活动联动入口，仅在该来源下刷新一次活动提示。
1079        if not self._luckyExpressTipRefreshPending then
1080            self._luckyExpressTipRefreshPending = true
1081            setTimeOut(function()
1082                self._luckyExpressTipRefreshPending = nil
1083                self:callPlug("Act", "updateTipView")
1084            end, 0)
1085        end
1086    end
1087end
