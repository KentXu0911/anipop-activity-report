397function MusicMasterModel:getRankInfoListAndSort()
398    local rankMemberInfoList = self:getRankInfoList{}
399
400    local profile = _G.amfBean("ProfileManager"):getProfile()
401    local headUrl = _G.amfBean("ProfileManager"):getHeadUrl()
402    local meUid = profile.xxlId
403
404    --找出自己
405    for i, v in ipairs(rankMemberInfoList) do
406        if v.uid == meUid  or (tonumber(v.uid) == tonumber(profile.uid)) then
407            v.isMe = true
408            v.avatarUrl = headUrl or "0"
409        elseif v.avatarUrl == nil then
410            v.avatarUrl = '0'
411        end
412    end
413
414    --把自己放入第一位，固定位置
415    local showRankInfoList = {}
416    for i, v in ipairs(rankMemberInfoList) do
417        if v.isMe then
418            local userInfo = table.clone(v)
419            userInfo.showPos = #showRankInfoList+1
420            table.insert( showRankInfoList, userInfo)
421            break
422        end
423    end
424    for i, v in ipairs(rankMemberInfoList) do
425        if not v.isMe then
426            local userInfo = table.clone(v)
427            userInfo.showPos = #showRankInfoList+1
428            table.insert( showRankInfoList, userInfo)
429        end
430    end
431
432    --加入空玩家
433    for i=1, 5 do
434        if not rankMemberInfoList[i] then
435            local userInfo = {}
436            userInfo.showPos = #showRankInfoList+1
437            userInfo.isEmpty = true
438            userInfo.score = 0
439            userInfo.uid = '0'--区分真人 机器人
440            userInfo.id = -1 --区分真人 机器人
441            userInfo.timeOfArrival = ""..(TimeKit:time()+1000)
442            userInfo.avatarUrl = ''
443            table.insert( showRankInfoList, userInfo)
444        end
445    end
446
447    ---拍名次
448    local scoreList = {}
449    
450    --按分数排序
451    table.sort( showRankInfoList, function (a, b)
452        if a and b and a.score and b.score then
453            if a.score==b.score then
454                if tonumber(a.timeOfArrival) == tonumber(b.timeOfArrival) then
455                    return a.uid > b.uid
456                else
457                    return tonumber(a.timeOfArrival) < tonumber(b.timeOfArrival)
458                end
459            else
460                return a.score > b.score
461            end
462        end
463
464        return false
465    end)
466
467    --给排名
468    for i,v in ipairs(showRankInfoList) do
469        v.rank = i
470    end
471
472    --重新按顺序排序
473    table.sort( showRankInfoList, function(a,b)
474        return a.showPos < b.showPos
475    end)
476
477    --填入旧分数
478    local rankMemberOldProgressList = self:getRankMemberOldProgressList()
479    local function getPlayerOldScore( id, uid, score )
480        local oldScore = score
481        if rankMemberOldProgressList then
482            for i,v in ipairs(rankMemberOldProgressList) do
483                if tonumber(uid) ~= 0 then
484                    if tonumber(v.uid) == uid then
485                        oldScore = v.score
486                        break
487                    end
488                else
489                    if tonumber(v.id) == id then
490                        oldScore = v.score
491                        break
492                    end
493                end
494            end
495        end
496        return oldScore
497    end
498    for i,v in ipairs(showRankInfoList) do
499        local oldScore = getPlayerOldScore( v.id, v.uid, v.score )
500        v.oldScore = oldScore
501    end
502
503    return showRankInfoList
504end

589function MusicMasterModel:getProgressCurCanGetRewardInfo( )
590    --获取进度奖励队列
591    local rewardedMusicNum = self:getRewardedMusicNum()
592    local musicNum = self:getCollectItem()
593    local progressRewardList = self:getProgressRewardListConfig()
594
595    local canGetRewardInfoList = {}
596    local bCanStartInset = false
597
598    for i,v in ipairs(progressRewardList) do
599
600        local startNeedNum = 0
601        local endNeedNum = v.threshold
602        if progressRewardList[i-1] then
603            startNeedNum = progressRewardList[i-1].threshold
604        end
605
606        if not bCanStartInset and rewardedMusicNum >= startNeedNum and rewardedMusicNum < endNeedNum then
607            bCanStartInset = true
608        end
609
610        if bCanStartInset then
611            local info = table.clone(v)
612
613            local needNum = endNeedNum - startNeedNum
614            local startShowNum = rewardedMusicNum - startNeedNum
615            if startShowNum < 0 then startShowNum = 0 end
616
617            local endShowNum = musicNum - startNeedNum
618            if endShowNum > needNum then endShowNum = needNum end
619
620            local bCanGetReward = false
621            if endShowNum >= needNum then
622                bCanGetReward = true
623            end
624
625
626            info.progressInfo = {
627                needNum = needNum,
628                startShowNum = startShowNum,
629                endShowNum = endShowNum,
630                bFinal = i==#progressRewardList,
631                bCanGetReward = bCanGetReward,
632                index = i,
633            }
634            table.insert( canGetRewardInfoList, info )
635        end
636
637        if musicNum < endNeedNum then
638            break
639        end
640    end
641    return canGetRewardInfoList
642end

746            if v.uid == meUid  or (tonumber(v.uid) == tonumber(profile.uid)) then
747                if v.score < targetWinNum then
748                    if bAdd then
749                        v.score = v.score + 1
750                        v.timeOfArrival = ""..TimeKit:time()
751                    else
752                        v.score = 0
753                        v.timeOfArrival = ""..TimeKit:time()
754                    end
755                end
756                break
757            end
758        end
759        self:updateRankInfoList(rankInfoList)
760    end
761end
