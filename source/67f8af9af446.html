6local TempActPanelHandler = class("TempActPanelHandler")
7local ActivityEventCenter = require "amf.middleware.activity.event.ActivityEventCenter"
8local AutoPopupDebugLog = require "amf.middleware.autopop.AutoPopupDebugLog"
9
10function TempActPanelHandler:create(autoPopupMgr)
11	local handler = TempActPanelHandler.new()
12	handler:init(autoPopupMgr)
13	return handler
14end
15
16function TempActPanelHandler:init(autoPopupMgr)
17    self._autoPopupMgr = autoPopupMgr
18	AutoPopEventCenter:register(AutoPopEvent.TMP_ACT_PANEL, self.onHandlePopup, self)
19end
20
21function TempActPanelHandler:onHandlePopup(evt)
22    local data = evt:getData()
23	local popupCheckPoint = data.popupCheckPoint
24	local params = data.params
25    
26	lua_switch(popupCheckPoint){
27		[PopupCheckPoint.ACT_MAIN_PANEL_STAGE_2] = function ()
28			self:handleActMainPanel(params, popupCheckPoint)
29		end,
30        [PopupCheckPoint.ACT_MAIN_PANEL_STAGE_4] = function ()
31            self:handleActMainPanel(params, popupCheckPoint)
32        end,
33		default = function ()
34			assert(false, "TempActPanelHandler invalid popupCheckPoint : " .. tostring(popupCheckPoint))
35		end
36	}
37end
38
39function TempActPanelHandler:handleActMainPanel(params, popupCheckPoint)
40    local metaId = params.metaId
41    local actId = params.actId
42
43    local function onHandleOver()
44        ActivityEventCenter:dispatch(ActivityEvent.ACTIVITY_POPUP_OVER,{
45            metaId = metaId
46        })
47        AutoPopEventCenter:dispatch(AutoPopEvent.SINGLE_POP_OVER, popupCheckPoint)
48    end
49
50    local function onStopAutoPopup()
51        --停掉强弹
52        self._autoPopupMgr:resetPopup()
53    end
54    
55    local function creator()
56        local activityMgr = _G.amfBean("ActivityMgr")
57        if not activityMgr:checkHasAutoPopupByMetaId(metaId) then
58            AutoPopupDebugLog:log(metaId .. "checkHasAutoPopupByMetaId return false, stop auto popup")
59            onHandleOver()
60            return
61        end
62
63        ActivityEventCenter:dispatch(ActivityEvent.OPEN_ACTIVITY,{
64            actId = actId,
65            userClick = false,
66        })
67
68        safeRunCall(onStopAutoPopup)
69        ActivityEventCenter:dispatch(ActivityEvent.ACTIVITY_POPUP_OVER,{
70            metaId = metaId
71        })
72    end
73    
74    AutoPopupDC:setDcExtraParam1(popupCheckPoint, actId)
75    self._autoPopupMgr:addPopupConfig(popupCheckPoint, creator, actId)
76end
77
78return TempActPanelHandler
