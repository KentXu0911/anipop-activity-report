87function Module:createFarmStarPanelTip(onCountDownStop)
88    if not self:isOpen() then return nil end
89
90    local content = getOtherPartBuilder():createChild("contentText")
91    if not content then return nil end
92
93    local manager = _G.amfBean("LuckyFarmStarManager")
94    content:setString(string.format(
95        "活动期间\n权杖每日幸运增加 <color=#E96B33FF>%d</color><color=#7CFF21FF> </color>关",
96        manager:getScepterNum()
97    ))
98
99    local countDown = getOtherPartBuilder():createChild("countDown")
100    if countDown then
101        countDown:setOnStopCallback(function()
102            safeRunCall(onCountDownStop)
103        end)
104        countDown:setLeftTime(math.max(0, manager:getEndTimeInSec() - TimeKit:timeInSec()))
105    end
106    local tip = UITip.Builder.new()
107        :setContent(content, true)
108        :setSkin(UITip.SkinType.NPC)
109        :setAutoCloseTime(0)
110        :setDirection(UITip.Direction.RIGHT)
111        :buildSingleTip()
112    tip:addChild(countDown)
113    countDown:setPositionXY(185, 60)
114    return tip
115end
