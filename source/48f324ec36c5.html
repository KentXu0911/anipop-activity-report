102function MainPanelVM:onEnterPanelRun()
103    local isCanReward = self._mgr:isCanReward()
104    if isCanReward then
105        self:lock()
106    end
107    deferred.success()
108    :next(function()
109        if self.isDisposed then return end
110        local td = deferred.new()
111        TimerUtil.addAlarm(function()
112            if self.isDisposed then return end
113            td:resolve()
114        end, 0.3)
115        return td
116    end)
117    :next(function()
118        if self.isDisposed then return end
119        local td = deferred.new()
120        if self.needPlayDayNum then
121            self._mgr:setLocalValue("popCardSigned_anim_"..self.needPlayDayNum, true)
122            self:callViewFuncWithCallback("playSignAnim", function()
123                td:resolve()
124            end, self.signedListCount, isCanReward)
125            self.needPlayDayNum = nil
126        else
127            td:resolve()
128        end
129        return td
130    end)
131    :next(function()
132        if self.isDisposed then return end
133        local td = deferred.new()
134        if isCanReward then
135            self:callViewFuncWithCallback("playBoxActiveEff", function()
136                td:resolve()
137            end)
138        else
139            td:resolve()
140        end
141        return td
142    end)
143    :next(function()
144        if self.isDisposed then return end
145        local td = deferred.new()
146        if isCanReward then
147            self:onGetRewardFlow(function()
148                td:resolve()
149            end)
150        else
151            td:resolve()
152        end
153        return td
154    end)
155    :next(function()
156        if self.isDisposed then return end
157        self:updateView()
158        if not self.userClick then
159		    self:callViewFunc("addJumpLayer")
160        end
161        self:unlock()
162    end)
