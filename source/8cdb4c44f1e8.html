326function AddSidebarManager:sendHttp(delegate)
327    local successCallback = delegate.successCallback
328    local failCallback = delegate.failCallback
329    local cancelCallback = delegate.cancelCallback
330    local type = delegate.type or 1
331    local noLoading = delegate.noLoading
332    local request = require("amf.feature.minihub.addsidebar.http.SidebarMainInfo")
333    local function onSuccess(evt)
334        local data = evt:getData()
335        safeRunCall(successCallback, data)
336    end
337
338    local function onFail(response)
339        if not response:isCancel() then
340            if not noLoading then
341                CommonTip:showErrorTip(response)
342            end
343            safeRunCall(failCallback)
344        else
345            safeRunCall(cancelCallback)
346        end
347    end
348
349    local param = {
350        type = type,
351    }
352
353    request:post(param, noLoading):next(onSuccess, onFail)
354end

357function AddSidebarManager:queryActInfoHttp(delegate)
358    delegate.type = 1
359    if delegate.noLoading == nil then
360        delegate.noLoading = true
361    end
362    local oriSuccessCallback = delegate.successCallback
363    local successCallback = function(data)
364        DYMiniGameLog:log("queryActInfoHttp", data)
365        self.model:setValue("_isActOpen", data.open)
366        self.model:setValue("_isTaskFinished", data.finishTask)
367        self.model:setValue("_hasGotReward", data.gotRewards)
368        if data.rewards then
369            self.model:setValue("_rewardConfig",  data.rewards[1])
370        end
371        safeRunCall(oriSuccessCallback, data)
372    end
373    delegate.successCallback = successCallback
374    self:sendHttp(delegate)
375end
