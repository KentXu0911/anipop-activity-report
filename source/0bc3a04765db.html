192        self._lastTipTime = curTime

194        local key = self.actInfo:getValue("mainBeginTime") .. "_hasEnterCollectWordAct"

438    local receiveWordInfoListA = delegate.rewards or {}
439    local shareType = delegate.shareType
440    local tempData = table.clone(receiveWordInfoListA)
441
442    local rewardMap = {}
443    local personMap = {}
444    local rewardList = {}
445    local personList = {}
446    for _, v in pairs(tempData) do
447        if not personMap[tostring(v.xxlId)] then
448            personMap[tostring(v.xxlId)] = 1
449        end
450    end
451
452    local personNum = table.size(personMap)
453
454    for _, v in pairs(tempData) do
455        table.insert(personList, v)
456    end
457
458    table.sort(personList, function(a, b)
459        return tonumber(a.time or 0) < tonumber(b.time or 0)

463
464    local personName = cutNameByteSensitive(personList[1].name, 12, nil, true)
