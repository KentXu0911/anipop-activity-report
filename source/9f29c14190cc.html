35local function addToSetIndex(index, key, value)
36    if key == nil then return end
37    key = tostring(key)
38    index[key] = index[key] or {}
39    index[key][value] = true
40end
41
42local function removeFromSetIndex(index, key, value)
43    if key == nil then return end
44    key = tostring(key)
45    local values = index[key]
46    if not values then return end
47    values[value] = nil
48    if not next(values) then
49        index[key] = nil
50    end
51end

53local function sortCandidatesBySchedule(autoPopupMgr, popStage, candidates)
54    local indexed = {}
55    for index, candidate in ipairs(candidates) do
56        indexed[candidate] = index
57        candidate._scheduleRank = autoPopupMgr:getPopupScheduleRank(popStage, candidate.checkPoint)
58    end
59    table.sort(candidates, function(left, right)
60        local leftRank = left._scheduleRank or math.huge
61        local rightRank = right._scheduleRank or math.huge
62        if leftRank ~= rightRank then return leftRank < rightRank end
63        return indexed[left] < indexed[right]
64    end)
65    return candidates
66end

96--- 注册声明式强弹。
97-- 每个注册项必须声明自己的业务 checkpoint、场景和弹出函数。
98-- checkpoint 的 PopType/组内优先级由 AutoPopupCheckPoints 统一反查，
99-- 每日次数上限由 AutoPopupLimit 静态定义。
100-- @param config table: { checkPoint, stages, check?, pop?, actId?, manualOpen?, popType? }
101--   checkPoint 即唯一身份；同一 checkPoint 重复注册视为覆盖。
102function PopupRegistration:register(config)
103    local checkPoint = config.checkPoint
104    assert(checkPoint, "PopupRegistration:register requires checkPoint")
105    assert(config.stages and #config.stages > 0, "PopupRegistration:register requires stages")
106    local pop = config.pop or config.creator
107    assert(pop, "PopupRegistration:register requires pop/creator")
108
109    local oldEntry = self._entries[checkPoint]
110    if oldEntry then
111        self:_removeEntryIndexes(oldEntry)
112        AutoPopupDebugLog:log("PopupRegistration:register overwrite", checkPoint)
113    end
114
115    local entry = {
116        checkPoint = checkPoint,
117        config = config,
118    }
119
120    self._entries[checkPoint] = entry
121    self:_addEntryIndexes(entry)
122    if config.actId ~= nil then
123        self._autoPopupMgr:removeConfiguredPopupsForActivity(config.actId)
124    end
125    -- 候选集合变化后，场景调度需要重新解析。
126    self._autoPopupMgr:invalidatePopupSchedule()
127    AutoPopupDebugLog:log("PopupRegistration:register", checkPoint)
128    return checkPoint
129end

131function PopupRegistration:unregister(checkPoint)
132    local entry = self._entries[checkPoint]
133    if not entry then return end
134
135    self:_removeEntryIndexes(entry)
136    self._entries[checkPoint] = nil
137    -- 队列槽位即 checkPoint，卸载时按 checkPoint 清理已入队弹窗。
138    self._autoPopupMgr:removePopupConfig(checkPoint)
139    self._autoPopupMgr:invalidatePopupSchedule()
140    AutoPopupDebugLog:log("PopupRegistration:unregister", checkPoint)
141end

143function PopupRegistration:_addEntryIndexes(entry)
144    local checkPoint = entry.checkPoint
145    local config = entry.config
146    for _, popStage in ipairs(config.stages) do
147        local list = self._stageIndex[popStage]
148        if not list then
149            list = {}
150            self._stageIndex[popStage] = list
151        end
152        table.insert(list, checkPoint)
153    end
154
155    addToSetIndex(self._manualOpenIndex, config.manualOpen, checkPoint)
156end
157
158function PopupRegistration:_removeEntryIndexes(entry)
159    local checkPoint = entry.checkPoint
160    local config = entry.config
161    for _, popStage in ipairs(config.stages) do
162        local list = self._stageIndex[popStage]
163        if list then
164            for i = #list, 1, -1 do
165                if list[i] == checkPoint then
166                    table.remove(list, i)
167                end
168            end
169        end
170    end
171
172    removeFromSetIndex(self._manualOpenIndex, config.manualOpen, checkPoint)
173end

175function PopupRegistration:_onActivityPopupOver(event)
176    if not self._autoPopupMgr:isEnabled() then return end
177
178    local data = event and event:getData() or {}
179    local checkPoints = {}
180    local function collect(values)
181        for checkPoint in pairs(values or {}) do
182            checkPoints[checkPoint] = true
183        end
184    end
185
186    if data.actId ~= nil then
187        collect(self._manualOpenIndex[tostring(data.actId)])
188    end
189
190    for checkPoint in pairs(checkPoints) do
191        local entry = self._entries[checkPoint]
192        if entry then
193            self:_markPopupHandled(checkPoint)
194        end
195    end
196end

198function PopupRegistration:_markPopupHandled(checkPoint)
199    local limitNum = self._autoPopupMgr:getCheckPointLimitNum(checkPoint)
200    local currentNum = self._autoPopupMgr:getCheckPointPopNum(checkPoint)
201    if limitNum == -1 or currentNum < limitNum then
202        AutoPopupDebugLog:log("PopupRegistration:_markPopupHandled", checkPoint)
203        self._autoPopupMgr:increaseCheckPointPopNum(checkPoint)
204    end
205    self._autoPopupMgr:removePopupConfig(checkPoint)
206end

212function PopupRegistration:hasRegisteredPopupForActivity(actId)

227function PopupRegistration:_onStageEnter(featureEvent, stageParams)
228    local popStage = _featureToPopStage[featureEvent]
229    if not popStage then return end
230
231    local checkPoints = self._stageIndex[popStage]
232    if not checkPoints or #checkPoints == 0 then return end
233
234    local candidates = {}
235    for _, checkPoint in ipairs(checkPoints) do
236        local entry = self._entries[checkPoint]
237        if entry then
238            local params = table.clone(stageParams or {})
239            params.popStage = popStage
240            local candidate = {
241                checkPoint = checkPoint,
242                actId = entry.config.actId,
243                metaId = entry.config.metaId,
244                popType = entry.config.popType,
245                entry = entry,
246                stageParams = params,
247            }
248            candidates[#candidates + 1] = candidate
249        end
250    end
251
252    local function checkCandidate(candidate)
253        local entry = candidate.entry
254        local sceneLogIdentity = getSceneLogIdentity(entry, {
255            popType = entry.config.popType,
256        })
257        local passed = true
258        local checkReason
259        local checkResult = "success"
260        if entry.config.check then
261            AutoPopupDebugLog:recordPopupCheckStarted(sceneLogIdentity)
262            local ok, result, reason = pcall(entry.config.check, candidate.stageParams)
263            passed = ok and not not result
264            if not ok then
265                checkResult = "fail"
266                checkReason = LogReason.Check.ERROR
267                AutoPopupDebugLog:log("PopupRegistration:check error", candidate.checkPoint, tostring(result))
268            elseif not passed then
269                checkResult = "fail"
270                checkReason = reason
271            end
272        else
273            checkResult = "none"
274        end
275        if checkResult == "none" then
276            AutoPopupDebugLog:recordPopupCheckNotRequired(sceneLogIdentity)
277        else
278            AutoPopupDebugLog:recordPopupCheck(sceneLogIdentity, checkResult, checkReason)
279        end
280        return passed
281    end
282
283    sortCandidatesBySchedule(self._autoPopupMgr, popStage, candidates)
284    for _, candidate in ipairs(candidates) do
285        if checkCandidate(candidate) then self:_enqueueRegisteredPopup(candidate) end
286    end
287end

289function PopupRegistration:_enqueueRegisteredPopup(candidate)
290    local entry = candidate.entry
291    local config = entry.config
292    local checkPoint = entry.checkPoint
293	local function onPopOver()
294		AutoPopEventCenter:dispatch(
295			AutoPopEvent.SINGLE_POP_OVER,
296			checkPoint
297		)
298    end
299
300    local pop = config.pop or config.creator
301    local ok, creator = pcall(pop, candidate.stageParams, onPopOver)
302    if not ok or type(creator) ~= "function" then
303        AutoPopupDebugLog:log("PopupRegistration:creator error", entry.checkPoint, tostring(creator))
304        AutoPopupDebugLog:recordPopupEnqueue({
305            checkPoint = checkPoint,
306            actId = config.actId,
307            source = "registered",
308        }, "fail", LogReason.Enqueue.CREATOR_ERROR, checkPoint)
309        return
310    end
311
312    local enqueued = self._autoPopupMgr:addPopupConfig(
313        checkPoint, creator, config.actId, "registered"
314    )
315    if enqueued and candidate.stageParams then
316        local popupQueue = self._autoPopupMgr:getPopupQueueByCheckPoint(checkPoint)
317        local popupEntry = popupQueue and popupQueue[checkPoint]
318        if popupEntry then
319            popupEntry.popupStageId = candidate.stageParams.popStage
320        end
321    end
322    return enqueued
323end

328function PopupRegistration:getSceneLogDefinitions(popStage)
329    local result = {}
330    local stageKey = getMappedPopStage(popStage)
331    local checkPoints = self._stageIndex[stageKey] or self._stageIndex[popStage] or {}
332    for _, checkPoint in ipairs(checkPoints) do
333        local entry = self._entries[checkPoint]
334        if entry then
335            result[#result + 1] = getSceneLogIdentity(entry, {
336                popType = entry.config.popType,
337            })
338        end
339    end
340    return result
341end
