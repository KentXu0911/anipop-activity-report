24--强弹结束
25function AutoPopupProcessor:popupOverFunc(data)
26    local metaId = data.metaId
27    local actId = data.actId
28    local popupConfig
29
30    ActivityDebugLog:logAutoPop(metaId, actId, "强弹结束，申请取消")
31    if metaId then
32        popupConfig = self._autoPopupConfig[tostring(metaId)]
33    elseif actId then
34        for k, v in pairs(self._autoPopupConfig) do
35            if tonumber(v:getActId()) == tonumber(actId) and not v:isStage3CustomizedPopup() and not v:isStage4CustomizedPopup() then
36                popupConfig = v
37                metaId = v:getMetaId()
38                break
39            end
40        end
41    end
42
43    if popupConfig then
44        local needSync = popupConfig:needSyncPopupNum()
45        if needSync then
46            popupConfig:syncPopupNum()
47        end
48        local shouldRemove = popupConfig:shouldRemoveAfterPop()
49        ActivityDebugLog:logAutoPop(popupConfig:getMetaId(), popupConfig:getActId(), "强弹结束，实际取消"
50            ,"需要同步到后端" .. tostring(needSync), "需要移除强弹" .. tostring(shouldRemove))
51        if shouldRemove then
52            self._autoPopupConfig[tostring(metaId)] = nil
53        end
54    end
55end

57function AutoPopupProcessor:process()
58    if not self:getEventHandled() then
59        self:handleConfigPopup()
60        self:setEventHandled(true)
61        if self._autoPopupConfig and next(self._autoPopupConfig) ~= nil then
62            return true
63        end
64    end
65end

75-- 配置强弹处理入口
76function AutoPopupProcessor:handleConfigPopup()
77    local config = require("activity/" .. self._source)
78    -- 活动状态切换会重跑本函数（ActivityNode:handleAutoPopup），先清掉上一次的配置，
79    -- 避免活动已 ignore / 无 AutoPopupMgr 时残留旧强弹配置。
80    self:removeOldConfig()
81    if self:ignoreAutoPop(config) then
82        return
83    end
84
85    if not _G.amfBean("AutoPopupMgr") then
86        return
87    end
88
89    self:addConfigToPopQueue(config, self._source)
90end
91
92function AutoPopupProcessor:removeOldConfig()
93    self._autoPopupConfig = {}
94end
95
96function AutoPopupProcessor:addConfigToPopQueue(config, source)
97    local _popupConfigs = self:genActAutoPopupConfig(config.autoPopConfigs, config)
98
99    local function insertToTable(tbl, popupConfig)
100        tbl[tostring(popupConfig:getMetaId())] = popupConfig
101    end
102    for _, popupConfig in ipairs(_popupConfigs) do
103        popupConfig:setActSource(source)
104        insertToTable(self._autoPopupConfig,popupConfig)
105    end
106end

108function AutoPopupProcessor:genActAutoPopupConfig(autoPopConfigs, actConfig)
109    local ret = {}
110    for i, v in ipairs(autoPopConfigs) do
111        local config = table.clone(v)
112        config.autoPopCustomizedCheck = self:getFunctionFromActConfig(actConfig, config.autoPopCustomizedCheck)
113        config.autoPopCustomizedCreator = self:getFunctionFromActConfig(actConfig, config.autoPopCustomizedCreator)
114        local popupConfig = APMetaConfigRef:create()
115        popupConfig:updateConfigData(config)
116        if popupConfig:isTimeEffective() and popupConfig:isRuleEffective() and popupConfig:isPayTypeEffective() and
117            popupConfig:isActiveTypeEffective() and popupConfig:isGroupSwitchEffective() then
118            table.insert(ret, popupConfig)
119        end
120    end
121    return ret
122end

130function AutoPopupProcessor:getFunctionFromActConfig(actConfig, funcKey)
131    funcKey = tostring(funcKey)
132    if not funcKey then
133        return
134    end
135    if funcKey == "" or funcKey == "nil" then
136        return
137    end
138    if not actConfig[funcKey] then
139        return
140    end
141    if type(actConfig[funcKey]) ~= "function" then
142        return
143    end
144
145    return actConfig[funcKey]
146end

158function AutoPopupProcessor:checkHasAutoPopupByMetaId(metaId)
