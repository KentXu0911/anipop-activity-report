262function _G.WeeklyRaceConst:getDressSlotNamesAndAttachmentsName( dressId )
263    if not _G.WeeklyRaceConst.DressSpineInfo[dressId]  then return {}, {} end
264    local slotNames = string.split( _G.WeeklyRaceConst.DressSpineInfo[dressId].slotName, ":")

266    return slotNames, attachmentsNames
267end
