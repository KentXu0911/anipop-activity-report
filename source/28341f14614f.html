29function WideScreenLogic:updateData()
30    if not self:isPlatformSupport() then return end
31    local data = self:getData()
32    local key = self:getScreenKey()
33
34    if data[key] then return end
35
36    data[key] = {
37        initialType = Const.SHOW_TYPE.SMALL,
38        nowType = self:isExpand() and Const.SHOW_TYPE.BIG
39            or Const.SHOW_TYPE.SMALL
40    }
41    local view = CCDirector:sharedDirector():getOpenGLView()
42    local curA = view:getDesignResolutionSize()
43    local minA = view:get_minimum_design_size()
44    local maxA = view:get_maximum_design_size()
45
46    --防止NAN
47    if isNaN(curA.height) or isNaN(minA.height) or isNaN(maxA.height) then
48        return
49    end
50    local canExpand = curA.width > minA.width
51    local meetAspectRatio = safeAspectFlag(curA.width, curA.height)
52    local baseRatio = (maxA.width == minA.width) and 1
53        or (curA.width - minA.width) / (maxA.width - minA.width)
54
55    data[key] = {
56        initialType = Const.SHOW_TYPE.SMALL,
57        nowType = self:isExpand() and Const.SHOW_TYPE.BIG or Const.SHOW_TYPE.SMALL,
58        canExpand = canExpand,
59        meetAspectRatio = meetAspectRatio,
60        baseRatio = baseRatio,
61    }
62    self:setData(data)
63
64    if data[key].nowType == Const.SHOW_TYPE.SMALL then
65        self:changeToOriginState()
66    else
67        self:changeToUserSetState()
68    end
69end

134function WideScreenLogic:changeDisplayMode(displayMode)
135    if not self:isEnabled() then return deferred.error() end
136    if self:getInitialType() == displayMode then
137        self:changeToOriginState()
138    else
139        self:changeToUserSetState()
140    end
141
142    local data = self:getData()
143    local key = self:getScreenKey()
144    if data[key].nowType ~= displayMode then
145        data[key].nowType = displayMode
146        self:setData(data)
147        return deferred.success()
148    else
149        return deferred.error()
150    end

168function WideScreenLogic:getUserCanSetResolution()
169    local view = CCDirector:sharedDirector():getOpenGLView()
170    local cur = view:getDesignResolutionSize()
171    local min = view:get_minimum_design_size()
172    local max = view:get_maximum_design_size()
173
174    local minW, nowW, maxW = min.width, cur.width, max.width
175    local useW
176    if self:getInitialType() == Const.SHOW_TYPE.SMALL then
177        useW = math.max(nowW / Const.MAX_CHANGE_RATIO, minW)
178    else
179        useW = math.min(nowW * Const.MAX_CHANGE_RATIO, maxW)
180    end
181
182    local ratio = (useW - minW) / (maxW - minW)
183    return ratio
184end

193function WideScreenLogic:tryAddPop()
194    if not self:isEnabled() or self:hasPopped() then return end
195
196    local key = PopupCheckPoint.WIDE_SCREEN_ADVICE
197
198    local function onHandleOver()
199        AutoPopEventCenter:dispatch(AutoPopEvent.SINGLE_POP_OVER, key)
200    end
201
202    local creator = function ( ... )
203        if not self:isEnabled() then
204            onHandleOver()
205            return
206        end
207        _G.amfBean("AutoPopupMgr"):sendPermanentData(key)
208        self:openNoticePanel(onHandleOver)
209    end
210
211    _G.amfBean("AutoPopupMgr"):addPopupConfig(key, creator)
212end
