96    end
97    local oriFunc = panelVM.onCustomCloseAllPanel
98    panelVM.onCustomCloseAllPanel = function(obj)
99        safeRunCall(oriFunc, obj)
100        if obj.isDisposed then
101            return
102        end
103        if obj.close then
104            obj:close(true)
105        else
106            obj:_close()
107        end
108    end
109
110    self:addObserver(panelVM)

538-- 返回值
539-- 当前进行中的任务0设首，1~7复访 >=8 已结束
540-- 任务状态 1未完成，2已完成 3已领取
541function Manager:getMissionStatus()
542    local dayOfSetHomePage = tonumber(self:getModel():getDataByKey("dayOfGetSetHomePageReward", 0))
543    if dayOfSetHomePage == 0 then
544        local statusOfSetHomePage = self:getModel():getDataByKey("statusOfSetHomePage", false)
545        return 0, statusOfSetHomePage and PhaseType.FINISH or PhaseType.UNFINISH
546    else
547        local curDayStartTime = getDayStartTimeByTS(time or TimeKit:timeInSec())
548        local numOfCanFollowUpDay = self:getModel():getDataByKey("numOfCanFollowUpDay", 0)
549        local today = math.ceil(curDayStartTime/86400)
550        local dayIdx = today - dayOfSetHomePage + 1
551        if dayIdx > numOfCanFollowUpDay then
552            return dayIdx, PhaseType.FINISH
553        end
554        local indexSetOfFollowedUp = self:getModel():getDataByKey("indexSetOfFollowedUp", {})
555        if not table.exist(indexSetOfFollowedUp, dayIdx) then
556            return dayIdx, PhaseType.UNFINISH
557        end
558        local indexSetOfHadReward = self:getModel():getDataByKey("indexSetOfHadReward", {})
559        if not table.exist(indexSetOfHadReward, dayIdx) then
560            return dayIdx, PhaseType.FINISH
561        end
562        return dayIdx, PhaseType.HASGOT
563    end
564end
