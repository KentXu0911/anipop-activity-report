87function ShopRewardAddStepView:refreshExtraGoldNum()
88    if not isSafeCocosObj(self.HorizontalContainer) then return end
89
90    local extraGoldNum = math.max(tonumber(self:getExtraGoldNum()) or 0, 0)
91    if self._extraGoldNum == extraGoldNum then return end
92    self._extraGoldNum = extraGoldNum
93
94    self.HorizontalContainer:removeChildren(true)
95    local numStr = tostring(extraGoldNum)
96    for i = 1, string.len(numStr) do
97        local digit = string.sub(numStr, i, i)
98        local sprite = Sprite:createWithSpriteFrameName(Res.DigitFramePrefix .. digit)
99        if sprite then
100            sprite:setScale(0.61)
101            sprite:setAnchorPoint(ccp(0.5, 0.5))
102            local zOrderIndex = i
103            sprite.getZOrderIndex = function()
104                return zOrderIndex
105            end
106            sprite:setPositionY(34)
107            self.HorizontalContainer:addChild(sprite)
108        end
109    end
110end
