176-- 按当前倍率判定幸运产出，命中且目标棋子图鉴已开放时返回等级和棋子 ID。
177function MergeManager:getLuckyLaunchLevel(powerMultiplier, itemId)
178    if not itemId then return nil end
179
180    local upgradeCount = POWER_MULTIPLIER_UPGRADE_COUNT[tonumber(powerMultiplier)]
181    if upgradeCount == nil then return nil end
182
183    local luckyLaunchConfig = self:getData():getLuckyLaunchConfig(upgradeCount + 1)
184    if not luckyLaunchConfig then return nil end
185
186    local showLuckRate = tonumber(luckyLaunchConfig.showLuckRate) or 0
187    local luckRandomNum = math.random(1, 100)
188    local launchCount = tonumber(self:getData():getLaunchCount()) or 0
189    local showSuperCount = tonumber(self:getData():getShowSuperCount()) or 0
190    if launchCount - showSuperCount <= self:getConst().LUCKY_LAUNCH_CD or luckRandomNum > showLuckRate then
191        return nil
192    end
193
194    local weightedLevels = {}
195    local totalWeight = 0
196    for level, weight in pairs(luckyLaunchConfig.luckyFoodLevelWeights or {}) do
197        local normalizedLevel = tonumber(level)
198        local normalizedWeight = math.max(tonumber(weight) or 0, 0)
199        if normalizedLevel and normalizedLevel >= 1 and normalizedWeight > 0 then
200            table.insert(weightedLevels, {
201                level = normalizedLevel,
202                weight = normalizedWeight,
203            })
204            totalWeight = totalWeight + normalizedWeight
205        end
206    end
207    if totalWeight <= 0 then return nil end
208    table.sort(weightedLevels, function(left, right)
209        return left.level < right.level
210    end)
211
212    local levelRandomNum = math.random() * totalWeight
213    local accumulatedWeight = 0
214    local findLevel
215    for _, weightedLevel in ipairs(weightedLevels) do
216        accumulatedWeight = accumulatedWeight + weightedLevel.weight
217        if levelRandomNum < accumulatedWeight then
218            findLevel = weightedLevel.level
219            break
220        end
221    end
222    if not findLevel then return nil end
223
224    local superMakeId = BoardMeta:getItemLevelIdByItemId(itemId, findLevel)
225    if not superMakeId then return nil end
226
227    local bOpen = self:getData():checkChartInfo(superMakeId)
228    if not bOpen then return nil end
229
230    return findLevel, superMakeId
231end

323        end

327        targetItemId = targetItemId,

487    local orderedItemIds = queryContext.orderedItemIds

490        or type(orderedItemIds) ~= "table"
491        or type(boardCountByItemId) ~= "table" then
492        return nil
493    end
494
495    local levelCountMap = {}
496    for level = 1, targetLevel - 1 do
497        levelCountMap[level] = tonumber(boardCountByItemId[tostring(orderedItemIds[level])]) or 0
498    end
499
500    for level = 1, targetLevel - 1 do
501        local currentCount = levelCountMap[level]
502        local upgradeCount = math.floor(currentCount / 2)
503        levelCountMap[level] = currentCount % 2
504        if level + 1 < targetLevel then
505            levelCountMap[level + 1] = levelCountMap[level + 1] + upgradeCount
506        end
507    end
508
509    for level = 1, targetLevel - 1 do
510        if levelCountMap[level] > 0 then
511            return orderedItemIds[level]
512        end
513    end
514    return nil

517-- 非引导饺子生成；powerCost 为本次主产物实际体力消耗。
518function MergeManager:getJiaoziLaunchResult(powerCost, makeExtraItem)
519    if not makeExtraItem or not makeExtraItem.id then
520        return nil, false
521    end
522
523    local data = self:getData()
524    local curWeek = data:getCurWeek()
525    local limitOrderInfo = data:getLimitOrderInfo() or {}
526    local orderIndex = (tonumber(limitOrderInfo.finishIndex) or -1) + 2
527    local specialItemId = limitOrderInfo.specialItemIds and limitOrderInfo.specialItemIds[orderIndex]
528    local specialItemData = specialItemId and BoardMeta:getItemDataByItemId(specialItemId)
529    local specialItemLevel = specialItemData and tonumber(specialItemData.level)
530    local bInPk = (tonumber(limitOrderInfo.pkRound) or 0) > 0
531    local specialConfig
532    if data:isInLimitOrderTime() and specialItemLevel then
533        local powerLevel = self:getConst().POWER_LEVEL_BY_MULTIPLIER[tonumber(powerCost)]
534        if powerLevel then
535            specialItemLevel = math.min(specialItemLevel, powerLevel)
536        end
537        specialConfig = data:getJiaoziSpecialLaunchConfig(powerLevel, specialItemLevel, bInPk)
538    end
539
540    if specialConfig then
541        local makeLevel = tonumber(specialConfig.makeLevel)
542        local percent = tonumber(specialConfig.percent)
543        local levelItemList = makeLevel and BoardMeta.jiaoziLevelList[makeLevel]
544        local jiaoziId = levelItemList and (levelItemList[curWeek] or levelItemList[1])
545        if percent and jiaoziId then
546            percent = math.max(0, math.min(percent, 100))
547            if math.random(1, 100) <= percent then
548                return jiaoziId, true
549            end
550            return nil, true
551        end
552    end
553
554    local makePercent = tonumber(makeExtraItem.percent) or 0
555    if bInPk then
556        makePercent = tonumber(makeExtraItem.pkPercent) or makePercent
557    end
558    makePercent = math.max(0, math.min(makePercent, 100))
559    if math.random(1, 100) > makePercent then
560        return nil, false
561    end
562
563    local jiaoziId = BoardMeta:getJiaoziIdByWeek(curWeek, makeExtraItem.id)
564    return jiaoziId, false
565end

796function MergeManager:hasRewards()
797    if not self:isEnabled2() then return false end
798
799    if self:getData():isInMainTime() then
800        -- 有实物且没填地址
801        local bHasNationRewards = self:getData():checkHasGetRealItem()
802        local bFillAddress = self:getData():getContact()
803
804        --进度奖励未领取
805        local bHasProgressReward = self:getData():checkHasProgressReward()
806        --体力比拼
807        local pkCanGetReward = self:getData():checkHasEnergyPKRankReward() or self:getData():checkHasEnergyPKCostReward()
808        --每日订单完成奖励
809        local limitCanGetReward = self:getData():checkCanGetLimitOrderReward()
810        --图鉴体力，头像框
811        local canShowBook = self:canShowBook()
812        local bChartCanGetReward = canShowBook and self:getData():checkChartCanGetReward()
813        local bCanGetChartAllReward = canShowBook and self:getData():checkCanGetHeadFrameReward()
814        --商店每日免费
815        local bCanGetDailyReward = self:canShowShop() and self:getData():checkFreeItemCanGet()
816        --旅行小鸡
817        local bCanGetTravalReward = self:getData():checkCanGetTravalReward()
818        -- 厨神
819        local bCanGetMatchReward = self:getData():checkCanGetMatchReward()
820
821        -- pk必须非同日且没结算pkReward
822        local pkReward = (not self:isCurLimitOrderSameDay()) and self:getData():checkLimitOrderPKReward()
823        local bContinueWinReward = self:getData():checkPKContinueWinReward()
824        local bOpenEggFinalReward = self:getData():checkOpenEggBigReward()
825        local bOpenEggFreeGift = self:getData():canGetOpenEggDataFreeGiftCnt() > 0
826        local bCanGetDailyRestaurantRatingReward = self:canGetDailyRestaurantRatingReward()
827        local bCanGetPraiseFeedbackReward = self:canGetPraiseFeedbackReward()
828
829        return (bHasNationRewards and not bFillAddress) or bHasProgressReward or pkCanGetReward or limitCanGetReward or bChartCanGetReward or bCanGetDailyReward or bCanGetChartAllReward
830            or bCanGetTravalReward or bContinueWinReward or bOpenEggFinalReward or bOpenEggFreeGift or pkReward or bCanGetMatchReward
831            or bCanGetDailyRestaurantRatingReward or bCanGetPraiseFeedbackReward
832    end
833
834    if self:getData():isInRewardTime() then
835        -- 有实物且没填地址
836        local bHasNationRewards = self:getData():checkHasGetRealItem()
837        local bFillAddress = self:getData():getContact()
838        -- 小组榜奖励且钱够
839        local groupRank = self:getData():getCurGroupRank()
840        local moneyNum = self:getData():getMoney()
841        local bHasGetGroupReward = self:getData():checkHasGetGroupRankReward()
842        local bHasGroupReward = false--not bHasGetGroupReward and groupRank > 0 and groupRank <= 7 and moneyNum > 1500
843        -- 进度奖励
844        local bHasProgressReward = self:getData():checkHasProgressReward()
845        -- 体力比拼奖励
846        local pkCanGetReward = false--self:getData():checkHasEnergyPKRankReward()
847        -- 头像框奖励
848        local bCanGetChartAllReward = self:getData():checkCanGetHeadFrameReward()
849        -- 体力折算奖励
850        local energy = self:getData():getPowerTotal()
851        local canConvert = energy > 0 and not self:getData():getHasConvert()
852        -- 厨神领奖期不结算
853        local bCanGetMatchReward = false -- self:getData():checkCanGetMatchReward()
854
855        return (bHasNationRewards and not bFillAddress) or bHasGroupReward or bHasProgressReward or pkCanGetReward
856            or bCanGetChartAllReward or canConvert or bCanGetMatchReward
857    end
858
859    return false
