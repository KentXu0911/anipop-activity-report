132function PetManager:_canPlayWingDailyProgressAnimation(frequencyContext)

134
135    local progress = self:getWingProgress()
136    progress = type(progress) == "table" and progress or {}
137    local continuousLoginDays = math.max(0, tonumber(progress.continuousLoginDays) or 0)
138    local unlockedDays = math.max(0, tonumber(progress.unlockProgressDays) or 0)
139    local remainingDays = math.max(0, tonumber(progress.needUnlockDays) or 0)
140    local totalDays = unlockedDays + remainingDays
141    local progressAnimationEligible = continuousLoginDays > 0

145
146    return not self:isWingActivated() or continuousLoginDays >= totalDays

228function PetManager:openWingAutoPop(candidate, onPopOver)
229    local completed = false
230    local shown = false
231    local completionReason = "panel_not_shown"
232
233    local function finish()
234        if completed then return end
235        completed = true
236        self:_dispatchWingAutoPopResult(candidate, shown and "shown" or "failed", completionReason)
237        safeRunCall(onPopOver, shown)
238    end
239
240    local function onWingAutoPopShown(displayMode)
241        if completed or shown or displayMode ~= candidate.displayMode then
242            if displayMode ~= candidate.displayMode then
243                completionReason = "display_mode_mismatch"
244            end
245            return
246        end
247        shown = true
248        completionReason = "panel_closed"
249    end
250
251    local function canOpenPanel()
252        if not self:_isWingAutoPopCandidateValid(candidate) then
253            completionReason = "candidate_invalid"
254            return false
255        end
256        return true
257    end
258
259    local function openPanel()
260        local currentCandidate = self:selectWingAutoPopCandidate(candidate.scene)
261        if not currentCandidate then
262            completionReason = "candidate_invalid"
263            finish()
264            return
265        end
266        candidate = currentCandidate
267        self:openActivity(finish, {
268            targetPanel = PetConfig.WingAutoPop.TARGET_PANEL,
269            wingDisplayMode = candidate.displayMode,
270            openReason = PetConfig.WingAutoPop.OPEN_REASON,
271            onWingAutoPopShown = onWingAutoPopShown,
272            mainInfoReady = true,
273            canOpen = canOpenPanel,
274        }, false)
275    end
276
277    if self._hasSyncedMainInfo then
278        openPanel()
279        return
280    end
281
282    local pending = { open = openPanel }
283    pending.timeout = TimerUtil.addAlarm(function()
284        if self._pendingWingAutoPopOpen ~= pending then return end
285        self._pendingWingAutoPopOpen = nil
286        completionReason = "main_info_wait_timeout"
287        finish()
288    end, WING_AUTO_POP_MAIN_INFO_WAIT_TIMEOUT, 1)
289    self._pendingWingAutoPopOpen = pending
290end

645    if type(decoInfo) ~= "table" or decoInfo.enable ~= true then

1212function PetManager:_trimNoHugeBuffFailList(failList, currentLevelId, topLevelId)
1213    if type(failList) ~= "table" then return {} end
1214
1215    local count = table.count(failList)
1216    if count <= 20 then
1217        return failList
1218    end
1219
1220    local deleteLevelId
1221    local deleteTs
1222    currentLevelId = tostring(currentLevelId or "")
1223    topLevelId = tostring(topLevelId or "")
1224    for levelId, info in pairs(failList) do
1225        local ts = type(info) == "table" and tonumber(info.ts) or 0
1226        if tostring(levelId) ~= currentLevelId and tostring(levelId) ~= topLevelId then
1227            if deleteTs == nil or ts < deleteTs then
1228                deleteLevelId = levelId
1229                deleteTs = ts
1230            end
1231        end
1232    end
1233
1234    if deleteLevelId == nil then
1235        for levelId, info in pairs(failList) do
1236            local ts = type(info) == "table" and tonumber(info.ts) or 0
1237            if tostring(levelId) ~= currentLevelId then
1238                if deleteTs == nil or ts < deleteTs then
1239                    deleteLevelId = levelId
1240                    deleteTs = ts
1241                end
1242            end
1243        end
1244    end
1245
1246    if deleteLevelId then
1247        failList[deleteLevelId] = nil
1248    end
1249    return failList
1250end

1252function PetManager:_buildNoHugeBuffUpdateByPassLevel(actData, passLevelData)
1253    local passLevelJson = passLevelData and passLevelData.passLevelJson or {}
1254    if not self:_isValidPetHugeBuffCarrier(passLevelJson) then
1255        if self:_isLogEnabled() then
1256            self:_log("[HUGE-BUFF] skip noHugeBuff local update: invalid carrier", passLevelJson and table.serialize(passLevelJson) or "nil")
1257        end
1258        return nil
1259    end
1260
1261    local petitPetLevelContext = actData and actData.petitPetPassLevelContext
1262    local endLevelData = petitPetLevelContext and petitPetLevelContext.endLevelData or actData and actData.endLevelData
1263    local levelId = endLevelData and endLevelData.levelId
1264        or petitPetLevelContext and petitPetLevelContext.levelId
1265    local levelScene = endLevelData and endLevelData.levelScene
1266        or petitPetLevelContext and petitPetLevelContext.levelScene
1267    if not levelId or not levelScene or not self:_isMainOrFsllLevel(levelScene) then
1268        self:_log("[HUGE-BUFF] skip noHugeBuff local update: invalid level scope", levelId, levelScene)
1269        return nil
1270    end
1271
1272    local noHugeBuffInfo = self:getNoHugeBuffInfo()
1273    local oldData = noHugeBuffInfo.noHugeBuffData or {}
1274    local failList = table.clone(oldData.noHugeBuffFailList or {}, true)
1275    local levelIdStr = tostring(levelId)
1276    local triggerHugeBuffType = tonumber(passLevelJson.triggerHugeBuffType)
1277    local isPass = endLevelData and endLevelData.win == true
1278    local isFail = petitPetLevelContext and petitPetLevelContext.isFail == true
1279    local topLevelId = endLevelData and endLevelData.topLevelId or _G.getTopLevelId()
1280    local nowTs = HeTimeUtil:getCurrentTimeMillis()
1281
1282    local canMaintainPassedCount = passLevelJson.validLevelForPassLevelPity == true
1283    local hasTriggeredHugeBuff = triggerHugeBuffType ~= nil
1284
1285    local updateData = {}
1286    if hasTriggeredHugeBuff then
1287        -- 触发后清除所有数据
1288        updateData["noHugeBuffFailList"] = {}
1289        updateData["noHugeBuffPassedCount"] = 0
1290        updateData["noHugeBuffDaysCount"] = 0
1291        -- updateData["noHugeBuffDaysTs"] = nowTs --这里前端没有用上，先不设置了，以后有需求和后端对一下他怎么实现的
1292    else
1293        -- 维护对应数据
1294        if isPass then
1295            failList[levelIdStr] = nil
1296        elseif isFail then
1297            local currInfo = type(failList[levelIdStr]) == "table" and failList[levelIdStr] or {}
1298            failList[levelIdStr] = {
1299                count = (tonumber(currInfo.count) or 0) + 1,
1300                ts = nowTs,
1301            }
1302            failList = self:_trimNoHugeBuffFailList(failList, levelIdStr, topLevelId)
1303        end
1304        updateData["noHugeBuffFailList"] = failList
1305
1306        if isPass and canMaintainPassedCount then
1307            updateData["noHugeBuffPassedCount"] = (tonumber(oldData.noHugeBuffPassedCount) or 0) + 1
1308        end
1309    end
1310
1311    if self:_isLogEnabled() then
1312        self:_log("[HUGE-BUFF] build noHugeBuff local update",
1313            "levelId=", levelIdStr,
1314            "levelScene=", levelScene,
1315            "isPass=", isPass,
1316            "isFail=", isFail,
1317            "validLevelForPassLevelPity=", passLevelJson.validLevelForPassLevelPity,
1318            "hasTriggeredHugeBuff=", hasTriggeredHugeBuff,
1319            "triggerHugeBuffType=", triggerHugeBuffType,
1320            "updateData=", table.serialize(updateData))
1321    end
1322    return updateData
1323end

1431function PetManager:_hasPetitPetClickCountOperation(opLogs)
1432    if type(opLogs) ~= "table" then return false end
1433
1434    for _, opLog in ipairs(opLogs) do
1435        local op = opLog
1436        if UBFOpLogMappingHelper and UBFOpLogMappingHelper.reflect then
1437            local ok, reflected = pcall(function()
1438                return UBFOpLogMappingHelper:reflect(opLog)
1439            end)
1440            if ok and reflected then
1441                op = reflected
1442            end
1443        end
1444
1445        if type(op) == "table" then
1446            local params = op.params or op.opParams
1447            if op.opName == PETIT_PET_CLICK_COUNT_OP_NAME
1448                or (type(params) == "table" and params.opName == PETIT_PET_CLICK_COUNT_OP_NAME)
1449                or (type(params) == "table" and type(params.data) == "table" and params.data.opName == PETIT_PET_CLICK_COUNT_OP_NAME) then
1450                return true
1451            end
1452        elseif type(op) == "string" and string.find(op, PETIT_PET_CLICK_COUNT_OP_NAME, 1, true) then
1453            return true
1454        end
1455    end
1456
1457    return false
1458end
