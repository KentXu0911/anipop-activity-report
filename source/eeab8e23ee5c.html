80function MainPanelVM:_initShiledState()
81    -- 记录被保护的数据列表，动画用
82    self._shieldDateList = {}
83    local useList = self._promoteDailyMgr:getShieldRecordList(true)
84    if #useList > 0 then
85        self._shieldDateList = self._promoteDailyMgr:getShiledDateInfoByLimit(#useList > 2 and 2 or #useList)
86        -- 获取使用的盾牌最早的获得时间
87        local minOwnDayNum
88        for i, v in ipairs(useList) do
89            local dayNum = TimeUtil:getDayIdxSince1970(v.showTime)
90            minOwnDayNum = minOwnDayNum and math.min(minOwnDayNum, dayNum) or dayNum
91        end
92        -- 如果比获取时间早就丢弃
93        for i = #self._shieldDateList, 1, -1 do
94            local data = self._shieldDateList[i]
95            local dayNum = data.dayNum
96            if minOwnDayNum and dayNum < minOwnDayNum then
97                table.remove(self._shieldDateList, i)
98            end
99        end
100        -- 这里判断有没有播过
101        for i = #self._shieldDateList, 1, -1 do
102            local data = self._shieldDateList[i]
103            local dayNum = data.dayNum
104            if self._promoteDailyMgr:isUseShieldShowed(dayNum) then
105                table.remove(self._shieldDateList, i)
106            end
107        end
108
109        if #self._shieldDateList > 0 then
110            self._initAddCount = #self._shieldDateList
111            self._promoteDailyMgr:addShieldNum(#self._shieldDateList)
112        end
113    end
114    local ownList = self._promoteDailyMgr:getShieldRecordList(false)
115    -- 先把数据改了，后边会再刷回来
116    if #ownList > 0 then
117        -- 有获得也有使用，直接把第一次赠送标记加上
118        if #useList > 0 then
119            self._promoteDailyMgr:addGuideIndex(PromoteDailyConst.GuideId.FIRST_OWN_SHIELD)
120        end
121        self._initMinusCount = #ownList
122        -- 如果获得了盾牌,先减去
123        self._promoteDailyMgr:addShieldNum(-#ownList)
124        self._promoteDailyMgr:setIconTipPopCount(0)
125        local title = ""
126        for i, v in ipairs(ownList) do
127            local dayNum = TimeUtil:getDayIdxSince1970(v.showTime)
128            local date = Utils:getYMDByDayCount(dayNum)
129            if title == "" then
130                title = date.month
131            else
132                if title ~= date.month then
133                    title = title .. "、" .. date.month
134                end
135            end
136        end
137        title = title .. "月免费赠送"
138        if not self._promoteDailyMgr:hasGuideIndex(PromoteDailyConst.GuideId.FIRST_OWN_SHIELD) then
139            self._promoteDailyMgr:addGuideIndex(PromoteDailyConst.GuideId.FIRST_OWN_SHIELD)
140            title = "免费赠送"
141        end
142        self.getShieldTitle = title     
143    end

232function MainPanelVM:_updateView(calendarDir)
233    if self.isDisposed then return deferred.success() end
234    local uiContext, contextErr = MainPanelConfig:resolveMonthContext(self._year, self._month)
235    if not uiContext then return deferred.error(contextErr) end
236    if self._viewTransition then
237        self._viewTransition:reject("promotedaily transition superseded")
238        self._viewTransition = nil
239    end
240    local transition = deferred.new()
241    self._viewTransition = transition
242    self:_updateArrow()
243    local param = {
244        year = self._year,
245        month = self._month,
246        day = self._day,
247        uiContext = uiContext,
248        direction = calendarDir,
249    }
250    local retiringYear = self._retiringResYear
251    self:callViewFuncWithCallback("updateWithTransition", function()
252        if self.isDisposed then return transition:reject("promotedaily main panel disposed") end
253        if self._viewTransition ~= transition then
254            return transition:reject("promotedaily transition superseded")
255        end
256        self._viewTransition = nil
257        if retiringYear and retiringYear == self._retiringResYear and retiringYear ~= self._activeResYear then
258            self:_releaseYearRes(retiringYear)
259            self._retiringResYear = nil
260        end
261        transition:resolve()
262    end, param)
263    self:_updateBackToCur()
264    self:_updateNewRecord()
265    self:_updateNewUIs()
266    return transition
267end

332function MainPanelVM:_tryShowKeyDayMedalPanel()
333    -- 如果今天未签到，并且是纪念日，强弹出纪念日界面
334    local td = deferred.new()
335    local medalCfg, idx = self._promoteDailyMgr:getFutureKeyDayByLimit()
336    if not medalCfg then
337        return td:resolve()
338    end
339    -- 不是今天
340    if idx > 0 then
341        -- 是否弹过
342        local hasPop = self._promoteDailyMgr:getLocalValue("mainPopMedalAlarm_"..medalCfg.medalId)
343        if hasPop then
344            return td:resolve()
345        end
346        -- 是否添加了推送，添加了就不弹了
347        if EnvUtil:isApp() then
348            if self._promoteDailyMgr:hasRegPush(medalCfg.medalId) then
349                return td:resolve()
350            end
351        elseif EnvUtil:isMiniGame() then
352            if not self._subMsgMgr or not self._subMsgMgr:canShowSubMsgNode(SubMessageConst.SceneType.PROMOTEDAILY) then
353                return td:resolve()
354            end
355        end
356        self._promoteDailyMgr:setLocalValue("mainPopMedalAlarm_"..medalCfg.medalId, true)
357        PromoteDailyEventCenter:dispatch(PromoteDailyEvent.SHOW_MEDAL_PANEL, {
358            medalId = medalCfg.medalId,
359            dayNum = self._promoteDailyMgr:getTodayNum() + idx,
360            closeCallback = function()
361                td:resolve()
362            end
363        })
364    else
365        local todaySigned = self._promoteDailyMgr:getTodaySignState() ~= PromoteDailyConst.SignState.NONE
366        local keyDayCfg = self._promoteDailyMgr:hasKeyDayMedal(self._promoteDailyMgr:getYear(), self._promoteDailyMgr:getMonth(), self._promoteDailyMgr:getDay())
367        -- 今天
368        if todaySigned then
369            return td:resolve()
370        end
371        self:_showMedalPanel({
372            showType = MedalShowPanelVM.ShowType.LEVEL,
373            medalId = keyDayCfg.medalId,
374            closeCallback = function()
375                td:resolve()
376            end
377        })
378    end      
379    return td
380end

382function MainPanelVM:_tryGetAllRewards()
383    local hasReward = false
384    local d = deferred.new()
385    deferred.success()
386    :next(function()
387        if self.isDisposed then return d:resolve() end
388        local td = deferred.new()

390        self._promoteDailyMgr:tryGetAllRewards(true):next(function(randomReward)
391            if self.isDisposed then return td:resolve() end
392            randomReward = randomReward or {}
393            if table.isEmpty(randomReward) then
394                return td:resolve()
395            end
396            hasReward = true
397            -- 高光时刻新字
398            self._promoteDailyMgr:setLocalValue("btnRecordNewTagShow", true)
399            local medalId = Utils:getMedalFromShopRewards(randomReward)
400            if medalId and medalId > 0 then
401                self:_showMedalPanel({
402                    medalId = medalId,
403                    showType = MedalShowPanelVM.ShowType.REWARD,
404                    closeCallback = function()
405                        self:openHighLightPanel({
406                            t1 = 2,
407                            closeCallback = function()
408                                td:resolve()
409                            end
410                        })
411                    end
412                })
413            else
414                self._promoteDailyMgr:showShopRewardsPanel(randomReward):next(function()
415                    td:resolve()
416                end)
417            end
418        end)
419        return td
420    end)
421    :next(function()
422        if self.isDisposed then return d:resolve() end
423        local td = deferred.new()
424        self._promoteDailyMgr:tryGetTaskReward():next(function(data)
425            if self.isDisposed then return td:resolve() end

435                        td:resolve()
436                    end)
437                end)
438            else
439                td:resolve()
440            end
441        end)
442        return td
443    end)
444    :next(function()
445        if self.isDisposed then return d:resolve() end
446        if hasReward then
447            self:callViewFunc("updateBottomNode", {
448                forceSelfRefresh = true
449            })
450            -- 先刷新一下

452            assert(uiContext, contextErr)
453            self:callViewFunc("updateCalendarNode", {
454                year = self._year,
455                month = self._month,
456                day = self._day,
457                uiContext = uiContext,
458                isForceUpdate = true
459            })
460        end
461        d:resolve()
462    end)
463    return d
464end

874function MainPanelVM:recycle()
875    if self._viewTransition then
876        self._viewTransition:reject("promotedaily main panel recycled")
877        self._viewTransition = nil
878    end
879    self:_nextNavigationGeneration()
880    PromoteDailyEventCenter:dispatch(PromoteDailyEvent.UPDATE_ICON)
881    self:unregisterEvent()
882    local UIEventCenter = require "amf.middleware.ui.event.UIEventCenter"
883    UIEventCenter:dispatch(UIEvent.Remove.AUTO_TIP)
884    -- 万一没播完动画，得还原回去
885    if self._initMinusCount and self._initMinusCount > 0 then
886        self._promoteDailyMgr:addShieldNum(self._initMinusCount)
887        self._initMinusCount = 0
888    end
889    if self._initAddCount and self._initAddCount > 0 then
890        self._promoteDailyMgr:addShieldNum(-self._initAddCount)
891        self._initAddCount = 0
892    end
893    if self._yearResLoadMap then
894        local loadingMap = self._yearResLoadMap
895        self._yearResLoadMap = {}
896        for _, loading in pairs(loadingMap) do
897            loading:reject("promotedaily main panel recycled")
898        end
899    end
900    if not table.isEmpty(self._retainResMap) then
901        for _, resList in pairs(self._retainResMap) do
902            self._promoteDailyMgr:releaseRes(resList)
903        end
904        self._retainResMap = {}
905    end
906    AMFBasePanelVM.recycle(self)
907end

1203function MainPanelVM:tryLoadYearRes(year)
1204    local yearKey = tostring(year)
1205    local yearList = MainPanelConfig:getYearResList(year)
1206    if #yearList == 0 then
1207        return deferred.success()
1208    end
1209    if self._retainResMap[yearKey] then
1210        return deferred.success()
1211    end
1212    if self._yearResLoadMap[yearKey] then
1213        return self._yearResLoadMap[yearKey]
1214    end
1215
1216    local d = deferred.new()
1217    self._yearResLoadMap[yearKey] = d
1218    local function _loadOver()
1219        if self._yearResLoadMap[yearKey] ~= d then return end
1220        self._yearResLoadMap[yearKey] = nil
1221        if self.isDisposed then
1222            return d:reject("promotedaily main panel disposed")
1223        end
1224        if not self._retainResMap[yearKey] then
1225            self._promoteDailyMgr:retainResOnce(yearList)
1226            self._retainResMap[yearKey] = yearList
1227        end
1228        d:resolve()
1229    end
1230    local function _loadFail(err)
1231        if self._yearResLoadMap[yearKey] ~= d then return end
1232        self._yearResLoadMap[yearKey] = nil
1233        d:reject(err)
1234    end
1235    if EnvUtil:isMiniGame() then
1236        self._assetsUtil:loadMiniFiles(yearList, false):next(function()
1237            _loadOver()
1238        end, _loadFail)
1239    else
1240        _loadOver()
1241    end
1242    return d
1243end

1245function MainPanelVM:_navigateTo(targetYear, targetMonth, targetDay, direction, transitionStarted)
1246    if not self:_isConfigAndDataEnabled(targetYear, targetMonth) then
1247        return deferred.error("promotedaily target month disabled")
1248    end
1249    local navigation = deferred.new()
1250    local generation = self:_nextNavigationGeneration()
1251    if targetYear ~= self._activeResYear and self._retiringResYear then
1252        self:_finishPendingYearTransition()
1253    end
1254    self._pendingNavigationYear = targetYear
1255    self._pendingNavigationMonth = targetMonth
1256    self:tryLoadYearRes(targetYear):next(function()
1257        if not self:_isNavigationCurrent(generation) then
1258            self:_releaseUnusedYearRes(targetYear)
1259            return self:_finishSupersededNavigation(navigation)
1260        end
1261        self:_activateYearRes(targetYear)
1262        self._year = targetYear
1263        self._month = targetMonth
1264        self._day = targetDay or self._day
1265        self._pendingNavigationYear = nil
1266        self._pendingNavigationMonth = nil
1267        local viewTransition = self:_updateView(direction)
1268        safeRunCall(transitionStarted)
1269        viewTransition:next(function()
1270            navigation:resolve()
1271        end, function(err)
1272            if not self:_isNavigationCurrent(generation) then
1273                return self:_finishSupersededNavigation(navigation)
1274            end
1275            navigation:reject(err)
1276        end)
1277    end, function(err)
1278        self:_releaseUnusedYearRes(targetYear)
1279        if self:_isNavigationCurrent(generation) then
1280            self._pendingNavigationYear = nil
1281            self._pendingNavigationMonth = nil
1282        end
1283        navigation:reject(err)
1284    end)
1285    return navigation
1286end
