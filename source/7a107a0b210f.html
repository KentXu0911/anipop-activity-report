426function Manager:checkStage3CanPop()
427    if self:ignoreForTopLevelPlayer() then return false end
428    if self:checkFirstDayCanPop() then return true end
429    if self:isInSignState() then return false end
430    local oldPoint = self:getData():getOldPoint()
431    local newPoint = self:getData():getProgressPoint()
432    -- 每个区间只会弹出一次，比如35-80区间
433    -- 当币消耗45时，判断消耗(45-35) < 45*2/3，所以能弹出
434    -- 此时记录35-80弹出过，后面该区间就不会弹出主面板了
435    if newPoint <= oldPoint or newPoint >= 3 then
436        self:log("新旧区间一致，或超出最后节点，不弹出主面板",newPoint,oldPoint)
437        return false
438    end
439    local lastNeed = math.max(0,self:getData():getPointNeed( newPoint ))
440    local nextNeed = math.max(0,self:getData():getPointNeed( newPoint + 1 ))
441	local nowValue = self:getData():getNowConsume() or 0
442    self:log("检测区间风车币消耗量",nowValue - lastNeed,2/3*(nextNeed - lastNeed))
443    if nowValue - lastNeed <= 0 then return false end
444    -- 进入新区间时消耗不到该区间的2/3就能弹出
445    local canPopMainPanel = (nowValue - lastNeed) <= 2/3*(nextNeed - lastNeed)
446    return canPopMainPanel
447end
