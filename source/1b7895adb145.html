66function IceBreakingGiftModule:isEnabled()
67    local ret = true
68    -- 客户端在判断 status == 1 && 当前时间 >= nextCallPullTs
69    -- && alreadyPassedLevel >= needPassLevel 时，认为可以尝试请求该礼包模块信息
70    if self.moduleExtra then
71        local status = self.moduleExtra.status or 0
72        local nextCallPullTs = self.moduleExtra.nextCallPullTs or 0
73        local alreadyPassedLevel = self.moduleExtra.alreadyPassedLevel or 0
74        local needPassLevel = self.moduleExtra.needPassLevel or 0
75        if status == 1 and TimeKit:timeInMillis() >= nextCallPullTs and alreadyPassedLevel >= needPassLevel then
76            ret = true
77        else
78            ret = false
79        end
80    end
81    if EnvUtil:isDevelopMode() then printx(48, "IceBreakingGiftModule:isEnable=",ret) end
82    return ret
83end

103function IceBreakingGiftModule:doCreateGiftView(autoPopup)
104    autoPopup = autoPopup or false
105    local commonGiftData = self.packageData
106    if not commonGiftData then return end
107    local view = nil
108    --根据场景判断使用view
109    if commonGiftData.stageType == GiftPackageConst.StageType.STORE then
110        local ViewCls = force_require "amf.feature.giftpackage.module.singleicon.icebreaking.view.IceBreakingStoreView"
111        view = ViewCls:create(self, commonGiftData)
112    else
113        -- CommonTip:showTip("礼包面板")
114        local userClick = not autoPopup
115        self:_initRmbTracer(userClick)
116
117        local VM = require "amf.feature.giftpackage.module.singleicon.icebreaking.view.IceBreakingGiftViewVM"
118        local vm = VM:create(commonGiftData, self, autoPopup)
119        return vm
120    end
121
122    return view
123end

195-- 创建折扣icon 复用SprintGiftPackageManager
196function IceBreakingGiftModule:createDiscountIcon(parentNode, isIdle, posX, posY, scale)
197    -- do return false end
198    local isShow = self:isShowDiscount()
199    if not isShow then return nil end
200    ---@type SprintGiftPackageManager
201    local mgr = _G.amfBean("SprintGiftPackageManager")
202    if not mgr then
203        printx(48, "SprintGiftPackageManager is not found!Error!")
204        return nil
205    end
206    scale = scale or 1
207    if not self.packageData then return nil end
208    local price = self.packageData:getPrice()
209
210    local discountIcon = mgr:createDiscountIcon()
211    if posX and posY then discountIcon:setPositionXY(posX, posY) end
212    discountIcon:setScale(scale)
213    local discountNum = price == 3 and 5000 or 3000
214    if isIdle then
215        discountIcon:playIdleAnim(discountNum)
216    else
217        discountIcon:playEnterAnim(discountNum)
218    end
219    parentNode:addChild(discountIcon)
220    return discountIcon
221end

290function IceBreakingGiftModule:tryAutoPopup()
291    if not self.packageData then return end
292    if self.packageData.stageType == GiftPackageConst.StageType.STORE then
293        return
294    end
295
296    local popupKey = self:getGiftTriggerTimes() == 2 and PopupCheckPoint.ICE_BREAKING_GIFT_POP_T2 or PopupCheckPoint.ICE_BREAKING_GIFT_POP
297    local function onHandleOver()
298        AutoPopEventCenter:dispatch(AutoPopEvent.SINGLE_POP_OVER, popupKey)
299    end
300
301    local creator = function ()
302        if not self:_isUserNotOpened() then
303            safeRunCall(onHandleOver)
304            return
305        end
306        ---@type UeAMFBasePanelVM
307        local vm = self:doCreateGiftView(true)
308        if not vm then
309            safeRunCall(onHandleOver)
310            return
311        end
312        if vm.setCloseCallback then
313            vm:setCloseCallback(onHandleOver)
314        else
315            printx(48, "[gk] IceBreakingGiftModule why logcenter vm.setCloseCallback is nil?")
316            safeRunCall(onHandleOver)
317        end
318
319        if vm.setLoadResFailCallback then vm:setLoadResFailCallback(onHandleOver) end
320        vm:popout()
321    end
322
323    -- _G.amfBean("AutoPopupMgr"):addPopupConfig(popupKey, creator)
324    local autoPopupMgr = _G.amfBean("AutoPopupMgr")
325    autoPopupMgr:addPopupConfig(popupKey, creator)
326    if not autoPopupMgr:isRunnning() then
327        autoPopupMgr:triggerStage2AutoPopup()
328    end
329end

381function IceBreakingGiftModule:requestBuy(succCallback, failCallback)
382    if not _G.amfBean("NetworkStatusManager"):isNetworkEnabled() then
383        CommonTip:showTip("网络异常，请检查网络~")
384        safeRunCall(failCallback)
385        return
386    end
387    local package = self.packageData
388    if not package then
389        safeRunCall(failCallback)
390        return
391    end
392
393    local goodsId = package:getGoodsId()
394
395    local function onSuccess(evt)
396        self._rmbTracer:paySuccess()
397        self:removeIcon()
398        safeRunCall(succCallback)
399
400        local showRewards = package:getItems()
401        local ShowRewardsPanelVM = require "amf.middleware.ui.kit.panel.reward.ShowRewardsPanelVM"
402        local vm = ShowRewardsPanelVM.Builder.new()
403            :setTitle("恭喜获得")
404            :setRewards(showRewards)
405            :setCloseCallback(function ()
406
407            end)
408            :setButtonText("收下")
409            :build()
410        vm:popout()
411    end
412
413    local function onFail(evt)
414        self._rmbTracer:payFail()
415        safeRunCall(failCallback)
416    end
417
418    local function onCancel()
419        self._rmbTracer:payCancel()
420        safeRunCall(failCallback)
421    end
422
423    self.newGift_stage = "bigVineScene"
424    self.moduleName = "discountBreakIce"
425
426    local buyContext = BuyContext.new(goodsId, MoneyType.kRmb)
427        :setFeatureAndSource(self.dcFeature, self.dcSource)
428        :setBuyCount(1)
429        :setCallback(onSuccess, onFail, onCancel)
430        :setUnifiedBuy(true, self.newGift_stage, self.moduleName)
431    _G.amfTransaction(TransactionEvent.PAY, buyContext)
432    self._rmbTracer:click()
433end
