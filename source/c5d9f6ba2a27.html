83function WinningStreakBottleManager:isEnabled()
84    if not self._model then return false end
85
86    local mainConfig = self:getMainConfig()
87    if not mainConfig or not mainConfig.metaId then return false end
88
89    if TimeKit:timeInMillis() >= mainConfig.endTs 
90        or TimeKit:timeInMillis() < mainConfig.beginTs
91    then return false end
92
93    if not self:isUserStateValid() then return false end
94
95    return true
96end

102function WinningStreakBottleManager:isAvailableLevel(levelId, levelScene)
103    local isMainLevel = LevelConfigBridge:isLevelTypeByID(levelId, GameLevelType.kMainLevel, levelScene)
104    if not isMainLevel then return false end
105
106    local userStateManager = _G.amfBean("UserStateManager")
107    if not userStateManager then return false end
108
109    if userStateManager:isFMJ() then 
110        local toplevelId = _G.getTopLevelId()
111        return toplevelId == levelId 
112	end
113
114    return not userStateManager:isColorfulStarFull() 
115end

399function WinningStreakBottleManager:popIconTipAfterActivate(done)
400    local icon = self:getIcon()
401    if not icon or icon.isDisposed then 
402        safeRunCall(done)
403        return 
404    end
405
406    if self._model:checkIsFullShowTip() then
407        safeRunCall(done)
408        return
409    end
410
411    self._model:setIsFullShowTip(true)
412
413    IconTipEventCenter:dispatch(IconTipEvent.HIDE_ICON_TIP, {
414        iconKey = _G.IconKey.kWinningStreakBottle,
415    })
416
417    local iconTip = HomeIconTipsFactory:createTextTips(icon,"保持连胜，加分不停！",{
418        endCallback = function() 
419            safeRunCall(done) 
420        end,
421        duration = 2.5,
422    })
423
424    if not iconTip then
425        safeRunCall(done)
426        return
427    end
428
429    iconTip:playCutInAnim(function() 
430        iconTip:playIdleAnim(function()
431            iconTip:playCutOutAnim(function()
432                iconTip:remove()
433            end)
434        end)
435    end)
436end
