231function PopupNewActMgr:presentDataDrivenNewActivity(candidates, hooks)
232    local context = nil
233    local cancelled = false
234    local control = {}
235
236    function control:cancel(reason)
237        cancelled = true
238        if context then
239            context:cancel(reason)
240        end
241    end
242
243    function control:skipToFinal(reason)
244        if context then
245            context:skipToFinal(reason)
246        end
247    end
248
249    local function startPanel()
250        if cancelled then
251            return
252        end
253        local ok, result = pcall(function()
254            local NewActBroadCastContext = require "amf.middleware.autopop.module.activity.NewActBroadCastContext"
255            context = NewActBroadCastContext:create(candidates, {
256                onVisible = hooks.onVisible,
257                onComplete = hooks.onComplete,
258                onFailure = hooks.onFailure,
259                canStart = function()
260                    if cancelled then
261                        return false
262                    end
263                    local popoutManager = _G.amfBean("PopoutManager")
264                    local hasPanel = popoutManager
265                        and popoutManager:hasAMFWindowOnScreen()
266                    return not self._autoPopupMgr:isRunnning() and not hasPanel
267                end,
268            })
269            context:build()
270            return true
271        end)
272        if not ok then
273            hooks.onFailure("notice_start_error:" .. tostring(result))
274        end
275    end
276
277    if EnvUtil:isWindows() then
278        startPanel()
279    else
280        local ok, promise = pcall(function()
281            return _G.amfBean("AssetsUtil"):loadMiniLabels({"auto_pop"}, false)
282        end)
283        if not ok or not promise or type(promise.next) ~= "function" then
284            hooks.onFailure("asset_load_start_failed")
285        else
286            promise:next(startPanel, function()
287                hooks.onFailure("asset_load_failed")
288            end)
289        end
290    end
291    return control
292end

316function PopupNewActMgr:onNoPanelBeforeNotification()

329function PopupNewActMgr:waitForPanelsBeforeNotification(stage, callback)
330    table.insert(self.pendingNotificationRequests, {
331        stage = stage,
332        callback = callback,
333    })
334    if self.waitingForPanelsBeforeNotification then

337    self.waitingForPanelsBeforeNotification = true
338    AutoPopEventCenter:registerGlobal(
339        NormalEvent.NO_PANEL_ON_SCENE,

344    self:onNoPanelBeforeNotification()

356function PopupNewActMgr:tryPopupFoundNewAct(stage, callback)
357    if type(stage) == "function" then
358        callback = stage
359        stage = AutoPopupConst.PopStage.STAGE_TWO
360    end
361    stage = stage or AutoPopupConst.PopStage.STAGE_TWO
362
363    if not self.notificationCoordinator then
364        if callback then callback() end
365        return
366    end
367    if stage == AutoPopupConst.PopStage.STAGE_FOUR then
368        self.notificationCoordinator:markStage2Unavailable("stage4_reached")
369    end
370    if self:hasPanelBeforeNotification() then
371        self:waitForPanelsBeforeNotification(stage, callback)
372        return
373    end
374    self:requestDataDrivenNotification(stage, callback)
375end
