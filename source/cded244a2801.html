61-- Unity 侧调试输出走 UError，非 Unity 侧保留 printx；后续若要换底层输出，只改这里。
62function Log.output(...)

86-- 设置内存日志最大保留条数；用于控制长时间试玩时的内存占用。
87function Log.setMaxRecordCount(count)
88    count = tonumber(count)

90    Log.Config.maxRecordCount = math.floor(count)

93-- 获取模块展示名；未知模块保留 ID，方便定位遗漏的模块定义。

98-- 对缺失模块配置输出一次提示；用于发现忘记加枚举或忘记配置开关的调用点。
99function Log.printMissingModuleTip(moduleId, reason)
100    if not Log.Config.enabled then return end
101    local key = tostring(reason) .. "_" .. tostring(moduleId)
102    if Log.missingModuleTips[key] then return end
103    Log.missingModuleTips[key] = true
104    Log.output("[PengPeng2026]", "[Log]", reason, tostring(moduleId))
105end

107-- 判断指定模块是否允许输出日志；必须有模块枚举名且单模块开关显式为 true。
108function Log.isEnabled(moduleId)
109    if not Log.Config.enabled then return false end
110    if Log.ModuleName[moduleId] == nil then
111        Log.printMissingModuleTip(moduleId, "undefined log module")
112        return false
113    end
114
115    local moduleEnabled = Log.Config.moduleSwitch[moduleId]
116    if moduleEnabled == nil then
117        Log.printMissingModuleTip(moduleId, "unconfigured log module")
118        return false
119    end
120    return moduleEnabled == true
121end

123-- 格式化单个日志参数；table 是否 tts 由调用方的 needTts 决定。
124function Log.formatValue(value, needTts)
125    if needTts == true and type(value) == "table" then
126        return tts(value)
127    end
128    return value
129end

131-- 组装最终输出参数；table 参数在这里按 needTts 决定是否转换。
132function Log.buildPrintArgs(moduleId, needTts, values)
133    local args = {
134        "[PengPeng2026]",
135        "[" .. Log.getModuleName(moduleId) .. "]",
136    }
137    for index = 1, #(values or {}) do
138        table.insert(args, Log.formatValue(values[index], needTts))
139    end
140    return args
141end

143-- 记录一条内存日志；即时不展示的模块也能在需要时统一 dump。
144function Log.record(moduleId, needTts, ...)
145    if not Log.Config.recordEnabled then return end
146
147    local values = {}
148    local count = select("#", ...)
149    for index = 1, count do
150        local value = select(index, ...)
151        table.insert(values, value)
152    end
153
154    table.insert(Log.records, {
155        time = TimeKit:timeInMillis(),
156        moduleId = moduleId,
157        needTts = needTts == true,
158        values = values,
159    })
160
161    while #Log.records > Log.Config.maxRecordCount do
162        table.remove(Log.records, 1)
163    end
164end

166-- 清空内存日志记录；用于新一轮调试前重置历史。
167function Log.clearRecords()
168    Log.records = {}
169end

171-- 输出内存日志记录；moduleId 为空时输出全部，needTtsOverride 可统一覆盖 tts 行为。
172function Log.dump(moduleId, needTtsOverride)
173    Log.output("[PengPeng2026]", "[Log]", "dump begin", "moduleId", tostring(moduleId), "count", #Log.records)
174    local dumpCount = 0
175    for _, record in ipairs(Log.records) do
176        if moduleId == nil or record.moduleId == moduleId then
177            dumpCount = dumpCount + 1
178            local needTts = record.needTts
179            if needTtsOverride ~= nil then
180                needTts = needTtsOverride == true
181            end
182            local args = Log.buildPrintArgs(record.moduleId, needTts, record.values)
183            table.insert(args, 3, "[History]")
184            table.insert(args, 4, tostring(record.time))
185            Log.output(unpack(args))
186        end
187    end
188    Log.output("[PengPeng2026]", "[Log]", "dump end", "dumpCount", dumpCount)
189    return dumpCount
190end

192-- 输出模块日志；table 参数可直接传入，只有 needTts 为 true 时才在最终输出前 tts。
193function Log.print(moduleId, needTts, ...)
194    Log.record(moduleId, needTts, ...)
195    if not Log.isEnabled(moduleId) then return end
196
197    local values = {}
198    local count = select("#", ...)
199    for index = 1, count do
200        local value = select(index, ...)
201        table.insert(values, value)
202    end
203    Log.output(unpack(Log.buildPrintArgs(moduleId, needTts, values)))
204end
