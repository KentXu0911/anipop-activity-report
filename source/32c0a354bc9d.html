61local function resolveMilestoneProgressRoi(itemId, options)

63    if not scene then
64        return nil, "running scene unavailable"
65    end
66    local builder = UEProgressMilestoneBuilder.new()
67        :setMaxValue(1000)
68        :setValue(0)
69        :setProgressWidth(options.progressWidth or 493)
70        :setScrollContainerWidth(options.scrollContainerWidth or 493)
71        :setScrollable(false)
72        :setMilestoneConfig({
73            autoClaim = true,
74            milestonesList = {
75                {
76                    threshold = 500,
77                    rewardConfig = { itemId = itemId, num = options.num or DEFAULT_NUM },
78                },
79            },
80        })
81    local success, probeOrError = pcall(function()
82        return WidgetManager:createWidgetObj("UEProgressMilestone", builder)
83    end)
84    if not success or not probeOrError then
85        return nil, tostring(probeOrError)
86    end
87
88    local probe = probeOrError
89    scene:addChild(probe)
90    local content = probe:_getContentNode()
91    local progress = probe:_getProgressNode()
92    local milestoneNode = probe:_getMilestoneNode(1)
93    local progressBounds = progress and progress:getGroupBounds(content, true)
94    local nodeBounds = milestoneNode and milestoneNode:getGroupBounds(content, true)
95    local result
96    if progressBounds and nodeBounds and progressBounds.origin and progressBounds.size
97        and nodeBounds.origin and nodeBounds.size then
98        local progressBottom = progressBounds.origin.y
99        local progressTop = progressBounds.origin.y + progressBounds.size.height
100        local nodeCenterY = nodeBounds.origin.y + nodeBounds.size.height * 0.5
101        local sourceCenterY = CAPTURE_HEIGHT * 0.5
102        -- manifest 保存运行时 scale=0.4 的坐标，Python 会映射回 scale=1 截图。
103        result = {
104            top = sourceCenterY + (progressTop - nodeCenterY),
105            bottom = sourceCenterY + (progressBottom - nodeCenterY),
106            scale = DEFAULT_MILESTONE_SCALE,
107        }
108        if result.top > result.bottom then
109            result.top, result.bottom = result.bottom, result.top
110        end
111    end
112    probe:removeFromParentAndCleanup(true)
113    if not result then
114        return nil, "milestone progress ROI unavailable"
115    end
116    return result

200-- 确保 imageInfo 对应的纹理可用：图集未加载时先加载 plist，独立 png 则按路径加载。
201-- 散图（png）不在运行时搜索路径时，回退到活动仓库查找；找不到则报错，交由上层询问用户。
202-- 避免 setFrame 未命中时精灵保留空白纹理、渲染成 256x256 空白色块却仍被当成成功。
203-- @param imageInfo table {Atlas, Frame, IsPropPng}
204-- @return boolean ok 帧/纹理是否可用
205-- @return string|nil reason 失败原因
206local function ensureImageLoaded(imageInfo)
207    if not imageInfo or not imageInfo.Frame or imageInfo.Frame == "" then
208        return false, "empty image frame"
209    end
210    local frameCache = CCSpriteFrameCache:sharedSpriteFrameCache()
211    if frameCache:spriteFrameByName(imageInfo.Frame) then
212        return true
213    end
214    -- fullPathForFilename 命中真实文件时返回绝对路径，缺失时原样返回入参
215    local fileUtils = CCFileUtils:sharedFileUtils()
216    local function resolvedPath(path)
217        local full = fileUtils:fullPathForFilename(path)
218        if full and full ~= "" and full:sub(1, 1) == "/" then
219            return full
220        end
221        return nil
222    end
223    local atlas = imageInfo.Atlas
224    if atlas and atlas ~= "" and imageInfo.IsPropPng ~= true then
225        -- 图集未加载：先确认真实存在，再加载 plist 并校验帧命中
226        if not resolvedPath(atlas) then
227            return false, "atlas file not found: " .. tostring(atlas)
228        end
229        frameCache:addSpriteFramesWithFile(atlas)
230        if frameCache:spriteFrameByName(imageInfo.Frame) then
231            return true
232        end
233        return false, "atlas loaded but frame not found: " .. tostring(imageInfo.Frame)
234    end
235    -- 独立 png：缺失时 addImage 会返回 32x32 占位图，必须先定位真实文件
236    local pngPath = resolvedPath(imageInfo.Frame)
237    if not pngPath then
238        -- 回退活动仓库：activity/<act>/... 或 amf/activity/<act>/... -> <repo>/<act>/...
239        local rel = tostring(imageInfo.Frame):gsub("^amf/activity/", ""):gsub("^activity/", "")
240        local repoRoot = resolveActivityRepoRoot()
241        local candidate = repoRoot and (repoRoot .. "/" .. rel) or nil
242        if not pathIsDir(candidate) then
243            if not repoRoot then
244                return false, "activity repo not found, ask user (dir: " .. ACTIVITY_REPO_DIR_NAME .. ")"
245            end
246            return false, "loose png not found (runtime & activity repo): " .. tostring(imageInfo.Frame)
247        end
248        pngPath = candidate
249        -- 渲染时必须用绝对路径，否则 Image:setFrame 仍按相对路径找不到文件
250        imageInfo.Frame = candidate
251        imageInfo.__renderDirect = true
252    end
253    local texture = CCTextureCache:sharedTextureCache():addImage(pngPath)
254    if not texture then
255        return false, "addImage failed: " .. tostring(pngPath)
256    end
257    return true
258end

260-- 复用 PropManager:getItemAttr 的 isSupported 语义判断图标是否异常。
261-- getItemAttr 在普通道具找不到图片时会走 getUnknownItemImage 兜底并置 isSupported=false，
262-- 此时 UEPropIcon 仍能成功创建但渲染的是“错误道具”占位图，像素补偿无意义，应记为失败项。
263-- @param itemId integer 道具 ID
264-- @return boolean ok 图标是否可正常解析
265-- @return string|nil reason 异常原因
266-- @return table|nil imageInfo 命中后的图片信息 {Atlas, Frame, IsPropPng}
267local function checkItemSupported(itemId)
268    -- 与 UEItemConvertor:setItemId 保持一致：先加载通用道具图集，避免 plist 帧查找遗漏
269    local commonDefine = _G.CommonDefine
270    if commonDefine and commonDefine.loadItemCommonPlist then
271        commonDefine.loadItemCommonPlist()
272    end
273    local propMgr = _G.amfBean and _G.amfBean("PropManager")
274    if not propMgr then
275        return false, "PropManager unavailable"
276    end
277    local ok, attr = pcall(function()
278        return propMgr:getItemAttr(itemId)
279    end)
280    if not ok then
281        return false, "getItemAttr error: " .. tostring(attr)
282    end
283    if not attr then
284        return false, "getItemAttr returned nil"
285    end
286    if attr.isSupported == false then
287        return false, "icon unsupported (getUnknownItemImage fallback), itemId=" .. tostring(itemId)
288    end
289    -- 新增：确认纹理/图集已加载并命中帧，未命中视为异常，避免采集到空白色块
290    local imageInfo = attr.image
291    if imageInfo then
292        local loaded, loadReason = ensureImageLoaded(imageInfo)
293        if not loaded then
294            return false, "image frame not resolved: " .. tostring(imageInfo.Frame) .. " (" .. tostring(loadReason) .. ")"
295        end
296    end
297    return true, nil, imageInfo
298end

300---判断道具是否需要传递 num：只有无限道具和限时道具需要。
301---无限道具用数量换算剩余时间，限时道具展示限时标签；普通道具传 num 会多出数量角标，
302---污染图标像素边界，因此不传。
303---@param itemId integer 道具 ID
304---@return boolean needNum 是否需要数量
305local function shouldPassNum(itemId)
306    local propMgr = _G.amfBean and _G.amfBean("PropManager")
307    if not propMgr then
308        return true
309    end
310    if propMgr:isInfiniteProp(itemId) or propMgr:isInfiniteTodayProp(itemId) then
311        return true
312    end
313    return propMgr:isTimeLimitProp(itemId) == true
314end

317-- （如 MergeManager 里 self:getData():getCurWeek()）。这里按监听同源配表直接取图，绕开 getData()。
318-- @param itemId integer 道具 ID
319-- @return table|nil imageInfo {Atlas, Frame, IsPropPng}
320local function resolveDynamicImage(itemId)
321    local BoardMeta = require "amf.feature.localactivity.merge.config.BoardMeta"

323        return nil
324    end
325    local atlas = BoardMeta.BoardItemsPlistPath
326    local frame = BoardMeta:getMergeItemSpName(itemId)

334    end

338local function captureIcon(itemId, options, cachePath)
339    -- 图标异常（走 getUnknownItemImage 兜底）直接记为失败，避免产出占位图的无意义补偿
340    local supported, reason, imageInfo = checkItemSupported(itemId)
341    local customImage = nil
342    if not supported then
343        -- RESOLVE_ICON 类动态图标失败时，按同源配表直接取图（方案 B），仍无法取到才记失败
344        local dynamicImage = resolveDynamicImage(itemId)
345        if not dynamicImage then
346            return nil, reason
347        end
348        local loaded, loadReason = ensureImageLoaded(dynamicImage)
349        if not loaded then
350            return nil, "dynamic image frame not resolved: " .. tostring(dynamicImage.Frame) .. " (" .. tostring(loadReason) .. ")"
351        end
352        customImage = dynamicImage
353    elseif imageInfo and imageInfo.__renderDirect then
354        -- 散图定位到活动仓库等非运行时路径时，直接按绝对路径设图
355        customImage = imageInfo
356    end
357
358    local root, rootError = getSceneRoot()
359    if not root then
360        return nil, rootError
361    end
362
363    local icon
364    local success, result = xpcall(function()
365        icon = UEPropIcon.create()
366        if not icon then
367            error("UEPropIcon.create returned nil")
368        end
369
370        -- 截图不设置外层 scale；UEPropIcon 内部资源适配逻辑仍按正常流程执行。
371        if customImage then
372            -- 动态图标 / 散图：直接设图，绕开 itemId → getItemAttr → RESOLVE_ICON 链路
373            icon:setItemImage(customImage)
374        else
375            -- 只有无限/限时道具才传 num，普通道具不传数量角标，避免影响像素边界测量。
376            local itemData = { itemId = itemId }
377            if shouldPassNum(itemId) then
378                itemData.num = options.num or DEFAULT_NUM
379            end
380            icon:setItem(itemData)
381        end
382        -- 采集图标统一使用已领取(Claimed)状态，与里程碑图标的展示状态保持一致。
383        if icon.setRewardState and _G.UEPropIcon_RewardState then
384            icon:setRewardState(_G.UEPropIcon_RewardState.Claimed)
385        end
386        root:addChild(icon)
387
388        local bounds = icon:getGroupBounds(root, true)
389        if not bounds or not bounds.origin or not bounds.size
390            or bounds.size.width <= 0 or bounds.size.height <= 0 then
391            error("icon bounds unavailable")
392        end
393
394        local boundsCenterX = bounds.origin.x + bounds.size.width * 0.5
395        local boundsCenterY = bounds.origin.y + bounds.size.height * 0.5
396        icon:setPositionXY(
397            icon:getPositionX() + CAPTURE_WIDTH * 0.5 - boundsCenterX,
398            icon:getPositionY() + CAPTURE_HEIGHT * 0.5 - boundsCenterY
399        )
400
401        local renderTexture = CCRenderTexture:create(
402            CAPTURE_WIDTH,
403            CAPTURE_HEIGHT,
404            kCCTexture2DPixelFormat_RGBA8888
405        )
406        if not renderTexture then
407            error("CCRenderTexture:create returned nil")
408        end
409        renderTexture:beginWithClear(0, 0, 0, 0, 0)
410        icon:visit()
411        renderTexture:endToLua()
412
413        local imagePath = cachePath .. "/item_" .. tostring(itemId) .. ".png"
414        if not renderTexture:saveToFile(imagePath) then
415            error("saveToFile failed: " .. imagePath)
416        end
417        if renderTexture.dispose then
418            renderTexture:dispose()
419        end
420
421        return {
422            itemId = itemId,
423            imagePath = imagePath,
424            captureScale = DEFAULT_CAPTURE_SCALE,
425            centerX = CAPTURE_WIDTH * 0.5,
426            centerY = CAPTURE_HEIGHT * 0.5,
427            progressTop = options.progressTop,
428            progressBottom = options.progressBottom,
429        }
430    end, function(errorMessage)
431        return debug.traceback(tostring(errorMessage))
432    end)
433
434    if icon and not icon.isDisposed then
435        icon:removeFromParentAndCleanup(true)
436    end
437    if root and not root.isDisposed then
438        root:removeFromParentAndCleanup(true)
439    end
440
441    if not success then
442        return nil, result
443    end
444    return result
445end

447-- 用 CCRenderTexture 只渲染进度条可视区域（ScrollContainer），不包含整屏背景。
448-- @param progress UEProgressMilestone 验证组件
449-- @param cachePath string 输出目录
450-- @param itemId integer 道具 ID
451-- @param state string 验证状态
452-- @return string|nil imagePath 截图路径
453-- @return string|nil errorMessage 错误信息
454local function captureProgressBar(progress, cachePath, itemId, state)
455    local scrollContainer = progress and progress:_getScrollContainer()
456    if not scrollContainer then
457        return nil, "scroll container unavailable"
458    end
459    local parent = progress:getParent()
460    if not parent then
461        return nil, "progress parent unavailable"
462    end
463
464    local size = scrollContainer:getContentSize()
465    if size.width <= 0 or size.height <= 0 then
466        return nil, "invalid scroll container size"
467    end
468
469    -- 让 ScrollContainer 可视区域左下角对齐渲染纹理原点，保证只截进度条区域。
470    local scWorld = scrollContainer:convertToWorldSpace(ccp(0, 0))
471    local scInParent = parent:convertToNodeSpace(scWorld)
472    local parentOrigin = parent:convertToNodeSpace(ccp(0, 0))
473    local offsetX = scInParent.x - parentOrigin.x
474    local offsetY = scInParent.y - parentOrigin.y
475    local originalX, originalY = progress:getPositionX(), progress:getPositionY()
476    progress:setPositionXY(originalX - offsetX, originalY - offsetY)
477
478    local renderTexture
479    local success, result = xpcall(function()
480        -- 进度条内部可能包含裁剪节点，使用带模板缓冲的渲染纹理避免裁剪失效。
481        renderTexture = CCRenderTexture:create(
482            size.width,
483            size.height,
484            kCCTexture2DPixelFormat_RGBA8888,
485            0x88F0
486        )
487        if not renderTexture then
488            error("CCRenderTexture:create returned nil")
489        end
490        renderTexture:beginWithClear(0, 0, 0, 0, 0)
491        progress:visit()
492        renderTexture:endToLua()
493
494        local imagePath = cachePath .. "/progress_" .. tostring(itemId)
495            .. "_" .. tostring(state) .. ".png"
496        if not renderTexture:saveToFile(imagePath) then
497            error("saveToFile failed: " .. imagePath)
498        end
499        return imagePath
500    end, function(errorMessage)
501        return debug.traceback(tostring(errorMessage))
502    end)
503
504    if renderTexture and renderTexture.dispose then
505        renderTexture:dispose()
506    end
507    progress:setPositionXY(originalX, originalY)
508
509    if not success then
510        return nil, result
511    end
512    return result
513end

569    local itemIds = getItemIds(options.itemIds)
570    if options.roiMode == "milestone" and options.progressTop == nil
571        and options.progressBottom == nil and itemIds[1] then
572        local roi = resolveMilestoneProgressRoi(itemIds[1], options)
573        if roi then
574            options.progressTop = roi.top
575            options.progressBottom = roi.bottom
576            options.progressScale = roi.scale
577        end
578    end
579    local task = {
580        running = true,
581        index = 1,
582        total = #itemIds,
583        itemIds = itemIds,
584        successes = {},
585        failures = {},
586        cachePath = cachePath,
587        options = options,
588    }
589    _G.__MilestonePropIconPixelConfigTask = task
590
591    local function step()
592        if not task.running then
593            return
594        end
595        local itemId = task.itemIds[task.index]
596        if itemId == nil then
597            task.running = false
598            local lines = {
599                "itemId\timagePath\tcaptureScale\tcenterX\tcenterY\tprogressTop\tprogressBottom",
600            }
601            for _, row in ipairs(task.successes) do
602                lines[#lines + 1] = serializeManifestRow(row)
603            end
604            writeText(cachePath .. "/manifest.tsv", table.concat(lines, "\n") .. "\n")
605            writeText(cachePath .. "/failures.lua", table.serialize(task.failures))
606            return
607        end
608
609        local result, errorMessage = captureIcon(itemId, task.options, cachePath)
610        if result then
611            task.successes[#task.successes + 1] = result
612        else
613            task.failures[#task.failures + 1] = {
614                itemId = itemId,
615                errorMessage = errorMessage,
616                inputIndex = task.index,
617            }
618        end
619        task.index = task.index + 1
620        setTimeOut(step, 0)
621    end
622    setTimeOut(step, 0)
623
624    return {
625        started = true,
626        total = task.total,
627        cachePath = task.cachePath,
628    }
629end

654---创建用于截图验证的真实 UEProgressMilestone。
655---@param itemId integer 道具 ID
656---@param beforePx number scale=1 的左侧补偿
657---@param afterPx number scale=1 的右侧补偿
658---@param value number 当前进度值
659---@param options table|nil maxValue、threshold、scale、position 等参数
660---@return table|nil result 验证组件信息
661---@return string|nil errorMessage 错误信息
662function MilestonePropIconPixelConfigDebug:createVerificationProgress(
663    itemId,
664    beforePx,
665    afterPx,
666    value,
667    options
668)
669    options = options or {}
670    local scene = Director:sharedDirector():getRunningSceneLua()
671    if not scene then
672        return nil, "running scene unavailable"
673    end
674    local oldProgress = _G.__MilestonePropIconPixelVerificationProgress
675    if oldProgress and not oldProgress.isDisposed then
676        oldProgress:removeFromParentAndCleanup(true)
677    end
678
679    local threshold = tonumber(options.threshold) or 500
680    local maxValue = tonumber(options.maxValue) or 1000
681    local milestoneScale = tonumber(options.milestoneScale) or DEFAULT_MILESTONE_SCALE
682    local builder = UEProgressMilestoneBuilder.new()
683        :setMaxValue(maxValue)
684        :setValue(tonumber(value) or 0)
685        :setProgressWidth(tonumber(options.progressWidth) or 900)
686        :setScrollContainerWidth(tonumber(options.scrollContainerWidth) or 520)
687        :setScrollable(false)
688        :setMilestoneConfig({
689            autoClaim = true,
690            showMilestoneLabels = true,
691            milestonesList = {
692                {
693                    threshold = threshold,
694                    rewardConfig = {
695                        itemId = itemId,
696                        num = tonumber(options.num) or DEFAULT_NUM,
697                    },
698                    pixelCompensation = {
699                        beforePx = (tonumber(beforePx) or 0) * milestoneScale,
700                        afterPx = (tonumber(afterPx) or 0) * milestoneScale,
701                    },
702                },
703            },
704        })
705
706    local success, progressOrError = pcall(function()
707        return WidgetManager:createWidgetObj("UEProgressMilestone", builder)
708    end)
709    if not success or not progressOrError then
710        return nil, tostring(progressOrError)
711    end
712
713    local progress = progressOrError
714    -- 截图验证时关闭进度条描边图片透明度，避免描边干扰可视区域判断（默认 0）。
715    if progress.setOutlineImageOpacity then
716        progress:setOutlineImageOpacity(tonumber(options.outlineImageOpacity) or 0)
717    end
718    progress:setName("MilestonePropIconPixelVerification")
719    progress:setPositionXY(
720        tonumber(options.positionX) or 360,
721        tonumber(options.positionY) or 500
722    )
723    scene:addChild(progress)
724    _G.__MilestonePropIconPixelVerificationProgress = progress
725    return {
726        itemId = itemId,
727        threshold = threshold,
728        value = tonumber(value) or 0,
729        beforePx = tonumber(beforePx) or 0,
730        afterPx = tonumber(afterPx) or 0,
731    }
732end

734---创建固定参数的里程碑进度条并按三状态截取进度条可视区域。
735---固定进度条长度 1000、最大值 1000、可视区域 500、里程碑值 100；
736---state 对应验证值：before=99、milestone=100、after=101。
737---截图使用 CCRenderTexture 只渲染进度条可视区域，不包含整屏。
738---@param itemId integer 道具 ID
739---@param beforePx number scale=1 的左侧补偿
740---@param afterPx number scale=1 的右侧补偿
741---@param state string 验证状态："before"|"milestone"|"after"
742---@param options table|nil cachePath 等参数
743---@return table|nil result 验证结果，含 state、value、imagePath
744---@return string|nil errorMessage 错误信息
745function MilestonePropIconPixelConfigDebug:createVerificationProgressShot(
746    itemId,
747    beforePx,
748    afterPx,
749    state,
750    options
751)
752    options = options or {}
753    local value = VERIFICATION_STATE_VALUES[state]
754    if not value then
755        return nil, "invalid verification state: " .. tostring(state)
756    end
757
758    local result, errorMessage = self:createVerificationProgress(itemId, beforePx, afterPx, value, {
759        threshold = VERIFICATION_FIXED.threshold,
760        maxValue = VERIFICATION_FIXED.maxValue,
761        progressWidth = VERIFICATION_FIXED.progressWidth,
762        scrollContainerWidth = VERIFICATION_FIXED.scrollContainerWidth,
763        positionX = VERIFICATION_FIXED.positionX,
764        positionY = VERIFICATION_FIXED.positionY,
765    })
766    if not result then
767        return nil, errorMessage
768    end
769
770    local cachePath = getCachePath(options)
771    ensureDirectory(cachePath)
772    local imagePath, shotError = captureProgressBar(
773        _G.__MilestonePropIconPixelVerificationProgress,
774        cachePath,
775        itemId,
776        state
777    )
778    if not imagePath then
779        return nil, shotError
780    end
781
782    result.state = state
783    result.value = value
784    result.imagePath = imagePath
785    return result
786end
