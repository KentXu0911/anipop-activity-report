671function Manager:refreshActivityIconV2(reason, icon)

673    if not isDataDrivenIcon(icon) then
674        return false, "notIconV2"
675    end
676
677    local ok, refreshed, detail = pcall(icon.refreshProvider, icon, reason)
678    if not ok then
679        return true, false, refreshed
680    end
681    return true, refreshed, detail

684function Manager:acquireActivityIconV2ReminderSuppression(icon, reason)
685    if not isDataDrivenIcon(icon) or type(icon.acquireReminderSuppression) ~= "function" then
686        return false
687    end
688
689    local ok = pcall(
690        icon.acquireReminderSuppression,
691        icon,
692        FRIEND_GO_TIP_REMINDER_OWNER
693    )
694    if not ok then
695        if type(icon.hideAllReminderUI) == "function" then
696            pcall(icon.hideAllReminderUI, icon)
697        end
698        return false
699    end
700    self._friendGoTipReminderIconV2 = icon
701    return true
702end

712    pcall(
713        icon.releaseReminderSuppression,

715        FRIEND_GO_TIP_REMINDER_OWNER

747            type = item.type,
748            headUrl = item.headUrl,
749            frame = item.frame and tonumber(item.frame),
750            add = item.add,
751			levelUp = item.levelUp
752        }
753
754        -- 根据 type 分类
755        if item.type == 1 then
756            table.insert(ret[TipEnum.GetNewInvitation], entry)
757        elseif item.type == 2 then
758            table.insert(ret[TipEnum.GetFriendInvitation], entry)
759        elseif item.type == 3 then
760            table.insert(ret[TipEnum.FriendGo], entry)
761        elseif item.type == 4 then
762            table.insert(ret[TipEnum.Upgrade], entry)
763        end
764    end
765    table.sort(ret[TipEnum.FriendGo], function(a, b)
766        return a.add > b.add
767    end)
768
769	table.sort(ret[TipEnum.Upgrade], function(a, b)
770			if a.levelUp > b.levelUp then
771				return true
772			else if a.levelUp < b.levelUp then
773				return false
774			else
775				return a.add > b.add
776			end
777		end
778    end)
779
780    if self:needFindFriend() then
781        ret[TipEnum.FindFriend]={{}}
782    end
783
784    for k, v in pairs(ret) do
785        if #v == 0 then
786            ret[k] = nil
787        end
788    end
789
790    local last = self._lastShowTipList or {}
791    local function reset()
792        self._lastShowTipList = ret
793        self._nextTipIndex = 1
794        self._nextEnumIndex = 1
795    end
796
797    local function compare(t1, t2)
798        -- 先比较键数量
799        if #t1 ~= #t2 then return false end
800
801        -- 比较每个键的值
802        for k, v in pairs(t1) do
803            if not t2[k] then return false end
804            for kk, vv in pairs(v) do
805                if vv ~= t2[k][kk] then return false end
806            end
807        end
808        return true
809    end
810
811    for _, v in pairs(TipEnum) do
812        if (ret[v] and last[v] and not compare(ret[v], last[v]))
813            or (last[v] and not ret[v]) or (not last[v] and ret[v]) then
814            reset()
815            break
816        end
817    end
818
819    return self._lastShowTipList
820end

826function Manager:checkIconTip(config)

1571function Manager:updatePassLevelData(data, collectionCount)
1572    -- 获得收集物为0时不必锁定icon的power
1573    if not collectionCount or collectionCount == 0 then
1574        self.iconCanChangePower = true
1575    end
1576    if not self:isOpen() then return end
1577    if data.settleMentState and collectionCount and collectionCount > 0 then
1578        local power = self:getPower()
1579        self:setPower(power + collectionCount)
1580        local icon = self:getActivityIcon()
1581        local isIconV2 = isDataDrivenIcon(icon)
1582        local isLegacyIcon = isLegacyActivityIcon(icon)
1583        if isLegacyIcon then
1584            icon:setPower(power)
1585        end
1586        local v2Finished = false
1587        local function finishIconV2Collection()
1588            if v2Finished then return end
1589            v2Finished = true
1590            self:refreshActivityIconV2("teamCampingCollectionAnimationEnded", icon)
1591            self.iconCanChangePower = true
1592        end
1593        local flyCallBack = function()
1594            if isIconV2 then
1595                finishIconV2Collection()
1596            elseif isLegacyIcon then
1597                icon:rollNumber(function()
1598                    self.iconCanChangePower = true
1599                end)
1600            else
1601                self.iconCanChangePower = true
1602            end
1603        end
1604        local sp = self:createIcon({
1605            itemId = Meta.collectionId,
1606            num = 1})
1607        local flyCollectionContext = FlyActCollectionContext.new(self:getFuncId(), {{
1608            icon = sp,
1609            num = collectionCount
1610        }})
1611        flyCollectionContext:setFinishCallback(flyCallBack)
1612        if isIconV2 then
1613            local transactionOk = pcall(
1614                _G.amfTransaction,
1615                TransactionEvent.FLY_ACT_COLLECTIONS,
1616                flyCollectionContext
1617            )
1618            if not transactionOk then
1619                flyCollectionContext:finish()
1620                if sp and not sp.isDisposed and type(sp.dispose) == "function" then
1621                    pcall(sp.dispose, sp)
1622                end
1623            end
1624            return
1625        end
1626        _G.amfTransaction(TransactionEvent.FLY_ACT_COLLECTIONS, flyCollectionContext)
1627    end
1628end
