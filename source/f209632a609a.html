12function CollectCardFriendSelectPanelVM.formatOnlineText(lastLoginTime, isOnline, now)
13    if isOnline then return "在线" end
14
15    local duration = math.max((now or TimeKit:timeInSec()) - (tonumber(lastLoginTime) or 0), 0)
16    if duration < 3600 then
17        return "1小时前在线"
18    elseif duration < 24 * 3600 then
19        return string.format("%d小时前登录", math.floor(duration / 3600))
20    elseif duration <= 7 * 24 * 3600 then
21        local hours = math.floor(duration / 3600)
22        return string.format("%d天%d小时前登录", math.floor(hours / 24), hours % 24)
23    end
24    return "7天以前登录过"
25end

58function CollectCardFriendSelectPanelVM:reloadFriends()
59    self.friendRows = {}
60    self.friendRowByUid = {}
61    for _, friend in ipairs(self._mgr:getAllFriendInfo()) do
62        local profile = friend:getProfile()
63        local uid = tostring(friend:getUid())
64        local lastLoginTime = friend:getLastLoginTime()
65        local row = {
66            uid = uid,
67            profile = profile,
68            name = nameDecode(profile.name),
69            xxlId = tostring(profile.xxlId),
70            lastLoginTime = lastLoginTime,
71            topLevel = friend:getTopLevelId(),
72            onlineText = self.formatOnlineText(lastLoginTime, friend:isOnline(), TimeKit:timeInSec()),
73            selectable = true,
74        }
75        table.insert(self.friendRows, row)
76        self.friendRowByUid[uid] = row
77    end
78    table.sort(self.friendRows, function(left, right)
79        if left.topLevel ~= right.topLevel then
80            return left.topLevel > right.topLevel
81        end
82        return left.uid < right.uid
83    end)
84    self:refreshFriendRows(false)
85end

87function CollectCardFriendSelectPanelVM:refreshFriendRows(updateView)
88    local keyword = self.searchText
89    self.visibleRows = {}
90    for _, row in ipairs(self.friendRows) do
91        if keyword == "" or row.xxlId == keyword then
92            row.selected = self.selectedUidSet[row.uid] == true
93            table.insert(self.visibleRows, row)
94        end
95    end
96    if updateView == false then return end
97    self:callViewFunc("updateFriendRows", self.visibleRows)
98end

124function CollectCardFriendSelectPanelVM:onKeyboardInput(evt)

127    local text = self.searchText or ""

155function CollectCardFriendSelectPanelVM:onFriendTapped(uid)
156    local changes = {}
157
158    local function setSelected(changedUid, selected)
159        self.selectedUidSet[changedUid] = selected and true or nil
160        self.friendRowByUid[changedUid].selected = selected
161        table.insert(changes, {uid = changedUid, selected = selected})
162    end
163
164    if self.selectedUidSet[uid] then
165        setSelected(uid, false)
166        for index, selectedUid in ipairs(self.selectedUids) do
167            if selectedUid == uid then
168                table.remove(self.selectedUids, index)
169                break
170            end
171        end
172    elseif self.mode == CollectCardConst.FriendSelectMode.ASK then
173        if #self.selectedUids >= MAX_ASK_SELECTION then
174            CommonTip:showTip("最多选择4名好友哦~")
175            return
176        end
177        setSelected(uid, true)
178        table.insert(self.selectedUids, uid)
179    else
180        local previousUid = self.selectedUids[1]
181        if previousUid then setSelected(previousUid, false) end
182        self.selectedUids = {uid}
183        setSelected(uid, true)
184    end
185    self:callViewFunc("updateSelection", changes, #self.selectedUids > 0, #self.selectedUids)
186end

203function CollectCardFriendSelectPanelVM:onContinueTapped()
204    if #self.selectedUids == 0 or self:isLocked() then return end
205
206    local payload = {
207        mode = self.mode,
208        cardId = self.cardId,
209        panelId = self.panelId,
210        sourcePanelId = self.sourcePanelId,
211        targetUids = table.simpleClone(self.selectedUids),
212        exchangeEnabled = self.mode == CollectCardConst.FriendSelectMode.GIVE and self.exchangeEnabled or false,
213    }
214    if self.mode == CollectCardConst.FriendSelectMode.GIVE then
215        payload.targetProfile = table.simpleClone(self.friendRowByUid[self.selectedUids[1]].profile)
216    end
217
218    self:lockWithoutBlock()
219    local result = self.onContinue(payload)
220    return result:next(function(value)
221        self:unlockWithoutBlock()
222        if self.isDisposed then return value end
223
224        local afterSuccessClose = self.afterSuccessClose
225        self:close()
226        safeRunCall(afterSuccessClose, value, payload)
227        return value
228    end, function(error)
229        self:unlockWithoutBlock()
230        if not self.isDisposed then CommonTip:showErrorTip(error) end
231        return deferred.error(error)
232    end)
233end
