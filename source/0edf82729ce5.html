579function Manager:onFlowerCreate(evt)
580    if not self:isEnabled() then return end
581    local data = evt:getData()
582    local levelId = data.levelId
583    local levelScene = data.levelScene
584    if UBFLevelSceneFunc:isMainLevel(levelScene) and self._userManager:getTopLevelId() < levelId then return end
585    if not levelId or not levelScene then return end
586    local levelIdList = self.mainSceneLevelMap[levelScene] or {}
587    local levelNeedRecheckList = self.needRecheckLevelMap[levelScene] or {}
588
589    local index = table.indexOf(levelNeedRecheckList, levelId)
590    if index then
591        table.remove(levelNeedRecheckList, index)
592        self.needRecheckLevelMap[levelScene] = levelNeedRecheckList
593    end
594
595    if not table.exist(levelIdList, levelId) then
596        return false
597    end
598
599    -- 非满级满星时 彩星关不展示
600    local score = self._levelAreaManager:getUserScore( levelId, levelScene )
601    local star = score and score.star or 0
602    if self._userStateManager:getUserSegmentState() ~= _G.UserStateConst.UserSegmentState.MHZC and star == 4 then
603        return
604    end
605
606    table.insert(data.dataList, {
607        priority = FlowerConfig.Priority.STAR_WAND_BUFF,
608        creator = function()
609            -- 主线top关
610            if UBFLevelSceneFunc:isMainLevel(levelScene) and self._userManager:getTopLevelId() == levelId and star == 0 then
611                local key = "star_wand_top"
612                local flower = data.pop(key)
613                if not flower then
614                    local TopFlower = require "amf.feature.fulllevel.starwandbuff.mainscene.TopFlower"
615                    flower = TopFlower:create(levelId, levelScene)
616                    flower.resKey = key
617                    flower:setName(key)
618                    data.add(flower)
619                end
620                return flower
621            else
622                -- 主线非top关&&隐藏关
623                local key = MainSceneConfig:getFlowerKey(levelId, levelScene)
624                local flower = data.pop(key)
625                if not flower then
626                    local NormalFlower = require "amf.feature.fulllevel.starwandbuff.mainscene.NormalFlower"
627                    flower = NormalFlower:create(levelId, levelScene)
628                    flower.resKey = key
629                    flower:setName(key)
630                    data.add(flower)
631                end
632                return flower
633            end
634        end
635    })
636end

1371    elseif levelScene == UBFLevelSceneType.kHide then
1372        pos = LevelNodePosConfig:getHiddenFlowerPositionInWorldSpace(levelId)

1376        if self._mainSceneButtonManager:isEnabled() then
1377            return 0, pos
1378        else
1379            return -1, pos
1380        end

1383            -- 在上方时避让区域满级小记
1384            return 0, pos
1385        else
