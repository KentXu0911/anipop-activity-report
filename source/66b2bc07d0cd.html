130-- 5级以下玩家首次启动直接进入关卡
131function Stage2EnterSceneHandler:checkAutoStartLevelUnder5()
132    local UserEnergyManager = _G.amfBean("UserEnergyManager")
133    local isInfiniteEnergy = UserEnergyManager:getEnergyState() == UserEnergyState.INFINITE
134    local isEnergyEnough = UserEnergyManager:getEnergy() >= 5
135    if not isEnergyEnough then return end
136    local topLevelId = _G.getTopLevelId()
137    if topLevelId < 1 or topLevelId > 5 then return false end
138
139    -- Feed自动进关目前是前5关自动进关的一个子逻辑分支
140    if self:checkFeedAutoStart() then return true end
141
142    if _G.noviceContinueFiveAttackEnable == nil then
143        --app全开，小游戏读开关
144        local on = EnvUtil:isApp() or MaintenanceManager:getInstance():isEnabledInGroup('keepinlevel', 'ON')
145        _G.noviceContinueFiveAttackEnable = on and true or false
146    end
147
148    if not _G.noviceContinueFiveAttackEnable then return false end
149
150
151    local costType = StartLevelCostType.kEnergy
152    if isInfiniteEnergy then costType = StartLevelCostType.kInfiniteEnergy end
153
154    if topLevelId == 1 and self._isFirstEnterMainScene and FeedContext.isVideoGuideLevelFlow() and FeedContext.isNewUser() then
155        local mgr = _G.amfBean("RecommendationFeedManager")
156        if mgr and mgr.prepareFeedPreplayVideo then
157            mgr:prepareFeedPreplayVideo()
158        end
159        if mgr and mgr.showFeedPreplayVideo then
160            mgr:showFeedPreplayVideo()
161        end
162    end
163
164    AutoPopEventCenter:dispatch(AutoPopEvent.CRASH_RESUME, {
165        popupCheckPoint = PopupCheckPoint.AUTO_START_LEVEL,
166        params = {
167            levelId = topLevelId,
168            costType = costType,
169            needAlert = false,
170            source = AutoPopupConst.StartLevelSource.LOW_LEVEL
171        }
172    })
173    return true
174end
