90function HappyDeliveryManager:registerActEndTimer()
91    if self.__passActMainNextTimer then
92        TimerUtil.removeAlarm(self.__passActMainNextTimer)
93        self.__passActMainNextTimer = nil
94    end
95
96    local mainEndTime = self:getModel():getMainEndTime()
97    if not mainEndTime or mainEndTime <= 0 then return end
98
99    local nowTime = TimeKit:timeInSec()
100    if nowTime >= mainEndTime then return end
101
102    self.__passActMainNextTimer = TimerUtil.addAlarm(function()
103        self.__passActMainNextTimer = nil
104        self:onPassActMainNext()
105    end, mainEndTime - nowTime, 1)
106end

315-- 关卡内在行为树判断通过、passlevel 协议发送前调用，用于快照创建 UI 所需数据
316-- 若活动不可用返回 nil
317-- 返回结构：
318--   taskIndex     int       当前任务是每天第几次任务
319--   curProgress   int       单任务当前进度（passlevel 前的值）
320--   totalProgress int       单任务总进度要求
321--   passCount     int       当天总通关数（passlevel 前的值）
322--   boxInfo       table     宝箱配置，对应后端 BoxInfoBean：
323--     baseLevel     int       宝箱初始等级（升级动画起点）
324--     finalLevel    int       宝箱最终等级（L1~L5 或 S1~S5）
325--     splitCount    int       分裂次数（0~5）
326--     scriptResult  list<int> 剧本：3次点击升级值序列
327--     rewards       list      奖励列表，每项 {rewardId, num}
328--     rewarded      bool      奖励是否已领取
329function HappyDeliveryManager:getUBFDisplayData()
330    if not self:isEnabled() then return nil end
331
332    local taskIndex, isCompleted = self:getCurTaskIndex()
333    if isCompleted then return nil end
334
335    local currentDay = self:getDataByKey("currentDay")
336    local dayInfo = self:getCurDayInfo()
337    local boxCfg = dayInfo.boxInfoList[taskIndex]
338    local prevAccumulated = taskIndex > 1 and self:getModel():getAccumulatedRequired(taskIndex - 1) or 0
339    local passCount = dayInfo.passCount
340    local curProgress = passCount - prevAccumulated
341
342    self.pendingAddPass = true
343
344    return {
345        currentDay = currentDay,
346        taskIndex = taskIndex,
347        curProgress = curProgress,
348        totalProgress = boxCfg.passRequired,
349        passCount = passCount,
350        boxInfo = boxCfg,
351    }
352end

354function HappyDeliveryManager:dcQualify()
355    if self:getLocalCache(self:getConst().DCKey.HAS_QUALIFY, true) then return end
356    self:setLocalCache(self:getConst().DCKey.HAS_QUALIFY, true, true)
357
358    local dayInfo = self:getCurDayInfo()
359    local levelNums = {}
360    for _, boxInfo in ipairs(dayInfo.boxInfoList) do
361        table.insert(levelNums, tostring(boxInfo.passRequired))
362    end
363    local t1 = table.concat(levelNums, "_")
364    local t2 = self:getDataByKey("userCrowdId")
365    self:dcPlug():dcQualify(t1, t2)
366end
