261function FinishAlbumPanel:playRewardFlyAnim()
262    local d = deferred.new()
263    if table.isEmpty(self.bubbleList) then
264        d:resolve()
265        return d
266    end
267
268    local needFlyNum = 0
269    for i, reward in ipairs(self.rewards) do
270        if not reward.doNotFly then
271            needFlyNum = needFlyNum + 1
272        end
273    end
274
275
276    local function onFlyEnd()
277        needFlyNum = needFlyNum - 1
278        if needFlyNum <= 0 then
279            d:resolve()
280        end
281    end
282
283    if needFlyNum == 0 then
284        d:resolve()
285        return d
286    end
287
288    for i, bubble in ipairs(self.bubbleList) do
289        local reward = self.rewards[i]
290        local worldPos = bubble:getPositionInWorldSpace()
291        bubble:playHideAnim(function()
292            if self.isDisposed then return end
293            bubble:setVisible(false)
294            local flyContext = FlyItemContext.new({
295                {itemId = reward.itemId, num = reward.num}
296            }, worldPos)
297            :setFinishCallback(onFlyEnd)
298            _G.amfTransaction(TransactionEvent.FLY_ITEMS, flyContext)
299        end, nil, true, 0.1)
300    end
301
302    return d
303end
