59        local function creator()

62                return

65                stage = AutoPopupConst.PopStage.kEnterFromLevel2nd

67        end
68        _G.amfBean("AutoPopupMgr"):addPopupConfig(PopupCheckPoint.RECALL_2025_STAGE4, creator)
69        self:setLocalCache("stage4_fail_pop", true, true)
70    end
71
72    self._lastFailed = false
73end

110function Manager:onActActInfoUpdate(event)

221function Manager:openMainPanel(userClick, closeCallback, extra)
222    local isFirst = false
223    local dayCount = 0
224    local popedId = self:getModel():getLocalCache("popedId")
225    local roundId = self:getModel():getDataByKey("roundId", 0)
226
227    if extra and extra.stage == AutoPopupConst.PopStage.STAGE_TWO then
228        if not self:getModel():checkTodayRightsGot() then
229            local bonusDays = self:getModel():getDataByKey("bonusDays", {})
230            if table.isEmpty(bonusDays) and roundId ~= popedId then isFirst = true end
231        end
232    end
233    dayCount = self:getModel():getDayCount()
234    local showLong = self.diffTime > 3
235
236    local delegate = {
237        userClick = userClick,
238        -- 首次弹出展示
239        isFirst = isFirst,
240        -- 第几天
241        dayCount = dayCount,
242
243        closeCallback = closeCallback,
244
245        showLong = showLong,
246
247        canEnter = extra and extra.canEnter
248    }
249
250    self:getModel():setLocalCache("popedId", nil, roundId)
251
252    local vm = require("amf.feature.recall.recall2025.view.MainPanelVM")
253    vm:create(delegate):popout()
254end

273function Manager:tryOpenMainPanel(userClick, closeCallback, extra)
274    if not self:isEnabled2() then
275        safeRunCall(closeCallback)
276        return
277    end
278
279    -- 领完权益就开展示面板
