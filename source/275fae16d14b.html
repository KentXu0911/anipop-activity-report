104function ShopRewardMainPopupLogic:doMainPopup(popupType, closeCallback)

110        safeRunCall(closeCallback)
111        return
112    end
113
114    self:markMainPopuped(popupType)
115    local MainPanelVM = require "amf.feature.localactivity.shopreward.view.ShopRewardMainPanelVM"
116    local vm = MainPanelVM:create({
117        popupType = popupType,
118        manager = self.manager,
119        mainPopupLogic = self,
120        closeCallBack = closeCallback,
121        maxExtraGoldNum = self:getMaxExtraGoldNum(),
122        mainEndTime = self.manager:getMainEndTime(),
123    })
124    vm:setLoadResFailCallback(closeCallback)
