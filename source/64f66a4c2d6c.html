176    local retValue = defaultValue

181            return ansValue
182        end
183        return t
184    end
185	
186	local data = self:getValue("data")
187
188    if not key or not data or table.isEmpty(data) then
189        return retValue
190    end
191
192    if data[key] ~= nil then
193        retValue = data[key]
194        retValue = copy(retValue)
195        return retValue
196    end
197
198	if data['userBean'] and data['userBean'][key] ~= nil then
199        retValue = data['userBean'][key]

678function CanteenModel:getA()
679	local makeTimes = 0
680	local menuList = self:getDataByKey('menuList',{})
681	for k,v in pairs(menuList) do
682		makeTimes = makeTimes + v.makeTimes
683	end
684
685	printx(229,'厨神食堂:已经制作的数量',makeTimes)
686	local checkN = 5
687	local canGetA,makeIdMpas= self:checkCanMakeTimes(checkN  - makeTimes)
688	if not canGetA then return nil end
689	printx(229,'厨神食堂:至少可以制作的菜谱列表',tts(makeIdMpas))
690	local menuList = self:getDataByKey('menuList',{})
691	local materialMap = table.clone( self:getDataByKey('materialMap',{},false))
692	local Mlist = {}
693	for k,v in pairs(menuList) do
694		local M1 = 0
695		local M2 = 0
696		for id,needNum in pairs(v.needMaterialMap) do
697			M2 = M2 + needNum
698			if not materialMap[id] or  materialMap[id] < needNum then
699				M1 = M1 + ( materialMap[id] or 0)
700			else
701				M1 = M1 + needNum
702			end
703		end
704		local M = M1*1.0/M2
705		if M < 1 then
706			table.insert(Mlist,{
707				M = M,
708				menuBean = v
709			})
710		end
711		printx(229,"厨神食堂:食谱ID" .. v.menuId .. "的完成度M = ",M)
712	end
713
714	local randomList = {}
715	table.sort(Mlist,function(a,b)
716		return a.M > b.M
717	end)
718	for k,v in ipairs(Mlist) do
719		if #randomList > 0 then break end
720		local maxMBean = v.menuBean
721		local isInLimit = self:getLimitMenu() and true or false
722		if maxMBean.menuType == 2 and not isInLimit then
723			--donothing
724		else
725			for id,needNum in pairs(maxMBean.needMaterialMap) do

727					table.insert(randomList,id)
728				end
729			end
730		end
731	end
732	if #randomList == 0 then return 0 end
733	return randomList[math.random(1,#randomList)]
734end

736--判断能做的菜是否大于这个值
737function CanteenModel:checkCanMakeTimes(needNum)
738	local menuList = self:getDataByKey('menuList',{})
739	local makeIdMap = {}
740	local materialMap = table.clone( self:getDataByKey('materialMap',{},false))
741
742
743	local checkCanMake = function(materialMap,needMaterialMap)
744		for id,needNum in pairs(needMaterialMap) do
745			if not materialMap[id] or  materialMap[id] < needNum then
746				return false
747			end
748		end
749	end
750
751	local doMake = function(materialMap,needMaterialMap,menuId)
752		for id,needNum in pairs(needMaterialMap) do
753			materialMap[id] = materialMap[id] - needNum
754		end
755		table.insert(makeIdMap,menuId)
756		return materialMap
757	end
758
759	local revertMake = function(materialMap,needMaterialMap,menuId)
760		for id,needNum in pairs(needMaterialMap) do
761			materialMap[id] = materialMap[id] + needNum
762		end
763		table.remove(makeIdMap,#makeIdMap)
764		return materialMap
765	end
766
767	local find
768	find = function(materialMap)
769		if #makeIdMap > needNum then
770			printx(29,"可制作的菜谱",tts(makeIdMap))
771			return false
772		end
773		for k,menuBean in pairs(menuList) do
774			local isInLimit = self:getLimitMenu() and true or false
775			if menuBean.menuType == 2 and not isInLimit then
776				--donothing
777			elseif checkCanMake(materialMap,menuBean.needMaterialMap) then
778				doMake(materialMap,menuBean.needMaterialMap)
779				if find(materialMap) then return true end
780				revertMake(materialMap,menuBean.needMaterialMap)
781			end
782		end
783		return true
784	end
785
786
787	return find(materialMap),makeIdMap
788
789end
