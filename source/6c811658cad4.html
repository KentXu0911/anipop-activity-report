292------------------ 场景4强弹（声明式注册，替代活动配置 autoPopConfigs） ------------------
293-- 需求来源：拾光成歌「强弹规则 / 全局弹窗优先级」。
294-- 1. 场景4外层仲裁：每次回到场景4只展示一个活动弹窗，优先级 引流1 > 引流2 > 活动主面板每日强弹。
295--    所以这里只声明 1 个 checkPoint，三个面板的优先级在 popStage4 内部仲裁；声明成多个 checkPoint 会被框架在同一次回到场景4里依次弹完。
296-- 2. 引流1：当日免费奖励待领取时弹；引流2：有新增好友助力生命值待查看时弹。
297-- 3. 主面板每日强弹：场景4每日 1 次；当日可收集线索（含当天与之前天数）全部找到时不弹。
298-- 4. 领奖期不弹：由 isEnabled() 的正式期判定拦截。
299-- 5. 次数限制：框架 CheckPointNumLimit 配置每日 3 次、次日刷新，对应需求「每日最多弹出3次」；
300--    主面板分支另有每日 1 次限制，框架无法在同一个 checkPoint 内按分支计数，沿用活动本地缓存（MainPanelPanelPopCount）自控。
301function Manager:_getAutoPopConfigList()
302    local PCP = _G.PopupCheckPoint
303    local PopStage = AutoPopupConst.PopStage
304
305    return {
306        -- 场景4强弹（同一次回到场景4只弹这一个）：仅场景4触发；每日 3 次由框架 CheckPointNumLimit 控制。
307        -- check：引流1 / 引流2 / 活动主面板每日强弹任一满足；creator：按上面的优先级只弹其中一个。
308        {
309            checkPoint = PCP.UNDERTHESIN_STAGE4_POP,
310            stages = { PopStage.STAGE_FOUR },
311            actId = self:getConst().actId,
312            check = function()
313                return self:checkCanPopStage4()
314            end,
315            creator = function(_, onPopOver)
316                return function()
317                    self:popStage4(onPopOver)
318                end
319            end,
320        },
321    }
322end

324-- 场景4强弹可弹条件：正式期，且引流（每日最多 3 次）或主面板每日强弹（每日 1 次）任一满足
325function Manager:checkCanPopStage4()
326    if not self:isEnabled() then return false end
327    return self:checkCanPopYinLiu() or self:checkCanPopMainPanel()
328end

330-- 场景4强弹弹出：外层仲裁，三个面板只弹一个，优先级 引流1 > 引流2 > 活动主面板每日强弹
331function Manager:popStage4(onPopOver)
332    local _, canPopYinLiu1, canPopYinLiu2 = self:checkCanPopYinLiu()
333    if canPopYinLiu1 then
334        self:popYinLiu1Panel(onPopOver)
335    elseif canPopYinLiu2 then
336        -- 引流面板2 依赖活动文案标签，先加载再弹；加载失败也要结束本次强弹，避免强弹 FSM 卡在 CheckClose
337        _G.amfBean("AssetsUtil"):loadMiniLabels({"underthesin"}):next(function()
338            self:popYinLiu2Panel(onPopOver)
339        end, function()
340            safeRunCall(onPopOver)
341        end)
342    elseif self:checkCanPopMainPanel() then
343        self:popMainPanel(onPopOver)
344    else
345        -- 入队后状态变化（如已领取免费奖励）导致三个面板都不可弹时，结束本次强弹
346        safeRunCall(onPopOver)
347    end
348end

352function Manager:checkCanPopMainPanel()
353    if not self:isEnabled() then
354        -- printx(41, Manager.ACT_CH_NAME.." 场景4强弹, 活动不可见")
355        return false
356    end
357    -- 超过每日上限1次
358    if self:getModel():getFlagFromLocalByKeyInt("MainPanelPanelPopCount", true) >= 1 then
359        -- printx(31, Manager.ACT_CH_NAME.." 场景4强弹, 主面板 今日达到上限1次")
360        return false
361    end
362
363    -- 能找的线索都找了
364    local handbooks = self:getModel():getDataByKey("handbooks", {})
365    local handbookMetas = self:getModel():getConfigByKey("handbookMetas", {})
366    for k,v in pairs(handbooks) do
367        local openTs = table.find(handbookMetas, function( vv )
368            return vv.id == v.id
369        end).openTs
370
371        if (openTs < TimeKit:timeInMillis()) and (#v.gotClueIds < 7) then
372            return true
373        end
374    end
375
376    -- printx(41, Manager.ACT_CH_NAME.." 场景4强弹, 主面板 所有能找的线索已找完")
377
378    return false
379end

381-- 引流面板是否可以弹出
382function Manager:checkCanPopYinLiu()
383    if not self:isEnabled() then
384        -- printx(41, Manager.ACT_CH_NAME.." 场景4强弹, 活动不可见")
385        return false, false, false
386    end
387    -- 超过每日上限3次
388    if self:getModel():getFlagFromLocalByKeyInt("YinLiuPanelPopCount", true) >= 3 then
389        -- printx(31, Manager.ACT_CH_NAME.." 场景4强弹, 引流 今日达到上限3次")
390        return false, false, false
391    end
392    
393    local canPop1, canPop2 = false, false
394    -- 引流1
395    canPop1 = not self:gotTodayFreeReward()
396    -- printx(41, Manager.ACT_CH_NAME.." 场景4强弹, 引流1", canPop1)
397
398    -- 引流2
399    local newHp = self:getHpNum()
400    -- printx(41, Manager.ACT_CH_NAME .. " 场景4强弹, 引流2", newHp)
401    if newHp ~= 0 then canPop2 = true end
402
403    return canPop1 or canPop2, canPop1, canPop2
404end
