6local RIGHT_SIDE_POOL = IconMeta.PoolType.kRightSide
7-- Product gate: keep the creation-time business priority, but freeze it for the icon lifetime.

10local function getPoolType(priority)
11    return math.floor(priority / IconMeta.TypeMagicNum)
12end

14local function isFiniteNumber(value)
15    return type(value) == "number"
16        and value == value
17        and value ~= math.huge
18        and value ~= -math.huge
19end

21local function safeLog(logger, level, event, data)
22    local logMethod = logger and (logger[level] or logger.log)
23    if type(logMethod) == "function" then
24        pcall(logMethod, logger, event, data)
25        return
26    end
27
28    local alog = rawget(_G, "ALog")
29    if alog and type(alog.strict_log_error) == "function" then
30        pcall(
31            alog.strict_log_error,
32            "Icon V2 priority " .. tostring(event)
33                .. ": " .. tostring(data and data.error or data and data.value)
34        )
35    end
36end

38function IconPriorityController.new(icon, options)
39    options = options or {}
40    local basePriority = assert(icon:getBasePriority(), "ICON_V2_BASE_PRIORITY_REQUIRED")
41    return setmetatable({
42        icon = icon,
43        logger = options.logger,
44        iconManager = options.iconManager,
45        mountedResolver = options.mountedResolver,
46        originalPoolType = icon:getPoolType() or getPoolType(basePriority),
47        effectivePriority = basePriority,
48        dynamicPriorityResolved = false,
49    }, IconPriorityController)
50end

52function IconPriorityController.create(selfOrIcon, iconOrOptions, options)
53    if selfOrIcon == IconPriorityController then
54        return IconPriorityController.new(iconOrOptions, options)
55    end
56    return IconPriorityController.new(selfOrIcon, iconOrOptions)
57end

67function IconPriorityController:_isMounted()
68    if type(self.mountedResolver) == "function" then
69        local ok, mounted = pcall(self.mountedResolver, self.icon)
70        if ok then
71            return mounted == true
72        end
73        safeLog(self.logger, "error", "icon_priority_mounted_check_failed", {
74            error = mounted,
75            iconKey = self.icon and self.icon.iconKey,
76        })
77        return false
78    end
79
80    local capability = self.icon and self.icon.activityEventCapability
81    return capability ~= nil and capability.mounted == true
82end

84function IconPriorityController:_validate(priority)
85    if not isFiniteNumber(priority) then
86        return false, "priority_not_finite_number"
87    end
88    if priority < IconMeta.PriorityBottom or priority > IconMeta.PriorityTop then
89        return false, "priority_out_of_range"
90    end
91    if getPoolType(priority) ~= self.originalPoolType then
92        return false, "priority_crosses_pool_type"
93    end
94    return true
95end

97function IconPriorityController:_resolveIconManager()
98    if self.iconManager then
99        return self.iconManager
100    end
101    if type(_G.amfBean) ~= "function" then
102        return nil
103    end
104    local ok, manager = pcall(_G.amfBean, "IconManager")
105    if ok then
106        return manager
107    end
108    safeLog(self.logger, "error", "icon_priority_manager_resolve_failed", {
109        error = manager,
110        iconKey = self.icon and self.icon.iconKey,
111    })
112end

114function IconPriorityController:_requestReorder(reason)
115    local manager = self:_resolveIconManager()
116    if not manager or type(manager.requestReorderIcon) ~= "function" then
117        safeLog(self.logger, "error", "icon_priority_reorder_api_missing", {
118            error = "IconManager:requestReorderIcon unavailable",
119            iconKey = self.icon and self.icon.iconKey,
120            reason = reason,
121        })
122        return
123    end
124
125    local ok, requested, requestError = pcall(manager.requestReorderIcon, manager, self.icon, reason)
126    if not ok or requested == false then
127        safeLog(self.logger, "error", "icon_priority_reorder_request_failed", {
128            error = ok and requestError or requested,
129            iconKey = self.icon and self.icon.iconKey,
130            reason = reason,
131        })
132        return false
133    end
134    return true
135end

137function IconPriorityController:refresh(state, reason, payload)
138    local basePriority = self.icon:getBasePriority()
139    local dynamicPriorityEnabled = self:_isDynamicPriorityEnabled()
140    local initialStaticResolution = not dynamicPriorityEnabled and not self.dynamicPriorityResolved
141    local dynamicPriority = self.effectivePriority
142    if dynamicPriorityEnabled or initialStaticResolution then
143        local dynamicMethod = self.icon and self.icon.getDynamicPriority
144        local ok
145        ok, dynamicPriority = pcall(dynamicMethod, self.icon, state, reason, payload)
146        if not ok then
147            safeLog(self.logger, "error", "icon_priority_dynamic_callback_failed", {
148                error = dynamicPriority,
149                iconKey = self.icon and self.icon.iconKey,
150                reason = reason,
151            })
152            dynamicPriority = basePriority
153        elseif dynamicPriority == nil then
154            dynamicPriority = basePriority
155        else
156            local valid, invalidReason = self:_validate(dynamicPriority)
157            if not valid then
158                safeLog(self.logger, "warn", "icon_priority_dynamic_value_rejected", {
159                    error = invalidReason,
160                    value = dynamicPriority,
161                    iconKey = self.icon and self.icon.iconKey,
162                    reason = reason,
163                })
164                dynamicPriority = basePriority
165            end
166        end
167        self.dynamicPriorityResolved = true
168    end
169
170    if self.originalPoolType ~= RIGHT_SIDE_POOL and dynamicPriority ~= basePriority then
171        safeLog(self.logger, "warn", "icon_priority_dynamic_value_rejected", {
172            error = "dynamic_priority_unsupported_pool",
173            value = dynamicPriority,
174            iconKey = self.icon and self.icon.iconKey,
175            reason = reason,
176        })
177        dynamicPriority = basePriority
178    end
179
180    local oldPriority = self.effectivePriority
181    if oldPriority == dynamicPriority then
182        return false
183    end
184
185    self.effectivePriority = dynamicPriority
186    if self.originalPoolType == RIGHT_SIDE_POOL
187        and self:_isMounted()
188        and not initialStaticResolution
189        and not self:_requestReorder(reason)
190    then
191        self.effectivePriority = oldPriority
192        return false
193    end
194    return true
195end
