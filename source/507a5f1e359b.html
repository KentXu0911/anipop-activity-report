110local function includes(values, target)
111    for _, value in ipairs(values or {}) do
112        if value == target then
113            return true
114        end
115    end
116    return false
117end

119local function getPlayerStage()
120    local userStateManager = _G.amfBean("UserStateManager")
121    if userStateManager:isFMJ() then
122        return PlayerStage.NON_FULL
123    elseif userStateManager:isColorfulStarFull() then
124        return PlayerStage.FULL_COLOR
125    end
126    return PlayerStage.FULL_NOT_COLOR
127end

129local function matchesRule(rule, context)
130    if rule.playerStage ~= context.playerStage
131        or not includes(rule.activationTagIds, context.activationTag)
132        or not includes(rule.livenessTagIds, context.livenessTag)
133        or not includes(rule.payTagIds, context.payTag) then
134        return false
135    end
136
137    if context.playerStage == PlayerStage.NON_FULL then
138        local startLevel = UBFActHelper:getDropConfigLevelId(rule.startLevel, context.maxLevelId)
139        local endLevel = UBFActHelper:getDropConfigLevelId(rule.endLevel, context.maxLevelId)
140        return context.levelId >= startLevel and context.levelId <= endLevel
141    elseif context.playerStage == PlayerStage.FULL_NOT_COLOR then
142        return context.oldStar >= rule.minStar and context.oldStar <= rule.maxStar
143    end
144
145    return true
146end

148local function resolveCountPerBird(levelId, runtime)
149    if not levelId or not runtime then return end
150
151    local levelScene = runtime.levelScene
152    local context = {
153        playerStage = getPlayerStage(),
154        activationTag = UserTagManager:getUserTagBySeries(UserTagNameKeyFullMap.kActivation),
155        livenessTag = UserTagManager:getUserTagBySeries(UserTagNameKeyFullMap.kBiLivenessTag),
156        payTag = UserTagManager:getUserPaymentTag2ServerVersion(),
157    }
158    if context.playerStage == PlayerStage.NON_FULL then
159        context.levelId = UBFActHelper:getHiddenLevelStartNormalLevel(levelId, levelScene)
160        context.maxLevelId = _G.amfBean("LevelUnLockManager"):getUserMaxLevelId()
161    elseif context.playerStage == PlayerStage.FULL_NOT_COLOR then
162        context.oldStar = UBFActHelper:getOldStar(levelId, levelScene)
163    end
164
165    local matchedRule
166    for _, rule in ipairs(TestConfig.countPerBirdRules) do
167        if matchesRule(rule, context) then
168            matchedRule = rule
169            break
170        end
171    end
172
173    local countPerBird = matchedRule and matchedRule.countPerBird
174    if type(countPerBird) ~= "number" or countPerBird < 0 or countPerBird % 1 ~= 0 then
175        return
176    end
177    return countPerBird
178end

200function Manager:isEnabledInLevel(data)
201    if getDailyPassLevelCount(self) >= TestConfig.dailyPassLevelLimit then
202        return false
203    end
204    local superMagicBirdManager = _G.amfBean("SuperMagicBird2024Manager")
205    if not superMagicBirdManager
206        or not superMagicBirdManager:isEnabledInLevel(data)
207        or not superMagicBirdManager:hasSuperPower()
208    then
209        return false
210    end
211
212    if EnvUtil:isDevelopMode() then
213        return self.debugEnabled == true
214    end
215
216    return false
217end
