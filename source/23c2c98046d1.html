142function Model:getScore()
143    if not self:isDataEnabled() then return end
144    local mapId = self:getMapId()
145    local meta = self:getConfigByKey("blockMapMetaList", {})
146    local score = self:getDataByKey("blockMapData.eliminateScore", 0)
147    if not self:checkInInfinite()then
148        for _, v in pairs(meta) do
149            if v.mapId == mapId then
150                return score, v.progressScore
151            end
152        end
153    else
154        local lastCost, lastRound = 0, 0
155        local endLessRound = self:getEndLessRound()
156        for _, v in ipairs(meta) do
157            if v.mapId == mapId and v.endLess then
158                if v.endLessRound == endLessRound then
159                    return score, v.progressScore
160                elseif v.endLessRound < endLessRound then
161                    score = score - v.progressScore
162                    lastCost = v.progressScore
163                    lastRound = v.endLessRound
164                end
165            end
166        end
167        if lastCost > 0 then
168            return score - (endLessRound - lastRound - 1) * lastCost, lastCost
169        end
170    end
171
172    return 0, 0
173end
