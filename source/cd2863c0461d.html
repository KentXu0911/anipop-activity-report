103function Config:getTaskMeta( taskType )

112function Config:getPlayLevelType(  )
113    --获取闯关类型
114    local playLevelType = Config.PlayLevelType.kTopLevel
115    local UserStateManager = _G.amfBean("UserStateManager")
116    if UserStateManager:isFMJ() then
117        playLevelType = Config.PlayLevelType.kTopLevel
118    elseif UserStateManager:isColorfulStarFull() then
119        playLevelType = Config.PlayLevelType.kFullLevelLib
120    else
121        playLevelType = Config.PlayLevelType.kRefreshColorStar
122    end
123    return playLevelType
124end

126function Config:getTaskTipStr( taskType, target, taskFlag )
127
128    if taskFlag == nil or taskFlag == 0 then 
129        taskFlag = self:getPlayLevelType() 
130    end
131
132    local showStr = ""
133    local bSucess, taskMeta = self:getTaskMeta(taskType)
134    if bSucess then
135        local tipStr = taskMeta.taskTip
136        if taskType == Config.TaskType.PASS_LEVEL then
137            local playLevelType = taskFlag
138            
139            if playLevelType == Config.PlayLevelType.kTopLevel then
140                tipStr = taskMeta.taskTip
141            elseif playLevelType == Config.PlayLevelType.kFullLevelLib then
142                tipStr = taskMeta.taskTip3
143            elseif playLevelType == Config.PlayLevelType.kRefreshColorStar then
144                tipStr = taskMeta.taskTip2
145            end
146        end
147        showStr = string.format( tipStr, target )
148    end
149    return showStr
150end

178function Config:getTaskResPath( taskType, target, taskFlag )
179
180    if taskFlag == nil or taskFlag == 0 then 
181        taskFlag = self:getPlayLevelType() 
182    end
183    
184    local resPath = ""
185    local bSucess, taskMeta = self:getTaskMeta(taskType)
186    if bSucess then
187        if taskType == Config.TaskType.PASS_LEVEL then
188            local playLevelType = taskFlag
189            if playLevelType == Config.PlayLevelType.kTopLevel then
190                resPath = "weektaskicons/"..taskMeta.picPath
191            elseif playLevelType == Config.PlayLevelType.kFullLevelLib then
192                resPath = "weektaskicons/"..taskMeta.picPath3
193            elseif playLevelType == Config.PlayLevelType.kRefreshColorStar then
194                resPath = "weektaskicons/"..taskMeta.picPath2
195            else
196                resPath = "weektaskicons/"..taskMeta.picPath
197            end
198        elseif taskType == Config.TaskType.LITTLE_MERGE_BOOK or taskType == Config.TaskType.LITTLE_MERGE_COST_ENERGY then
199            --小merge
200            if _G.amfBean("MergeEvolutionManager") and _G.amfBean("MergeEvolutionManager"):isEnabled() then
201                resPath = "weektaskicons/"..taskMeta.picPath2
202            elseif _G.amfBean("MusicMergeManager") and _G.amfBean("MusicMergeManager"):isEnabled() then
203                resPath = "weektaskicons/"..taskMeta.picPath
204            else
205                resPath = "weektaskicons/"..taskMeta.picPath2
206            end
207        else
208            resPath = "weektaskicons/"..taskMeta.picPath
209        end
210    end
211    return resPath
212end
