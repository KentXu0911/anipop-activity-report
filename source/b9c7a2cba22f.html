449function ReturnGift2Panel:updateRewardingAwardLayout(rewarding, rewardCount)
450    self:cacheAwardNodeOriginPos()
451
452    if rewardCount == 2 then
453        self:setAwardNodeVisible(1, true)
454        self:setAwardNodeVisible(2, true)
455        self:setAwardNodeVisible(3, false)
456        local pos1 = self.awardNodeOriginPos and self.awardNodeOriginPos[1]
457        local pos2 = self.awardNodeOriginPos and self.awardNodeOriginPos[2]
458        local pos3 = self.awardNodeOriginPos and self.awardNodeOriginPos[3]
459        if pos1 and pos2 and pos3 then
460            local gap = pos1.y - pos2.y
461            local centerY = (pos1.y + pos3.y) / 2
462            self:setAwardNodePosition(1, pos1.x, centerY + gap / 2 + TWO_REWARD_FIRST_OFFSET_Y)
463            self:setAwardNodePosition(2, pos2.x, centerY - gap / 2)
464        end
465    else
466        self:restoreAwardNodeOriginPos()
467        for index = 1, 3 do
468            self:setAwardNodeVisible(index, index <= rewardCount)
469        end
470    end
471
472    for index = 1, 3 do
473        local addImg = self["addImg_" .. tostring(index)]
474        if addImg then addImg:setVisible(not rewarding and index <= rewardCount) end
475    end
476end

631function ReturnGift2Panel:initGiftCarAnim(playAppearAnim, enterAnimFinishCallback)
632    local frameImg = self.frameImg
633    if not frameImg or not UICreateUtil then return end
634
635    local parent = frameImg:getParent()
636    if not parent or not parent.getChildIndex or not parent.addChildAt then return end
637
638    if isSafeCocosObj(self.giftCarBackAnim) then
639        self.giftCarBackAnim:removeFromParentAndCleanup(true)
640        self.giftCarBackAnim = nil
641    end
642    if isSafeCocosObj(self.giftCarFontAnim) then
643        self.giftCarFontAnim:removeFromParentAndCleanup(true)
644        self.giftCarFontAnim = nil
645    end
646
647    local index = parent:getChildIndex(frameImg)
648    if playAppearAnim == nil then playAppearAnim = self:shouldPlayGiftCarAppearAnim() end
649    self:prepareContentNodeEnterAnim(playAppearAnim)
650    local backAnimName = playAppearAnim and "appear_back2" or "idle_back2"
651    local frontAnimName = playAppearAnim and "appear_front2" or "idle_front2"
652    local backAnim = self:createGiftCarAnim(backAnimName, not playAppearAnim)
653    local fontAnim = self:createGiftCarAnim(frontAnimName, not playAppearAnim, function()
654    end)
655    setTimeOut(function ()
656        if self.isDisposed then return end
657        self:playContentNodeEnterAnim(enterAnimFinishCallback)
658    end, 1)
659    if not backAnim or not fontAnim then
660        if playAppearAnim then safeRunCall(enterAnimFinishCallback) end
661        return
662    end
663
664    backAnim:setPositionXY(360, -780)
665    fontAnim:setPositionXY(360, -780)
666    parent:addChildAt(backAnim, 0)
667    parent:addChildAt(fontAnim, index + 2)
668
669    self.giftCarBackAnim = backAnim
670    self.giftCarFontAnim = fontAnim
671end

701function ReturnGift2Panel:playContentNodeEnterAnim(finishCallback)
702    if self.isDisposed or not self.contentNode then return end
703    if not self.contentNodeOriginX or not self.contentNodeOriginY then
704        safeRunCall(finishCallback)
705        return
706    end
707
708    self.contentNode:stopAllActions()
709    self.contentNode:setVisible(true)
710    self.contentNode:runAction(CCSequence:createWithTwoActions(
711        CCMoveTo:create(7/30, ccp(self.contentNodeOriginX, self.contentNodeOriginY+10)),
712        CCMoveTo:create(5/30, ccp(self.contentNodeOriginX, self.contentNodeOriginY))
713    ))
714    self:playRewardItemsEnterAnim()
715    self:playUnpurchasedNodeEnterAnim()
716    --延迟20帧
717    setTimeOut(function ()
718        if self.isDisposed then return end
719            safeRunCall(finishCallback)
720    end, 0.67)
721end
