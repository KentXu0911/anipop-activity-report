122function WaterfallGiftProgressManager:_scheduleIconV2Boundary()
123    self:_cancelIconV2BoundaryTimer()
124    if not self:isV2IconActive() then
125        return true
126    end
127
128    local now = TimeKit and type(TimeKit.timeInSec) == "function" and tonumber(TimeKit:timeInSec()) or nil
129    local inAwardTime = self:isInAwardTime()
130    local boundaryTime = tonumber(inAwardTime and self:getRewardEndTime() or self:getEndTime())
131    if not now or not boundaryTime then
132        return false, "WATERFALL_PROGRESS_ICON_V2_BOUNDARY_TIME_REQUIRED"
133    end
134    if boundaryTime <= now then
135        if inAwardTime then
136            return self:updateIcon()
137        end
138        return self:onEnterRewardTime()
139    end
140
141    local delay = math.max(1, math.floor(boundaryTime - now) + 1)
142    local timerId
143    local timerSource
144    local function onBoundary()
145        if self._iconV2BoundaryTimerId ~= timerId then
146            return
147        end
148        self._iconV2BoundaryTimerId = nil
149        self._iconV2BoundaryTimerSource = nil
150        if inAwardTime then
151            self:updateIcon()
152        else
153            self:onEnterRewardTime()
154        end
155    end
156    if type(setTimeOut) == "function" then
157        local ok, result = pcall(setTimeOut, onBoundary, delay)
158        if ok and result ~= nil then
159            timerId = result
160            timerSource = "timeout"
161        end
162    end
163    if not timerId and TimerUtil and type(TimerUtil.addOnce) == "function" then
164        local ok, result = pcall(TimerUtil.addOnce, onBoundary, delay)
165        if ok and result ~= nil then
166            timerId = result
167            timerSource = "timer_util"
168        end
169    end
170    if not timerId then
171        return false, "WATERFALL_PROGRESS_ICON_V2_BOUNDARY_TIMER_REQUIRED"
172    end
173    self._iconV2BoundaryTimerId = timerId
174    self._iconV2BoundaryTimerSource = timerSource
175    return true
176end

183    local adapterLoaded, PackageIconV2Adapter = pcall(require, PACKAGE_ICON_V2_ADAPTER_MODULE)
184    if not adapterLoaded or type(PackageIconV2Adapter.create) ~= "function" then

197        PackageIconV2Adapter.create,
198        PackageIconV2Adapter,

202        return nil, adapter, nil, nil, false
203    end

264        res_map = IconV2Config.getPreloadFiles()

270            return
271        end
272        if self.iconAdapter then
273            self:_refreshOwnedIcon("waterfall_progress_asset_ready")
274            return

277            self:_addLegacyIconIfNotExist()
278            return

280

282        if not adapter then
283            if cleanupOwner then
284                self.iconAdapter = cleanupOwner
285                self._iconAdapterCleanupPending = true
286            elseif safeLegacy then
287                self:_addLegacyIconIfNotExist()

295        self.iconDataProvider = provider
296        self._iconAdapterCleanupPending = false
297        self.icon = adapter:getIcon()
298        if self:isV2IconActive() then
299            self:dcQualify()
300            local scheduled, scheduleError = self:_scheduleIconV2Boundary()
301            if scheduled == false then
302                logIconV2Error("schedule boundary after create failed: " .. tostring(scheduleError))

511function WaterfallGiftProgressManager:requestGetAwardTimeRewards(onSuccess, onFail, onCancel)
512    dataModel:requestGetAwardTimeRewards():next(function(response)
513        local data = response:getData() or {}
514        local rewards = data.rewards or {}
515
516        if data.nodeRewardDetails and data.progressRewardDetails then
517
518            local nodeRewardDetails = data.nodeRewardDetails or {}
519            local progressRewardDetails = data.progressRewardDetails or {}
520
521            for i,v in ipairs(progressRewardDetails) do
522                local extraMap = { pay_type = DcPayType.kRmb }
523                self:addRewards(v.rewards, "falls_gift", "rewardtime", extraMap)
524            end
525
526            local freeNum = 2 -- 理应使用dataModel:getRealFreeNum()，但是领奖期不使用配置了。强行写死2个
527            for i,v in ipairs(nodeRewardDetails) do
528                local extraMap = {}
529                if v.nodeId <= freeNum then
530                    extraMap = { pay_type = DcPayType.kFree }
531                else
532                    extraMap = { pay_type = DcPayType.kRmb }
533                end
534                self:addRewards(v.rewards, "falls_gift", "rewardtime", extraMap)
535            end
536        else
537            -- nodeRewardDetails/progressRewardDetails 缺失时走兜底，无法区分 pay_type
538            self:addRewards(rewards, "falls_gift", "rewardtime")
539        end
540
541        dataModel:setRewardTimeReward()
542        rewards = table.mergeRewards(rewards)
543        local nodeRewardIdList = data.nodeRewardIdList or {}
544        local progressRewardIdList = data.progressRewardIdList or {}
545        local t1, t2
546        if #nodeRewardIdList > 0 then
547            t1 = table.concat(nodeRewardIdList, "_")
548        end
549        if #progressRewardIdList > 0 then
550            t2 = table.concat(progressRewardIdList, "_")
551        end
552        safeRunCall(onSuccess, rewards, t1, t2)
553    end, function(response)
554        if response:isCancel() then
555            safeRunCall(onCancel)
556        else
557            local errCode = response:getErrorCode()
558            if errCode then
559                CommonTip:showTip(localize("error.tip."..tostring(errCode)))
560            end
561            safeRunCall(onFail)
562        end
563    end)
564end

718function WaterfallGiftProgressManager:checkNeedAutoPopout(sceneType)
719    if not self:isEnabled() then return false end
720    local config = dataModel:getAutoPopoutConfig()
721
722    -- 强弹时间
723    local autoPopoutCd = config.autoPopoutCd
724    local lastPopTime = dataModel:getLastPopoutTime()
725    if (TimeKit:timeInSec()-lastPopTime) < autoPopoutCd*3600 then
726        log("checkNeedAutoPopout time in cd", TimeKit:timeInSec(), lastPopTime, autoPopoutCd)
727        return false
728    end
729
730    -- 条件
731    local conditionList = config.conditionList
732    local AutoPopDefinition = Meta.AutoPopDefinition
733    local conditionMetaList = config.conditionMetaList
734    log("checkNeedAutoPopout condition check", tts(conditionList), sceneType)
735    for i, conditionId in ipairs(conditionList) do
736        local conditionMeta = conditionMetaList[conditionId] or {}
737        local sceneId = conditionMeta.sceneId
738        if sceneId == sceneType then
739            local value = conditionMeta.conditionValue or 0
740            local id = conditionMeta.conditionId
741            if AutoPopDefinition[id] then
742                local checkFunc = AutoPopDefinition[id]
743                local res = checkFunc(value)
744                if res ~= false then
745                    log("checkNeedAutoPopout condition check pass", res)
746                    return res
747                end
748            end
749        end
750    end
751    return false
752end

928function WaterfallGiftProgressManager:findThenQueryDyOffer()
929    local deferred = require "amf.script.sugar.deferred"
930    local d = deferred.new()
931    local userData = self:getUserData() or {}
932    local actConfig = self:getActConfig() or {}
933    local fixedGiftCfg = actConfig.fixRewardNodeList
934    local curIdx = userData.gotRewardNodeIndex
935    local needQuery = false
936    for i = 1, 6 do
937        local cfg = fixedGiftCfg[curIdx + i]
938        if cfg then
939            if not cfg.freeNode and cfg.goodsId then
940                local goodsObj = GoodsManager:getGoods(cfg.goodsId)
941                if goodsObj then
942                    local price = goodsObj:getPrice()
943                    self:queryDyOffer(price, d)
944                    needQuery = true
945                    break
946                end
947            end
948        else
949            break
950        end
951    end
952    if not needQuery then
953        d:reject()
954    end
955    return d
956end

958function WaterfallGiftProgressManager:queryDyOffer(price, d)
959    local Req = require "amf.middleware.purchase.http.DouyinTiktokPrepayConsult"
960    local req = Req:post({
961        price = price,
962        goodsTagActKey = Meta.ModuleName,
963    })
964    if not req then
965        d:reject()
966        return
967    end
968
969    req:next(function(response)
970        local data = response:getData()
971        if not data or not data.viewData or not data.viewData.operationTip then
972            d:reject()
973            return
974        end
975
976        local offerInfo = {
977            price = price,
978            operationTip = data.viewData.operationTip,
979            assignDiscountKey = data.assignDiscountKey,
980            goodsTag = data.goodsTag,
981        }
982
983        if data.goodsTag then
984            local tags = table.deserialize(data.goodsTag)
985            if tags and tags.dyType then
986                offerInfo.dyType = tags.dyType
987            end
988        end
989
990        local discount
991        if data.viewData.operationMinAmount then
992            if data.viewData.operationMinAmount == data.viewData.operationMaxAmount then
993                discount = tonumber(data.viewData.operationMinAmount)
994            end
995        else
996            discount = tonumber(string.match(data.viewData.operationTip, "^立减(%d+%.?%d*)元$"))
997        end
998
999        if discount then
1000            offerInfo.discount = discount
1001            offerInfo.finalPrice = (price * 1000 - discount * 1000) / 1000
1002        end
1003        d:resolve(offerInfo)
1004    end, function()
1005        d:reject()
1006    end)
1007end
