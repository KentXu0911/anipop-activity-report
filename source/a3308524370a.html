249function UBFWinningActsBannerView:initTrophyExtraRewardsView()
250    if table.isEmpty(self.extraRewards) then
251        return
252    end
253    if not self:isViewSafe("trophy") then
254        return
255    end
256    for k, v in pairs(self.extraRewards) do
257        if not v.streakWinNum then
258            v.streakWinNum = self.curWinNum 
259        end
260        local isCanCreate = v.streakWinNum >= self.curWinNum
261        if v.streakWinNum > 5 then
262            isCanCreate = self.curWinNum == v.streakWinNum
263        end
264        if isCanCreate then
265            --当大于5连胜的奖励放在第五个奖杯处
266            local index = math.clamp(v.streakWinNum, 1, 5)
267            local extraRewardViewName = string.format("trophyExtraReward_%s_%s", v.rewardId, index)              
268            if not self:isViewSafe(extraRewardViewName) then
269                if not self.trophyExtraRewardViewData[index] then
270                    self.trophyExtraRewardViewData[index] = {}
271                end
272                table.insert(self.trophyExtraRewardViewData[index], v)
273                local updateData = {parentViewName = "trophy_" .. index}
274                if v.isMultiple then
275                    --倍数
276                    local viewData = UBFHelper:shallowClone(v.pendantViewData)
277                    UBFHelper:shallowMerge(viewData, updateData)
278                    self:createAndAddViewAsync(viewData, extraRewardViewName):next(function(anim)
279                        if not isSafeCocosObj(anim) then return end
280                        if not self:isViewSafe() then return end
281                        local posList = {ccp(28, 49), ccp(34, 55), ccp(36, 63), ccp(41, 70), ccp(25, 76)}
282                        anim:setPosition(posList[index])
283                    end)
284                else
285                    --给数量
286                    self:createAndAddPreparedViewAsync("extraRewardNode", extraRewardViewName, updateData):next(function(view)
287                        if not self:isViewSafe() or not isSafeCocosObj(view) then
288                            return
289                        end
290                        local posList = {ccp(12, 49), ccp(18, 55), ccp(22, 63), ccp(25, 70), ccp(30, 76)}
291                        view:setPosition(posList[index])
292                        view:getChildByName("tipNum"):setString("+" .. v.num)
293                        self:createAndAddViewAsync(v.viewData):next(function(node)
294                            if not self:isViewSafe() or not isSafeCocosObj(node) then
295                                return
296                            end
297                            local pos = view:getChildByName("tipIcon"):getPosition()
298                            node:setPosition(ccp(pos.x, pos.y))
299                            node:setScale(0.4)
300                            view:addChildAt(node, 1)
301                            view:getChildByName("tipIcon"):removeFromParentAndCleanup(false)
302                            self:referenceView("extraRewardTipIcon" .. index, node)
303                        end)
304                    end)
305                end
306            else
307                UBFThrow("该奖励节点已存在" .. v.rewardId)
308            end
309        end
310    end
311end

408function UBFWinningActsBannerView:initRewardView()

418    local spacingX = 180

432            self:getView("reward"):addChild(bubble)

467        local smallBallViewName = string.format("ball_%d_smallBall", lastWinNum)

469        self:callView(smallBallViewName, "setVisible", true)
470        local bigBallFrontViewName = string.format("ball_%d_bigBall_bigBallFront", lastWinNum)
471        local node1 = self:getView(bigBallFrontViewName)
472        local node2 = self:getView(smallBallViewName)
473        local arrayA, arrayB = ActionUtil:getIconChangeActionArray(node1, node2)
474        node1:runAction(arrayA)
475        node2:runAction(arrayB)
476    end
477
478    if self.curWinNum > 0 and self.curWinNum <= 5 then
479        local curWinNum = math.clamp(self.curWinNum, 1, 5)
480        local animateTime = self.lastWinNum > 0 and FRAME_TIME * 12 or 0
481        self:refreshProgressBar(curWinNum, animateTime, function()
482            local bigBallViewName = string.format("ball_%d_bigBall", curWinNum)
483            local node = self:getView(bigBallViewName)
484            local bigBallFrontViewName = string.format("ball_%d_bigBall_bigBallFront", curWinNum)
485            local bigBallNumFrontViewName = string.format("ball_%d_bigBall_bigBallNumFront", curWinNum)
486            self:callView(bigBallFrontViewName, "setVisible", true)
487            self:callView(bigBallNumFrontViewName, "setVisible", true)
488            ActionUtil:playCommonAppearAnim(node)
489            safeRunCall(done)
490        end)
491    else
492        safeRunCall(done)
493    end

1031        local found = false

1034                found = true
1035                v2.num = v2.num + v.num
1036                break
1037            end
1038        end
1039        if not found then
1040            table.insert(rewards, v)
1041        end
1042    end
1043    return rewards
