453function UBFPlpBottomWinRepeatRank:switchNodeRank(node, rank, lastRank)
454	local listItemCenter = node:getChildByName("listItemCenter")
455	local rankPart = listItemCenter:getChildByName("rankPart")
456	local topOther = rankPart:getChildByName("topOther")
457
458	local info = node.info
459	info.rank = rank 
460
461	local disapearUseTime = 2*FRAMETIME
462	local showUseTime = 3*FRAMETIME
463	local allUseTime = disapearUseTime + showUseTime
464	if lastRank > 3 and rank > 3 then
465		--数字变数字
466		local function changeRank()
467			--替换排名
468			if not self:isViewSafe() then return end
469			topOther:getChildByName("topOtherLabel"):setString(''..rank)
470		end
471
472		topOther:runAction(UIHelper:sequence{
473			ActionUtil:fadeToU3D(topOther, disapearUseTime, 100),
474			CCCallFunc:create(changeRank),
475			ActionUtil:fadeToU3D(topOther, showUseTime, 255),
476		})
477	else
478		--数字变皇冠
479		--旧的消失
480		if lastRank > 3 then
481			topOther:runAction(UIHelper:sequence{
482				ActionUtil:fadeOutU3D(topOther, allUseTime),
483			})
484		else
485			rankPart:getChildByName("top" .. lastRank):runAction(CCFadeOut:create(allUseTime))
486		end
487
488		--新的出现
489		if rank > 3 then
490			topOther:getChildByName("topOtherLabel"):setString(''..rank)
491			topOther:runAction(CCFadeIn:create(allUseTime))
492			topOther:setVisible(true)
493		else
494			rankPart:getChildByName("top" .. rank):runAction(CCFadeIn:create(allUseTime))
495			rankPart:getChildByName("top" .. rank):setVisible(true)
496		end
497	end
498
499	--礼盒
500	local giftBox = listItemCenter:getChildByName("giftBox")
501	local bCurHasBoxShow = giftBox:isVisible() 
502	local bHasReward = self:checkHasRankRewards(rank)
503	if bCurHasBoxShow then
504		local function changeBoxSp()
505			--替换礼盒icon
506			if not self:isViewSafe() then return end
507			if bHasReward then
508				giftBox:setVisible(true)
509				giftBox:changeDisplayByFrameName( self:getGixtBoxSpriteFrameName(rank) )
510			else
511				giftBox:setVisible(false)
512			end
513		end
514		--礼盒出现
515		giftBox:runAction(UIHelper:sequence{
516			ActionUtil:fadeToU3D(giftBox, disapearUseTime,100),
517			CCCallFunc:create(changeBoxSp),
518			ActionUtil:fadeToU3D(giftBox, showUseTime,255),
519		})
520	else
521		--替换礼盒icon，同时出现
522		if bHasReward then
523			giftBox:setVisible(true)
524			giftBox:setOpacity(0)
525			giftBox:changeDisplayByFrameName(self:getGixtBoxSpriteFrameName(rank))
526			giftBox:runAction(UIHelper:sequence{
527				CCFadeIn:create(allUseTime),
528			})
529		end
530	end
