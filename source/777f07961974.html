46function ReturnGift2StartPanelView:mergeRewardingExtra(target, source)
47    if not target or not source then return end
48    local packageType = tonumber(source.packageType) or Extra.getPackageType(source)
49    local status = tonumber(source.status) or 0
50    local state = tonumber(source.state) or 0
51    if packageType ~= Const.PackageType.PACKAGE_2
52        and status ~= Const.Status.REWARDING
53        and state ~= Const.State.PACKAGE_2_REWARDING then
54        return

61function ReturnGift2StartPanelView:initRewardingLeftTime()
62    local endTime = Extra.getRewardEndTime(self:getModuleExtra())
63    if endTime <= 0 then return end
64
65    if self._tickId then
66        TimerUtil.removeAlarm(self._tickId)
67        self._tickId = nil
68    end
69
70    local remain = endTime - TimeKit:timeInSec()
71    if remain > 0 then
72        self._tickId = TimerUtil.addAlarm(function()
73            if self.isDisposed then return end
74            remain = remain - 1
75            self:updateByTime(remain)
76            if remain <= 0 then
77                TimerUtil.removeAlarm(self._tickId)
78                self._tickId = nil
79                self:onCountDownEnd()
80            end
81        end, 1, 0)
82        self:updateByTime(remain)
83    else
84        self:updateByTime(0)
85        self:onCountDownEnd()
86    end
87end

146function ReturnGift2StartPanelView:refreshRewardBlocks()
147    local rewardNodes = Extra.getRewardNodes(self:getModuleExtra())
148    self:updateRewardBlocksLayout(#rewardNodes)
149    for index, block in ipairs(self.rewardBlocks or {}) do
150        local rewardNode = rewardNodes[index]
151        if block.node then block.node:setVisible(rewardNode ~= nil) end
152        local alreadyGot = rewardNode and rewardNode.alreadyGot == true
153        if block.receivedNode then block.receivedNode:setVisible(alreadyGot) end
154        if block.unreceivedNode then block.unreceivedNode:setVisible(rewardNode ~= nil and not alreadyGot) end
155        local reward = rewardNode and rewardNode.rewards and rewardNode.rewards[1]
156        if block.awardGroupIcon then
157            block.awardGroupIcon:setVisible(reward ~= nil)
158            if reward then
159                self:getGoodsUIOperator():setIconItem(block.awardGroupIcon, reward)
160            end
161        end
162        if block.timelabelText and block.timelabelText.setString and reward then
163            block.timelabelText:setString(self:getRewardTimeText(reward))
164        elseif block.timelabelText and block.timelabelText.setString then
165            block.timelabelText:setString("")
166        end
167    end

170function ReturnGift2StartPanelView:updateRewardBlocksLayout(rewardCount)

175        local pos3 = self.rewardBlockOriginPos and self.rewardBlockOriginPos[3]

180        end

220function ReturnGift2StartPanelView:onRewardBlockTapped(index)

222
223    local rewardNodes = Extra.getRewardNodes(self:getModuleExtra())
224    local rewardNode = rewardNodes[index]
225    if not rewardNode or rewardNode.alreadyGot then return end
226    if not rewardNode.rewards or #rewardNode.rewards == 0 then return end
227
228    local rewardIndex = tonumber(rewardNode.index)
229    if rewardIndex == nil then rewardIndex = index - 1 end
230    if not self.delegate or not self.delegate.claimReward then return end
231
232    self.delegate:claimReward(rewardIndex, function(rewards)
233        if self.isDisposed then return end
234        self:playClaimRewardFlyToBag(rewards, index, function()
235            if self.isDisposed then return end
236            self:refreshRewardState()
237            self:removeSelfIfAllClaimed()
238        end)
239    end)
240end

242function ReturnGift2StartPanelView:playClaimRewardFlyToBag(rewards, blockIndex, finishCallback)
243    local flyItems = self:getClaimRewardFlyItems(rewards, blockIndex)
244    if not flyItems or #flyItems == 0 then
245        safeRunCall(finishCallback)
246        return
247    end
248
249    --小游戏飞动画特殊处理
250    if EnvUtil:isMiniGame() then
251        local flyNum = #flyItems
252        for index, value in ipairs(flyItems) do
253            setTimeOut(function()
254                if self.isDisposed then return end
255                local flyItem = table.clone(value)
256                local flyContext = FlyItemContext.new(flyItem, ccp(360, 720))
257                if index == flyNum then
258                    flyContext:setFinishCallback(function()
259                        safeRunCall(finishCallback)
260                    end)
261                end
262                _G.amfTransaction(TransactionEvent.FLY_ITEMS, flyContext)
263            end, (index - 1) * Const.FlyItemDealy)
264        end
265        return
266    end
267
268    local flyContext = FlyItemContext.new(flyItems, ccp(360, 720))
269    flyContext:setFinishCallback(function()
270        safeRunCall(finishCallback)
271    end)
272    _G.amfTransaction(TransactionEvent.FLY_ITEMS, flyContext)
273end

275function ReturnGift2StartPanelView:getClaimRewardFlyItems(rewards, blockIndex)
276    local flyItems = {}
277    local block = self.rewardBlocks and self.rewardBlocks[blockIndex]
278    local sourceNode = block and (block.awardGroupIcon or block.node)
279    for index, item in ipairs(rewards or {}) do
