20--- 请求奖励
21function Manager:getReward(params, successHandler, sucCb, failCb, cancelCb)
22    local function onSuccess(evt)
23        if not evt.data then return end
24        local rewards = evt.data.rewards or {}
25        self:addRewards(rewards, params.type, params.dcSource)
26        safeRunCall(successHandler, rewards, evt)
27        self:checkDataAfterHttp(evt)
28        local t2
29        if params.type == 3 then
30            t2 = 2
31        end
32        Dc:featureDc("help_success", params.inviteCode, t2)
33        safeRunCall(sucCb, rewards, params)
34    end
35
36    local function onFail(evt)
37        if not params or not (params.type == 3 or params.type == 4) then
38            self:showErrorTip(evt)
39        end
40        local t2
41        if params.type == 3 then
42            t2 = 2
43        end
44        Dc:featureDc("help_failed", params.inviteCode, t2)
45        safeRunCall(failCb, evt)
46    end
47
48    Manager:requestServerData("amf.feature.localactivity.energyactv2.http.EnergyActV2RequestShareRewards",
49            params, onSuccess, onFail, cancelCb)
50end

52-- 在线请求统一入口
53-- 强强表示必须一个协议一个Request文件，自己建request去吧
54function Manager:requestServerData(requestClassPath, reqParams, onSuccess, onFail, onCancel, noLoading)
55    if noLoading == nil then noLoading = true end
56    local request = require(requestClassPath)
57    request:post(reqParams, nil, not noLoading):next(onSuccess, function(response)
58        if response:isCancel() then
59            safeRunCall(onCancel, response)
60        else
61            safeRunCall(onFail, response)
62        end
63    end)
64end

66--- 添加奖励
67function Manager:addRewards(rewardItems, type, source)
68    for _, reward in pairs(rewardItems or {}) do
69        if reward.itemId == Meta.ACT_ENERGY then
70            local oldBuff = _G.amfBean("UserEnergyManager"):getInfiniteEnergyEndTime()
71            local newBuff = 0
72            if oldBuff < TimeKit:time() then
73                newBuff = TimeKit:time() + 60 * 1000 * reward.num
74            else
75                newBuff = oldBuff + 60 * 1000 * reward.num
76            end
77            _G.amfBean("UserEnergyManager"):setInfiniteEnergyEndTime(newBuff)
78            self:addMultiRewards({reward}, source)
79        elseif reward.itemId == Meta.OTHER_ENERGY then
80            self:addMultiRewards({{itemId = Meta.OTHER_ENERGY, num = 1}}, source)
81        else
82            local propSource = 0
83            _G.amfBean("UserManager"):addReward(reward, propSource)
84            self:addMultiRewards({reward}, source)
85        end
86    end
87end

99function Manager:checkDataAfterHttp(evt)

101    if data.energyRewarded ~= nil then
102        -- 助力导致 能量值 变化，领取状态变化

107--- 显示错误提示
108function Manager:showErrorTip(evt, callback)
109    local errCode = self:getErrorCode(evt)
110    if errCode then
111        local text = Localization:getInstance():getText("error.tip." .. tostring(errCode))
112        if errCode >= 731776 and errCode <= 731779 then
113            CommonTipWithBtn:showTip({
114                tipStr = text,
115                okInfo = { btnStr = "知道了" },
116                colorType = PanelColorConst.ColorType.kBlue
117            }, "negative", nil, nil, nil, true)
118        else
119            CommonTip:showTip(text, "negative", callback)
120        end
121    else
122        safeRunCall(callback)
123    end
124end
