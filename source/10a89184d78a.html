411function Manager:_getTaskRewardReceivedProgress()
412    local taskMetaList = self:getTaskMetaList() or {}
413    local totalTaskCount = NUMBER_VALUE.ZERO
414    local receivedTaskCount = NUMBER_VALUE.ZERO
415    for _, taskMeta in ipairs(taskMetaList) do
416        if type(taskMeta) == "table" then
417            local taskId = toNumberOrNil(taskMeta.taskId) or taskMeta.taskId
418            if taskId ~= nil and self:isTaskTypeSupported(toNumberOrNil(taskMeta.taskType)) then
419                totalTaskCount = totalTaskCount + NUMBER_VALUE.ONE
420                if self:isTaskRewardReceived(taskId) then
421                    receivedTaskCount = receivedTaskCount + NUMBER_VALUE.ONE
422                end
423            end
424        end
425    end
426    return receivedTaskCount, totalTaskCount
427end

583function Manager:doShareTicketHelpByMini(panelId, shareConfig)
584    shareConfig = shareConfig or {}
585    if not self:_checkCanShareTicketHelp(shareConfig) then return end
586
587    local const = self:getConst()
588    local text, textIndex = self:_getShareText(self:getTicketHelpMiniCardTextList())
589    local onSuccess, onFail, onCancel = self:_makeTicketHelpShareCallback(false, shareConfig)
590    local originOnSuccess = onSuccess
591    local originOnFail = onFail
592    onSuccess = function(...)
593        safeRunCall(originOnSuccess, ...)
594        if shareConfig.needDcShareHelpResultVx == true then
595            self:_dcShareHelpResultVx()
596        end
597        if shareConfig.needDcShareHelpResultDy == true then
598            self:_dcShareHelpResultDy(SHARE_HELP_RESULT_TYPE.SUCCESS)
599        end
600    end
601    onFail = function(...)
602        safeRunCall(originOnFail, ...)
603        if shareConfig.needDcShareHelpResultDy == true then
604            self:_dcShareHelpResultDy(SHARE_HELP_RESULT_TYPE.FAIL)
605        end
606    end
607    local context = OpenShareContext.new(OpenShareConst.ShareType.HELP_ACTIVITY, OpenShareConst.ShareTarget.MINI)
608        :setActId(const.actId)
609        :setTextParams(text)
610        :setMiniParams({
611            imageUrl = shareConfig.miniImageUrl,
612            templateId = shareConfig.dyTemplateId,
613        })
614        :setAskHelpParams(nil, {}, const.SHARE_ID)
615        :setKVData(self:_getTicketHelpKVData(textIndex, panelId))
616        :setShareCallback(onSuccess, onFail, onCancel)
617
618    if EnvUtil:isWXMiniGame() and not EnvUtil:isEntryFromQQ() then
619        local askHelpPanelVM = require "amf.feature.localactivity.happymatch.view.HappyMatchAskHelpPanelVM"
620        context:setAskPanelParams(OpenShareConst.FriendType.WECHAT, nil, nil, nil, askHelpPanelVM)
621    end
622
623    _G.amfTransaction(TransactionEvent.SHARE, context)
624end

833function Manager:_normalizeTaskProgressReward(value)

845function Manager:_getYinLiuTaskData()
846    local taskMetaList = self:_getRawConfigByKey("happyMatchTaskMetaList")
847    if type(taskMetaList) ~= "table" or table.isEmpty(taskMetaList) then
848        return nil
849    end
850    for _, taskMeta in ipairs(taskMetaList) do
851        if type(taskMeta) ~= "table" then
852            return nil
853        end
854        if not TASK_PROGRESS_FIELD_BY_TYPE[toNumberOrNil(taskMeta.taskType)] then
855            return nil
856        end
857        if taskMeta.taskId == nil then
858            return nil
859        end
860        if toNumberOrNil(taskMeta.taskTarget) == nil then
861            return nil
862        end
863        if type(taskMeta.taskRewardList) ~= "table" or type(taskMeta.taskRewardList[NUMBER_VALUE.ONE]) ~= "table" then
864            return nil
865        end
866    end
867
868    local taskProgressData = self:_getRawDataByKey("taskProgressData")
869    if type(taskProgressData) ~= "table" then
870        return nil
871    end
872    for _, fieldName in pairs(TASK_PROGRESS_FIELD_BY_TYPE) do
873        if toNumberOrNil(taskProgressData[fieldName]) == nil then
874            return nil
875        end
876    end
877
878    local taskMap = self:_getRawDataByKey("taskMap")
879    if type(taskMap) ~= "table" then
880        return nil
881    end
882
883    local taskProgressReward = self:_normalizeTaskProgressReward(self:_getRawDataByKey("taskProgressReward"))
884    if taskProgressReward == nil then
885        return nil
886    end
887
888    return {
889        taskMetaList = taskMetaList,
890        taskProgressData = taskProgressData,
891        taskMap = taskMap,
892        taskProgressReward = taskProgressReward,
893    }
894end

1011    local rewardList = type(taskMeta) == "table" and taskMeta.taskRewardList or nil

1022function Manager:_getClaimableYinLiuRewards(taskData)
1023    taskData = taskData or self:_getYinLiuTaskData()
1024    if not taskData then
1025        return nil
1026    end
1027    if type(taskData.taskMetaList) ~= "table" then
1028        return nil
1029    end
1030    if type(taskData.taskProgressData) ~= "table" then
1031        return nil
1032    end
1033    if type(taskData.taskMap) ~= "table" then
1034        return nil
1035    end
1036
1037    local rewards = {}
1038    local totalTaskCount = NUMBER_VALUE.ZERO
1039    local receivedTaskCount = NUMBER_VALUE.ZERO
1040    for _, taskMeta in ipairs(taskData.taskMetaList) do
1041        if type(taskMeta) == "table" then
1042            local taskType = toNumberOrNil(taskMeta.taskType)
1043            local fieldName = TASK_PROGRESS_FIELD_BY_TYPE[taskType]
1044            local taskId = toNumberOrNil(taskMeta.taskId) or taskMeta.taskId
1045            local target = toNumberOrNil(taskMeta.taskTarget)
1046            if fieldName and taskId ~= nil and target ~= nil then
1047                totalTaskCount = totalTaskCount + NUMBER_VALUE.ONE
1048                local current = toNumberOrNil(taskData.taskProgressData[fieldName]) or NUMBER_VALUE.ZERO
1049                local received = self:_isYinLiuTaskRewardReceived(taskData.taskMap, taskId)
1050                if received then
1051                    receivedTaskCount = receivedTaskCount + NUMBER_VALUE.ONE
1052                elseif current >= target then
1053                    local taskRewardList = self:_buildYinLiuTaskRewardList(taskMeta)
1054                    for _, reward in ipairs(taskRewardList) do
1055                        table.insert(rewards, reward)
1056                    end
1057                end
1058            end
1059        end
1060    end
1061
1062    if totalTaskCount > NUMBER_VALUE.ZERO and receivedTaskCount >= totalTaskCount and taskData.taskProgressReward ~= true then
1063        table.insert(rewards, table.clone(TASK_PROGRESS_REWARD) or TASK_PROGRESS_REWARD)
1064    end
1065
1066    return rewards
1067end
