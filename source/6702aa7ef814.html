470function Manager:createThreeForOnePackageIconAdapter()

472        require,
473        "amf.feature.mainscene.module.icon.v2.runtime.PackageIconV2Adapter"
474    )

476        return {
477            state = ADAPTER_CREATE_SAFE_LEGACY,
478            error = adapterLoaded and "NEWBIE_GIFT_CHOICE_ICON_V2_ADAPTER_CREATE_REQUIRED"
479                or PackageIconV2Adapter,
480        }
481    end
482
483    local giftType = self.GIFT_TYPE.THREEFORONE
484    local configOk, options, provider = pcall(function()
485        local IconV2Config = require(THREE_FOR_ONE_ICON_V2_CONFIG_MODULE)
486        local configOptions, configProvider = IconV2Config.create(self, giftType, IconEventCenter)
487        assert(type(configOptions) == "table", configProvider or "NEWBIE_GIFT_CHOICE_ICON_V2_CONFIG_REQUIRED")
488        return configOptions, configProvider
489    end)
490    if not configOk then
491        local path = self.ICONSPINE[giftType]
492        local configError = options
493        options = {
494            ownerIconKey = self.ICONKEY[giftType],
495            ownerIdentity = self,
496            iconEventCenter = IconEventCenter,
497            legacyFactory = function()
498                return self:createLegacyActivityIcon(giftType, path)
499            end,
500            v2PreflightError = configError,
501        }
502        provider = nil
503    end
504
505    local createOk, adapter, modeOrError, cleanupOwner = pcall(
506        PackageIconV2Adapter.create,
507        PackageIconV2Adapter,
508        options
509    )
510    if not createOk then
511        return {
512            state = ADAPTER_CREATE_WAIT_OWNER,
513            error = adapter,
514        }
515    end
516    if not adapter then
517        return {
518            state = ADAPTER_CREATE_WAIT_OWNER,
519            error = modeOrError,
520            cleanupOwner = cleanupOwner,
521        }
522    end
523    return {
524        state = ADAPTER_CREATE_READY,
525        adapter = adapter,
526        provider = provider,
527        mode = modeOrError,
528    }
529end

531function Manager:_disposeThreeForOneIconAdapter(reason)
532    local adapter = self._threeForOneIconAdapter
533    if not adapter then
534        self._threeForOneIconAdapterRecoveryPending = false
535        return true
536    end
537    local ok, disposed, disposeError = pcall(adapter.dispose, adapter)
538    if not ok or disposed ~= true then
539        self._threeForOneIconAdapterRecoveryPending = true
540        log("dispose three-for-one icon adapter failed", reason, ok and disposeError or disposed)
541        return false, ok and disposeError or disposed
542    end
543    self._threeForOneIconAdapter = nil
544    self._threeForOneIconDataProvider = nil
545    self._threeForOneIconAdapterRecoveryPending = false
546    return true
547end
548
549function Manager:_recoverThreeForOneIconAdapter()
550    if not self._threeForOneIconAdapterRecoveryPending then
551        return true
552    end
553    local adapter = self._threeForOneIconAdapter
554    if not adapter then
555        self._threeForOneIconAdapterRecoveryPending = false
556        return true
557    end
558    local cleanupRequired = true
559    if type(adapter.needsDisposalRecovery) == "function" then
560        local cleanupOk, required = pcall(adapter.needsDisposalRecovery, adapter)
561        cleanupRequired = not cleanupOk or required == true
562    end
563    if not cleanupRequired and type(adapter.completeOwnerAdoption) == "function" then
564        local adoptedOk, adopted = pcall(adapter.completeOwnerAdoption, adapter)
565        if adoptedOk and adopted == true then
566            if self._threeForOneIconAdapter ~= adapter then
567                self._threeForOneIconAdapterRecoveryPending = false
568                return true
569            end
570            local mountedOk, mounted = pcall(adapter.isMounted, adapter)
571            if mountedOk and mounted == true then
572                self._threeForOneIconAdapterRecoveryPending = false
573                return true
574            end
575        end
576    end
577    local disposed = self:_disposeThreeForOneIconAdapter("recovery")
578    if disposed then
579        self.iconList[self.ICONKEY[self.GIFT_TYPE.THREEFORONE]] = nil
580    end
581    return disposed
582end

584function Manager:_adoptThreeForOneIconAdapter(createResult, giftType)
585    local adapter = createResult.adapter or createResult.cleanupOwner
586    if adapter then
587        self._threeForOneIconAdapter = adapter
588    end
589    if createResult.state ~= ADAPTER_CREATE_READY then
590        self._threeForOneIconDataProvider = nil
591        self._threeForOneIconAdapterRecoveryPending = true
592        log("three-for-one icon adapter owner pending", createResult.error)
593        return nil
594    end
595
596    local readyOk, ready = pcall(adapter.isReadyForOwnerAdoption, adapter)
597    local iconOk, icon = pcall(adapter.getIcon, adapter)
598    if not readyOk or ready ~= true or not iconOk or not icon or icon.isDisposed then
599        self._threeForOneIconDataProvider = nil
600        self._threeForOneIconAdapterRecoveryPending = true
601        log("three-for-one icon adapter not ready for owner adoption")
602        return nil
603    end
604
605    icon.giftType = giftType
606    self.iconList[icon.iconKey] = icon
607    self._threeForOneIconDataProvider = createResult.mode == "v2" and createResult.provider or nil
608    self._threeForOneIconAdapterRecoveryPending = false
609    if createResult.mode == "v2" then
610        if type(adapter.completeOwnerAdoption) ~= "function" then
611            self._threeForOneIconAdapterRecoveryPending = true
612            log("three-for-one icon adapter owner adoption required")
613            return nil
614        end
615        local adoptedOk, adopted, adoptError = pcall(adapter.completeOwnerAdoption, adapter)
616        if not adoptedOk or adopted ~= true then
617            self._threeForOneIconAdapterRecoveryPending = true
618            log("three-for-one icon adapter owner adoption failed", adoptedOk and adoptError or adopted)
619            return nil
620        end
621        if self._threeForOneIconAdapter ~= adapter then
622            return nil
623        end
624    end
625    self:dcQualify()
626    return icon
627end

629function Manager:refreshActivityIcon(giftType)
630    if giftType == self.GIFT_TYPE.THREEFORONE
631        and self._threeForOneIconAdapterRecoveryPending
632        and not self:_recoverThreeForOneIconAdapter() then
633        return false
634    end
635    local icon = self:getActivityIcon(giftType)
636    if not icon then return false end
637    if giftType == self.GIFT_TYPE.THREEFORONE and self._threeForOneIconAdapter then
638        local activeOk, active = pcall(
639            self._threeForOneIconAdapter.isV2Active,
640            self._threeForOneIconAdapter
641        )
642        if activeOk and active == true then
643            local ok, refreshed, refreshError = pcall(
644                self._threeForOneIconAdapter.refresh,
645                self._threeForOneIconAdapter,
646                "newbie_gift_update"
647            )
648            if not ok or refreshed == false then
649                log("refresh three-for-one icon adapter failed", ok and refreshError or refreshed)
650                return false
651            end
652            return true
653        end
654    end
655    if type(icon.updateIcon) == "function" then
656        icon:updateIcon()
657    end
658    return true
659end

899function Manager:checkNeedAutoPopout(giftType, sceneType)
900    if not self:checkTypeEnabled(giftType) then return false end
901    local config = self:getAutoPopoutConfig(giftType)
902    if not config or table.isEmpty(config) then
903        log("checkNeedAutoPopout config is empty", giftType)
904        return false
905    end
906
907    -- 强弹时间
908    local autoPopoutCd = config.autoPopoutCd
909    local lastPopTime = self:getLastPopoutTime(giftType)
910    if (TimeKit:timeInSec()-lastPopTime) < autoPopoutCd*3600 then
911        log("checkNeedAutoPopout time in cd", giftType, TimeKit:timeInSec(), lastPopTime, autoPopoutCd)
912        return false
913    end
914
915    -- 条件
916    local conditionList = config.conditionList
917    local AutoPopDefinition = Meta.AutoPopDefinition
918    local conditionMetaList = config.conditionMetaList
919    -- log("checkNeedAutoPopout condition check", giftType, tts(conditionList), sceneType)
920    for i, conditionId in ipairs(conditionList) do
921        local conditionMeta = conditionMetaList[conditionId] or {}
922        local sceneId = conditionMeta.sceneId
923        if sceneId == sceneType then
924            local value = conditionMeta.conditionValue or 0
925            local id = conditionMeta.conditionId
926            if AutoPopDefinition[id] then
927                local checkFunc = AutoPopDefinition[id]
928                local res = checkFunc(giftType, value)
929                if res ~= false then
930                    log("checkNeedAutoPopout condition check pass", giftType, res)
931                    return res
932                end
933            end
934        end
935    end
936    return false
937end

1106            return
