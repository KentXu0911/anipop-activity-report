35function ActivityCenterPanelVM:initData()
36    local datas = table.clone(self._activityCenterManager:getVisibleDatas())
37    local newActData
38    self.itemDatas = {}
39
40    for _, data in pairs(datas) do
41        data.redTag = nil
42        data.panel_id = self.panel_id
43        self.itemDatas[#self.itemDatas + 1] = data
44        local beginTime1 = data.getBeginTime()
45        local beginTime2 = newActData and newActData.getBeginTime() or nil
46        if not newActData or (beginTime1 > beginTime2 or (beginTime1 == beginTime2 and data.actId > newActData.actId)) then
47            newActData = data
48        end
49    end
50    if newActData then
51        newActData.yellowTag = false
52        newActData.redTag = true
53    end
54    table.sort(self.itemDatas, function(a, b)
55        local top_a = a.needTop()
56        local top_b = b.needTop()
57        if top_a ~= top_b then return top_a end
58        local beginTime_a = a.getBeginTime()
59        local beginTime_b = b.getBeginTime()
60        if beginTime_a ~= beginTime_b then return beginTime_a > beginTime_b end
61        return a.actId > b.actId
62    end)
63end

65-- 有活动结束，移除刷新
66function ActivityCenterPanelVM:refreshView()
67    if self.isDisposed then return end
68    local deletedDatas = {}
69    for _, v in ipairs(self.itemDatas) do
70        if not v:isSupport() then
71            table.insert(deletedDatas, v)
72        end
73    end
74
75    if #deletedDatas > 0 then
76        for _, v in ipairs(deletedDatas) do
77            local data, index = table.findItem(self.itemDatas, v)
78            self:callViewFunc("removeItemNode", data, index)
79            table.remove(self.itemDatas, index)
80        end
81        self:callViewFunc("refreshView")
82
83        if #self.itemDatas == 0 then
84            CommonTip:showTip(localize("activitycenter.end.desc"))
85            self:onCloseBtnTapped()
86            self._activityCenterManager:removeActivityCenterIcon()
87        else
88            CommonTip:showTip("资讯福利站内容已更新~")
89        end
90    end
91end

93-- ViewModel层：处理业务逻辑，决定显示什么类型的面板
94function ActivityCenterPanelVM:handleItemPanel(data)
95    if self.isDisposed then return end
96
97    if not data.panelCls then
98        self:callViewFunc("showLoadingPanel", data)
99        self._activityCenterManager:loadActivityRes(data.actId, function(data)
100            if self.isDisposed then return end
101            self:callViewFunc("showItemPanel", data)
102        end, function()
103            ALog.error("ActivityCenterPanelVM:handleItemPanel() - Failed to load activity resources for actId:",
104                data.actId)
105        end)
106    elseif data.panelCls then
107        self:callViewFunc("showItemPanel", data)
108    end
109end

138function ActivityCenterPanelVM:showActDc(positionY, height, itemHeight)
139    if self.isDisposed then return end
140    self.actDcList = self.actDcList or {}
141    local itemVerticalMargin = 5
142    for i, v in ipairs(self.itemDatas) do
143        if not self.actDcList[v.actId] then
144            local interval = (i - 1) * itemVerticalMargin
145            local interval2 = math.max(interval - itemVerticalMargin, 0)
146            local maxY = i * itemHeight + interval - 22
147            local minY = (i - 1) * itemHeight + interval2 - height + 6.5
148            if positionY >= minY and positionY <= maxY then
149                self.actDcList[v.actId] = true
150                ActivityCenterDC:dc("brilliant", "message_show", self._activityCenterManager:getActName(v), nil, nil,
151                    nil, { panel_id = self.panel_id })
152            end
153        end
154    end
155end
