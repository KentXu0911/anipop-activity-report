288function Meta:getName(goodsId)
289    for k, v in pairs(self.GoodsName) do
290        if v.goodsId == goodsId then
291            return v.showName
292        end
293    end
294end
295
296function Meta:getJinduHalfGift(payTag)
297    return table.clone(self.JinduHalfLibao[payTag])
298end
299
300function Meta:getRecommendGift(payTag)
301    return table.clone(self.GiftRecommend[payTag])
302end
303
304function Meta:checkIsLimit(goodsId)
305    return table.exist(self.LimitedGoods, goodsId)
306end
307
308function Meta:getPrice(goodsId)
309    for k, v in pairs(self.LimitedGoods) do
310        if v == goodsId then
311            local number = string.match(k, "(%d+)$")
312            return tonumber(number)
313        end
314    end
315
316    for k, v in pairs(self.UnlimitedGoods) do
317        if v == goodsId then
318            local number = string.match(k, "(%d+)$")
319            return tonumber(number)
320        end
321    end
322end
323
324return Meta
