506-- 返回今天已开启且尚未到截止时间的有效当前订单；毫秒 endTime 统一经 tonumber 归一化。
507function MergeModel:getActiveGoodGiftOrderInfo()
508	local orderInfo = self:getGoodGiftOrderInfo()
509	if type(orderInfo) ~= 'table' then
510		return nil
511	end
512	if not self:checkGoodGiftHasOpenedToday() then
513		return nil
514	end
515
516	local orderIndex = orderInfo.orderIndex
517	if type(orderIndex) ~= 'number'
518		or orderIndex ~= math.floor(orderIndex)
519		or orderIndex < 0
520		or orderIndex >= 5 then
521		return nil
522	end
523	local endTime = tonumber(orderInfo.endTime)
524	if not endTime or TimeKit:timeInMillis() >= endTime then
525		return nil
526	end
527
528	local orderDataList = orderInfo.orderDataList
529	local currentOrder = type(orderDataList) == 'table' and orderDataList[orderIndex + 1]
530	if type(currentOrder) ~= 'table'
531		or type(currentOrder.foodIds) ~= 'table' then
532		return nil
533	end
534	return orderInfo
535end
