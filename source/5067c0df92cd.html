394function PetModel:getWingFeatureTiming(currentTime)
395    local now = toSeconds(currentTime) or TimeKit:timeInSec()
396    local beginTimeValue = self:getDataByKey("wingBeginTs")
397    local endTimeValue = self:getDataByKey("wingEndTs")
398    local beginTime = toSeconds(beginTimeValue)
399    local endTime = toSeconds(endTimeValue)
400    local contextAvailable = self:_isWingContextAvailable()
401    local afterBegin = beginTime == nil or beginTime == 0 or now >= beginTime
402    local beforeEnd = endTime == 0 or (endTime ~= nil and now < endTime)
403    local featureEnabled = self:_isWingSwitchEnabled()
404        and contextAvailable
405        and afterBegin
406        and beforeEnd
407
408    return {
409        currentTs = now,
410        beginTs = beginTime,
411        endTs = endTime,
412        featureEnabled = featureEnabled,
413        countdownVisible = featureEnabled and endTime ~= nil and endTime > 0,
414        remainingSeconds = featureEnabled and endTime and endTime > 0 and math.max(0, endTime - now) or 0,
415    }

690function PetModel:isFoodNumEnough()
691    local levelMetas = self:getConfigByKey("levelMetas", {})
692    if type(levelMetas) ~= "table" or #levelMetas <= 0 then
693        return false
694    end
695
696    local targetNeedExp = 0
697    for _, cfg in ipairs(levelMetas) do
698        targetNeedExp = math.max(targetNeedExp, tonumber(cfg.needTotalExp) or 0)
699    end
700
701    if targetNeedExp <= 0 then
702        return false
703    end
704
705    local totalExp = self:getExp()
706    for _, foodMeta in ipairs(self:getFoodMetas()) do
707        local foodItemId = foodMeta.foodItemId or foodMeta.foodCollectionId
708        local foodNum = self:getFoodNum(foodItemId)
709        local feedExp = self:getFeedExp(foodItemId)
710        totalExp = totalExp + foodNum * feedExp
711    end
712
713    return totalExp >= targetNeedExp
714end

900function PetModel:updateWearingDecos(wearingDecos)
901    if type(wearingDecos) ~= "table" then return false end
902
903    local mainData = table.clone(self:getData(), true)
904    local enableDecos = table.clone(mainData.enableDecos or {}, true)
905    local existingPartTypes = {
906        PetConfig.DecoPartType.Image,
907        PetConfig.DecoPartType.Accessory,
908        PetConfig.DecoPartType.Eye,
909        PetConfig.DecoPartType.Wing,
910    }
911    for _, partType in ipairs(existingPartTypes) do
912        local decoInfo = self:_getDecoByPart(enableDecos, partType)
913        if type(decoInfo) == "table" then
914            decoInfo.itemId = 0
915        end
916    end
917
918    for _, wearingDeco in ipairs(wearingDecos) do
919        local partType = type(wearingDeco) == "table" and wearingDeco.partType
920        local decoId = type(wearingDeco) == "table" and wearingDeco.decoId
921        local decoInfo = self:_getDecoByPart(enableDecos, partType)
922        if type(decoInfo) ~= "table" then return false end
923        decoInfo.itemId = tonumber(decoId) or decoId or 0
924    end
925
926    mainData.enableDecos = enableDecos
927    mainData[PetConfig.DeprecatedDataKey.WEARING_DECOS] = nil
928    self:setValue("userInfo", mainData)
929    self:flush()
930    return true
931end
