157function ClimbTowerManager:addTicket(num)

159    if not self:isEnabled() then return end
160
161    local cur = self:getDataByKey("ticket") or 0
162    local total = cur + num
163    self:getModel():setDataByKey("ticket", total)
164
165    local actInfo = self.activityMgr:getActInfo(self:getConst().actId)
166    if actInfo and actInfo.extra then
167        local extraData = table.deserialize(actInfo.extra) or {}
168        extraData.ticket = total

174

176        addNum = num,
177        total  = total,
178    })

291            end

293            if self:getModel():isInRewardTime() and self:getTotalTickets() > 0 then
294                return (self:getLocalCache(TRIGGER_REWARD_KEY, false) or 0) < 1
295            end
296            return false
297        end,
298        creator    = function(_, onPopOver)
299            return function()
300                -- 正式期和领奖期各弹一次，本地缓存记录（needTime=false 跨日持久），key 拼入 periodId 按期隔离
301                local periodId           = tostring(self:getPeriodId() or "")
302                local TRIGGER_MAIN_KEY   = "climb_tower_trigger_main_" .. periodId
303                local TRIGGER_REWARD_KEY = "climb_tower_trigger_reward_" .. periodId
304                -- 本地记录本期弹出，防止重复弹

562---商店购买成功事件回调：后端礼包 rewardList **不包含**我们的奖券，前端按配置补发。
563---流程：拿 packageId → 查 packageTicketRefund 配置 → 创建奖券道具飞行动画 → 飞行完成后
564---调 self:addTicket(num)（addTicket 内部已写 actInfo + dispatch TICKET_ADDED，cocos 主面板
565---订阅 TICKET_ADDED 后会自动 pushTickets + pushDrawButtonState）。
566---飞行落点优先用 banner 的「去抽奖」按钮（StorePanel:resolveStoreHeaderFlyPos）；如果商店关闭
567---或 banner 不在，则跳过飞行直接加券（保证 server-side 状态一致）。
568function ClimbTowerManager:onStoreAfterBuy(event)
569    if not self:isEnabled() then return end
570    if not self:isInMainTime() then return end
571
572    local payload = event and event:getData()
573    if not payload then return end
574
575    local giftData   = payload.giftData
576    local storePanel = payload.storePanel
577    if not giftData or not giftData.getGoodsId then return end
578
579    local packageId = giftData:getGoodsId()
580    local num = self:getTicketRefundByPackageId(packageId) or 0
581    if num <= 0 then return end
582
583    self:playTicketRefundFly(num, storePanel, function()
584        _G.GainPropData:create()
585            :setFeature("happy_climb")
586            :setSource(tostring(packageId))
587            :setProps({ {itemId = self:getConst().TicketItemId, num = num} })
588            :setExtraMap({ pay_type = "3" })
589            :execute()
590    end)
591end

593---创建奖券道具的飞行动画，飞向 banner「去抽奖」按钮。
594---startWorldPos 取屏幕中心（与主面板入场动画一致）；targetWorldPos 走 StorePanel
595---:resolveStoreHeaderFlyPos(TicketItemId) —— banner 内 setFlyItemId 已注入 TicketItemId
596---只接收奖券飞行，其他 itemId 返回 nil。
597---FlyItemContext / TransactionEvent.FLY_ITEMS 自动挂到 scene.topLayer，跨 panel 也能飞。
598function ClimbTowerManager:playTicketRefundFly(num, storePanel, callback)
599    local TicketItemId = self:getConst().TicketItemId
600
601    local targetWorldPos
602    if storePanel and not storePanel.isDisposed and storePanel.resolveStoreHeaderFlyPos then
603        targetWorldPos = storePanel:resolveStoreHeaderFlyPos(TicketItemId)
604    end
605
606    if not targetWorldPos then
607        safeRunCall(callback)
608        return
609    end
610
611    local visibleArea = _G.fullScreenSizeManager:getVisibleArea()
612    local startWorldPos = ccp(
613        visibleArea.origin.x + visibleArea.size.width  * 0.5,
614        visibleArea.origin.y + visibleArea.size.height * 0.5
615    )
616
617    require "amf.middleware.ui.kit.animation.fly.transaction.FlyItemContext"
618    local flyContext = FlyItemContext.new({
619        {
620            itemId     = TicketItemId,
621            num        = num,
622            targetPos  = ccp(targetWorldPos.x, targetWorldPos.y),
623            duration   = 0.5,
624            scale      = 1.0,
625            shandowNum = 3,
626        }
627    }, ccp(startWorldPos.x, startWorldPos.y)):setFinishCallback(function()
628        self:playEffect(self:getConst().Sound.TICKET_COLLECT)
629        safeRunCall(callback)
630    end)
631    _G.amfTransaction(TransactionEvent.FLY_ITEMS, flyContext)
632end
