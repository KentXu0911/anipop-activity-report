44local function callAdapterMethod(adapter, methodName, ...)
45    local method = adapter and adapter[methodName]
46    if type(method) ~= "function" then
47        return false, nil, "PHONE_BIND_MINI_ICON_ADAPTER_METHOD_REQUIRED:" .. tostring(methodName)
48    end
49    local ok, result, detail = pcall(method, adapter, ...)
50    if not ok then
51        return false, nil, result
52    end
53    return true, result, detail
54end

56local function resolveAdapterState(adapter)
57    if not adapter then
58        return ADAPTER_STATE_NONE
59    end
60    local mountedOk, mounted, mountedError = callAdapterMethod(adapter, "isMounted")
61    if not mountedOk then
62        return ADAPTER_STATE_BLOCKED, mountedError
63    end
64    if mounted ~= true then
65        return ADAPTER_STATE_CLEANUP
66    end
67    local activeOk, active, activeError = callAdapterMethod(adapter, "isV2Active")
68    if not activeOk then
69        return ADAPTER_STATE_BLOCKED, activeError
70    end
71    return active == true and ADAPTER_STATE_V2 or ADAPTER_STATE_LEGACY
72end

74local function getAdapterIcon(adapter)
75    local called, icon, callError = callAdapterMethod(adapter, "getIcon")
76    if not called then
77        return nil, callError
78    end
79    return icon
80end

82local function disposeAdapter(adapter)
83    local called, disposed, disposeError = callAdapterMethod(adapter, "dispose")
84    if not called then
85        return false, disposeError
86    end
87    if disposed ~= true then
88        return false, disposeError or "PHONE_BIND_MINI_ICON_ADAPTER_DISPOSE_FAILED"
89    end
90    return true
91end

93local function refreshAdapter(adapter, reason, payload)
94    local called, refreshed, refreshError = callAdapterMethod(adapter, "refresh", reason, payload)
95    if not called then
96        return false, refreshError
97    end
98    if refreshed == false then
99        return false, refreshError
100    end
101    return refreshed, refreshError
102end

288function MiniAccountBindingManager:createPackageIconAdapter()
289    local adapterLoaded, PackageIconV2Adapter = pcall(
290        require,
291        "amf.feature.mainscene.module.icon.v2.runtime.PackageIconV2Adapter"
292    )
293    if not adapterLoaded then
294        return {
295            state = ADAPTER_CREATE_SAFE_LEGACY,
296            error = PackageIconV2Adapter,
297        }
298    end
299    if type(PackageIconV2Adapter.create) ~= "function" then
300        return {
301            state = ADAPTER_CREATE_SAFE_LEGACY,
302            error = "PHONE_BIND_MINI_ICON_V2_ADAPTER_CREATE_REQUIRED",
303        }
304    end
305
306    local ownerIconKey = IconKey.kPhoneBind
307    local configOk, options, provider = pcall(function()
308        local IconV2Config = require "amf.feature.accountbind.minibind.icon.IconV2Config"
309        local configOptions, configProvider = IconV2Config.create(self, IconEventCenter)
310        assert(type(configOptions) == "table", configProvider or "PHONE_BIND_MINI_ICON_V2_CONFIG_REQUIRED")
311        return configOptions, configProvider
312    end)
313    if not configOk then
314        local configError = options
315        options = {
316            ownerIconKey = ownerIconKey,
317            iconEventCenter = IconEventCenter,
318            legacyFactory = function()
319                return self:createLegacyIcon()
320            end,
321            v2PreflightError = configError,
322        }
323        provider = nil
324    end
325
326    local createOk, adapter, modeOrError, cleanupOwner = pcall(
327        PackageIconV2Adapter.create,
328        PackageIconV2Adapter,
329        options
330    )
331    if not createOk then
332        return {
333            state = ADAPTER_CREATE_WAIT_OWNER,
334            error = adapter,
335        }
336    end
337    if not adapter then
338        return {
339            state = ADAPTER_CREATE_WAIT_OWNER,
340            error = modeOrError,
341            cleanupOwner = cleanupOwner,
342        }
343    end
344    return {
345        state = ADAPTER_CREATE_READY,
346        adapter = adapter,
347        provider = provider,
348        mode = modeOrError,
349    }
350end

352function MiniAccountBindingManager:_adoptIconAdapter(adapter, provider, recoveryPending)
353    self.iconAdapter = adapter
354    local adapterState, stateError = resolveAdapterState(adapter)
355    local icon, iconError = getAdapterIcon(adapter)
356    self.icon = icon
357    self.iconDataProvider = adapterState == ADAPTER_STATE_V2 and provider or nil
358    self._iconAdapterRecoveryPending = recoveryPending == true
359        or adapterState == ADAPTER_STATE_CLEANUP
360        or adapterState == ADAPTER_STATE_BLOCKED
361        or iconError ~= nil
362    if stateError then
363        logIconV2Error("adapter state failed: " .. tostring(stateError))
364    end
365    if iconError then
366        logIconV2Error("adapter icon failed: " .. tostring(iconError))
367    end
368    if adapterState == ADAPTER_STATE_V2 then
369        self:_registerV2LabelRefresh()
370    else
371        self:_unregisterV2LabelRefresh()
372    end
373end

375function MiniAccountBindingManager:_disposeIconAdapter()
376    local adapter = self.iconAdapter
377    if not adapter then
378        return true
379    end
380    local disposed, disposeError = disposeAdapter(adapter)
381    if not disposed then
382        local icon, iconError = getAdapterIcon(adapter)
383        if not iconError then
384            self.icon = icon
385        end
386        self._iconAdapterRecoveryPending = true
387        logIconV2Error("dispose failed: " .. tostring(disposeError or iconError))
388        return false, disposeError or iconError
389    end
390    self.iconAdapter = nil
391    self.iconDataProvider = nil
392    self.icon = nil
393    self._iconAdapterRecoveryPending = false
394    self:_unregisterV2LabelRefresh()
395    return true
396end

405function MiniAccountBindingManager:addIconIfNotExist()
406    local recovered = self:_recoverPendingIconAdapter()
407    if recovered == false then
408        return self.icon
409    end
410    if self.iconAdapter then
411        local adapterState, stateError = resolveAdapterState(self.iconAdapter)
412        if adapterState == ADAPTER_STATE_V2 or adapterState == ADAPTER_STATE_LEGACY then
413            local icon, iconError = getAdapterIcon(self.iconAdapter)
414            if iconError then
415                self._iconAdapterRecoveryPending = true
416                logIconV2Error("mounted adapter icon failed: " .. tostring(iconError))
417                return self.icon
418            end
419            self.icon = icon
420            self._iconAdapterRecoveryPending = false
421            return self.icon
422        end
423        if stateError then
424            logIconV2Error("adapter state failed before recovery: " .. tostring(stateError))
425        end
426        local disposed = self:_disposeIconAdapter()
427        if disposed == false then
428            return self.icon
429        end
430    end
431    if isIconAlive(self.icon) then
432        return self.icon
433    end
434
435    if self:isV2IconExperimentEnabled() or self._iconAdapterRecoveryPending then
436        if not self._iconV2AssetReady and EnvUtil:isUnity() then
437            return self:_loadV2IconAsset()
438        end
439        local createResult = self:createPackageIconAdapter()
440        local ownerAdapter = createResult.adapter or createResult.cleanupOwner
441        if ownerAdapter then
442            self:_adoptIconAdapter(
443                ownerAdapter,
444                createResult.provider,
445                createResult.state == ADAPTER_CREATE_WAIT_OWNER
446            )
447            if not createResult.adapter then
448                logIconV2Error("create pending cleanup: " .. tostring(createResult.error))
449            end
450            return self.icon
451        end
452        if createResult.state ~= ADAPTER_CREATE_SAFE_LEGACY then
453            self._iconAdapterRecoveryPending = true
454            logIconV2Error("create waiting for owner recovery: " .. tostring(createResult.error))
455            return nil
456        end
457        self._iconAdapterRecoveryPending = false
458        logIconV2Error("adapter unavailable, fallback legacy: " .. tostring(createResult.error))
459    end
460
461    self._iconAdapterRecoveryPending = false
462    self.icon = self:createLegacyIcon()
463    self:_unregisterV2LabelRefresh()
464    IconEventCenter:addIconEvent(self.icon)
465    return self.icon
466end

535---场景4强弹 ,目前只有支付宝抖音有
536function MiniAccountBindingManager:onActEnterStage4(evt)
537
538    local popupFlag = EnvUtil:isHuaweiMiniGame() or EnvUtil:isALIPAYMiniGame() or EnvUtil:isUnionOppoMiniGame() or EnvUtil:isUnionMiMiniGame()  or EnvUtil:isKuaiShouMiniGame() or EnvUtil:isJdMiniGame() or EnvUtil:isDYMiniGame()
539    if not popupFlag then
540        return
541    end
542
543	-- 实验组的0～30关用户已改由场景2每日触发，场景4必须跳过，避免同一次登录出现两个手机号绑定强弹。
544	-- 31～1998关仍沿用原场景4和 phone_bind_popup 终身一次标记；OFF组也保持原行为。
545	if self._snsBindLogic:isDyPhoneBindExperimentEnabled() and _G.getTopPassedLevelId() <= 30 then
546		return
547	end
548
549    if not self._snsBindLogic:canShowPhoneBindButton() then
550        return
551    end
552    local isAlreadyPop = UserManager:getInstance():getClientCustomData("phone_bind_popup") ---永久弹一次，是否弹过的标识
553    if isAlreadyPop ~= nil then
554        return
555    end
556    local function onHandleOver()
557        --UWarning("MiniAccountBindingManager onActEnterStage4 onHandleOver")
558        AutoPopEventCenter:dispatch(AutoPopEvent.SINGLE_POP_OVER, PopupCheckPoint.PHONE_BIND)
559    end
560    local function creator()
561        --UWarning("MiniAccountBindingManager onActEnterStage4 creator")
562        self._snsBindLogic:onPhoneBindButtonTapped(true,onHandleOver)
563        UserManager:getInstance():setClientCustomDataHttp("phone_bind_popup",true)
564    end
565    _G.amfBean("AutoPopupMgr"):addPopupConfig(PopupCheckPoint.PHONE_BIND, creator)
566end

568function MiniAccountBindingManager:canGetPhoneNumber()
569    if EnvUtil:isALIPAYMiniGame() then
570        if UserManager:getClientCustomData("phone_bind_aliApiQueryBanned") then
571            return false
572        end
573        local todayDay = getDayIndexByTs(TimeKit:timeInSec())
574        local num = UserManager:getClientCustomData("phone_bind_aliApiQueryToday")
575        if num and todayDay == tonumber(num) then
576            return false
577        end
578        return true
579    ---微信的快速验证只有ios支持，微信那客服那头反馈的,如果将来支持了安卓PhoneBindLogic:addTouchListener()注册和去掉的ios判断也删掉
580    elseif EnvUtil:isWXMiniGame() and EnvUtil:isMiniGameIOS() then
581        local canGetPhoneNumber = CS.CallJS.InvokeSync("globalThis.animalPlugin.sdkManager.isInterfaceAvailable","getPhoneNumber")
582        return table.deserialize(canGetPhoneNumber)
583    end
584
585    return false
