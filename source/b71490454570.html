164function IconModule:createGiftPackageIconAdapter()
165    local adapterLoaded, PackageIconV2Adapter = pcall(
166        require,
167        "amf.feature.mainscene.module.icon.v2.runtime.PackageIconV2Adapter"
168    )
169    if not adapterLoaded or type(PackageIconV2Adapter.create) ~= "function" then
170        return {
171            state = ADAPTER_CREATE_SAFE_LEGACY,
172            error = adapterLoaded and "COLLECT_CARD_GIFT_ICON_V2_ADAPTER_CREATE_REQUIRED"
173                or PackageIconV2Adapter,
174        }
175    end
176
177    local configOk, options, provider = pcall(function()
178        local IconV2Config = require "amf.feature.localactivity.collectcard.icon.CollectCardGiftIconV2Config"
179        local configOptions, configProvider = IconV2Config.create(self.manager, IconEventCenter)
180        assert(type(configOptions) == "table", configProvider or "COLLECT_CARD_GIFT_ICON_V2_CONFIG_REQUIRED")
181        return configOptions, configProvider
182    end)
183    if not configOk then
184        local configError = options
185        options = {
186            ownerIconKey = IconKey.kCollectCardGiftIcon,
187            ownerIdentity = self,
188            iconEventCenter = IconEventCenter,
189            legacyFactory = function()
190                return self:createLegacyGiftIcon()
191            end,
192            v2PreflightError = configError,
193        }
194        provider = nil
195    end
196
197    local createOk, adapter, modeOrError, cleanupOwner = pcall(
198        PackageIconV2Adapter.create,
199        PackageIconV2Adapter,
200        options
201    )
202    if not createOk then
203        return {
204            state = ADAPTER_CREATE_WAIT_OWNER,
205            error = adapter,
206        }
207    end
208    if not adapter then
209        return {
210            state = ADAPTER_CREATE_WAIT_OWNER,
211            error = modeOrError,
212            cleanupOwner = cleanupOwner,
213        }
214    end
215    return {
216        state = ADAPTER_CREATE_READY,
217        adapter = adapter,
218        provider = provider,
219        mode = modeOrError,
220    }
221end

223function IconModule:_adoptGiftIconAdapter(createResult)
224    local adapter = createResult.adapter or createResult.cleanupOwner
225    if adapter then
226        self.giftIconAdapter = adapter
227    end
228    if createResult.state ~= ADAPTER_CREATE_READY then
229        self.giftIconDataProvider = nil
230        self._giftIconAdapterRecoveryPending = true
231        logIconV2Error("adapter owner pending: " .. tostring(createResult.error))
232        return nil
233    end
234
235    local readyOk, ready = pcall(adapter.isReadyForOwnerAdoption, adapter)
236    local iconOk, icon = pcall(adapter.getIcon, adapter)
237    if not readyOk or ready ~= true or not iconOk or not icon or icon.isDisposed then
238        self.giftIconDataProvider = nil
239        self._giftIconAdapterRecoveryPending = true
240        logIconV2Error("adapter owner not ready for adoption")
241        return nil
242    end
243
244    self.giftIcon = icon
245    self.giftIconDataProvider = createResult.mode == "v2" and createResult.provider or nil
246    self._giftIconAdapterRecoveryPending = false
247    if createResult.mode == "v2" then
248        self:_registerGiftIconEvent()
249    else
250        self:_unregisterGiftIconEvent()
251    end
252    if createResult.mode == "v2" then
253        if type(adapter.completeOwnerAdoption) ~= "function" then
254            self._giftIconAdapterRecoveryPending = true
255            logIconV2Error("adapter owner adoption required")
256            return nil
257        end
258        local adoptedOk, adopted, adoptError = pcall(adapter.completeOwnerAdoption, adapter)
259        if not adoptedOk or adopted ~= true then
260            self._giftIconAdapterRecoveryPending = true
261            logIconV2Error(
262                "adapter owner adoption failed: " .. tostring(adoptedOk and adoptError or adopted)
263            )
264            return nil
265        end
266        if self.giftIconAdapter ~= adapter then
267            return nil
268        end
269    end
270    return icon
271end

273function IconModule:_disposeGiftIconAdapter(reason)
274    local adapter = self.giftIconAdapter
275    if not adapter then
276        self._giftIconAdapterRecoveryPending = false
277        self:_unregisterGiftIconEvent()
278        return true
279    end
280    local ok, disposed, disposeError = pcall(adapter.dispose, adapter)
281    if not ok or disposed ~= true then
282        self._giftIconAdapterRecoveryPending = true
283        logIconV2Error(
284            "dispose failed reason=" .. tostring(reason)
285                .. " error=" .. tostring(ok and disposeError or disposed)
286        )
287        return false, ok and disposeError or disposed
288    end
289    self.giftIconAdapter = nil
290    self.giftIconDataProvider = nil
291    self._giftIconAdapterRecoveryPending = false
292    self:_unregisterGiftIconEvent()
293    return true
294end

296function IconModule:_recoverGiftIconAdapter()
297    if not self._giftIconAdapterRecoveryPending then
298        return true
299    end
300    local adapter = self.giftIconAdapter
301    if not adapter then
302        self._giftIconAdapterRecoveryPending = false
303        return true
304    end
305    local cleanupRequired = true
306    if type(adapter.needsDisposalRecovery) == "function" then
307        local cleanupOk, required = pcall(adapter.needsDisposalRecovery, adapter)
308        cleanupRequired = not cleanupOk or required == true
309    end
310    if not cleanupRequired and type(adapter.completeOwnerAdoption) == "function" then
311        local adoptedOk, adopted = pcall(adapter.completeOwnerAdoption, adapter)
312        if adoptedOk and adopted == true then
313            if self.giftIconAdapter ~= adapter then
314                self._giftIconAdapterRecoveryPending = false
315                return true
316            end
317            local mountedOk, mounted = pcall(adapter.isMounted, adapter)
318            if mountedOk and mounted == true then
319                self._giftIconAdapterRecoveryPending = false
320                return true
321            end
322        end
323    end
324    local disposed = self:_disposeGiftIconAdapter("recovery")
325    if disposed then
326        self.giftIcon = nil
327    end
328    return disposed
329end

331function IconModule:refreshGiftIconAdapter(reason)

334    local activeOk, active = pcall(adapter.isV2Active, adapter)
335    if not activeOk or active ~= true then
336        return false

341        return false
342    end
343    return true

364function IconModule:updateGiftIcon()
365    printx(33, "IconModule:updateGiftIcon")
366    if self:needShowGiftIcon() then
367        if not self._iconSystemReady then return end
368        if self._giftIconAdapterRecoveryPending and not self:_recoverGiftIconAdapter() then
369            return
370        end
371        if self.giftIconAdapter then
372            local activeOk, active = pcall(self.giftIconAdapter.isV2Active, self.giftIconAdapter)
373            if activeOk and active == true then
374                self:refreshGiftIconAdapter("collect_card_gift_update")
375            else
376                IconEventCenter:refreshIconEvent(IconKey.kCollectCardGiftIcon)
377            end
378            return
379        end
380        self.giftIcon = _G.amfBean("IconManager"):getIconByIconKey(IconKey.kCollectCardGiftIcon)
381        if self.giftIcon then
382            IconEventCenter:refreshIconEvent(IconKey.kCollectCardGiftIcon)
383        else
384            self:createGiftIcon()
385        end
386    else
387        self:clearGiftIcon()
388    end
389end

391function IconModule:createGiftIcon()
392    if self._isCreatingGiftIcon then
393        return
394    end
395    self._isCreatingGiftIcon = true
396    self._giftStrongIdle = true
397    self._giftIconCreateGeneration = self._giftIconCreateGeneration + 1
398    local generation = self._giftIconCreateGeneration
399    local legacyIconPath = "amf/activity/collectcard/icon/jika_icon.skel"
400    local iconRes = { legacyIconPath }
401    if self:isV2IconExperimentEnabled() then
402        local IconV2Config = require
403            "amf.feature.localactivity.collectcard.icon.CollectCardGiftIconV2Config"
404        iconRes = IconV2Config.getPreloadFiles(legacyIconPath)
405    end
406    AssetsUtil:loadMiniFiles(iconRes):next(function()
407        if generation ~= self._giftIconCreateGeneration then return end
408        if not self:needShowGiftIcon() then
409            self._isCreatingGiftIcon = false
410            return
411        end
412        if self:isV2IconExperimentEnabled() then
413            local createResult = self:createGiftPackageIconAdapter()
414            if createResult.state == ADAPTER_CREATE_SAFE_LEGACY then
415                IconEventCenter:addIconIfNotExist(IconKey.kCollectCardGiftIcon, function()
416                    self.giftIcon = self:createLegacyGiftIcon()
417                    return self.giftIcon
418                end)
419                self.giftIcon = _G.amfBean("IconManager"):getIconByIconKey(IconKey.kCollectCardGiftIcon)
420            else
421                self:_adoptGiftIconAdapter(createResult)
422            end
423        else
424            IconEventCenter:addIconIfNotExist(IconKey.kCollectCardGiftIcon, function()
425                self.giftIcon = self:createLegacyGiftIcon()
426                return self.giftIcon
427            end)
428            self.giftIcon = _G.amfBean("IconManager"):getIconByIconKey(IconKey.kCollectCardGiftIcon)
429        end
430        self._isCreatingGiftIcon = false
431    end, function(loadError)
432        if generation ~= self._giftIconCreateGeneration then return end
433        logIconV2Error("load asset failed: " .. tostring(loadError))
434        self._isCreatingGiftIcon = false
435    end)
436end
