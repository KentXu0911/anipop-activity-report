301function LuckyDeliveryPassManager:dcRewardTimeReward(expressOpenResults)
302    local records = {}
303    local rewardDays = {}
304
305    for _, openResult in ipairs(expressOpenResults or {}) do
306        local rewardType = openResult.vip and 2 or 1
307
308        local finalLevel = openResult.finalLevel 
309        if openResult.splitNum > 1 then
310            finalLevel = finalLevel .. "_" .. openResult.splitNum
311        end
312
313        local record = table.concat({
314            tostring(openResult.initLevel),
315            tostring(finalLevel),
316            tostring(rewardType),
317        }, "~")
318
319        table.insert(records, record)
320        table.insert(rewardDays, self:_formatRewardTimeRewardDay(openResult.day))
321    end
322
323    self:dcPlug():dcRewardTimeReward(table.concat(records, ";"), table.concat(rewardDays, ";"))
324end

353        moduleName = PASS_GIFT_MODULE_NAME,

386function LuckyDeliveryPassManager:receiveCompensateReward(onComplete, onFail)
387    if self._isReceivingCompensateReward then return end

391    end

395        self._isReceivingCompensateReward = false
396        self:refreshMainSceneButton()

399        self:dcRewardTimeReward(data.expressOpenResults)
400        self:playExpressOpenResults(data.expressOpenResults, function()
401            safeRunCall(onComplete)
402        end, "rewardtime_reward")
403    end, function(...)
404        self._isReceivingCompensateReward = false
405        safeRunCall(onFail, ...)
406    end)
