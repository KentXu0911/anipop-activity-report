36local function callAdapterMethod(adapter, methodName, ...)
37    local method = adapter and adapter[methodName]
38    if type(method) ~= "function" then
39        return false, nil, "PHONE_BIND_APP_ICON_ADAPTER_METHOD_REQUIRED:" .. tostring(methodName)
40    end
41    local ok, result, detail = pcall(method, adapter, ...)
42    if not ok then
43        return false, nil, result
44    end
45    return true, result, detail
46end

48local function resolveAdapterState(adapter)
49    if not adapter then
50        return ADAPTER_STATE_NONE
51    end
52    local mountedOk, mounted, mountedError = callAdapterMethod(adapter, "isMounted")
53    if not mountedOk then
54        return ADAPTER_STATE_BLOCKED, mountedError
55    end
56    if mounted ~= true then
57        return ADAPTER_STATE_CLEANUP
58    end
59    local activeOk, active, activeError = callAdapterMethod(adapter, "isV2Active")
60    if not activeOk then
61        return ADAPTER_STATE_BLOCKED, activeError
62    end
63    return active == true and ADAPTER_STATE_V2 or ADAPTER_STATE_LEGACY
64end

66local function getAdapterIcon(adapter)
67    local called, icon, callError = callAdapterMethod(adapter, "getIcon")

74local function disposeAdapter(adapter)
75    local called, disposed, disposeError = callAdapterMethod(adapter, "dispose")
76    if not called then
77        return false, disposeError
78    end
79    if disposed ~= true then
80        return false, disposeError or "PHONE_BIND_APP_ICON_ADAPTER_DISPOSE_FAILED"
81    end
82    return true
83end
84
85local function refreshAdapter(adapter, reason, payload)
86    local called, refreshed, refreshError = callAdapterMethod(adapter, "refresh", reason, payload)
87    if not called then
88        return false, refreshError
89    end
90    if refreshed == false then
91        return false, refreshError
92    end
93    return refreshed, refreshError
94end

155function PhoneBindActManager:tryUpdateIcon()
156    if not self._phoneBindManager or
157            not self._phoneBindManager:checkPhoneBindEnabled() then
158        self:removeIconIfExist()
159        return
160    end
161    if self._iconAdapterRecoveryPending then
162        return self:addIconIfNotExist()
163    end
164    if self.iconAdapter then
165        local adapterState = resolveAdapterState(self.iconAdapter)
166        if adapterState == ADAPTER_STATE_V2 then
167            local refreshed, refreshError = refreshAdapter(self.iconAdapter, "eligibility_update")
168            if refreshed == false then
169                logIconV2Error("refresh failed: " .. tostring(refreshError))
170            end
171            return refreshed, refreshError
172        end
173        if adapterState == ADAPTER_STATE_LEGACY then
174            return self.icon
175        end
176        return self:addIconIfNotExist()
177    end
178    return self:addIconIfNotExist()
179end

194function PhoneBindActManager:createPackageIconAdapter()
195    local adapterLoaded, PackageIconV2Adapter = pcall(
196        require,
197        "amf.feature.mainscene.module.icon.v2.runtime.PackageIconV2Adapter"
198    )
199    if not adapterLoaded then
200        return {
201            state = ADAPTER_CREATE_SAFE_LEGACY,
202            error = PackageIconV2Adapter,
203        }
204    end
205    if type(PackageIconV2Adapter.create) ~= "function" then
206        return {
207            state = ADAPTER_CREATE_SAFE_LEGACY,
208            error = "PHONE_BIND_APP_ICON_V2_ADAPTER_CREATE_REQUIRED",
209        }
210    end
211
212    local ownerIconKey = IconKey.kPhoneBind
213    local configOk, options, provider = pcall(function()
214        local IconV2Config = require "amf.feature.accountbind.phonebind.module.phonebindact.icon.IconV2Config"
215        local configOptions, configProvider = IconV2Config.create(self, IconEventCenter)
216        assert(type(configOptions) == "table", configProvider or "PHONE_BIND_APP_ICON_V2_CONFIG_REQUIRED")
217        return configOptions, configProvider
218    end)
219    if not configOk then
220        local configError = options
221        options = {
222            ownerIconKey = ownerIconKey,
223            iconEventCenter = IconEventCenter,
224            legacyFactory = function()
225                return self:createLegacyIcon()
226            end,
227            v2PreflightError = configError,
228        }
229        provider = nil
230    end
231
232    local createOk, adapter, modeOrError, cleanupOwner = pcall(
233        PackageIconV2Adapter.create,
234        PackageIconV2Adapter,
235        options
236    )
237    if not createOk then
238        return {
239            state = ADAPTER_CREATE_WAIT_OWNER,
240            error = adapter,
241        }
242    end
243    if not adapter then
244        return {
245            state = ADAPTER_CREATE_WAIT_OWNER,
246            error = modeOrError,
247            cleanupOwner = cleanupOwner,
248        }
249    end
250    return {
251        state = ADAPTER_CREATE_READY,
252        adapter = adapter,
253        provider = provider,
254        mode = modeOrError,
255    }
256end

258function PhoneBindActManager:_adoptIconAdapter(adapter, provider, recoveryPending)
259    self.iconAdapter = adapter
260    local adapterState, stateError = resolveAdapterState(adapter)
261    local icon, iconError = getAdapterIcon(adapter)
262    self.icon = icon
263    self.iconDataProvider = adapterState == ADAPTER_STATE_V2 and provider or nil
264    self._iconAdapterRecoveryPending = recoveryPending == true
265        or adapterState == ADAPTER_STATE_CLEANUP
266        or adapterState == ADAPTER_STATE_BLOCKED
267        or iconError ~= nil
268    if stateError then
269        logIconV2Error("adapter state failed: " .. tostring(stateError))
270    end
271    if iconError then
272        logIconV2Error("adapter icon failed: " .. tostring(iconError))
273    end
274end

276function PhoneBindActManager:_disposeIconAdapter()
277    local adapter = self.iconAdapter
278    if not adapter then
279        return true
280    end
281    local disposed, disposeError = disposeAdapter(adapter)
282    if not disposed then
283        local icon, iconError = getAdapterIcon(adapter)
284        if not iconError then
285            self.icon = icon
286        end
287        self._iconAdapterRecoveryPending = true
288        logIconV2Error("dispose failed: " .. tostring(disposeError or iconError))
289        return false, disposeError or iconError
290    end
291    self.iconAdapter = nil
292    self.iconDataProvider = nil
293    self.icon = nil
294    self._iconAdapterRecoveryPending = false
295    return true
296end

305function PhoneBindActManager:addIconIfNotExist()
306    local recovered = self:_recoverPendingIconAdapter()
307    if recovered == false then
308        return self.icon
309    end
310    if self.iconAdapter then
311        local adapterState, stateError = resolveAdapterState(self.iconAdapter)
312        if adapterState == ADAPTER_STATE_V2 or adapterState == ADAPTER_STATE_LEGACY then
313            local icon, iconError = getAdapterIcon(self.iconAdapter)
314            if iconError then
315                self._iconAdapterRecoveryPending = true
316                logIconV2Error("mounted adapter icon failed: " .. tostring(iconError))
317                return self.icon
318            end
319            self.icon = icon
320            self._iconAdapterRecoveryPending = false
321            return self.icon
322        end
323        if stateError then
324            logIconV2Error("adapter state failed before recovery: " .. tostring(stateError))
325        end
326        local disposed = self:_disposeIconAdapter()
327        if disposed == false then
328            return self.icon
329        end
330    end
331    if isIconAlive(self.icon) then return self.icon end
332
333    if self:isV2IconExperimentEnabled() or self._iconAdapterRecoveryPending then
334        if not self._iconV2AssetReady and EnvUtil:isUnity() then
335            return self:_loadV2IconAsset()
336        end
337        local createResult = self:createPackageIconAdapter()
338        local ownerAdapter = createResult.adapter or createResult.cleanupOwner
339        if ownerAdapter then
340            self:_adoptIconAdapter(
341                ownerAdapter,
342                createResult.provider,
343                createResult.state == ADAPTER_CREATE_WAIT_OWNER
344            )
345            if not createResult.adapter then
346                logIconV2Error("create pending cleanup: " .. tostring(createResult.error))
347            end
348            return self.icon
349        end
350        if createResult.state ~= ADAPTER_CREATE_SAFE_LEGACY then
351            self._iconAdapterRecoveryPending = true
352            logIconV2Error("create waiting for owner recovery: " .. tostring(createResult.error))
353            return nil
354        end
355        self._iconAdapterRecoveryPending = false
356        logIconV2Error("adapter unavailable, fallback legacy: " .. tostring(createResult.error))
357    end
358
359    self._iconAdapterRecoveryPending = false
360    self.icon = self:createLegacyIcon()
361    IconEventCenter:addIconEvent(self.icon)
362    return self.icon
363end

393function PhoneBindActManager:popBindPhonePanel(delegate)
394    local onBindSuccess = delegate.onBindSuccess
395    local onBindFail = delegate.onBindFail
396    local closeCallback = delegate.closeCallback
397    local source = delegate.source
398    local ignorePopTime = delegate.ignorePopTime
399    if ignorePopTime == nil then
400        ignorePopTime = false
401    end
402    if not self._phoneBindManager:popTimeUpdate() and not ignorePopTime then
403        safeRunCall(closeCallback)
404        return false
405    end
406    local tbl = self._phoneBindManager:getRewardsChoice()
407    if table.isEmpty(tbl) or table.isEmpty(tbl[1].rewards) or table.isEmpty(tbl[1].rewards.list) then
408        safeRunCall(closeCallback)
409        return false
410    end
411    local reward = tbl[1].rewards.list[1]
412    if self._phoneBindManager:hasGotBindPhoneReward() then
413        reward = {}
414    end
415    self._phoneBindManager:showPhoneBindPanel(reward, function()
416        self:removeIconIfExist()
417        safeRunCall(onBindSuccess)
418    end, onBindFail, closeCallback)
419    PhoneBindDC:dc("phone2021", "phone2021_pop", source, 3)
420    return true
421end

447function PhoneBindActManager:showExitGameBindPanel(event)
448    local choice = self._phoneBindManager:getRewardsChoice()
449    if self._phoneBindManager:checkPhoneBindEnabled() and
450            not table.isEmpty(choice) and
451            self._phoneBindManager:popTimeUpdate() then
452        local reward = choice[1].rewards.list[1]
453        local context = event:getData() or {}
454        local onExit = context.onExit
455        local onCancelExit = context.onCancelExit
456        local vm = require "amf.feature.accountbind.phonebind.module.phonebindact.view.ExitGameBindPanelVM"
457                :create(reward, onExit)
458        vm:setCloseCallback(function()
459            if onCancelExit then onCancelExit() end
460        end)
461        vm:popout()
462    else
463        event:redeliver()
464    end
