74-- 活动node是否可以启动，需要在启动器下完之后才知道，即loadLauncher之后才知道是否可以
75function ActivityNode:isEnabled()
76    if not self._iconSystemReady then
77        ActivityDebugLog:logWarning("iconSystemNotReady，暂不启动后续流程")
78        return false
79    end
80    
81    -- 活动启动器未加载，不能启动
82    if not self._launcherLoadSuccess then
83        ActivityDebugLog:logWarning(self._actMeta:getActSource() .. "启动器加载失败，isEnabled判定不过")
84        return false
85    end
86
87    local config
88    pcall(function()
89        config = require("activity/" .. self._actMeta:getActSource())
90    end)
91
92    if not config then
93        ActivityDebugLog:logWarning(self._actMeta:getActSource() .. "没有加载上config，无法启动")
94        return false
95    end
96
97    return true
98end

100function ActivityNode:isConfigAndActInfoSupport()
101    local config = require("activity/" .. self._actMeta:getActSource())
102    local result
103    if type(config) == "table" and config.isSupport then
104        pcall(function(...)
105            result = config.isSupport()
106        end)
107    end
108
109    self._configAndActInfoSupported = result == true
110    if not self._configAndActInfoSupported then
111        return false
112    end
113
114    if self._actInfo and not self._actInfo.see then
115        self._configAndActInfoSupported = false
116        return false
117    end
118
119    return true
120end

281-- 流程 启动器下载后判断活动是否可以开启，开启的话去创建icon并发liteStart，但loadLauncher和后面两步本身可以剥离，无关
282function ActivityNode:processing()
283    self:setProcessing()
284    local configVersion = self._configVersion
285    if not EnvUtil:isWindows() and self._launcherLoadSuccess then
286        self:afterLauncher(configVersion, function()
287            self:finishProcessing()
288        end)
289        return
290    end
291    self:loadLauncher():next(function()
292        self._launcherLoadSuccess = true
293        self:afterLauncher(configVersion, function()
294            self:finishProcessing()
295        end)
296    end,function()
297        ActivityDebugLog:logWarning("启动器加载失败" .. self._actMeta:getActId() .. "活动无法启动")
298        self:processOver(false, configVersion)
299        self:finishProcessing()
300    end)
301end

367function ActivityNode:afterLauncher(configVersion, callback)
368    -- 资源没下完，或者没config，活动不启动
369    if not self:isEnabled() then
370        safeRunCall(callback)
371        self:loadActFail("活动not enabled, 不能启动 ", configVersion)
372        return
373    end
374
375    self:pushDataIntoConfig()
376
377    -- config.isSupport没过或者actInfo没过，活动不启动
378    if not self:isConfigAndActInfoSupport() then
379        safeRunCall(callback)
380        self:loadActFail("活动not isSupport, 不能启动 ", configVersion)
381        return
382    end
383    
384    self:loadLauncherMain():next(function()
385        return self:loadCodeAndIcon()
386    end):next(function()
387        self:initProcessorAfter()
388        self:handleIcon()
389        self:handleLiteStart()
390        self:handleAutoPopup()
391
392        safeRunCall(callback)
393        if self.iconProcessor then
394            self:processOver(true, configVersion)
395        else
396            self:processOver(false, configVersion)
397        end
398    end, function()
399        safeRunCall(callback)
400        self:loadActFail("活动icon或代码加载失败, 不能启动", configVersion)
401    end)
402end

491function ActivityNode:openActivity(delegate)
492    if not delegate then
493        delegate = {}
494    end
495
496    if self._isResLoading then
497        ActivityDebugLog:logWarning("资源加载中，稍后再点" .. self._actMeta:getActId())
498        safeRunCall(delegate.failCallback)
499        return
500    end
501
502    if self:isEnabled() and self:isConfigAndActInfoSupport() then
503        if not EnvUtil:isWindows() and self._isResLoaded then
504            self:pushDataIntoConfig()
505            self:loadCodeAndIcon():next(function()
506                self:initProcessorAfter()
507                self:beforeLaunch(delegate)
508                self.launchProcessor:process(delegate)
509            end,function()
510                safeRunCall(delegate.failCallback)
511            end)
512            return
513        end
514
515        self._isResLoading = true
516        self._isResLoaded = false
517        self:loadCodeAndIcon():next(function()
518            self:initProcessorAfter()
519            return self:loadRes()
520        end):next(function()
521            --资源重新加载会导致config所需数据消失，需要重新push，目前windows特供
522            self:pushDataIntoConfig()
523            self._isResLoaded = true
524            self._isResLoading = false
525            if self.launchProcessor then
526                self:beforeLaunch(delegate)
527                self.launchProcessor:process(delegate)
528            else
529                safeRunCall(delegate.failCallback)
530                ActivityDebugLog:logError("活动" .. self._actMeta:getActId() .. "launcher被释放，无法打开")
531            end
532        end, function()
533            ActivityDebugLog:logWarning("活动" .. self._actMeta:getActId() .. "完整资源下载失败, 无法开启后续流程")
534            self._isResLoaded = false
535            self._isResLoading = false
536            safeRunCall(delegate.failCallback)
537        end)
538    else
539        ActivityDebugLog:logWarning("活动" .. self._actMeta:getActId() .. "不满足条件，不能开启")
540        safeRunCall(delegate.failCallback)
541    end
542end

544function ActivityNode:downloadResSilent()
545    if self._isResLoaded then return end
546    if self:isEnabled() and self:isConfigAndActInfoSupport() then
547        self:loadRes(true):next(function()

553            self._isResLoaded = false
