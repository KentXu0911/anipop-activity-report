104function APMetaConfigRef:getPopScene()
105    local _popScene = self:getValue("_popScene")
106    if not _popScene then
107        local tmp = {}
108        local autoPopScene = self:getValue("autoPopScene")
109        if autoPopScene and type(autoPopScene) == "table" then
110            for i, v in ipairs(self.autoPopScene) do
111                local stageIndex = tostring(v)
112                if AutoPopupConst.PopStageKeyMapping[stageIndex] then
113                    table.insert(tmp, v)
114                end
115            end
116        end
117        if table.isEmpty(tmp) then
118            tmp = {AutoPopupConst.PopStageId.kEnterFromOutside}
119        end
120        _popScene = tmp
121        self:setValue("_popScene", tmp)
122    end
123    return _popScene
124end

186function APMetaConfigRef:createUniqueKey()
187    local actId = self:getActId()
188    local metaId = self:getMetaId()
189    local autoPopRule = self:getPopRule()
190    local uniqueKey
191    if autoPopRule == AutoPopupConst.MetaPopupRules.EVERY_DAY then
192        local todayStartTime = TimeKit:getTodayStart()
193        uniqueKey = string.format("%s_%s_%s_%s", autoPopRule, actId, metaId, todayStartTime)
194    elseif autoPopRule == AutoPopupConst.MetaPopupRules.A_LIFE_TIME then
195        local endTime = self:getValue("endTime")
196        uniqueKey = string.format("%s_%s_%s_%s", autoPopRule, actId, metaId, endTime)
197    elseif autoPopRule == AutoPopupConst.MetaPopupRules.EVERY_TIME then
198        uniqueKey = string.format("%s_%s_%s", autoPopRule, actId, metaId)
199    else
200        assert(false, "invalid auto popup rule" .. autoPopRule)
201    end
202    return uniqueKey
203end

205function APMetaConfigRef:isRuleEffective()
206    local uniqueKey = self:getCacheUniqueKey()

213        local limitNum = self:getPopRuleParam1() and tonumber(self:getPopRuleParam1()) or 1
214        if curPopNum < limitNum then
215            return true
216        else
217            return false
218        end
219    elseif autoPopRule == AutoPopupConst.MetaPopupRules.A_LIFE_TIME then
220        local curPopNum = _G.amfBean("AutoPopupMgr"):getPermanentShowNum(uniqueKey)
221        local limitNum = self:getPopRuleParam1() and tonumber(self:getPopRuleParam1()) or 1
222        if curPopNum < limitNum then
223            return true
224        else
225            return false
226        end
227    elseif autoPopRule == AutoPopupConst.MetaPopupRules.EVERY_TIME then
228        return true
229    else
230        assert(false, "invalid auto popup rule" .. autoPopRule)

377-- 支持多开关配置"switch1|ON1;switch1|ON2" or "switch1|ON;switch2|ON1;switch2|ON2"
378function APMetaConfigRef:isGroupSwitchEffective()
379    local groupSwitch = self:getValue("groupSwitch")
380    if groupSwitch then
381        local switches = splite(groupSwitch, ";")
382        if #switches == 0 then
383            return true
384        end
385        for _, singleSwitch in ipairs(switches) do
386            local groupSwitchTb = splite(singleSwitch, "|")
387            local switchName = groupSwitchTb[1]
388            local switchGroup = groupSwitchTb[2]
389            if not switchName or not switchGroup then
390                return true
391            end
392            local isEnabled, _, src = MaintenanceManager.getInstance():isEnabledInGroup(switchName, switchGroup)
393            if not isEnabled then
394                -- 开关不存在或者配错时，视为开关生效
395                isEnabled = src == 5
396            end
397            if isEnabled then
398                return true
399            end
400        end
401    else
402        return true
403    end
404    return false
405end

415                return true
416            else
417                AutoPopupDebugLog:log("APMetaConfigRef:isLevelEffective 1 false " .. self:getActId() .. "_" .. userState)
418                return false
419            end
420        else
421            local topLevel = _G.getTopLevelId()
422            if topLevel >= autoPopMinLevel and topLevel <= autoPopMaxLevel then
423                return true
424            else
425                AutoPopupDebugLog:log("APMetaConfigRef:isLevelEffective 2 false " .. self:getActId() .. "_" .. topLevel)
426                return false
427            end
428        end

478function APMetaConfigRef:isUnlimitedEnergyEffective()
479    local autoPopUnlimitedEnergy = self:getValue("autoPopUnlimitedEnergy")
480    if autoPopUnlimitedEnergy == nil or autoPopUnlimitedEnergy == "" then
481        return true
482    end
483
484    local energyState = _G.amfBean("UserEnergyManager"):getEnergyState()
485    local isInfinite = energyState == _G.UserEnergyState.INFINITE
486    local isEnabled = false
487    if autoPopUnlimitedEnergy == "true" then
488        isEnabled = isInfinite
489    else
490        isEnabled = not isInfinite
491    end
492
493    if not isEnabled then
494        AutoPopupDebugLog:log("APMetaConfigRef:isUnlimitedEnergyEffective false " .. self:getActId() .. "_" ..
495                                  energyState)
496    end
497
498    return isEnabled

526-- 检测并准备强弹数据
527function APMetaConfigRef:checkAndGetPopData(data)
528    local levelData = data.levelData
529    local popupStage = data.popStage
530    local isMetaPop = data.isMetaPop
531    local popData
532
533    --配置化强弹自定义不触发
534    if isMetaPop and not self:isMetaPopConfig() then
535        return
536    end
537
538    if not table.exist(self:getPopScene(), popupStage) then
539        return
540    end
541
542    if isMetaPop and _G.amfBean("ActivityParticipationMgr")
543        :shouldReduceConfiguredMainPanelAutoPopup(self:getActId()) then
544        return
545    end
546
547    if self:isLevelEffective() and self:isStarEffective() and self:isMinEnergyEffective() and
548        self:isUnlimitedEnergyEffective() and self:isCustomizedCheckFulfill(levelData) then
549        local metaId = self:getMetaId()
550        local actId = self:getActId()
551
552        -- 主面板强弹
553        local function setPopData(type)
554            popData = {
555                metaId = metaId,
556                actId = actId,
557                levelData = data,
558                priority = self:getPopRelativePriority(),
559                autoPopupConfig = self,
560                type = type,
561            }
562        end
563
564        if isMetaPop then
565            --剩余时间足够触发
566            if self:isRemainTimeEnough() then
567                setPopData(ActivityConst.PopType.MAIN_PANEL)
568            end
569        else
570            if popupStage == AutoPopupConst.PopStageId.STAGE_THREE and self:isStage3CustomizedPopup() then
571                setPopData(ActivityConst.PopType.CUSTOMIZE)
572            elseif popupStage == AutoPopupConst.PopStageId.STAGE_FOUR and self:isStage4CustomizedPopup() then
573                setPopData(ActivityConst.PopType.CUSTOMIZE)
574            end
575        end
576    end
577
578    return popData
579end
