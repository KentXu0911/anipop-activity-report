32function FixBirdNestModel:replaceRewards(rewards)
33    local miniGameRewardConvertConfigList = self:getConfigByName("miniGameRewardConvertConfigList")
34    
35    local function replaceReward(reward)
36        for k, v in pairs(miniGameRewardConvertConfigList) do
37            if v.beReplaceId == reward.itemId then
38                reward.itemId = v.replaceId
39                reward.num = v.ratio
40                break
41            end
42        end
43    end
44
45    for k, reward in pairs(rewards) do
46        replaceReward(reward)
47    end
48
49    return rewards
50end

390-- checkIsSameDay检查是否同一天，如果不是同一天就返回空
391function FixBirdNestModel:getRecord(key, checkIsSameDay)
392    if not key then return end
393    
394    local recordData = self:getValue("_recordData")
395    local saveJsonStr = recordData[key]
396    if saveJsonStr == nil then return end
397    if type(saveJsonStr) ~= "string" then return end
398    local saveTable = table.deserialize(saveJsonStr)
399
400    local nowTime = getDayStartTimeByTS(TimeKit:timeInSec())
401    local bTimeIsOk = true
402
403    -- 需要检查是否同一天 且 时间不同
404    if checkIsSameDay and saveTable.time ~= nowTime then
405        bTimeIsOk = false
406    end
407
408    -- 检查是否同一天 且 不是同一天 返回空
409    if checkIsSameDay and not bTimeIsOk then
410        return
411    else
412        return saveTable.value
413    end
414end
