324function Manager:onHandlePassLevel(evt)

374function Manager:registerEndTimeCallback()
375    local endTime = self:getEndTimeInSec()
376    if endTime <= 0 or TimeKit:timeInSec() >= endTime then
377        return
378    end
379
380    local callback
381    callback = function()
382        if self._endTimeCallback ~= callback then return end
383        self._registeredEndTime = nil
384        self._endTimeCallback = nil
385        self:refreshActivity()
386    end
387    self._registeredEndTime = endTime
388    self._endTimeCallback = callback
389    PassDayTimer:registerEndTimeCallback(endTime, callback)
390end

396    end
397end
