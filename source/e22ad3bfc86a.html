203function MarkManager:showMainPanel(param)
204    if not self:isEnabled() then return end
205    local dcLocation = param.dcLocation or -1
206    local isClick = param.isClick
207    local closeCallback = param.closeCallback
208    local skipIfTodayMarked = param.skipIfTodayMarked == true
209    local forceRequest = param.forceRequest == true
210
211    self.dcLocation = dcLocation
212    -- 没网
213    if not self._networkStatusManager:isNetworkEnabled() then
214        CommonTip:showTip("网络异常，请检查网络设置~")
215        safeRunCall(closeCallback)
216        return
217    end
218    -- 点击签到icon，但有界面弹出，不弹
219    if isClick and not self:_checkCanPop(isClick) then
220        safeRunCall(closeCallback)
221        return
222    end
223
224    local function _showMainPanel(delegate)
225        if skipIfTodayMarked and self:isTodayMarked() then
226            safeRunCall(closeCallback)
227            return
228        end
229        if not self:_checkCanPop() then
230            safeRunCall(closeCallback)
231            return
232        end
233        local function closeCall()
234            -- 刷新签到上的红点
235            self:updateMarkButton()
236            self.mainPanelVM = nil
237            safeRunCall(closeCallback)
238
239            -- if EnvUtil:isOhos() then
240            --     local OhosCard = require "ohos_card"
241            --     if OhosCard then
242            --         OhosCard:flush()
243            --     end
244            -- end

247        delegate.closeCallback = closeCall
248        delegate.autoCloseAfterSignIn = param.autoCloseAfterSignIn == true
249        delegate.signInRequest = param.signInRequest
250        delegate.panelShownCallback = param.panelShownCallback
251        local MarkMainPanelVM = require_debug "amf.feature.dailyreward.mark.view.main.MarkMainPanelVM"
252        self.mainPanelVM = MarkMainPanelVM:create(delegate)
253        self.mainPanelVM:setLoadResFailCallback(closeCallback)
254        self.mainPanelVM:popout(nil, not EnvUtil:isMiniGame())
255
256        -- 打点
257        self:dc("signPanel_popUp")
258        self:dc("newPanel_trigger",{
259                t1=tostring(self:getMarkValue("totalMark")),

277                dcLocation = dcLocation,

295    -- 如果是点击icon，一定先请求数据
296    if isClick or forceRequest then
297        -- self:log("点击icon")

300        -- 如果数据可用就

510        -- 找到补单的那天

514            for _,vv in ipairs(lastDayList) do
515                if v.day == vv.day and v.markType ~= MarkConst.MarkType.NONE and vv.markType == MarkConst.MarkType.NONE then
516                    needDay = v.day
517                    find = true
518                    break
519                end
520            end
521            if find then
522                break
523            end
524        end
525        -- 打点
526        local count = data.rewards and #data.rewards

528
529        -- 丢单数量清零
530        -- 协议里已经发了
531        -- privateNS.markModel:setValue("v3MakeUpChances", 0)
532        self:_checkOtherRewards(data, true, true)
533        d:resolve({
534            data = data,
535            needDay = needDay
536        })
537
538        MarkEventCenter:dispatch(MarkEvent.MARK_MARKUP_SUCCESS, {
539            day = needDay,
540            hasMonthRewards = not table.isEmpty(data.monthRewards or {}),

704        end
705        d:reject(response:getErrorCode(), response:isCancel())
706    end)
707    return d
708end

726function MarkManager:_requestBuy(goodsId)
727    local d = deferred.new()
728    self:setMarkValue("isHttpError", true)
729    local function onSuccess()
730        self:doTracer("paySuccess")
731        self:setMarkValue("isHttpError", false)
732        setTimeOut(function()
733            d:resolve()
734        end, 1)
735    end
736    local function onFail(errorCode, errorMsg, payInfo)
737        self:doTracer("payFail")
738        d:reject(errorCode)
739    end
740    local function onCancel(payInfo)
741        self:setMarkValue("isHttpError", false)
742        self:doTracer("payCancel")
743        d:reject(nil, true)
744    end
745    -- 开始购买
746    local buyContext = BuyContext.new(goodsId, MoneyType.kRmb)
747        :setFeatureAndSource(DcFeatureType.kSignIn, DcSourceType.kSign2020MakeUp)
748        :setCallback(onSuccess, onFail, onCancel)
749    _G.amfTransaction(TransactionEvent.PAY, buyContext)
750    return d
751end

816function MarkManager:onActUBFAddScoreBuff(event)
817    local eventData = event:getData()
818    if eventData.isAskForHelp then return false end
819    local scoreBuffRate, buffEndTime = self:getCurAvailableBuffAndEndTime()
820    local hasScoreBuff = buffEndTime > TimeKit:timeInSec()
821
822    local levelId = eventData.levelId
823    local levelScene = eventData.levelScene
824
825    local levelSupportType = UBFActHelper:getPlayLevelModeType(levelId, nil, levelScene)
826    local levelOk = levelSupportType == LevelSupportModeType.kNormal or levelSupportType == LevelSupportModeType.kExceedStar or levelSupportType == LevelSupportModeType.kExceedScore
827
828    if not levelOk then return end
829
830    if hasScoreBuff then
831        local builder = UBFPlayImproveBuffBuilder:create()
832        builder:setModSourceId(MarkConst.ACT_ID)
833        builder:addBuffWithOld(InitBuffType.CUSTOM_SCORE_BUFF, InitBuffCreateType.BUFF_ACTIVITY, scoreBuffRate - 1)
834        table.insert(eventData.buffList, builder:toBuffCreateData() )
835    end
836end
