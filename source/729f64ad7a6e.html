124function UBFUESlpBottomWinRepeatRank:cleanupContent()
125    self.contentVersion = (self.contentVersion or 0) + 1
126    if self.scrollView == self:getView("ScrollView") then
127        self:removeView("ScrollView")
128    elseif isSafeCocosObj(self.scrollView) then
129        self.scrollView:removeFromParentAndCleanup(true)
130    end
131    self.scrollView = nil
132
133    for _, node in ipairs(self.nodeList or {}) do
134        local cellName = node and node.cellName
135        self:unreferenceCellView(cellName, node)
136        if cellName then
137            if self._cachedViews then
138                self._cachedViews[cellName] = nil
139            end
140            if self._pendingViewCreateData then
141                self._pendingViewCreateData[cellName] = nil
142            end
143        end
144        if isSafeCocosObj(node) then
145            node:removeFromParentAndCleanup(true)
146        end
147    end
148
149    local playerCount = #(self.playerList or EMPTY_TABLE)
150    local oldPlayerCount = math.max(self.lastPlayerCellCount or 0, playerCount)
151    for i = 1, oldPlayerCount do
152        local cellName = "playerCell_" .. i
153        self:unreferenceCellView(cellName, self:getView(cellName))
154        if self._cachedViews then
155            local cachedView = self._cachedViews[cellName]
156            self:unreferenceCellView(cellName, cachedView)
157            if isSafeCocosObj(cachedView) then
158                cachedView:removeFromParentAndCleanup(true)
159            end
160            self._cachedViews[cellName] = nil
161        end
162        if self._pendingViewCreateData then
163            self._pendingViewCreateData[cellName] = nil
164        end
165    end
166
167    self.nodeList = {}
168    self.myNode = nil
169    self.myNodeIndex = nil
170    self.lastPlayerCellCount = playerCount
171    return self.contentVersion
172end

435function UBFUESlpBottomWinRepeatRank:onGiftBoxClick(node, giftBox)

456function UBFUESlpBottomWinRepeatRank:checkHasRankRewards(rank)
457    local rewardInfo = self.rankRewardMap[rank] or self.rankRewardMap[tostring(rank)]
458    local rewards = rewardInfo and rewardInfo.rewards or {}
459    local extraRewards = rewardInfo and rewardInfo.extraRewards
460    local mergeRewards = table.union(rewards, extraRewards)
461    if table.isEmpty(mergeRewards) then
462        return false, {}, nil
463    end
464    return true, rewards, extraRewards, mergeRewards
465end
