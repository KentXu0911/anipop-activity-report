166function ShopRewardMainPanelVM:closeWithShrinkAnim(afterClose)
167    if self.isClosingWithShrink then return end
168
169    self.isClosingWithShrink = true
170    local clickCloseBtn = self._clickCloseBtn
171
172    local target = self:getShrinkTarget()
173    if not isSafeCocosObj(target) then
174        self:removePanelForShrinkClose()
175        safeRunCall(afterClose, clickCloseBtn)
176        return
177    end
178
179    local sourceNode = self:getShrinkSourceNode()
180    if not isSafeCocosObj(sourceNode) then
181        self:removePanelForShrinkClose()
182        safeRunCall(afterClose, clickCloseBtn)
183        return
184    end
185
186    local anim = PanelShrinkToTargetAnim.prepare({
187        sourceNode = sourceNode,
188        targetIcon = target,
189        config = ShrinkAnimConfig,
190    })
191    if not anim then
192        self:removePanelForShrinkClose()
193        safeRunCall(afterClose, clickCloseBtn)
194        return
195    end
196
197    self:removePanelForShrinkClose()
198    anim:play()
199    TimerUtil.addAlarm(function()
200        safeRunCall(afterClose, clickCloseBtn)
201    end, ShrinkCallbackDelay, 1)
202end
