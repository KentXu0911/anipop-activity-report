220    local collectCardMgr = _G.amfBean and _G.amfBean("CollectCardManager")
221    if collectCardMgr then
222        local normalRewards = {}
223        local linkRewards = {}
224        for _, v in ipairs(rewardItems) do
225            if collectCardMgr:isCardPack(v.itemId) then
226                table.insert(normalRewards, v)
227            else
228                table.insert(linkRewards, v)
229            end

231        if #normalRewards > 0 then
232            self:addRewardsNormal(normalRewards, params)
233        end
234        if #linkRewards > 0 then
235            self:callPlug("ActExchange", "addLinkRewards", linkRewards, params)
236        end

296function Manager:getEnterLevelCollectBaseCount(levelId, levelScene, payTag)
297    local baseDropCount = 0
298    local collectionConfig = UBFActHelper:getSpecificCollectionIdConfig(3078, "183")
299    local collectionBaseCountConfig = collectionConfig['baseDropRule'] and collectionConfig['baseDropRule']['collectionBaseCountConfig']
300    if collectionBaseCountConfig then
301        local function getCollectBaseCount(config, countConfig, currTravelIndex)
302            if not countConfig then return 0 end
303            local baseCount = 0
304            local configData = string.split(countConfig, ",")
305            -- 按照固定值取
306            if #configData == 1 then
307                baseCount = tonumber(configData[1])
308             -- 按照付费标签取       
309            elseif #configData == 7 then
310                payTag = payTag or 1
311                baseCount = tonumber(configData[payTag])
312            else
313                UBFThrow("收集物基础数量配置格式有误：" .. tts(countConfig))
314            end
315            return tonumber(baseCount)
316        end
317        local maxLevelId = _G.amfBean("LevelUnLockManager"):getUserMaxLevelId()
318        local levelId = UBFActHelper:getHiddenLevelStartNormalLevel(levelId, levelScene)
319        local levelSupportType = self:getPlayLevelModeType(levelId, nil, levelScene)
320        for i, config in ipairs(collectionBaseCountConfig or {}) do
321            local startLevel = UBFActHelper:getDropConfigLevelId(config.startLevel, maxLevelId)
322            local endLevel = UBFActHelper:getDropConfigLevelId(config.endLevel, maxLevelId)
323            if levelId >= startLevel and levelId <= endLevel then
324                if LevelSceneFunc:isFullLevelLib(levelScene) then
325                    baseDropCount = getCollectBaseCount(config, config.fsllLevelDrop)
326                elseif levelSupportType == LevelSupportModeType.kNormal then
327                   baseDropCount = getCollectBaseCount(config, config.topDrop)
328                elseif levelSupportType == LevelSupportModeType.kExceedStar then
329                    baseDropCount = getCollectBaseCount(config, config.refreshStarDrop)
330                elseif levelSupportType == LevelSupportModeType.kExceedScore then
331                    baseDropCount = getCollectBaseCount(config, config.refreshScoreDrop)
332                end
333            end
334        end
335    end
336
337    return baseDropCount
338end

340function Manager:getPlayLevelModeType(levelId, actLevelType, levelScene)
341    if _G.editorMode then return LevelSupportModeType.kNormal end
342    if LevelSceneFunc:isFullLevelLib(levelScene) then
343        return LevelSupportModeType.kNormal
344    end
345    local levelType = UBFLevelConfigManager:userConfig():getLevelType(levelId, levelScene)
346    if actLevelType and levelType == actLevelType then
347        return LevelSupportModeType.kNormal
348    end
349    if UBFLevelSceneFunc:isActLevelScene(levelScene) then
350        return LevelSupportModeType.kNormal
351    end
352    local isFMJState, isTopLevel
353    if EnvUtil:isAMFEnabled() then
354        local userState = _G.amfBean("UserStateManager"):getUserLevelState()
355        isFMJState = userState == _G.UserStateConst.UserLevelState.NORMAL
356        local levelAreaMgr = _G.amfBean("LevelAreaManager")
357        local topLevelId = _G.getTopLevelId(true)
358        if levelAreaMgr:isAreaBlocked(topLevelId) then
359            isTopLevel = levelId >= _G.getTopLevelId() and not _G.amfBean("LevelAreaManager"):hasPassedLevelEx(levelId, levelScene)
360        else
361            isTopLevel = levelId == _G.getTopLevelId() and not _G.amfBean("LevelAreaManager"):hasPassedLevelEx(levelId, levelScene)
362        end
363    else
364        isFMJState = ActUserStateManager:getInstance():isFMJ()
365        isTopLevel = levelId == UserManager:getInstance().user:getTopLevelId() and not UserManager:getInstance():hasPassedLevelEx(levelId, levelScene)
366    end
367    if not UBFLevelSceneFunc:isMainLevel(levelScene) then isTopLevel = false end
368    if isFMJState then
369        if isTopLevel then
370            return LevelSupportModeType.kNormal
371        else
372            if not UBFActHelper:isFullStar(levelId, levelScene) then
373                return LevelSupportModeType.kExceedStar
374            end
375        end
376    else
377        if UBFActHelper:isFullStar(levelId, levelScene) then
378            --由活动开启行为树去判断刷分是否需要打推荐关还是任意关，此处不在判断了
379            return LevelSupportModeType.kExceedScore
380        else
381            return LevelSupportModeType.kExceedStar
382        end
383    end
384end

435function Manager:popIconTip(reason, params)
436    local icon = self:callPlug("Act", "getActIcon")
437    if isSafeCocosObj(icon) and type(icon.refreshProvider) == "function" then
438        return
439    end
440
441    local delegate = {}
442    local data = {}
443    local befTaskId = self:getBefTask()
444    local maxClaimableId = self:getMaxClaimableTaskId() or befTaskId
445    delegate.iconKey = tostring(self:getConst().actId)
446    delegate.iconType = IconTipType.MESSAGE
447    delegate.priority = "S"
448    data.tipsType = HomeIconTipsType.PROGRESS
449    data.value = (maxClaimableId == befTaskId) and (self:getBefLeftCoin() or 0) or 0
450    data.min = 0
451    data.max = self:getModel():getNeedCoin(maxClaimableId) or 0
452    data.targetValue = self:getModel():getNeedCoin(maxClaimableId) or 0
453    data.textMode = ProgressBar.TextMode.kFrection
454    data.iconCreate = function()
455        local rewards = self:getBoxRewards(maxClaimableId)
456        local item = rewards and rewards[1]
457        if not item then return Node:create() end
458        local node = Node:create()
459        local icon = GameItemCreator:getPropIcon({itemId = item.itemId, num = 1})
460        icon:setScale(0.5)
461        node:addChild(icon)
462        return node
463    end
464    delegate.creatorDelegate = data
465    delegate.ignoreNewLevel = true
466    -- printx(41,"popIconTip",tts(delegate))
467    IconTipEventCenter:dispatch(IconTipEvent.ADD_ICON_TIP, delegate)
468end
