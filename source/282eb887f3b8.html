234function Model:getBuildProgress(id)
235    local buildList = self:getDataByKey("buildList",{})
236	local lastRewardProgress = 0
237	local allProgress = 0
238	for k,v in pairs(buildList) do 
239		if v.buildId == id then 
240			lastRewardProgress = v.lastRewardProgress
241			allProgress = v.mateProgress + v.selfProgress
242		end 
243	end 
244	local buildProgressRewards = self:getConfigByKey("buildProgressRewards",{})
245	local min = 0	
246	local max = 999999999
247	for k,v in pairs(buildProgressRewards) do 
248		local needNum = tonumber(v.needNum)
249		if lastRewardProgress >= needNum then
250			min = math.max(min,needNum)
251		else
252			max = math.min(max,needNum)
253		end 
254	end
255	local isFullLevel = max == 999999999
256	return allProgress,min,max ,isFullLevel
257	
258end

279function Model:getBuffRateAndTime()
280	local collectBuffList = self:getDataByKey("collectBuffList", {})
281    local nowTime = TimeKit:timeInSec()
282    local nextTime
283    local rate = 1
284    for i, v in pairs(collectBuffList) do

286        if t > nowTime then
287            if not nextTime then
288                nextTime = t
289            else
290                nextTime = math.min(nextTime, t)
291            end
292
293            if self:getConst().buffTimesMap[tonumber(v.itemId)] then
294                rate = rate * self:getConst().buffTimesMap[tonumber(v.itemId)]
295            end

416function Model:_getProgressRewardIndexByProgress(totalProgress)
417    local normalConfig = self:getTotalProgressRewards()
418    local lastMeta = normalConfig[table.length(normalConfig)] or {}
419    local lastNum = lastMeta.needNum
420
421    if totalProgress < lastNum then
422        for k, v in ipairs(normalConfig) do
423            if v.needNum > totalProgress then
424                return k
425            end
426        end
427    else
428        totalProgress = totalProgress - lastNum
429        local progressMax = self:getTotalProgressLoopPeriod()
430        local index = math.floor(totalProgress / progressMax) + 1
431        return index + table.length(normalConfig)
432    end
