124-- 如果要自己继承插件改的话自己传plugPath, 可以继承调用
125-- 如果有自己的插件模块要添加也可以这么整，在自己的活动功能文件夹下整插件加上去。
126function BaseFeatureManager:_initPlug(key, data)
127    local plugPath
128    if data and data.plugPath then
129        plugPath = data.plugPath
130    else
131        plugPath = self:_getPlugPath(key)
132    end
133
134    local holdName = TextUtil:lowercaseFirstLetter(key) .. "Plug"
135
136    self:_installPlug(key, data, plugPath):as(holdName)
137
138
139    -- 插件函数调用
140    self["call" .. key .. "Func"] = function(obj, funcName, ...)
141        return obj:callPlug(key, funcName, ...)
142    end
143end

153function BaseFeatureManager:_installPlug(plugType, installData, scriptPath)
154    local plug = self:_getPlugContainer():install(plugType, installData, scriptPath)
155    return {
156        as = function(_, name)
157            if name then
158                if self[name] == plug then
159                    self[name] = nil
160                end
161                self[name] = plug
162            end
163        end
164    }
165end

190-- 跨天刷新数据
191function BaseFeatureManager:_checkAndNotifyPassDayOnInit()
192    if not self.passDayRefreshData then return end
193
194    local fileObj = self:getBaseFileObject()
195    if not fileObj then return end
196
197    local last_save_key = self:getClassName() .. "_lastSavedTime"
198    local lastSavedTime = fileObj:getValue(last_save_key)
199    local nowTime = tostring(TimeKit:getTodayStart())
200    if lastSavedTime ~= nil and lastSavedTime ~= nowTime then
201        self:passDayRefreshData()
202    end
203    fileObj:setValue(last_save_key, nowTime)
204    fileObj:saveAsync(0.5)
205end
