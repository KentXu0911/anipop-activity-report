71function GiftModule:getPackageIdsByGroup(group)
72    local packages = {}
73    for _, package in ipairs(self._packages or {}) do
74        local limited =
75            package.isLimited and package:isLimited()
76        if package:getExtraValue("group") == group
77            and not limited then
78            table.insert(packages, package)
79        end
80    end
81    table.sort(packages, function(left, right)
82        local leftPriority = tonumber(
83            left:getExtraValue("displayPriority")
84        ) or math.huge
85        local rightPriority = tonumber(
86            right:getExtraValue("displayPriority")
87        ) or math.huge
88        if leftPriority ~= rightPriority then
89            return leftPriority < rightPriority
90        end
91        local leftGoodsId =
92            tonumber(left:getGoodsId()) or math.huge
93        local rightGoodsId =
94            tonumber(right:getGoodsId()) or math.huge
95        return leftGoodsId < rightGoodsId
96    end)
97    local packageIds = {}
98    for _, package in ipairs(packages) do
99        table.insert(packageIds, package:getGoodsId())
100    end
101    return packageIds
102end

173function GiftModule:selectDailyGiftPackage(packages, payTag)
174    local recommendation =
175        DAILY_GIFT_RECOMMENDATIONS[tonumber(payTag) or 0]
176            or DAILY_GIFT_RECOMMENDATIONS[0]
177
178    local function findPackage(price, limited)
179        local targetPriceKey = getPriceKey(price)
180        for _, package in ipairs(packages or {}) do
181            if isDailyGiftPackage(package)
182                and isPackageAvailable(package)
183                and isLimitedPackage(package) == limited
184                and getPackagePriceKey(package) == targetPriceKey then
185                return package
186            end
187        end
188    end
189
190    for _, price in ipairs(recommendation.limited) do
191        local package = findPackage(price, true)
192        if package then return package end
193    end
194    return findPackage(recommendation.unlimited, false)
195end
