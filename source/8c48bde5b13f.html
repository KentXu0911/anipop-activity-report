111function ActPlug:openActivity(delegate)
112    if not delegate then
113        delegate = {}
114    end
115
116    if not self:getActivityMgr():isEnabled() then
117        safeRunCall(delegate.failCallback)
118        return
119    end
120
121    if delegate.userClick == nil then
122        delegate.userClick = false
123    end
124
125    delegate.actId = delegate.actId or self.actId
126
127
128    ActivityEventCenter:dispatch(ActivityEvent.OPEN_ACTIVITY, delegate)
129end

132function ActPlug:registerClearData()
133    local ActivityEventCenter = require "amf.middleware.activity.event.ActivityEventCenter"
134
135    local function onClearData(evt)
136        local data = evt:getData()
137        if data.actId == self.actId then
138            self.model:reset()
139            CommonTip:showTip("活动" .. self.actId .. "本地数据已清除，最好重启一下游戏")
140        end
141    end
142
143    self.onClearData = onClearData  
144    ActivityEventCenter:register(ActivityEvent.ACT_DATA_DELETE, onClearData)
145end
