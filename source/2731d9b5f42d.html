30local function emitDiagnostic(logger, code, data)
31    if logger == nil then
32        return
33    end
34    if type(logger) == "function" then
35        pcall(logger, code, data)
36        return
37    end
38    if type(logger) ~= "table" then
39        return
40    end
41    local method = logger.warn or logger.log
42    if type(method) == "function" then
43        pcall(method, logger, code, data)
44    end
45end

47local function resolveCycleKey(config, iconConfig, options)
48    if isNonEmptyString(iconConfig.newActivityCycleKey) then
49        return iconConfig.newActivityCycleKey
50    end
51    if isNonEmptyString(options.fallbackCycleKey) then
52        return options.fallbackCycleKey
53    end
54    if type(options.resolveCycleKey) == "function" then
55        local ok, value = pcall(options.resolveCycleKey, config)
56        if ok and isNonEmptyString(value) then
57            return value
58        end
59    end
60end

62local function isConsumed(records, key)
63    if key == nil or records == nil then
64        return false
65    end
66    local getter = records.isConsumed or records.get
67    if type(getter) ~= "function" then
68        return false
69    end
70    local ok, value = pcall(getter, records, key)
71    return ok and (value == true or (type(value) == "number" and value > 0))
72end

74function NewActivityPolicy.resolve(config, options)
75    options = options or {}
76    local iconConfig = type(config) == "table" and type(config.icon) == "table" and config.icon or {}
77    local rawType = iconConfig.newActivityType
78    local activityType = rawType
79    if not ValidTypes[activityType] then
80        activityType = "A"
81        emitDiagnostic(options.logger, "ICON_V2_NEW_ACTIVITY_TYPE_INVALID", {
82            value = rawType,
83            fallback = activityType,
84        })
85    end
86
87    local rawFirstOpen = iconConfig.firstOpenUseCNotification
88    if rawFirstOpen ~= nil and type(rawFirstOpen) ~= "boolean" then
89        emitDiagnostic(options.logger, "ICON_V2_FIRST_OPEN_C_FLAG_INVALID", {
90            valueType = type(rawFirstOpen),
91        })
92    end
93
94    local cycleKey = resolveCycleKey(config, iconConfig, options)
95    if activityType ~= "none" and not isNonEmptyString(cycleKey) then
96        emitDiagnostic(options.logger, "ICON_V2_NEW_ACTIVITY_CYCLE_KEY_MISSING", {
97            activityType = activityType,
98        })
99        cycleKey = nil
100    end
101
102    local featureKey = isNonEmptyString(iconConfig.newActivityFeatureKey)
103        and iconConfig.newActivityFeatureKey or nil
104    local firstOpenUseCNotification = activityType == "B" and rawFirstOpen == true
105    if firstOpenUseCNotification and featureKey == nil then
106        emitDiagnostic(options.logger, "ICON_V2_NEW_ACTIVITY_FEATURE_KEY_MISSING", {
107            activityType = activityType,
108        })
109        firstOpenUseCNotification = false
110    end
111
112    local policy = {
113        activityType = activityType,
114        enabled = activityType ~= "none",
115        reminder = copyReminder(ReminderByType[activityType]),
116        cycleKey = cycleKey,
117        featureKey = featureKey,
118        firstOpenUseCNotification = firstOpenUseCNotification,
119    }
120    if cycleKey ~= nil then
121        policy.newClearedRecordKey = cycleKey
122        policy.noticeShownRecordKey = cycleKey .. "_icon_v2_notice_shown"
123        policy.legacyPanelShownRecordKey = cycleKey .. "_panel"
124    end
125    if featureKey ~= nil then
126        policy.firstCRecordKey = featureKey .. "_icon_v2_first_c_notification"
127    end
128    return policy
129end

131function NewActivityPolicy.evaluate(policy, records)
132    assert(type(policy) == "table", "ICON_V2_NEW_ACTIVITY_POLICY_REQUIRED")
133    local newVisible = policy.enabled == true
134        and policy.newClearedRecordKey ~= nil
135        and not isConsumed(records, policy.newClearedRecordKey)
136    local noticeShown = isConsumed(records, policy.noticeShownRecordKey)
137        or isConsumed(records, policy.legacyPanelShownRecordKey)
138    local firstCConsumed = isConsumed(records, policy.firstCRecordKey)
139    local temporaryC = policy.activityType == "B"
140        and policy.firstOpenUseCNotification == true
141        and policy.firstCRecordKey ~= nil
142        and not firstCConsumed
143    local notificationEligible = newVisible
144        and not noticeShown
145        and (policy.activityType == "C" or temporaryC)
146
147    return {
148        newVisible = newVisible,
149        reminder = newVisible and copyReminder(policy.reminder) or nil,
150        notificationEligible = notificationEligible,
151        temporaryC = temporaryC,
152        noticeShown = noticeShown,
153        firstCConsumed = firstCConsumed,
154    }
155end

157function NewActivityPolicy.buildNotificationCandidate(policy, records, metadata)
158    local evaluation = NewActivityPolicy.evaluate(policy, records)
159    if not evaluation.notificationEligible then
160        return nil
161    end
162    local candidate = {}
163    for key, value in pairs(metadata or {}) do
164        candidate[key] = value
165    end
166    candidate.activityType = policy.activityType
167    candidate.cycleKey = policy.cycleKey
168    candidate.featureKey = policy.featureKey
169    candidate.newClearedRecordKey = policy.newClearedRecordKey
170    candidate.noticeShownRecordKey = policy.noticeShownRecordKey
171    candidate.firstCRecordKey = policy.firstCRecordKey
172    candidate.temporaryC = evaluation.temporaryC
173    return candidate
174end
