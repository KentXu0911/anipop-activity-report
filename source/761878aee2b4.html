530function UBFHappyDeliveryPassLevelView:_playLabelIdle(label)
531    if not isSafeCocosObj(label) then return end
532    self:_stopLabelAnim(label)
533
534    local idleKF = {
535        {0, 1}, {6, 1.1}, {11, 0.95}, {18, 1}, {180, 1},
536    }
537    local totalFrame = 180
538    local frame = 0
539
540    local function lerpKF(kfList, f)
541        local pf, pv, nf, nv = 0, kfList[1][2], kfList[#kfList][1], kfList[#kfList][2]
542        for i = 1, #kfList do
543            if kfList[i][1] <= f then pf, pv = kfList[i][1], kfList[i][2] end
544            if kfList[i][1] >= f then nf, nv = kfList[i][1], kfList[i][2]; break end
545        end
546        if nf == pf then return nv end
547        return pv + (nv - pv) * (f - pf) / (nf - pf)
548    end
549
550    label._labelAnimEntry = self:_scheduleEntry(function()
551        if not isSafeCocosObj(label) then return true end
552        frame = frame + 1
553        if frame > totalFrame then
554            label:setScale(1)
555            frame = 0
556            return
557        end
558        if frame % 2 ~= 0 then return end
559        label:setScale(lerpKF(idleKF, frame / 2))
560    end)
561end
