45function LocalServer:fromServer( serverData,skipSaveLocal )
46	local parseConfig = function ( extra )
47		if not extra then return end
48		self:setValue("localConfig",extra)
49		for k,v in pairs(extra) do
50			if k == 'configMd5' then
51				self:setValue("configMd5",v.serverMd5)
52			end 
53		end
54	end
55	for k,v in pairs(serverData) do
56		if k == "extra" then
57			parseConfig(table.deserialize(v))
58		elseif k == "info" then 
59			for _k,_v in pairs(v) do 
60				self:setValue(_k,_v)
61			end 
62		else
63			self:setValue(k,v)
64		end
65	end
66
67	if not skipSaveLocal then
68		self:flush()
69	end
70
71	local payTag = self:getDataByKey("payTagId",0)
72	self:setDataByKey("payTagIdToday", payTag, true, true)
73end

289function LocalServer:getMapIdList()
290	local mapIdList = {}
291	local mapIdSet = {}
292	local gemMapMetas = self:getDataByKey("digGemMapMetaList", {})
293	for _, mapMeta in pairs(gemMapMetas or {}) do
294		local mapId = tonumber(mapMeta.mapId)
295		if mapId and not mapIdSet[mapId] then
296			mapIdSet[mapId] = true
297			table.insert(mapIdList, mapId)
298		end
299	end
300
301	if #mapIdList == 0 then
302		local gemMaps = self:getDataByKey("gemMaps", {})
303		for mapId in pairs(gemMaps or {}) do
304			local numericMapId = tonumber(mapId)
305			if numericMapId and not mapIdSet[numericMapId] then
306				mapIdSet[numericMapId] = true
307				table.insert(mapIdList, numericMapId)
308			end
309		end
310	end
311
312	table.sort(mapIdList)
313	return mapIdList
314end

463    local cellStatus2 = mapBean.cellStatus2
464	local stoneList = {}
465    for k,v in pairs(cellStatus2) do 
466        local row = tonumber(k) - 1
467        local list = string.split(v,",")
468		for id = 0,#list do 
469			list[id] = list[id + 1 ]
470		end
471		stoneList[row] = list 
472    end 
473	local hasClickGem = false 
474	local hasClickSuperGem = false 
475	local isOpenSuper = false 
476	local hasAllOpen = true 
477	for _,gemBean in pairs(mapBean.pointMap ) do 
478		local gemId = gemBean.gemId 
479		local row = gemBean.x 
480		local column = gemBean.y 
481		local gemConfig = Meta:getGemConfig(gemId)
482		local oldOpen = gemBean.open 
483		gemBean.open = true 
484
485		for k,v in pairs(gemConfig.tapListBase) do
486			local rowB =  v.row + row
487			local columnB = v.column + column 
488			if rowB == rowClick and columnClick == columnB then 
489				hasClickGem = true
490				if gemBean.superGem then 
491					hasClickSuperGem = true 
492				end  
493			end 
494			if stoneList[rowB][columnB] == "0" then 
495				gemBean.open = false 
496			end 
497		end 
498		if not oldOpen and gemBean.open and gemBean.superGem then 
499			isOpenSuper = true 
500		end 
501		hasAllOpen = hasAllOpen and gemBean.open
502	end 
503	if hasClickGem then 
504		local list = self:getDataByKey("digOpRecordList",{})
505		table.insert(list,{x = rowClick,y = columnClick})
506        self:setDataByKey("digOpRecordList",list)
507	else 
508        self:setDataByKey("digOpRecordList",{})
509	end 
510
511	return mapBean,isOpenSuper,hasAllOpen,hasClickSuperGem,hasClickGem
