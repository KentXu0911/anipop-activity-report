87function CollectCardGiftModel:_getPayResultParams(cardId)

105function CollectCardGiftModel:isCanCompAlbumByCardId(cardId)
106    local cardData = self._mgr:getCardDataById(cardId)
107    if cardData == nil or cardData.getGotNum == nil then
108        printx(48, "cardData error.cardData=",tts(cardData))
109        return
110    end
111    if not cardData or cardData:getGotNum() > 0 then return false end
112    ---@type AlbumData
113    local albumData = self._mgr:getAlbumDataById(cardData.albumId)
114    if not albumData then return false end
115
116    local unIds = albumData:getUnCollectionCards()
117    if unIds and #unIds == 1 then
118        return unIds[1] == cardId
119    end
120    return false

126function CollectCardGiftModel:requestGiftData(callback, userClick)
127    if not _G.amfBean("NetworkStatusManager"):isNetworkEnabled() then
128        CommonTip:showTip("网络异常，请检查网络~")
129        safeRunCall(callback, false, nil)
130        return
131    end
132
133    --后端收到字段为true时会把tips的弹出状态清零
134    local needRefresh = true
135
136    local callbackPromise = deferred.new()
137    callbackPromise:next(function (packages)
138        -- if EnvUtil:isDevelopMode() then printx(48, "CollectCardGiftModel get data=", tts(packages)) end
139        self._packages = packages
140        local ret = true
141        if packages and #packages > 0  then
142            local moduleExtraMap = packages and #packages > 0 and packages[1]:getExtraValue("moduleExtra") or nil
143            -- moduleExtraMap = "{\"configId\":15,\"packageIdByCardId\":{\"21406\":4618,\"21503\":4618},\"cardIds\":[21406,21503],\"tips\":0}" --debug
144            local extraConfig = (moduleExtraMap) and table.deserialize(moduleExtraMap) or {}
145            self._moduleExtraData = extraConfig
146            if extraConfig then
147                self:_sortCardIdsByStar()
148                local ids = self:getCardIds()
149                if not ids or table.isEmpty(ids) or #ids < 1 then
150                    ret = false
151                    printx(48, "CollectCardGiftModel error ids=", tts(ids))
152                else
153                    self:_initRmbTracer(userClick)
154
155                    local curSelectIdx = self:getSelectedIdx()
156                    local resetSelectIdx = curSelectIdx
157                    if self:isCanShowBanner() then
158                        resetSelectIdx = 0
159                    end
160                    if #ids == 1 then --只有一个
161                        resetSelectIdx = 1
162                    end
163                    if resetSelectIdx ~= curSelectIdx then
164                        self:setSelectedIdx(resetSelectIdx)
165                        if EnvUtil:isDevelopMode() then
166                            printx(48, "CollectCardGiftModel 重置选择下标=", resetSelectIdx)
167                        end
168                    end
169
170                end
171            else
172                ret = false
173            end
174        else
175            ret = false
176        end
177        if ret then
178
179            CollectCardEventCenter:dispatch(CollectCardEvent.ON_GIFT_ICON_UPDATE, self._isHaveCompCard)
180        else
181            self:removeGiftIcon()
182        end
183        safeRunCall(callback, ret, packages)
184    end, function ()
185        CommonTip:showTip("网络异常，请稍后重试~")
186        safeRunCall(callback, false, nil)
187    end)
188
189    GiftEventCenter:dispatch(GiftEvent.PULL_ACTIVITY_PACKAGE, {
190        actId = 9087,
191        moduleName = self.moduleName,
192        callbackPromise = callbackPromise,
193        params = {
194            moduleVersion = 1,
195            needRefresh = needRefresh
196        },
197    })
198end

476function CollectCardGiftModel:_sortCardIdsByStar()
477    local ids = self:getCardIds()
478    if not ids or #ids <= 2 then return end
479
480    self._isHaveCompCard = false
481    local cardDatas = {}
482    for i, cardId in ipairs(ids) do
483        local cardData = self._mgr:getCardDataById(cardId)
484        local star = cardData and cardData:getStarLevel() or 0
485        if cardData:getCardType() == self._mgr:getConst().CardType.GOLD then star = star + 1 end
486        local albumData = self._mgr:getAlbumDataById(cardData and cardData.albumId or 0)
487        local collectedCount = albumData and albumData:getCollectedCount() or 0
488        table.insert(cardDatas, {id = cardId, priority = collectedCount * 100 + star})
489        local canComp = self:isCanCompAlbumByCardId(cardId)
490        if canComp then self._isHaveCompCard = true end
491    end
492    table.sort(cardDatas, function(a, b) return a.priority < b.priority end)
493
494    local n = #cardDatas
495    local sorted = {}
496    if n == 3 then
497        sorted[1] = cardDatas[1].id
498        sorted[2] = cardDatas[3].id
499        sorted[3] = cardDatas[2].id
500    else
501        for i = 1, n do
502            sorted[i] = cardDatas[i].id
503        end
504    end
505
506    self._moduleExtraData.cardIds = sorted
507end
