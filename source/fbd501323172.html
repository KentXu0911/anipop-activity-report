65--[[
66    创建 levelkey
67    levelScene, levelExp, levelGroup，均可以缺省
68    字符串会自动处理缺省，按尽可能 短 的规则创建
69        - 以下效果等效
70        - 2986:main:default:default
71        - 2986:main:default:default
72        - 2986:main
73        - 2986

75local _tmpLevelKeyList = {};
76function UBFLevelIdUtil:createLevelKey(levelId, levelScene, levelExp, levelGroup)
77    local _levelScene = levelScene
78    local _levelExp = levelExp
79    local _levelGroup = levelGroup
80    
81    if not levelExp then
82        _levelExp = "default"
83        _levelGroup = "default"
84    end
85
86    if not levelGroup then
87        _levelGroup = "default"
88    end
89
90    if not levelScene then
91        _levelScene = UBFLevelSceneType.kMain
92    end
93
94    for k, v in pairs(_tmpLevelKeyList) do _tmpLevelKeyList[k] = nil end
95
96    _tmpLevelKeyList[1] = tostring(levelId)
97    _tmpLevelKeyList[2] = _levelScene
98    _tmpLevelKeyList[3] = _levelExp
99    _tmpLevelKeyList[4] = _levelGroup
100    
101    if _levelGroup == "default" then 
102        _tmpLevelKeyList[4] = nil
103        if _levelExp == "default" then
104            _tmpLevelKeyList[3] = nil
105            if _levelScene == UBFLevelSceneType.kMain then
106                _tmpLevelKeyList[2] = nil
107            end
108        else
109            if __WIN32 then 
110                UBFThrow("levelExp怎么回事？要是default都应该是default！")
111            end
112        end
113    end
114
115    return table.concat(_tmpLevelKeyList, ":")
116end

136function UBFLevelIdUtil:traversalDefault(levelId, levelScene, expGroup, func)
137    if not func then func = alwaysTrue end
138
139    local config = UBFLevelConfigManager:getLevelGroupConfig()
140    local groupRule = config.ruleMap[expGroup]
141    if not groupRule then return nil end
142
143    local function getMappingUUID(expGroup)
144        return UBFLevelConfigManager:getLevelUuidByLevelKeyParams(levelId, levelScene, expGroup)
145    end
146
147    local mLevelId, mLevelScene = UBFLevelIdUtil:mappingLevelIdScene(levelId, levelScene)
148
149    local uuid = getMappingUUID(expGroup)
150    if uuid then 
151        local ret = func(expGroup, mLevelId, mLevelScene)
152        if ret then return ret, expGroup, uuid end 
153    end
154
155    local routing = groupRule.fallbackRules or {}
156    
157    for i = 1, #routing do 
158        local currExpGroup = routing[i]
159        local uuid = getMappingUUID(currExpGroup)
160        if uuid then 
161            local ret = func(currExpGroup, mLevelId, mLevelScene)
162            if ret then return ret, currExpGroup, uuid end 
163        end
164    end
165
166    return nil
167end

173function UBFLevelIdUtil:createLevelKeyData(levelId, levelScene, levelExp, levelGroup)
174    local levelKeyData = {
175        levelId = levelId,
176        levelScene = levelScene or UBFLevelSceneType.kMain,
177        levelExp = levelExp or "default",
178        levelGroup = levelGroup or "default",
179    }
180    if levelKeyData.levelExp == "default" then 
181        levelKeyData.levelExpGroup = "default"
182    else
183        levelKeyData.levelExpGroup = levelExp .. ":" .. levelGroup
184    end
185    setmetatable(levelKeyData, { __index = LEVEL_KEY_DATA_METHODS })
186    return levelKeyData
187end

201function UBFLevelIdUtil:createLevelKeyDataWithAutoLevelGroup(levelId, levelScene)
202    levelScene = levelScene or UBFLevelSceneType.kMain
203    local levelKeyData = { levelId = levelId, levelScene = levelScene }
204    local levelExpGroup = UBFLevelConfigManager:userConfig():getLevelGroup(levelId, levelScene)
205    if levelExpGroup == "default" then 
206        levelKeyData.levelExp = "default"
207        levelKeyData.levelGroup = "default"
208        levelKeyData.levelExpGroup = "default"
209    else
210        local _arr = string.split(levelExpGroup, ":")
211        levelKeyData.levelExp = _arr[1]
212        levelKeyData.levelGroup = _arr[2]
213        levelKeyData.levelExpGroup = levelExpGroup
214    end
215    setmetatable(levelKeyData, { __index = LEVEL_KEY_DATA_METHODS })
216    return levelKeyData
