588local function formatSpecial( num, divisor, decimalPlaces )
589    decimalPlaces = decimalPlaces or 2
590    -- string.format("%.2f",num/10000)的写法会进位... 999999990000
591    local text = tostring(num/divisor)
592    local dotIndex = string.find(text, "%.")
593    if dotIndex then
594        text = string.sub(text, 1, dotIndex+decimalPlaces)
595        -- 补足后导零
596        if string.len(text) < dotIndex+decimalPlaces then
597            text = text..string.rep("0", dotIndex+decimalPlaces-string.len(text))
598        elseif decimalPlaces == 0 then
599            text = string.sub(text, 1, dotIndex-1)
600        end
601    end
602    local text1  = string.sub(text, string.len(text)-decimalPlaces)
603    if text1 == "."..string.rep("0", decimalPlaces) then
604        text = string.sub(text, 1, string.len(text)-decimalPlaces-1)
605    end
606    return text
607end

1585function Manager:getCanMakeMenuNum(menuBean,materialMap,addMap)
1586    local map = table.clone(materialMap)
1587    if addMap then
1588        for k,v in pairs(addMap) do
1589            map[k] = map[k] or 0
1590            map[k] = map[k] + addMap[k]
1591        end
1592    end
1593    local nowMap = map
1594
1595
1596    local multipleMap = function(materialMultiple,defaultMaterialArr)
1597        local map = {}
1598        for k,v in pairs(defaultMaterialArr) do
1599            if type(v) == "table" and v.baseNeedCount then
1600                map[tostring(v.materialId)] = math.ceil(tonumber(v.baseNeedCount) * materialMultiple)
1601            else
1602                map[k]  = math.ceil(v * materialMultiple)
1603            end
1604        end
1605        return map
1606    end
1607
1608    local decMap = function(map1,map2)
1609        for k,v in pairs(map2) do
1610            if map1[k] then
1611                map1[k] = map1[k] - map2[k]
1612            end
1613        end
1614        return map1
1615    end
1616    local getCanMakeTimes = function(needMap,haveMap)
1617        local canMakeNum = 999999
1618		for id,needNum in pairs(needMap) do
1619            local numFromData = haveMap[id] or 0
1620            local oldMakeNum = math.floor(numFromData / needNum)
1621            canMakeNum = math.min(oldMakeNum,canMakeNum)
1622		end
1623        return canMakeNum
1624    end
1625
1626    local menuMetaList = self:getData():getDataByKey('menuMetaList')
1627    local menuId = menuBean.menuId
1628    local nowMakeTime = menuBean.makeTimes
1629    local allCanMakeTime = 0
1630    for id,v in ipairs(menuMetaList) do
1631        if v.menuId and v.menuId == menuId then
1632            local expandMeta = v.expandMeta
1633            local defaultMaterialArr = v.defaultMaterialArr
1634            local multiple = expandMeta and expandMeta.defaultMaterialMultiple or 1
1635            if not expandMeta or not expandMeta.detailMetaList or table.isEmpty(expandMeta.detailMetaList) then
1636                return getCanMakeTimes(menuBean.needMaterialMap,nowMap)
1637            end
1638
1639            local useMetalist = table.union(
1640                {
1641                    {
1642                        needTimes = 0 ,
1643                        materialMultiple = multiple
1644                    },
1645                },
1646                table.clone(expandMeta.detailMetaList)
1647            )
1648
1649            for k,meta in ipairs(useMetalist) do
1650                local nextNeedTimes = useMetalist[k+1] and useMetalist[k+1].needTimes
1651                local canMakeThisExpand = false
1652                if nowMakeTime >= meta.needTimes and (not nextNeedTimes or nextNeedTimes > nowMakeTime ) then
1653                    canMakeThisExpand = true
1654                    multiple = meta.materialMultiple
1655                end
1656                if canMakeThisExpand then
1657                    local needMap = multipleMap(multiple,defaultMaterialArr)
1658                    printx(29,"当前的食材需求应该是",tts(needMap),"当前的数据显示",tts(menuBean.needMaterialMap ),v.menuId)
1659                    local canMakeTime = getCanMakeTimes(needMap,nowMap)
1660                    if nextNeedTimes and canMakeTime > (nextNeedTimes - nowMakeTime) then
1661                        allCanMakeTime = allCanMakeTime + nextNeedTimes - nowMakeTime
1662                        nowMap = decMap(nowMap,multipleMap(nextNeedTimes - nowMakeTime,needMap))
1663                        printx(29,'增加了A',allCanMakeTime,nextNeedTimes,nowMakeTime)
1664                        nowMakeTime = nextNeedTimes
1665                    else
1666                        allCanMakeTime  = allCanMakeTime + canMakeTime
1667                        printx(29,'增加了B',allCanMakeTime,canMakeTime)
1668                    end
1669                end
1670            end
1671            break
1672        end
1673    end
1674    return allCanMakeTime
1675end
