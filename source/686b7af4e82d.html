220function ServerData:getAddScore(winCount)

222        winCount = self:getDataByKey("streakNum") or 0
223    end
224
225    if winCount == 0 then
226        return 0
227    end
228
229    local maxNum = 0
230    local streakWinScoreTable = self:getConfigByKey("streakWinScoreTable") or {}
231    for k, v in ipairs(streakWinScoreTable) do
232        if winCount == v.streakWin then
233            return v.score
234        end
235        maxNum = math.max(maxNum, v.score)
236    end
237
