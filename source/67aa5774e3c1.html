75function OpenCardFlowPanelVM:onCloseBtnTapped()

78        openCardFlowId = self.openCardFlowId,

149function OpenCardFlowPanelVM:onCardPackAnimOpening(event)
150    if self.isDisposed then return end
151    if not self:isMyOpenCardAnimEvent(event) then return end
152    local data = event:getData()
153    local openingIdx = data.openingIdx
154    local totalNum = self.totalNum
155    if totalNum == 1 then
156        self:callViewFunc("setSubTitleStr", "恭喜你获得以下卡片")
157    else
158        self:callViewFunc("setSubTitleStr", string.format("开启第%d/%d个卡包，获得以下卡片", openingIdx, totalNum))
159    end
160end

165end

167function OpenCardFlowPanelVM:onCardPackAnimSkip(event)
168    if self.isDisposed then return end
169    if not self:isMyOpenCardAnimEvent(event) then return end
170    local data = event:getData() or {}
171    local skippedNum = data.skippedNum or self.totalNum
172    self:callViewFunc("setSubTitleStr", string.format("开启了剩余%d个卡包，获得以下卡片", skippedNum))
173end
