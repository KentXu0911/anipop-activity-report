65function ProcessGiftPanel:initData(delegate)
66    self.goodsId = delegate.goodsId
67    self.goodsObj = GoodsManager:getGoods(self.goodsId)
68    self.buyHandler = delegate.buyHandler
69    self.panelId = UICreateUtil:getPanelId(
70        "LayeredTileMatchProcessGiftPanel"
71    )
72    self.items = table.clone(self.goodsObj:getItems() or {})
73    self.displayItems = {}
74
75    local displayedItemIds = {}
76    for _, item in ipairs(self.items) do
77        if item.itemId == COIN_ITEM_ID then
78            self.coinNum = item.num
79        elseif not displayedItemIds[item.itemId]
80            and #self.displayItems < MAX_EXTRA_REWARD_KINDS then
81            displayedItemIds[item.itemId] = true
82            table.insert(self.displayItems, item)
83        end
84    end
85
86end

175function ProcessGiftPanel:initButtons()
176    self.closeBtn:setTouchHandler(function()
177        self:dcPaymentResult(4)
178        self:sendCommand("close")
179    end)
180    self.mainButton:setTouchHandler(function()
181        if self.isDisposed or self.buying then return end
182        self.buying = true
183        self:dcPaymentClick()
184        safeRunCall(
185            self.buyHandler,
186            function()
187                self:dcPaymentResult(1)
188                if self.isDisposed then return end
189                self:sendCommand("close")
190            end,
191            function()
192                self:dcPaymentResult(2)
193                self.buying = false
194            end,
195            function()
196                self:dcPaymentResult(3)
197                self.buying = false
198            end
199        )
200    end)
201    self:setBuyText()
202end

204function ProcessGiftPanel:initRewards()
205    self.title:setString(PANEL_TITLE)
206    self.titleBg:setString(PANEL_TITLE)
207    if self.mainText then
208        self.mainText:setString(PANEL_SUB_TITLE)
209    end
210    local coinNum = tonumber(self.coinNum) or 0
211    local hasCoin = coinNum > 0
212    if self.coin then
213        self.coin:setVisible(hasCoin)
214    end
215    if hasCoin then
216        self.coinText:setString(tostring(coinNum))
217    end
218
219    self.rightRoot:setPositionXY(
220        GIFT_ROOT_POSITION.x,
221        GIFT_ROOT_POSITION.y
222    )
223    local positionConfig = hasCoin
224        and GiftPosCfg
225        or NoCoinGiftPosCfg
226    local positions = positionConfig[#self.displayItems] or {}
227    for index, item in ipairs(self.displayItems) do
228        item.needNum = true
229        local icon = WidgetManager:createWidgetObj("UEPropIcon")
230        icon:setItem(item)
231        icon:setScale(0.5)
232        self.rightRoot:addChild(icon)
233        local position = positions[index]
234        if position then
235            icon:setPositionXY(position.x, position.y)
236        end
237    end
