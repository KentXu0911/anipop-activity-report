270-- 播放完成、被跳过或被丢弃的 context 会被 handler 标记，不能再被复用。
271local function _isFlyCollectionContextConsumed(context)
272    if not context then return true end
273    return context.handled == true or context._finished == true
274end

276-- 同一 icon 同时挂两个待播 context 时会各建一个图标动画会话，后建会话 settle
277-- 前一个会话的瞬间会把 provider 的当前进度直接提交到进度条上，表现为进度条
278-- 先被清空、再从 0 涨满、最后又清空。所以多段上涨计划要复用 UBF 过关时已经
279-- 送出的收集物 context，不能再补发第二个 context。
280-- 事务总线回调：暂存 UBF 为本活动 icon 建好的待播收集物 context。
281function Manager:onReceiveTransaction(event)
282    if event:getName() ~= TransactionEvent.FLY_ACT_COLLECTIONS then return end
283
284    local context = event:getData()
285    if tostring(context.iconKey) ~= tostring(self:getActId()) then return end
286    -- 自己补发的计划 context 只负责承载计划，不参与暂存，避免被后续流程复用。
287    if context._layeredTileMatchPlanContext then return end
288    if _isFlyCollectionContextConsumed(context) then return end
289
290    self._pendingEntryTokenFlyContext = context
291end

293-- 取出暂存的收集物 context：只消费一次，已播放/过期的对象直接丢弃。
294function Manager:takePendingEntryTokenFlyContext()
295    local context = self._pendingEntryTokenFlyContext
296    self._pendingEntryTokenFlyContext = nil
297    if _isFlyCollectionContextConsumed(context) then return nil end
298    return context
299end

301-- 通过过关飞收集物事务把多段进度动画交给 Icon V2 播放。
302-- 优先把计划挂到 UBF 已经建好的收集物 context 上，保证一个 icon 只播一个会话；
303-- 没有可复用的 context（例如本次过关没有收集物飞行）时，再补发一个只带计划的 context。
304function Manager:submitEntryTokenProgressAnimationPlan(plan, pendingContext)
305    if not plan then
306        return false
307    end
308
309    pendingContext = pendingContext or self:takePendingEntryTokenFlyContext()
310    if pendingContext then
311        pendingContext:setProgressAnimationPlan(plan)
312        return true
313    end
314
315    local context = FlyActCollectionContext.new(self:getActId())
316    context:setProgressAnimationPlan(plan)
317    -- 标记为活动自己补发的计划 context，事务总线回调据此跳过暂存。
318    context._layeredTileMatchPlanContext = true
319    _G.amfTransaction(TransactionEvent.FLY_ACT_COLLECTIONS, context)
320    return true
321end

566function Manager:addActItem(reward)
567    reward = type(reward) == "table" and reward or {}
568    local itemId = tonumber(reward.itemId)
569    local itemIds = self:getConst().ItemId
570    local countKeyByItemId = {
571        [itemIds.CHALLENGE_TOKEN] = "challengeTokenNum",
572        [itemIds.UNDO] = "undoItemNum",
573        [itemIds.SHUFFLE] = "shuffleItemNum",
574        [itemIds.REMOVE] = "removeItemNum",
575        [itemIds.COMPLETE] = "completeItemNum",
576    }
577    local countKey = countKeyByItemId[itemId]
578    local gainNum = math.max(0, tonumber(reward.num) or 0)
579    if not countKey or gainNum <= 0 then
580        return false
581    end
582
583    local currentNum = math.max(
584        0,
585        tonumber(self:getDataByKey(countKey, 0)) or 0
586    )
587    self:setDataByKey(countKey, currentNum + gainNum, true)
588    if itemId == itemIds.CHALLENGE_TOKEN then
589        self:updateTipView()
590    end
591    return true
592end

599-- UBF 在组装 passLevel 数据时回调，collectionCount 与服务端本局入账口径一致。
600function Manager:updatePassLevelData(
601    data,
602    collectionCount,
603    collectionMap,
604    passLevelData
605)
606    -- UBF 在本流程开头已经送出收集物 context，这里一次性取出：
607    -- 无论本次是否真的提交动画计划，都不再跨流程保留这个引用。
608    local pendingContext = self:takePendingEntryTokenFlyContext()
609    local gainNum = math.max(0, tonumber(collectionCount) or 0)
610    local isPassLevel = data and data.settleMentState
611    if not isPassLevel or gainNum <= 0 then
612        return
613    end
614
615    local oldTokenNum = self:getChallengeTokenNum()
616    local newTokenNum = oldTokenNum + gainNum
617    -- 过关代币可能一次跨过多段，交给 Icon V2 播放多段上涨动画。
618    -- 必须先提交计划（开启 icon 显示延迟），再入账并刷新 provider，
619    -- 否则入账触发的刷新会把新进度提前刷到界面，动画会从新值起跳。
620    local planSubmitted =
621        self:playEntryTokenProgressAnimation(
622            oldTokenNum,
623            newTokenNum,
624            pendingContext
625        )
626    self:addActItem({
627        itemId = self:getConst().ItemId.CHALLENGE_TOKEN,
628        num = gainNum,
629    })
630    -- 只有真的提交了计划（显示延迟已开启）才需要补一次刷新；
631    -- 未跨段时保持原逻辑，交给 Icon V2 默认单段动画。
632    if planSubmitted then
633        self:refreshEntryTokenProgressIcon()
634    end
635end

985function Manager:claimBattlePassReward(id, vip, rewardUniteExtraIds, onOK, onFail)
986    local protocol = self:getConst().Protocol.CLAIM_BATTLE_PASS_REWARD
987    local params = {
988        id = id,
989        vip = vip == true,
990        rewardUniteExtraIds = rewardUniteExtraIds or {},
991    }
992    local plug = self:_getPlug("ActSend")
993    if plug then
994        params = plug:handleRequestParams(protocol, params)
995    end
996
997    self:getModel():requestClaimBattlePassReward(params):next(function(evt)
998        local data = _extractResponseData(evt)
999        if plug then
1000            plug:handleRewardByProtocol(protocol, data)
1001        end
1002        local rewards = data.rewards or {}
1003        if not table.isEmpty(rewards) then
1004            self:addRewardsNormal(rewards, {
1005                dcSource = "feedback_reward",
1006                extraMap = {
1007                    pay_type = vip
1008                        and DcPayType.kRmb
1009                        or DcPayType.kFree,
1010                },
1011            })
1012        end
1013        local challengeTokenNum = tonumber(data.challengeTokenNum)
1014        if challengeTokenNum ~= nil then
1015            self:setDataByKey(
1016                "challengeTokenNum",
1017                math.max(0, challengeTokenNum),
1018                true
1019            )
1020        end
1021        self:updateTipView()
1022        safeRunCall(onOK, data)
1023    end, function(resp)
1024        safeRunCall(onFail, resp)
1025    end)
1026end

1173function Manager:_registerCrossPeriodCallbacks()
1174    local PassDayTimer = require "amf.script.tool.timer.PassDayTimer"
1175    local model = self:getModel()
1176
1177    self:_unregisterCrossPeriodCallbacks()
1178
1179    local mainBegin = model:getMainBeginTimeInSec()
1180    local mainEnd = model:getMainEndTimeInSec()
1181    local actEnd = model:getEndTimeInSec()
1182    local now = TimeKit:timeInSec()
1183    local periodId = model:getPeriodId()
1184
1185    if mainBegin and mainBegin > 0 then
1186        if mainBegin > now then
1187            self._cbMainBegin = function()
1188                if self:getPeriodId() == periodId then
1189                    self:onActPassMainBegin()
1190                end
1191            end
1192            self._cbMainBeginTime = mainBegin
1193            PassDayTimer:registerEndTimeCallback(mainBegin + 1, self._cbMainBegin)
1194        end
1195        if mainEnd and mainEnd > now and mainEnd ~= mainBegin then
1196            self._cbMainEnd = function()
1197                if self:getPeriodId() == periodId then
1198                    self:onActPassMainEnd()
1199                end
1200            end
1201            self._cbMainEndTime = mainEnd
1202            PassDayTimer:registerEndTimeCallback(mainEnd + 1, self._cbMainEnd)
1203        end
1204        if actEnd and actEnd > now and actEnd ~= mainEnd then
1205            self._cbActEnd = function()
1206                if self:getPeriodId() == periodId then
1207                    self:onActPassActEnd()
1208                end
1209            end
1210            self._cbActEndTime = actEnd
1211            PassDayTimer:registerEndTimeCallback(actEnd, self._cbActEnd)
1212        end
1213    end
1214end
1215
1216function Manager:_unregisterCrossPeriodCallbacks()
1217    local PassDayTimer = require "amf.script.tool.timer.PassDayTimer"
1218    if self._cbMainBegin then
1219        PassDayTimer:unregisterEndTimeCallback(self._cbMainBeginTime, self._cbMainBegin)
1220        self._cbMainBegin = nil
1221        self._cbMainBeginTime = nil
1222    end
1223    if self._cbMainEnd then
1224        PassDayTimer:unregisterEndTimeCallback(self._cbMainEndTime, self._cbMainEnd)
1225        self._cbMainEnd = nil
1226        self._cbMainEndTime = nil
1227    end
1228    if self._cbActEnd then
1229        PassDayTimer:unregisterEndTimeCallback(self._cbActEndTime, self._cbActEnd)
1230        self._cbActEnd = nil
1231        self._cbActEndTime = nil
1232    end
1233end
