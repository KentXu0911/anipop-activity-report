167function Manager:onActInfosUpdate(event)
168    local data = event and event:getData() or {}
169    local actInfos = data.actInfos or {}
170    local actInfo = actInfos[tostring(self:getActId())] or actInfos[self:getActId()]
171    if not actInfo then
172        for _, item in pairs(actInfos) do
173            if type(item) == "table" and tonumber(item.actId) == self:getActId() then
174                actInfo = item
175                break
176            end
177        end
178    end
179    if not data.fromServer then
180        return
181    end
182
183    self:syncActInfo(actInfo)
184    if not actInfo then
185        self:updateTipView()
186    end
187end
