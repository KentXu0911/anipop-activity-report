48function Manager:getCurAvailableBuffAndEndTime()
49    local buffExtraRate, buffEndTime = 0, 0
50    if self:getModel():getScoreBuff(1) > TimeKit:timeInSec() then
51        buffExtraRate = 0.2
52        buffEndTime = self:getModel():getScoreBuff(1)*1000
53    end
54
55    if self:getModel():getScoreBuff(2) > TimeKit:timeInSec() then
56        buffExtraRate = buffExtraRate + 0.5
57        buffEndTime = buffEndTime > 0 and math.min(self:getModel():getScoreBuff(1)*1000, self:getModel():getScoreBuff(2)*1000) or self:getModel():getScoreBuff(2)*1000
58    end
59    UWarning("getCurAvailableBuffAndEndTime", buffExtraRate, buffEndTime)

100function Manager:_getAutoPopConfigList()
101    return {
102        {
103            checkPoint = PopupCheckPoint.BLOCK_STAGE4_POP,
104            actId = self:getConst().actId,
105            stages = {AutoPopupConst.PopStage.STAGE_FOUR},
106            check = function()
107                if not self:isActivitySupported(self:getConst().actId) then return false end
108                if not self:isEnabled() then return false end
109                return not shouldReduceAutoPopup(self)
110                    and (self:checkCanPop() == true
111                        or not self:getFlagFromLocalByKey("MainPanelShow", true))
112            end,
113            pop = function(_, onPopOver)
114                return function()
115                    if not self:isEnabled() then
116                        safeRunCall(onPopOver)
117                        return
118                    end
119                    if not self:getFlagFromLocalByKey("MainPanelShow", true) then
120                        self:setFlagToLocalByKey("MainPanelShow", true, true)
121                        self:callActFunc("openActivity", {
122                            failCallback = onPopOver,
123                            endCallback = onPopOver,
124                            extra = {},
125                        })
126                        return
127                    end
128                    if self.showYinliuStage4 then
129                        self:showYinliuStage4(onPopOver)
130                    else
131                        safeRunCall(onPopOver)
132                    end
133                end
134            end,
135        },
136    }

139function Manager:onActInfoUpdate(evt)
140    local data = evt:getData()
141    local fromServer = data.fromServer
142    if not fromServer then return end
143    local actInfos = data.actInfos or {}
144    local actInfo = actInfos[tostring(self:getConst().actId)]
145    if not actInfo then return end
146
147    local extra = self:callActFunc("getActInfoExtraData")
148    if extra then
149        local time = {
150            beginTime = extra.beginTime,
151            mainBeginTime = extra.beginTime,
152            mainEndTime = extra.endTime,
153            endTime = extra.rewardEndTime
154        }
155        local oldMainBeginTime = self:getModel():getMainBeginTime()
156        if math.floor(tonumber(extra.beginTime) / 1000) ~= oldMainBeginTime then
157            self:getModel():clearData()
158            self:getModel():updateTimeInfo(time)
159            self:getModel():setPeriodId(extra.periodId)
160            -- icon倒计时 拿到时间后再刷一次
161            local actIcon = self:callActFunc("getActIcon")
162            if actIcon and not actIcon.isDisposed then
163                actIcon:updateCountDown()
164            end
165        end
166    end
167end

353function Manager:getNowProgressRewards()
354    local bGetFinal = self:getModel():getFinalReward()
355    local rewards
356    local rewardPoint
357    if not bGetFinal then
358        rewardPoint = 2
359        rewards = self:getModel():getFinalRewards()
360    else
361        rewardPoint = 3
362        local bFMJ = _G.amfBean("UserStateManager"):isFMJ()
363        rewards = self:getModel():getProgressRewards(nil, not bFMJ)
364    end
365    local plug = self:_getPlug("ActSend")
366    rewards = plug:replaceRewards(rewards,rewardPoint,self.mapId)
367    local extraRewards = plug:getExtraRewards(rewardPoint,self.mapId)
368    rewards = table.union(rewards,extraRewards)
369    return rewards
370end

431function Manager:checkCanPop()
432    if not self:isEnabled() then return end
433    local yinliuTimes = self:getModel():getLocalCache('yinliuTimes') or 0
434    local count = self:getModel():getLocalCache('yinliuCount') or 0
435    local num = self:getModel():getCollectionNum()
436    if yinliuTimes == 0 then
437        return num >= 30
438    else
439        return yinliuTimes <= 2 and (count >= 20 and num >= 30)
440    end
441end
