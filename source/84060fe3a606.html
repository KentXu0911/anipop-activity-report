127function WaterfallGiftManager:_scheduleIconV2Boundary()
128    self:_cancelIconV2BoundaryTimer()
129    if not self:isV2IconActive() then
130        return true
131    end
132
133    local now = TimeKit and type(TimeKit.timeInSec) == "function" and tonumber(TimeKit:timeInSec()) or nil
134    local endTime = tonumber(self:getEndTime())
135    if not now or not endTime then
136        return false, "WATERFALL_GIFT_ICON_V2_BOUNDARY_TIME_REQUIRED"
137    end
138    if endTime <= now then
139        return self:onDeprecated()
140    end
141
142    local delay = math.max(1, math.floor(endTime - now) + 1)
143    local timerId
144    local timerSource
145    local function onBoundary()
146        if self._iconV2BoundaryTimerId ~= timerId then
147            return
148        end
149        self._iconV2BoundaryTimerId = nil
150        self._iconV2BoundaryTimerSource = nil
151        self:updateIcon()
152    end
153    if type(setTimeOut) == "function" then
154        local ok, result = pcall(setTimeOut, onBoundary, delay)
155        if ok and result ~= nil then
156            timerId = result
157            timerSource = "timeout"
158        end
159    end
160    if not timerId and TimerUtil and type(TimerUtil.addOnce) == "function" then
161        local ok, result = pcall(TimerUtil.addOnce, onBoundary, delay)
162        if ok and result ~= nil then
163            timerId = result
164            timerSource = "timer_util"
165        end
166    end
167    if not timerId then
168        return false, "WATERFALL_GIFT_ICON_V2_BOUNDARY_TIMER_REQUIRED"
169    end
170    self._iconV2BoundaryTimerId = timerId
171    self._iconV2BoundaryTimerSource = timerSource
172    return true
173end

528function WaterfallGiftManager:checkNeedAutoPopout(sceneType)
529    if not self:isEnabled() then return false end
530    local config = dataModel:getAutoPopoutConfig()
531
532    -- 强弹时间
533    local autoPopoutCd = config.autoPopoutCd
534    local lastPopTime = dataModel:getLastPopoutTime()
535    if (TimeKit:timeInSec()-lastPopTime) < autoPopoutCd*3600 then
536        log("checkNeedAutoPopout time in cd", TimeKit:timeInSec(), lastPopTime, autoPopoutCd)
537        return false
538    end
539
540    -- 条件
541    local conditionList = config.conditionList
542    if TimeKit:timeInSec() < 1769356740 then
543        table.insertIfNotExist(conditionList, 3) --强行加场景2，因为配置没配，已触发活动的玩家无法更新配置
544    end
545    local AutoPopDefinition = Meta.AutoPopDefinition
546    local conditionMetaList = config.conditionMetaList
547    log("checkNeedAutoPopout condition check", tts(conditionList), sceneType)
548    for i, conditionId in ipairs(conditionList) do
549        local conditionMeta = conditionMetaList[conditionId] or {}
550        local sceneId = conditionMeta.sceneId
551        if sceneId == sceneType then
552            local value = conditionMeta.conditionValue or 0
553            local id = conditionMeta.conditionId
554            if AutoPopDefinition[id] then
555                local checkFunc = AutoPopDefinition[id]
556                local res = checkFunc(value)
557                if res ~= false then
558                    log("checkNeedAutoPopout condition check pass", res)
559                    return res
560                end
561            end
562        end
563    end
564    return false
565end

734function WaterfallGiftManager:findThenQueryDyOffer()
735    local deferred = require "amf.script.sugar.deferred"
736    local d = deferred.new()
737    local userData = self:getUserData() or {}
738    local actConfig = self:getActConfig() or {}
739    local fixedGiftCfg = actConfig.fixRewardNodeList
740    local curIdx = userData.gotRewardNodeIndex
741    local needQuery = false
742    for i = 1, 6 do
743        local cfg = fixedGiftCfg[curIdx + i]
744        if cfg then
745            if not cfg.freeNode and cfg.goodsId then
746                local goodsObj = GoodsManager:getGoods(cfg.goodsId)
747                if goodsObj then
748                    local price = goodsObj:getPrice()
749                    self:queryDyOffer(price, d)
750                    needQuery = true
751                    break
752                end
753            end
754        else
755            break
756        end
757    end
758    if not needQuery then
759        d:reject()
760    end
761    return d
762end

764function WaterfallGiftManager:queryDyOffer(price, d)
765    local Req = require "amf.middleware.purchase.http.DouyinTiktokPrepayConsult"
766    local req = Req:post({
767        price = price,
768        goodsTagActKey = "waterfall2025",
769    })
770    if not req then
771        d:reject()
772        return
773    end
774
775    req:next(function(response)
776        local data = response:getData()
777        if not data or not data.viewData or not data.viewData.operationTip then
778            d:reject()
779            return
780        end
781
782        local offerInfo = {
783            price = price,
784            operationTip = data.viewData.operationTip,
785            assignDiscountKey = data.assignDiscountKey,
786            goodsTag = data.goodsTag,
787        }
788
789        if data.goodsTag then
790            local tags = table.deserialize(data.goodsTag)
791            if tags and tags.dyType then
792                offerInfo.dyType = tags.dyType
793            end
794        end
795
796        local discount
797        if data.viewData.operationMinAmount then
798            if data.viewData.operationMinAmount == data.viewData.operationMaxAmount then
799                discount = tonumber(data.viewData.operationMinAmount)
800            end
801        else
802            discount = tonumber(string.match(data.viewData.operationTip, "^立减(%d+%.?%d*)元$"))
803        end
804
805        if discount then
806            offerInfo.discount = discount
807            offerInfo.finalPrice = (price * 1000 - discount * 1000) / 1000
808        end
809        d:resolve(offerInfo)
810    end, function()
811        d:reject()
812    end)
813end

847function WaterfallGiftManager:popDyOfferTipPanel()
848    if not self:isDouyinMarketing() then return end

850    local canPop = not self:hasGuideIndex(Meta.DyOfferGuideId.PAY_END_4)
851    if canPop then

853
854        local function onHandleOver()
855            AutoPopEventCenter:dispatch(AutoPopEvent.SINGLE_POP_OVER, PopupCheckPoint.WATERFALL_GIFT_DY_OFFER)
856        end
857
858        local creator = function ()
859            log("stage4popup")
860            queryPromise:next(function(dyOfferInfo)
861                self:showDyOfferTipPanel(dyOfferInfo, onHandleOver)
862            end, onHandleOver)
863
864            setTimeOut(function()
865                queryPromise:reject()
866            end, 1.5)
867        end
868        _G.amfBean("AutoPopupMgr"):addPopupConfig(PopupCheckPoint.WATERFALL_GIFT_DY_OFFER, creator)
869    end
