13local function cloneTable(source)
14    local result = {}
15    for key, value in pairs(source or {}) do
16        result[key] = type(value) == "table" and cloneTable(value) or value
17    end
18    return result
19end

87local function createMapData(mapCount, claimedThrough)
88    local mapMetas = {}
89    local gemMaps = {}
90    for mapId = 0, mapCount - 1 do
91        mapMetas[#mapMetas + 1] = {
92            mapId = mapId,
93        }
94        gemMaps[tostring(mapId)] = {
95            progressReward = mapId <= claimedThrough,
96            pointMap = {},
97        }
98    end
99    return mapMetas, gemMaps
100end
