291function DesktopWidgetManager:requestInstallActGetReward(params, noLoading)
292    if not self:isActEnabled() then return deferred.error() end

404function DesktopWidgetManager:checkPopAct(today)
405    if not self:isActEnabled() or self:checkMainPanelVM() then return end
406    today = today or time2day(TimeKit:timeInSec())
407    local model = self:getModel()
408
409    local guideKey = self:getConst().GuideKey
410    local guideId = self:getConst().GuideId
411
412    local canPop
413    if not self:hasGuideIndex(guideId.TRIGGER)then
414        local trigger = self:checkTrigger()
415        if trigger then
416            self:addGuideIndex(guideId.TRIGGER)
417        else
418            canPop = guideId.TRIGGER
419        end
420    end
421
422    -- 第二、三天未安装且未点击，强弹
423    if not canPop and not self:isWidgetInstall()
424        and not model:checkGuide(guideKey.DAILY_PANEL_OPEN, today) then
425        local triggerDay = time2day(model:getTriggerTime())
426        local delta = today - triggerDay
427        if delta == 1 or delta == 2 and not self:hasGuideIndex(delta + 1) then
428            canPop = delta + 1
429        end
430    end
431
432    -- 当日有奖
433    if not canPop and not model:checkGuide(guideKey.DAILY_PANEL_REWARD, today) then
434        local todayIndex = self:checkHaveEnterFromWidget()
435        if not todayIndex then
436            todayIndex = self._launchFromWidget and (model:getLoginTimeListFromWidgetLength() + 1)
437        end
438        if todayIndex and not table.indexOf(self:getModel():getHasGetRewardList(), todayIndex)
439            and not self:hasGuideIndex(todayIndex + 10) then
440            canPop = todayIndex + 10
441        end
442    end
443    if not canPop then return end
444
445    local function onHandleOver()
446		AutoPopEventCenter:dispatch(AutoPopEvent.SINGLE_POP_OVER, PopupCheckPoint.WIDGET_REWARD_ACT)
447	end
448
449	local function creator()
450        self:addGuideIndex(canPop)
451
452        self:callActFunc("openActivity", {
453            failCallback = onHandleOver,
454            endCallback = onHandleOver,
455            extra = {
456                from = 3
457            },
458            userClick = true,
459        })
460	end
461    self._autoPopupMgr:addPopupConfig(PopupCheckPoint.WIDGET_REWARD_ACT, creator)
462    return true
463end

494function DesktopWidgetManager:tryRequestGetWidgetEncryptionKey()
495    if not self:isEnabled() or self._retryGetTime > 2 then return deferred.error() end
496    if self._getEncryptionKeySuccess then return deferred.error() end
497    self._retryGetTime = self._retryGetTime + 1
498    local d = deferred.new()
499    self:getModel():requestGetWidgetEncryptionKey():next(function()
500        self._getEncryptionKeySuccess = true
501        d:resolve()
502    end, function()
503        d:reject()
504    end)
505    return d
506end
