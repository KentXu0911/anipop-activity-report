157function YYBWashingPackageManager:getEndTime()
158    local actInfo = self:getActInfo()
159    if actInfo then
160        local extraConfig = table.deserialize(actInfo.extra)
161        local endTime = math.ceil2(tonumber(extraConfig.endTime)/1000)
162        local actEndTime = math.ceil2(tonumber(extraConfig.actEndTime)/1000)
163        if endTime == 0 then
164            return actEndTime
165        else
166            return endTime
167        end
168    end
169    return -1
170end
