35local function _normalizeTaskDataMap(userBean, localConfig)
36    userBean = userBean or {}
37    if userBean.taskDataMap ~= nil then return userBean.taskDataMap end
38    local progressMap = userBean.exploreTaskProgressMap or {}
39    local claimedIds = userBean.claimedExploreTaskIds or {}
40    local taskMetas = (localConfig and localConfig.exploreTaskMetas) or {}
41    local out = {}
42
43    for _, meta in ipairs(taskMetas) do
44        if meta and meta.id ~= nil then
45            local taskId = meta.id
46            local current = progressMap[taskId] or progressMap[tostring(taskId)] or 0
47            out[taskId] = {
48                current = current,
49                collections = {},
50                finished = current >= (meta.targetCount or 1),
51                claimed = _contains(claimedIds, taskId),
52            }
53        end
54    end
55
56    for taskId, current in pairs(progressMap) do
57        local k = tonumber(taskId) or taskId
58        if out[k] == nil then
59            out[k] = {
60                current = current or 0,
61                collections = {},
62                finished = false,
63                claimed = _contains(claimedIds, k),
64            }
65        end
66    end
67
68    return out
69end

503function Model:applyRewardHeapTaskProgress()
504    local cfg = self:getLocalConfig() or {}
505    local Const = self:getConst()
506    local actionType = Const and Const.ExploreTaskActionType and Const.ExploreTaskActionType.REWARD_HEAP_ALL_REWARDED
507    if not actionType then return end
508    local current = self:getStoneRewardHeapClaimedAllCount()
509    for _, task in ipairs(cfg.exploreTaskMetas or {}) do
510        if task and task.actionType == actionType then
511            -- 服务端不回 taskData 时，本地用当前已全领石台数同步 REWARD_HEAP_ALL_REWARDED 任务。
512            local taskId = task.id
513            local td = self:getTaskData(taskId) or {}
514            local targetCount = tonumber(task.targetCount) or 0
515            local progress = current
516            if targetCount > 0 and progress > targetCount then progress = targetCount end
517            self:_mergeOneTaskData(taskId, {
518                current = progress,
519                finished = targetCount > 0 and progress >= targetCount or td.finished,
520                claimed = td.claimed,
521                collections = td.collections,
522            })
523        end
524    end
525end

1196        local k = tonumber(itemId) or itemId

1198        local sk = tostring(k)
1199        if v > 0 then
1200            bag[k] = v
1201            if sk ~= k then
1202                bag[sk] = nil
1203            end
1204        else
1205            bag[k] = nil
1206            if sk ~= k then
1207                bag[sk] = nil

1209        end
1210    end
