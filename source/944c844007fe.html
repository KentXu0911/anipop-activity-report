54function ActInfosModel:onResponseGetActInfos2024(response)

56    if response:isSuccess() then
57        local data = response:getData()
58        local actInfos = data.actInfos
59        local distributionData = {}
60        for k, v in ipairs(actInfos) do
61            distributionData[tostring(v.actId)] = v
62        end
63        self:setValue("_actInfos",distributionData)
64        self:setValue("_activityParticipationTags", data.activityParticipationTags)
65
66        ActivityEventCenter:dispatch(ActivityEvent.ACT_INFOS_UPDATE, {
67            actInfos = distributionData,
68            activityParticipationTags = data.activityParticipationTags,
69            -- 从服务器更新的
70            fromServer = true,
71        })
72    end
73end
