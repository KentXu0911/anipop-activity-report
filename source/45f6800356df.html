171-- 限流归属按 popupCheckPoint 对应的 PopType 计算；popupCheckPoint 同时是队列寻址键。
172function AutoPopupMgr:getNextPopupEntryAcrossPopTypes(popStage)
173	local normalizedPopStage = AutoPopupConst.PopStageKeyMapping[tostring(popStage)] or popStage
174	local orderedCheckPoints = self._popupSchedule:getOrderedCheckPoints(normalizedPopStage)
175	local checkList = self:getPopupCheckList(normalizedPopStage) or {}
176	local listed = {}
177	for _, popType in ipairs(checkList) do listed[popType] = true end
178
179	for _, popupCheckPoint in ipairs(orderedCheckPoints) do
180		local popType = self:getPopType(popupCheckPoint)
181		local allowed = listed[popType] ~= false
182		-- popType 级限流命中时，该类型下所有候选都不参与本轮竞争，逐条记录原因。
183		local popTypeFailReason
184		if allowed then
185			if self:isLimitedByStageChange(normalizedPopStage, popType) then
186				popTypeFailReason = LogReason.Enqueue.STAGE_CHANGE_LIMIT
187			elseif self:isLimitedBySameStage(normalizedPopStage, popType) then
188				popTypeFailReason = LogReason.Enqueue.SAME_STAGE_LIMIT
189			elseif self:isLimitedByDailyPopType(normalizedPopStage, popType) then
190				popTypeFailReason = LogReason.Enqueue.POP_TYPE_DAILY_LIMIT
191			elseif self:isLimitedByPopType(normalizedPopStage, popType) then
192				popTypeFailReason = LogReason.Enqueue.POP_TYPE_ROUND_LIMIT
193			end
194		end
195		local popupEntry = self:getPopupQueueByPopType(popType)[popupCheckPoint]
196		if popTypeFailReason then
197			if popupEntry then
198				AutoPopupDebugLog:recordPopupEnqueue(
199					popupEntry, "fail", popTypeFailReason, popupCheckPoint
200				)
201			end
202		elseif popupEntry then
203			local failedReason
204			if self:isLimitedBySpecifiedPopStages(popupCheckPoint, normalizedPopStage) then
205				failedReason = LogReason.Enqueue.SPECIFIED_POP_STAGES
206			elseif self:isLimitedByStageOnly(popupCheckPoint, normalizedPopStage) then
207				failedReason = LogReason.Enqueue.STAGE_ONLY
208			elseif self:isCheckPointBlockedByRule(popupCheckPoint, normalizedPopStage) then
209				failedReason = LogReason.Enqueue.CHECK_POINT_BLOCK_RULE
210			end
211			if not failedReason then
212				AutoPopupDebugLog:recordPopupEnqueue(
213					popupEntry, "success", nil, popupCheckPoint
214				)
215				return {
216					checkPoint = popupCheckPoint,
217					popupEntry = popupEntry,
218				}
219			end
220			AutoPopupDebugLog:recordPopupEnqueue(
221				popupEntry, "fail", failedReason, popupCheckPoint
222			)
223		end
224	end
225	return nil
226end

794local function addPopupEntry(autoPopupMgr, popupCheckPoint, popupCreator, actId, source)
795	local popupQueue = autoPopupMgr:getPopupQueueByCheckPoint(popupCheckPoint)
796	assert(popupQueue, "unknown popupCheckPoint: " .. tostring(popupCheckPoint))
797	local popupEntry = {
798		checkPoint = popupCheckPoint,
799		creator = popupCreator,
800		actId = actId,
801		source = source or "config",
802	}
803	popupQueue[popupCheckPoint] = popupEntry
804	autoPopupMgr:_recordDynamicPopupScheduleCandidate(popupEntry)
805	AutoPopupDebugLog:recordPopupEnqueue(popupEntry, "success", nil, popupCheckPoint)
806end
807
808--- 入队一个强弹候选。
809-- popupCheckPoint 同时是身份、业务规则和队列槽位，所以入队只需要一个 key。
810-- @param popupCheckPoint see _G.PopupCheckPoint in AutoPopupCheckPoints.lua
811--@param popupCreator
812-- @param actId 归属活动；用于日志与“注册式优先”判定
813-- @param source "config"（配置中心/事件驱动，默认）或 "registered"（声明式注册）
814function AutoPopupMgr:addPopupConfig(popupCheckPoint, popupCreator, actId, source)
815	source = source or "config"
816	-- 配置式候选只要进入本次场景的评估，就要有调度位置：即使随后
817	-- 被限流或强弹规则挡在队列外，日志也要按真实调度位置排序。
818	self:_recordDynamicPopupScheduleCandidate({
819		checkPoint = popupCheckPoint,
820		actId = actId,
821		source = source,
822	})
823	local function reject(reason)
824		AutoPopupDebugLog:recordPopupEnqueue({
825			checkPoint = popupCheckPoint,
826			actId = actId,
827			source = source,
828		}, "fail", reason, popupCheckPoint)
829		return false
830	end
831
832	if popupCheckPoint == nil then
833		return reject(LogReason.Enqueue.REJECTED)
834	end
835	if not self:isEnabled() then
836		return reject(LogReason.Enqueue.SYSTEM_DISABLED)
837	end
838
839	if self:checkPointLimitedCheck(popupCheckPoint) then
840		AutoPopupDebugLog:log("AutoPopupMgr:addPopupConfig ==> cut off by num limit", popupCheckPoint)
841		return reject(LogReason.Enqueue.CHECK_POINT_LIMIT_NUM)
842	end
843
844	-- 同一活动一旦拥有声明式强弹，配置中心的同名活动强弹只保留注册式入口。
845	if source == "config" then
846		local hasRegisteredPopup = actId and self:hasRegisteredPopupForActivity(actId)
847		if not hasRegisteredPopup then
848			hasRegisteredPopup = self._popupRegistration:hasRegisteredPopupForCheckpoint(popupCheckPoint)
849		end
850		if hasRegisteredPopup then
851			AutoPopupDebugLog:log(
852				"AutoPopupMgr:addPopupConfig skipped because registered popup exists",
853				actId, popupCheckPoint
854			)
855			return reject(LogReason.Enqueue.REJECTED)
856		end
857	end
858
859	AutoPopupDebugLog:log("AutoPopupMgr:addPopupConfig", popupCheckPoint, source)
860	addPopupEntry(self, popupCheckPoint, popupCreator, actId, source)
861	self:invalidateInterruptibleEntries(popupCheckPoint)
862	self:_markPopupScheduleDirty()
863	return true
864end

943

945		return self:doPopup()

956		return

961		AutoPopupDebugLog:finishPopupScene("skipped", LogReason.Scene.GUIDE_ON_SCREEN)

966		return self:doPopup()
967	end

969	self.stage1WaitingPanelsClose = true
970
971	AutoPopupDebugLog:log("AutoPopupMgr:handlePopup ==> waiting panels close")
972	AutoPopEventCenter:registerGlobal(NormalEvent.NO_PANEL_ON_SCENE, self.onWaitPanelClose, self)

1050-- 场景 3 全部强弹登记完成后准备本轮资格。业务可声明允许共存的预期流程。
1051function AutoPopupMgr:prepareInterruptibleEntries(popupCheckPoint, context)
1052	local hasPopup = self:hasPopupConfig(popupCheckPoint)
1053	local hasMatchingEntry = false
1054	local hasEnabledEntry = false
1055	local blockedByOtherPopup
1056	local allowedRegisteredPopup
1057	for name, entry in pairs(self._interruptibleEntries) do
1058		local isMatching = entry.config.popupCheckPoint == popupCheckPoint
1059		local isEnabled = isMatching and entry.config.isEnabled(context)
1060		local allowedPopupCheckPoint = isMatching
1061			and self:getAllowedPopupConfig(entry.config.allowedPopupCheckPoints)
1062		local otherPopupCheckPoint = isMatching
1063			and self:getOtherPopupConfig(
1064				popupCheckPoint,
1065				entry.config.allowedPopupCheckPoints
1066			)
1067		local queueExclusive = hasPopup and not otherPopupCheckPoint
1068		entry.candidate = queueExclusive
1069			and isMatching
1070			and isEnabled
1071		entry.window = false
1072		entry.requested = false
1073		if isMatching then
1074			hasMatchingEntry = true
1075			blockedByOtherPopup = blockedByOtherPopup or otherPopupCheckPoint
1076			allowedRegisteredPopup = allowedRegisteredPopup or allowedPopupCheckPoint
1077		end
1078		if entry.candidate then hasEnabledEntry = true end
1079		AutoPopupDebugLog:log(
1080			"InterruptibleEntry:prepare",
1081			"name=" .. tostring(name),
1082			"candidate=" .. tostring(entry.candidate),
1083			"enabled=" .. tostring(isEnabled),
1084			"otherPopup=" .. tostring(otherPopupCheckPoint),
1085			"allowedPopup=" .. tostring(allowedPopupCheckPoint)
1086		)
1087	end
1088
1089	local reason = allowedRegisteredPopup
1090		and "allowed_popup_registered:" .. tostring(allowedRegisteredPopup)
1091		or "allowed"
1092	if not hasPopup then
1093		reason = "popup_not_registered"

1190		return false
1191	end

1193		AutoPopupDebugLog:log("InterruptibleEntry:requestDenied", "name=" .. tostring(name), "reason=window_closed")
1194		return false
1195	end

1197		AutoPopupDebugLog:log("InterruptibleEntry:requestAccepted", "name=" .. tostring(name), "reason=already_requested")
1198		return true
1199	end
1200
1201	entry.requested = true
1202	AutoPopupDebugLog:log("InterruptibleEntry:requestAccepted", "name=" .. tostring(name))
1203	entry.config.onInterrupt()
1204	return true
