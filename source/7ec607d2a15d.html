18function LevelAreaConfigParser:loadConfigForMainLevel()
19    local result = {}
20    local localConfigMaxLevel = 0
21    MetaConfigUtil:traverseConfigThenRelease("level_area", function(_, value)
22        local singleConfig = {
23            id = 0,
24            maxLevel = 0,
25            minLevel = 0,
26            star = 0,
27            unlockTaskLevelId = nil,
28            starList = {},
29        }
30        for singleConfigKey, singleConfigValue in pairs(value) do
31            if singleConfigKey == "starAreaABTestStr" then
32                for i,v2 in pairs(string.split(singleConfigValue,",")) do
33                    singleConfig.starList[i] = tonumber(v2)
34                end
35            elseif type(singleConfigKey) == "string" then
36                singleConfig[singleConfigKey] = tonumber(singleConfigValue)
37            end
38        end
39
40        if singleConfig.maxLevel > localConfigMaxLevel and singleConfig.maxLevel < _G.levelUnlockConst.ImpossibleMaxLevel then
41            localConfigMaxLevel = singleConfig.maxLevel
42        end
43
44        result[tonumber(singleConfig.id)] = singleConfig
45    end)
46    self.localConfigMaxLevel = localConfigMaxLevel
47    self.levelAreaConfig = result
48end
