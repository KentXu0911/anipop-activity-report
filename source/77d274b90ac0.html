675function BoardMeta:posTurnToRowCol( pos )
676	--位置转换行列
677	local col = pos%BoardMeta.BoardWidthNum

679

685function BoardMeta:rowColTurnToPos( row, col )
686	--行列转换 pos
687	if row <= 0 or col <= 0 then
688		return -1
689	end
690	if row > BoardMeta.BoardHeightNum or col > BoardMeta.BoardWidthNum then
691		return -1
692	end
693
694	local pos = (row-1)*BoardMeta.BoardWidthNum + col
695	return pos
696end
