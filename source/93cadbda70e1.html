189function PromoteLowMgr:requestReward(param)
190    local d = deferred.new()
191    self.model:requestReward(param):next(function(response)
192        local rewards = self:getValue("rewardList") or {}
193        if not table.isEmpty(rewards) then
194            GainPropData:create()
195                :setFeature(PromoteLowConst.DC_FEATURE)
196                :setSource(PromoteLowConst.DC_SOURCE)
197                :setProps(rewards)
198                :execute()
199        end 
200        self:setValue("rewarded", true) 
201        -- 刷新icon
202        self:removeActButton()
203        d:resolve(rewards)
204    end, function(response)
205        if not response:isCancel() then
206            self:showErrorTip(response)
207        end
208        local errCode = response:getErrorCode()
209        if tonumber(errCode) == 737139 then
210            self.model:setTempData("forceDisable", true)
211            self:removeActButton()
212        end
213        d:reject()
214    end)
215
216    return d
217end

352-- 是否可以强弹场景3/4
353function PromoteLowMgr:canAutoPopSignPanel()
354    local enable = self:isEnabled()
355    self:log("PromoteLowMgr: 开始检查强弹", enable, self:isModelEnabled())
356    if not enable then return false end
357    if not self:isModelEnabled() then return false end
358    local rewarded = self:getValue("rewarded")
359    -- 判断是否已经领奖了
360    if rewarded then
361        self:log("PromoteLowMgr: 已经领奖了 不弹")
362        return false
363    end
364    local curDayNum = TimeUtil:getDayIdxSince1970()
365    -- 配置成了一天弹一次，就不用自己存了， config那边突然不会计数了
366    if self:getLocalValue("autoPopPanel_"..curDayNum) then
367        self:log("PromoteLowMgr: 已经弹过了 不弹")
368        return false
369    end
370    -- 如果可以领奖，但是没网就不弹了
371    if self:isCanReward() then
372        if not self._networkStatusManager:isNetworkEnabled() then
373            self:log("PromoteLowMgr: 可以领奖没网 不弹")
374            return false
375        end
376    elseif not self:isTodaySigned() then
377        self:log("PromoteLowMgr: 没盖章 不弹")
378        return false
379    end
380    return true
381end

446--分数buff
447function PromoteLowMgr:onActUBFAddScoreBuff(event)
448    local eventData = event:getData()
449    if eventData.isAskForHelp then return false end
450    local scoreBuffRate, buffEndTime = self:getCurAvailableBuffAndEndTime()
451    local hasScoreBuff = buffEndTime > TimeKit:timeInMillis()
452    if hasScoreBuff then
453        local builder = UBFPlayImproveBuffBuilder:create()
454        builder:setModSourceId(PromoteLowConst.ACT_ID)
455        builder:addBuffWithOld(InitBuffType.CUSTOM_SCORE_BUFF, InitBuffCreateType.BUFF_ACTIVITY, scoreBuffRate - 1)
456        table.insert(eventData.buffList, builder:toBuffCreateData() )
457    end
458end
