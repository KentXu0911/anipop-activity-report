31-- 解析主 info：处理换期清理、本地配置、商品包注册与 actInfo 同步。
32function Model:parseMainInfo(data)
33    local dataBean = data.dataBean or {}
34    local serverVersionedConfig = data.serverVersionedConfig or {}
35    local localConfig = self:getValue("localConfig") or {}
36    local oldPeriodId = self:getValue("data.periodId")
37
38    if oldPeriodId and dataBean.periodId and tostring(oldPeriodId) ~= tostring(dataBean.periodId) then
39        self:setValue("data", {})
40        self:setValue("localCache", {})
41    end
42
43    if serverVersionedConfig.config and serverVersionedConfig.config ~= "" then
44        localConfig = table.deserialize(serverVersionedConfig.config) or {}
45    end
46
47    localConfig.serverMd5 = serverVersionedConfig.serverMd5 or self:getValue("localConfig.serverMd5") or ""
48    dataBean.buyPackageMap = dataBean.buyPackageMap or {}
49    self:parsePackagesPullResponse(data.packagesPullResponse)
50    data.packagesPullResponse = nil
51    ActInfo:syncMainInfo(dataBean)
52
53    self:setValue("data", dataBean)
54    self:setValue("localConfig", localConfig)
55    self:setValue("actTimeCfg.beginTime", dataBean.beginTime or 0)
56    self:setValue("actTimeCfg.mainEndTime", dataBean.mainEndTime or 0)
57    self:setValue("actTimeCfg.endTime", dataBean.endTime or 0)
58    self:setValue("data.fromServer", true)
59end

61-- 解析主 info 携带的 packagesPullResponse，仅注册到 GoodsManager，不落本地存储。
62function Model:parsePackagesPullResponse(packagesPullResponse)
63    if not packagesPullResponse then return end
64    for packageId, goods in pairs(packagesPullResponse.packageMetas or {}) do
65        GoodsManager:registerGoods({
66            goodsId = packageId,
67            packageMeta = goods.packageMeta,
68            productId = goods.productId,
69        })
70    end
71end
