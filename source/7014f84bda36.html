446function Manager:syncIconV2QuerySchedule(icon)
447    if not icon or icon.isDisposed or type(icon.refreshProvider) ~= "function" then
448        return
449    end
450
451    local phase = self:_getIconV2ActivityPhase()
452    if not phase or phase == ICON_V2_PHASE_PREHEAT then
453        self:_resetIconV2QueryFlow()
454        return
455    end
456
457    local model = self:getModel()
458    if not model:isIconV2PKDataReady() then
459        self:_resetIconV2QueryFlow()
460        return
461    end
462
463    local pkState = model:getPKStateSnapshot()
464    local pkStateConst = self:getConst().PKState
465    local matchEndTime = toSeconds(self:getDataByKey("matchRefBean.matchInfo.endTime")) + 1
466    if self._iconV2SettlementPending then
467        local sameMatch = self._iconV2PendingMatchEndTime == matchEndTime
468        local pendingState = pkState == pkStateConst.INGAME
469            or pkState == pkStateConst.TO_BE_SETTLED_1
470            or pkState == pkStateConst.TO_BE_SETTLED_2
471        if sameMatch and pendingState then
472            return
473        end
474        self:_resetIconV2QueryFlow()
475    end
476    if phase ~= ICON_V2_PHASE_MAIN and phase ~= ICON_V2_PHASE_REWARD then
477        self:_resetIconV2QueryFlow()
478        return
479    end
480    if pkState ~= pkStateConst.INGAME then
481        self:_resetIconV2QueryFlow()
482        return
483    end
484
485    if matchEndTime <= 1 then
486        return
487    end
488    if self._iconV2ActiveMatchEndTime ~= matchEndTime then
489        self:_activateIconV2QueryFlow(matchEndTime)
490    end
491    local generation = self._iconV2QueryGeneration
492    if self._iconV2QueryExhaustedEndTime == matchEndTime
493        or self._iconV2QueryInFlightGeneration == generation
494        or self._iconV2QueryRetryTimer then
495        return
496    end
497    if self._iconV2QueryEndTimer and self._iconV2ScheduledEndTime == matchEndTime then
498        return
499    end
500
501    self:_cancelIconV2EndTimer()
502    self._iconV2QueryRetryCount = 0
503    self._iconV2ScheduledEndTime = matchEndTime
504    self._iconV2QueryEndTimer = TimerUtil.addAlarm(function()
505        self:_onIconV2MatchEnded(matchEndTime, generation)
506    end, math.max(0, matchEndTime - TimeKit:timeInSec()), 1)
507end

755function Manager:checkTipShow(stage)
756    if not self:isEnabled() then return false end
757
758    local PopReason = self:getConst().PopReason
759    local state = self:getAndRefreshPKState()
760    local TipKey = self:getConst().TipKey
761
762    local pkState = self:getConst().PKState
763
764    -- 对战中
765    if state == pkState.INGAME then
766        if stage == "STAGE_TWO" then
767            local lastMyScore, lastEnemyScore = self:getModel():getLastScore()
768            local myScore, enemyScore = self:getMatchScore()
769
770            if myScore ~= lastMyScore or enemyScore ~= lastEnemyScore then
771                self:getModel():setLastScore(myScore, enemyScore)
772                return PopReason.COLLECTION, {
773                    myScore = myScore,
774                    enemyScore = enemyScore
775                }
776            end
777        end
778        if stage == "STAGE_THREE" then
779            -- 过关失败
780            local popValue1 = self:getLocalCache(TipKey.CONTINUE_REWARD_ROUND) or 0
781            local popValue2 = self:getLocalCache(TipKey.DOUBLE) or 0
782            if self:checkIsContinueRewardRound() and self:isPassLevelFail() and popValue1 < 2 then
783                return PopReason.CONTINUE_REWARD_ROUND, {
784                    key = TipKey.CONTINUE_REWARD_ROUND
785                }
786            elseif self:isInDoubleState() and popValue2 < 1 then
787                return PopReason.DOUBLE, {
788                    key = TipKey.DOUBLE
789                }
790            end
791        end
792        -- 准备阶段
793    elseif state == pkState.PREPARE and stage == "STAGE_THREE" then
794        local canBeInvited = self:getModel():checkCanBeInvited()
795        local popValue1 = self:getLocalCache(TipKey.BE_INVITED) or 0
796
797        if canBeInvited and popValue1 < 2 then
798            return PopReason.BE_INVITED, {
799                key = TipKey.BE_INVITED,
800                needTime = true
801            }
802        end
803
804        local popValue1 = self:getLocalCache(TipKey.CONTINUE_REWARD_ROUND) or 0
805        if self:checkIsContinueRewardRound() and self:checkHasLeftTimes() and popValue1 < 2 then
806            return PopReason.CONTINUE_REWARD_ROUND, {
807                key = TipKey.CONTINUE_REWARD_ROUND
808            }
809        end
810
811        local popValue2 = self:getLocalCache(TipKey.ENERGY) or 0
812        -- 参与立刻得精力！
813        if self:checkGetExtraPower() and popValue2 < 2 then
814            return PopReason.ENERGY, {
815                key = TipKey.ENERGY
816            } -- 不需要key，满足条件一直弹出
817        end
818    elseif state == pkState.PREPARE and stage == "STAGE_ONE" then
819        local popValue = self:getLocalCache(TipKey.ENERGY) or 0
820        -- 参与立刻得精力！
821        if self:checkGetExtraPower() and popValue < 2 then
822            return PopReason.ENERGY, {
823                key = TipKey.ENERGY
824            } -- 不需要key，满足条件一直弹出
825        end
826    end
827
828    return false
829end

1902-- 开始面板下方吊牌数据
1903function Manager:getSlpBottomData(params)
1904    local isPlp = params.isPlp
1905    local level_bank, matchEndTime
1906    local canShowMatchButton = self:isCanShowMatchButton() and not isPlp
1907    local myScore, enemyScore = self:getMatchScore()
1908    local isRetry = params.isRetry
1909    local lastPassLevelData = self:getLastPassLevelData()
1910    if isPlp and not lastPassLevelData then
1911        return
1912    end
1913
1914    if UBFLevelSceneFunc:isFullLevelLib(params.levelScene) then
1915        level_bank = params.fsllInfo and params.fsllInfo.levelIndex or 0
1916    end
1917    if not canShowMatchButton then
1918        matchEndTime = self:getMatchEndTime()
1919        if matchEndTime > 0  and matchEndTime < 10000000000 then
1920            -- 认为是秒数，转成毫秒
1921            matchEndTime = matchEndTime * 1000
1922        end
1923    end
1924
1925    local slpBottomData = {
1926        canShowMatchButton = canShowMatchButton,
1927        matchEndTime = matchEndTime,
1928        enemyScore = enemyScore,
1929        myScore = myScore,
1930        isPlp = params.isPlp,
1931        lastPassLevelData = lastPassLevelData,
1932        isFirstPlay = UBFActHelper:ubfCheckIsFirstPlay(params.levelId, params.topLevelFailCount, params.levelScene),
1933        levelSupportType = UBFActHelper:getPlayLevelModeType(params.levelId, nil, params.levelScene),
1934        isInDoubleState = self:isInDoubleState(),
1935        isGrandPrizeRound = self:checkIsContinueRewardRound(),
1936        isDoubleTimeFirstEnterSlp = self:checkFirstMatchDouble() and not isPlp,
1937        isWinningStreakGroup = self:isWinningStreakGroup(),
1938        winningStreakCount = isRetry and 0 or self:getWinningStreakCount(),
1939        enemyProfile = self:getEnemyProfile(),
1940        difficultyFlag = params.difficultyFlag,
1941        myProfile = _G.amfBean("ProfileManager"):getProfile(),
1942        onClickViewDetails = function ()
1943            self:createDescPanel()
1944        end,
1945        showRecordTip = function (target, callback)
1946            return self:createEnemyTip(target, callback)
1947        end,
1948        requestJoin = function (successCallback, failCallback)
1949            self:requestJoin(successCallback, failCallback, 2)
1950        end,
1951        enterViewCallBack = function ()
1952            if canShowMatchButton then
1953                self:setStartDownPartHignPriorityShowedToday(true)
1954            end
1955        end,
1956        -- 打点相关数据
1957        dcParams = {
1958            start_show = {
1959                t6 = params.sourceId,
1960                level_bank = level_bank
1961            },
1962        }
1963    }
1964    if LevelSceneFunc:isFullLevelLib(params.levelScene) then
1965        slpBottomData.isFirstPlay = UBFActHelper:checkIsFsllFirstPlay(params.levelId, params.levelScene)
1966    end
1967
1968    return slpBottomData
1969end

1971function Manager:onActBeforeStartLevel(event)
1972    if not self:isEnabled() then return end
1973    local data = event:getData()
1974    local sourceId = UBFActivityIdConst.kMagicRush
1975    local oldStar, oldScore = UBFActHelper:getOldStarAndScore(data.levelId)
1976    local runtime = {
1977        sourceId = sourceId,
1978        levelId = data.levelId,
1979        levelType = data.levelType,
1980        startLevelType = UBFStartLevelType.kCommon,
1981        oldScore = oldScore,
1982        oldStar = oldStar,
1983        levelScene = data.levelScene or LevelScene.kDefault,
1984    }
1985    if not UBFActHelper:checkActOpenCondition(runtime) then return end
1986    data.jsonTable[tostring(sourceId)] = {isMagic1v1Stage = true}
1987end

1989function Manager:onActCrashResumeFailed(event)
1990    local data = event:getData()
1991    if not data or not data.resumeData or not data.levelId then return end
1992    if not self:isEnabled() then return end
1993    local resumeData = data.resumeData
1994    if resumeData:hasGameMod(UBFActivityIdConst.kMagicRush) then
1995        if data.json then
1996            data.json[tostring(UBFActivityIdConst.kMagicRush)] = {isMagic1v1Stage = true}
1997        end
1998    end
1999end
