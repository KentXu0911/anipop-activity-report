121function Manager:onActInfoUpdate(evt)

124    if not fromServer then return end
125    local actInfos = data.actInfos or {}
126    local actInfo = actInfos[tostring(self:getConst().actId)]
127    if not actInfo then
128        self:clearData()
129        return
130    end
131
132    local extra = self:callActFunc("getActInfoExtraData")
133    if extra then
134        local beginTime = extra.periodMainBeginTs
135        local time = {
136            beginTime = beginTime,
137            mainBeginTime = beginTime,
138            mainEndTime = extra.periodMainEndTs,
139            endTime = extra.periodPostEndTs
140        }
141        local oldMainBeginTime = self:getModel():getMainBeginTime()
142        if math.floor(tonumber(beginTime) / 1000) ~= oldMainBeginTime then
143            self:getModel():clearData()

145            self:getModel():setPeriodId(extra.periodId)
146            local actIcon = self:callActFunc("getActIcon")
147            if actIcon and not actIcon.isDisposed then
148                actIcon:updateCountDown()
149            end
150        end

153        if extra.latestFlowcNotShowTodayTs then
154            if TimerUtil:isSameDay(TimeKit:timeInSec(),tonumber(extra.latestFlowcNotShowTodayTs)) then
155                self:getModel():setNotShowToday(true)

568function Manager:getEnterLevelCollectBaseCount(levelId, levelScene, payTag)
569    local baseDropCount = 0
570    local collectionConfig = UBFActHelper:getSpecificCollectionIdConfig(3080, "185")
571    local collectionBaseCountConfig = collectionConfig['baseDropRule'] and collectionConfig['baseDropRule']['collectionBaseCountConfig']
572    if collectionBaseCountConfig then
573        local function getCollectBaseCount(config, countConfig, currTravelIndex)
574            if not countConfig then return 0 end
575            local baseCount = 0
576            local configData = string.split(countConfig, ",")
577            -- 按照固定值取
578            if #configData == 1 then
579                baseCount = tonumber(configData[1])
580             -- 按照付费标签取       
581            elseif #configData == 7 then
582                payTag = payTag or 1
583                baseCount = tonumber(configData[payTag])
584            else
585                UBFThrow("收集物基础数量配置格式有误：" .. tts(countConfig))
586            end
587            return tonumber(baseCount)
588        end
589        local maxLevelId = _G.amfBean("LevelUnLockManager"):getUserMaxLevelId()
590        local levelId = UBFActHelper:getHiddenLevelStartNormalLevel(levelId, levelScene)
591        local levelSupportType = UBFActHelper:getPlayLevelModeType(levelId, nil, levelScene)
592        for i, config in ipairs(collectionBaseCountConfig or {}) do
593            local startLevel = UBFActHelper:getDropConfigLevelId(config.startLevel, maxLevelId)
594            local endLevel = UBFActHelper:getDropConfigLevelId(config.endLevel, maxLevelId)
595            if levelId >= startLevel and levelId <= endLevel then
596                if LevelSceneFunc:isFullLevelLib(levelScene) then
597                    baseDropCount = getCollectBaseCount(config, config.fsllLevelDrop)
598                elseif levelSupportType == LevelSupportModeType.kNormal then
599                   baseDropCount = getCollectBaseCount(config, config.topDrop)
600                elseif levelSupportType == LevelSupportModeType.kExceedStar then
601                    baseDropCount = getCollectBaseCount(config, config.refreshStarDrop)
602                elseif levelSupportType == LevelSupportModeType.kExceedScore then
603                    baseDropCount = getCollectBaseCount(config, config.refreshScoreDrop)
604                end
605            end
606        end
607    end
608
609    return baseDropCount
610end
