79function WaterfallPanelVM:getProgressNodeInfo()
80	local progressNodeConfig = self.mgr:getProgressNodeConfig()
81	local progressNodeId = self.mgr:getProgressNodeId()
82	local progressTotalScore = self.mgr:getProgressTotalScore()
83
84	local lastNodeTotalScore = 0
85	local startValue, endValue, curValue = 0, 0, 0
86	local rewardList
87	local canGetId = 0
88	for i, cfg in ipairs(progressNodeConfig) do
89		if progressTotalScore < cfg.nodeTotalScore or progressNodeId < i then
90			startValue = lastNodeTotalScore
91			endValue = cfg.nodeTotalScore - lastNodeTotalScore
92			curValue = progressTotalScore - lastNodeTotalScore
93			rewardList = cfg.rewardList
94			if progressTotalScore >= cfg.nodeTotalScore then
95				canGetId = cfg.nodeId
96			end
97			break
98		else
99			lastNodeTotalScore = cfg.nodeTotalScore
100			canGetId = cfg.nodeId
101		end
102	end
103	printx(33, "getProgressNodeInfo", progressNodeId, canGetId, #progressNodeConfig)
104	return {
105		startValue = startValue,
106		endValue = endValue,
107		curValue = curValue,
108		progressNodeId = progressNodeId,
109		canGetId = canGetId,
110		rewardList = rewardList,
111		isFinal = (canGetId+1) == #progressNodeConfig,
112	}
113end

124function WaterfallPanelVM:tryGetProgressReward(callback)
125	local progressNodeInfo = self:getProgressNodeInfo()
126	local canGetId = progressNodeInfo.canGetId
127	local progressNodeId = self.mgr:getProgressNodeId()
128	printx(33, "tryGetProgressReward", canGetId, progressNodeId)
129	if canGetId <= progressNodeId then
130		safeRunCall(callback)
131		return
132	end
133
134	deferred.success()
135	:next(function()
136		if self.isDisposed then return end
137		return self:requestGetProgressRewards(progressNodeId+1)
138	end)
139	:next(function(rewards)
140		if self.isDisposed then return end
141		if self.mgr:isFinishAll() then
142			return self:showFinalRewardPanel(rewards)
143		else
144			return self:flyProgressReward(rewards, progressNodeInfo.isFinal):next(function()
145				if self.isDisposed then return end
146				local progressInfo = self:getProgressNodeInfo()
147				self:callViewFunc("setProgressBarZero", progressInfo)
148				self:callViewFunc("setProgressBarRewardList", progressInfo)
149				return self:playProgressBarAddAnim(progressInfo)
150			end)			
151		end
152	end)
153	:next(function()
154		if self.mgr:isFinishAll() then
155			self.mgr:onDeprecated()
156		end
157		safeRunCall(callback)
158	end, function()
159		safeRunCall(callback)
160	end)
161end

483function WaterfallPanelVM:doUnifiedBuy(goodsId, onSuccess, onFail, onCancel)
484    local function successCallback(payInfo)
485		local rewards = payInfo and payInfo.serverRewards or {}
486        if onSuccess then

488        end
489
490		self.mgr:onPaySuccess(payInfo.payType)
491    end
492
493	local stageType = GiftPackageConst.StageType.ACTIVITY
494	local moduleName = self.mgr:getGiftModuleName()
495    local buyContext = BuyContext.new(goodsId, MoneyType.kRmb)
496                                 :setFeatureAndSource(self.feature, self.source)
497                                 :setUnifiedBuy(true, stageType, moduleName)
498                                 :setCallback(successCallback, onFail, onCancel)
499
500	if self.isDyOfferActOpen then
501		buyContext:setPriorityPayType(Payments.DOUYIN_ANDROID)
502		if self.dyOfferInfo and self.dyOfferInfo.assignDiscountKey then
503			buyContext:setPreOrderExtraInfo({
504				assignDiscountKey = self.dyOfferInfo.assignDiscountKey,
505				dyGoodsTagActKey = Meta.ModuleName,
506			})
507		end
508	end
