53function CardDisplayPanelVM:getCardState(helpRemain)
54    local State = Const.CardDisplayState
55    local gotNum = self.cardData:getGotNum()
56    local isGold = self.cardData:getCardType() == Const.CardType.GOLD
57    if isGold then
58        return State.UNGIFTABLE
59    end
60    if gotNum <= 0 then
61        return helpRemain > 0 and State.SEEK_HELP or State.COOLDOWN
62    end
63    return State.GIFTABLE
64end

75function CardDisplayPanelVM:refreshState()
76    local model = self.collectCardManager:getModel()
77    local state = self:getCardState(model:getHelpRemain())
78    local stateData = {}
79    self.giftDisabledTip = nil
80    if state == Const.CardDisplayState.COOLDOWN then
81        local cooldownLeftTime = model:getCdHelpEndTime() / 1000 - TimeKit:timeInSec()
82        if cooldownLeftTime <= 0 then
83            self:onHelpCooldownEnd()
84            return
85        end
86        stateData.cooldownLeftTime = cooldownLeftTime
87    end
88    if state == Const.CardDisplayState.GIFTABLE then
89        stateData.sendRemain = model:getSendRemain()
90        stateData.sendDailyLimit = model:getSendDailyLimit()
91        self.giftDisabledTip = self:getGiftDisabledTip(stateData.sendRemain)
92        stateData.giftEnabled = self.giftDisabledTip == nil
93    elseif state == Const.CardDisplayState.UNGIFTABLE then
94        stateData.ungiftableText = self.cardData:getGotNum() <= 0
95            and "金卡不能求助哦~" or "金卡不能赠送~"
96    elseif state == Const.CardDisplayState.SEEK_HELP then
97        self.askChannelState = AskShareLogic:getAskChannelState()
98        stateData = self.askChannelState
99        if self.askChannelState.showWechatFriend
100            and not self.hasAskShareShowDc
101        then
102            self.hasAskShareShowDc = true
103            self.collectCardManager:dcActPlug():dcAskShareShow(
104                self.panelId,
105                self.collectCardManager.ACT_ID,
106                self.cardData:getCardId()
107            )
108        end
109    end
110
111    self:callViewFunc("updateState", state, stateData)
112end

193function CardDisplayPanelVM:submitHelp(context)
194    return self.collectCardManager:createSwapMessage({
195        type = Const.SwapCreateType.HELP,
196        targetUids = context.targetUids,

200            context.panelId,
201            #context.targetUids,

207        self.collectCardManager:dcActPlug():dcFriendHelpResult(

217function CardDisplayPanelVM:onSeekHelpWechatBtnTapped()
218    if self.isWechatSharePending then return end
219    if not _G.amfBean("NetworkStatusManager"):isNetworkEnabled() then
220        CommonTip:showTip("网络异常，请检查网络~")
221        return
222    end
223
224    local channelState = self.askChannelState
225    if channelState.wechatInstalled == false then
226        CommonTip:showTip("没有检测到微信，请先安装~")
227        return
228    end
229
230    self.collectCardManager:dcActPlug():dcAskShareClick(
231        self.panelId,
232        self.collectCardManager.ACT_ID,
233        self.cardData:getCardId()
234    )
235
236    self.isWechatSharePending = true
237
238    local function finishShare(imageUrl)
239        if self.isDisposed then return end
240        if imageUrl and imageUrl ~= "" then
241            AskShareLogic:startWechatAsk(
242                self.cardData:getCardName(),
243                imageUrl,
244                self.cardData:getCardId(),
245                self.panelId
246            )
247        else
248            CommonTip:showTip("分享图片生成失败，请稍后重试")
249        end
250        self.isWechatSharePending = false
251    end
252
253    if channelState.imageMode == Const.ShareImageMode.DYNAMIC then
254        self:callViewFunc("captureAskShareImage", self.cardData, finishShare)
255    else
256        finishShare(AskShareLogic:getFixedShareImageUrl())
257    end
258end

267function CardDisplayPanelVM:onEnterPanelRun()
268    self:callViewFunc("initCard", self.cardData, self.worldPos, self.isNew)
269    if not self.swapEnabled then
270        self.collectCardManager:dcActPlug():dcCardDisplay(self.cardData:getCardId(), self.panelId)
271        return
272    end
273    self:refreshState()
274    local gotNum = self.cardData:getGotNum()
275    local cardState
276    if self.cardData:getCardType() == Const.CardType.GOLD then
277        cardState = 3
278    elseif gotNum > 1 then
279        cardState = 1
280    elseif gotNum == 1 then
281        cardState = 2
282    end
283    self.collectCardManager:dcActPlug():dcCardDisplay(
284        self.cardData:getCardId(),
285        cardState,
286        self.collectCardManager:getModel():getSendRemain(),
287        self.panelId
288    )
289end
