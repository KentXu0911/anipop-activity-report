83function SuperMagicBirdMoney2025Manager:getPlayLevelState(levelId, levelScene )
84    local levelState = 0
85    if LevelSceneFunc:isFullLevelLib(levelScene) then 
86        levelState = 3
87    elseif LevelSceneFunc:isMainLevel(levelScene) then 
88        levelState = UBFActHelper:getPlayLevelModeType(levelId, nil, levelScene) 
89
90        local isFMJ = _G.amfBean("UserStateManager"):isFMJ()
91        if levelState == 2 and isFMJ then
92            levelState = 0
93        end
94
95        if levelState ~= 1 and levelState ~= 2 then 
96            levelState = 0
97        end
98    end
99    return levelState
100end

111function SuperMagicBirdMoney2025Manager:onActBeforePlayLevelEnd(evt)
112
113    local data = evt.data
114    if not data or not data.levelId then return end
115    if _G.amfBean("AskForHelpManager"):hasAskForHelpInfo(data.levelId ) then return end
116
117    local inst = UBFGameSceneManager:getInstance()
118    local levelInfo = inst:getLevelInfo()
119    -- if levelInfo.levelId ~= levelInfo.topLevelId then return end
120
121    local levelState = 0
122    if levelInfo.outLevelAttachData and levelInfo.outLevelAttachData["super_bird"] then
123        levelState = levelInfo.outLevelAttachData["super_bird"]
124    end
125
126    if levelState == 0 then return end 
127
128    local pworth = inst:getSceneData().pworth or {}
129
130    local actInLevelManager = UBFGameModManagerOrganizer:getRunningModManager(8068)
131    local modEnv = actInLevelManager and actInLevelManager:getModEnvData()
132    local hasSuperPower = modEnv and modEnv.hasSuperPower
133
134    -- local endLevelData = inst:getEndLevelData()
135
136    data.jsonTable["8069"] = { propWorth = pworth["MX"], levelState = levelState, propWorthV2 = pworth["RY"] }
137
138    --
139    self:getData():passLevelAddGoldNum(levelInfo.levelId, pworth["MX"], hasSuperPower, data.isWin, levelState)
140    self:getData():passLevelAddGoldNumV2(levelInfo.levelId, pworth["RY"], hasSuperPower, data.isWin, levelState)
141    self:getData():addCurCost( levelInfo.levelId, levelInfo.levelScene, pworth["RY"] )
142end
