447function UserTagModel:updateTags( resp , source )
448
449	local userData = UserManager:getInstance():getUserRef()
450	local topLevel = userData:getTopLevelId()
451
452	local userTag = resp.userTag
453	local difficultyTag = resp.difficultyTag
454	local payTag = resp.payTag
455	local socialTag = resp.socialTag
456	local levelTag = resp.levelTag
457	local propTag = resp.propTag
458
459	if source == UserTagDCSource.kLaunch then
460		if resp.defaultTagStartTime then
461			self.localData.defaultTagStartTime = math.floor( tonumber( resp.defaultTagStartTime ) / 1000 )
462		else
463			self.localData.defaultTagStartTime = 0
464		end
465		self.localData.defaultTagEndTime = self.localData.defaultTagStartTime + DefaultUserTagDurationTime
466		self.localData.defaultTagTopLevelId = resp.defaultTagTopLevelId or 0
467	end
468
469	if userTag then self:updateTag( UserTagGroupKey.kUserTag , userTag , source ) end
470	if difficultyTag then self:updateTag( UserTagGroupKey.kDifficultyTag , difficultyTag , source ) end
471	if payTag then self:updateTag( UserTagGroupKey.kPayTag , payTag , source ) end
472	if socialTag then self:updateTag( UserTagGroupKey.kSocialTag , socialTag , source ) end
473	if levelTag then self:updateTag( UserTagGroupKey.kLevelTag , levelTag , source ) end
474	if propTag then self:updateTag( UserTagGroupKey.kPropTag , propTag , source ) end
475
476	-- printx( 1 , "UserTagModel:updateTags  ---------- self.localData = \n" , table.tostring(self.localData) )

990--获取用户当前的某个标签,自动校验过期时间和topLevel
991--nameKey in global table : UserTagNameKey
992function UserTagModel:getUserTag( nameKey )
993	-- RemoteDebug:uploadLog("getUserTag", "UserTagModel:getUserTag" , nameKey)
994
995	if not nameKey or type(nameKey) ~= "string" then return nil end
996
997	local resultValue = nil
998
999	if self.localData then
1000		local tagGroup = UserTagNameKeyToGroupMap[nameKey]
1001
1002		if tagGroup and self.localData[tagGroup] and self.localData[tagGroup][nameKey] then
1003			local tagBean = self.localData[tagGroup]
1004			local tagValue = tagBean[nameKey]
1005			local tagTopLevelId = tonumber( tagBean[nameKey .. topLevelIdSuffix] )
1006			local tagTopLevelIdLength = tonumber( tagBean[nameKey .. topLevelIdLengthSuffix] )
1007			local tagEndTime = tonumber( tagBean[nameKey .. endTimeSuffix] )
1008
1009
1010			-- RemoteDebug:uploadLog("getUserTag", "tagValue" , tagValue , "tagTopLevelId" , tagTopLevelId , "tagTopLevelIdLength" , tagTopLevelIdLength , "tagEndTime" , tagEndTime )
1011
1012			--判断topLevel是否合法
1013			if tagTopLevelId and tagTopLevelIdLength and tonumber(tagTopLevelId) ~= 0 and tonumber(tagTopLevelIdLength) ~= 0 then
1014				local userData = UserManager:getInstance():getUserRef()
1015				local topLevel = userData:getTopLevelId()
1016
1017				if topLevel < tagTopLevelId or topLevel > tonumber(tagTopLevelId + tagTopLevelIdLength - 1) then
1018					if type(tagValue) == "table" then
1019						-- RemoteDebug:uploadLog("getUserTag", "return 1" , nameKey)
1020						return nil , GetUserTagReturnType.kFailByTopLevel
1021					else
1022						-- RemoteDebug:uploadLog("getUserTag", "return 2" , nameKey)
1023						return 0 , GetUserTagReturnType.kFailByTopLevel
1024					end
1025				end
1026			end
1027
1028			--判断过期时间是否合法
1029			if tagEndTime and tonumber(tagEndTime) ~= 0 then
1030				if tagEndTime < Localhost:timeInSec() then
1031					if type(tagValue) == "table" then
1032						-- RemoteDebug:uploadLog("getUserTag", "return 3" , nameKey)
1033						return nil , GetUserTagReturnType.kFailByTime
1034					else
1035						-- RemoteDebug:uploadLog("getUserTag", "return 4" , nameKey)
1036						return 0 , GetUserTagReturnType.kFailByTime
1037					end
1038				end
1039			end
1040
1041			resultValue = tagValue
1042		end
1043	end
1044	-- RemoteDebug:uploadLog("getUserTag", "return kSuccess  resultValue " , resultValue , "nameKey" , nameKey)
1045	return resultValue , GetUserTagReturnType.kSuccess
1046end
