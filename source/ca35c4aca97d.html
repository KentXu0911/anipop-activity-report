429-- 生成主面板抽奖券提示文案。库存信息优先，缺失时按活动开始时间兜底。
430function Manager:getLotteryTicketTip(isInLevel)
431    local leftCount = self:getRealRewardLeftCount()
432    isInLevel = isInLevel or false
433    local inLevelLottery = "快来抽奖!"
434    local outLevelLottery = "得券抽奖!"
435    local finalLottery = isInLevel and inLevelLottery or outLevelLottery
436    if leftCount ~= nil and leftCount < REAL_REWARD_STOCK_LIMIT then
437        return string.format("只剩%d个，" .. finalLottery, leftCount)
438    end
439
440    local beginTime = self:getModel():getBeginTimeInSec()
441    if beginTime <= 0 then
442        beginTime = self:getModel():getMainBeginTimeInSec()
443    end
444    local elapsed = TimeKit:timeInSec() - beginTime
445    if beginTime > 0 and elapsed <= 2 * DAY_SECONDS then
446        return "精美实物，" .. finalLottery
447    end
448    if beginTime > 0 and elapsed <= 4 * DAY_SECONDS then
449        return "库存有限，" .. finalLottery
450    end
451    return "库存告急，" .. finalLottery
452end

679function Manager:_registerCrossPeriodCallbacks()
680    local PassDayTimer = require "amf.script.tool.timer.PassDayTimer"
681    self:_unregisterCrossPeriodCallbacks()
682
683    local model = self:getModel()
684    local mainBegin = model:getMainBeginTimeInSec()
685    local mainEnd = model:getMainEndTimeInSec()
686    local actEnd = model:getEndTimeInSec()
687    local now = TimeKit:timeInSec()
688
689    if mainBegin and mainBegin > now then
690        self._cbMainBegin = function() self:onActPassMainBegin() end
691        self._cbMainBeginTime = mainBegin
692        PassDayTimer:registerEndTimeCallback(mainBegin + 1, self._cbMainBegin)
693    end
694    if mainEnd and mainEnd > now and mainEnd ~= mainBegin then
695        self._cbMainEnd = function() self:onActPassMainEnd() end
696        self._cbMainEndTime = mainEnd
697        PassDayTimer:registerEndTimeCallback(mainEnd + 1, self._cbMainEnd)
698    end
699    if actEnd and actEnd > now and actEnd ~= mainEnd then
700        self._cbActEnd = function() self:onActPassActEnd() end
701        self._cbActEndTime = actEnd
702        PassDayTimer:registerEndTimeCallback(actEnd, self._cbActEnd)
703    end
704end
705
706function Manager:_unregisterCrossPeriodCallbacks()
707    local PassDayTimer = require "amf.script.tool.timer.PassDayTimer"
708    for _, item in ipairs({
709        { "_cbMainBeginTime", "_cbMainBegin" },
710        { "_cbMainEndTime", "_cbMainEnd" },
711        { "_cbActEndTime", "_cbActEnd" },
712    }) do
713        local time = self[item[1]]
714        local callback = self[item[2]]
715        if time and callback then
716            PassDayTimer:unregisterEndTimeCallback(time, callback)
717        end
718        self[item[1]] = nil
719        self[item[2]] = nil
720    end
721end

737function Manager:onActPassActEnd()
738    self:closeAllPanels()
739    self:updateTipView()
740    local ActivityEventCenter = require "amf.middleware.activity.event.ActivityEventCenter"
741    require "amf.middleware.activity.event.ActivityEvent"
742    ActivityEventCenter:dispatch(ActivityEvent.REMOVE_ACTIVITY, { actId = self:getActId() })
743end

745-- Base NotifyPlug:onInstall() 会把 Manager 注册到 amfEC feature bus。
746-- PassDayTimer 跨天时会发送 FeatureEvent.PASS_DAY，FeatureEventBus 按 onAct 前缀调用到这里。
747function Manager:onActPassDay()
748    if not self:isActivityDataValid() then
749        return
750    end
751    -- 先记录跨天时是否有活动面板，避免刷新入口或关闭回调移除 observer 后漏弹提示。
752    local hasPanel = self:getObserverNum() > 0
753    BaseFeatureManager.onActPassDay(self)
754    self:getModel():onPassDay()
755    self:updateTipView()
756
757    self:closeAllPanels()
758    if hasPanel then
759        -- TODO: 按需替换为业务定制提示文案
760        CommonTip:showTip("新的一天开始了，数据已更新，\n请尝试重进活动~")
761    end
762end

764function Manager:dispose()
765    if self._disposed then return end
766    self._disposed = true
767    self:closeAllPanels()
768    self:_unregisterCrossPeriodCallbacks()
769    self:getModel():dispose()
770    self:_getPlugContainer():uninstallAll()
771end
772
773return Manager
