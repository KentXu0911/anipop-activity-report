35function PopupModel:syncServerRecord(data)
36	if not data or type(data) ~= "table" then return end
37	if table.isEmpty(data) then return end
38	local popupInfo = data.popupInfo or {}
39	self:setValue("_permanentPopupOrders", popupInfo.popupOrders or {})
40	local dailyPopupInfo = data.dailyPopupInfo or {}
41	self:setValue("_dailyPopupOrders", dailyPopupInfo.popupOrders or {})

50local function normalizePopupSceneStage(popStage)
51    local stageId = tonumber(popStage)
52    if stageId == 2 or stageId == 3 or stageId == 4 then
53        return tostring(stageId)
54    end
55end

57-- 追加一轮调试快照：每个场景只保留最近 MAX_SCENE_LOG_ROUNDS 轮。
58function PopupModel:setLastPopupSceneSnapshot(popStage, snapshot)
59    local stageKey = normalizePopupSceneStage(popStage)
60    if not stageKey then return end
61
62    local rounds = self._lastPopupSceneSnapshots[stageKey] or {}
63    table.insert(rounds, 1, snapshot)
64    while #rounds > MAX_SCENE_LOG_ROUNDS do
65        table.remove(rounds)
66    end
67    self._lastPopupSceneSnapshots[stageKey] = rounds
68    self:setValue("_lastPopupSceneSnapshots", self._lastPopupSceneSnapshots, true)
69end

143-----------BEGIN 两次场景2之间弹出限制-----------
144function PopupModel:initStage2LastPopupInfo()
145	if self._lastLimitTime then
146        local currentTime = TimeKit:timeInSec()
147		if currentTime - self._lastLimitTime > AutoPopupLimit.Stage2PopupInterval then
148            self:setValue("_lastLimitTime", 0, false)
149            self:setValue("_lastPopupSum", 0, false)
150		end
151	else
152        self:setValue("_lastLimitTime", 0, false)
153        self:setValue("_lastPopupSum", 0, false)
154	end
155end

173function PopupModel:initDailyPopTypeInfo()
174	local today = TimeKit:getTodayStart()
175	if not self._dailyPopTypeInfo then
176        self:setValue("_dailyPopTypeInfo.updateDay", today, false)
177	else
178		local cacheDay = self._dailyPopTypeInfo.updateDay or 0
179		if today ~= cacheDay then
180            self:setValue("_dailyPopTypeInfo.updateDay", today, false)
181		end
182	end
183end
