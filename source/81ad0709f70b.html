10local function formatStr(str, params)
11    if not str then
12        return nil
13    end
14	if params == nil then return str end
15	local newStr = str
16	--字符串 参数 替换
17	for k, v in pairs(params) do
18		local param1 = '{'..k..'}'
19		local param2 = v
20		newStr = string.gsub( newStr, param1, param2 )
21	end
22
23	return newStr
24end

48			end

51			if #parts == 2 then
52				--说明是区间段

55                    local message = metaCfg['text'..i]

58                        :setTitle(formatStr(data.title, params) or formatStr(title, params))
59                        :setMessage(formatStr(data.message, params) or formatStr(message, params))
60                        -- :setPicId(pic)
61                        :setLabelType(i)
62					break
63				end
64			else
65				--说明是单个尾号

67                    local title = metaCfg['title'..i]
68                    local message = metaCfg['text'..i]
69                    local pic = metaCfg['pic'..i]
70					structAddAlarm:setTitle(title)
71                        :setTitle(formatStr(data.title, params) or formatStr(title, params))

73                        -- :setPicId(pic)
74                        :setLabelType(i)

83-- 修改即时推送时间
84function LocalPushUtil:makeStructChangePushTime(structAddAlarm, metaCfg)
85    -- 不是即时推送不处理
86    if metaCfg.groupId ~= LocalPushConst.AlarmType.NO_LIMIT then
87        return structAddAlarm
88    end
89    -- 修改推送时间, 即时推送
90    local dayStartTime = TimeKit:getDayStartTimeByTS(structAddAlarm.timeStamp)
91    if structAddAlarm.timeStamp < dayStartTime + 7 * 3600 then
92        structAddAlarm:setTimeStamp(dayStartTime + 7 * 3600) --7点前 改到7点推送
93    elseif structAddAlarm.timeStamp > dayStartTime + 22 * 3600 then
94        structAddAlarm:setTimeStamp(dayStartTime + 24*3600 + 7 * 3600) --如果22点后 次日7点推送
95    end
96    return structAddAlarm
97end
