153function MainSceneButtonManager:resolveButtonConfig(config)
154    local data = {}
155    for k,v in ipairs(config) do
156        if not data[v.uniqueId] then
157            data[v.uniqueId] = v
158            if v.uniqueId == _G.BuffTypeConfig.SCORE_BUFF then
159                data[v.uniqueId].iconCreator = function()
160                    return self:createScoreBuff(data[v.uniqueId].rate)
161                end
162            end
163        else
164            if v.uniqueId == _G.BuffTypeConfig.SCORE_BUFF then
165                data[v.uniqueId].rate = v.rate + data[v.uniqueId].rate - 1
166                data[v.uniqueId].endTime = math.min(v.endTime, data[v.uniqueId].endTime)
167                data[v.uniqueId].iconCreator = function()
168                    return self:createScoreBuff(data[v.uniqueId].rate)
169                end
170                data[v.uniqueId].actId = v.actId .. "_" .. data[v.uniqueId].actId
171            end
172        end
173    end
174
175    local retData = {}
176    -- 关卡库筛除加分
177    for k,v in pairs(data) do
178        if k == _G.BuffTypeConfig.SCORE_BUFF and self._fullStarLevelLibManager:canPlay() then
179        -- 筛掉时间为0的,去除鬼畜
180        elseif v.endTime > TimeKit:timeInSec() then
181            table.insert(retData, v)
182        end
183    end
184    return retData
185end

501-- 计算关卡推荐数据
502function MainSceneButtonManager:calLevelData()
503    local fullLevelState = self._userStateManager:getUserSegmentState()
504    local levelId, levelFlag, areaLocked, levelScene, showStarWand
505    local buttonState
506    if self._userStateManager:isFMJ() then
507        levelId = _G.getTopLevelId()
508        local topPassedMainLevelId = _G.getTopPassedLevelId()
509        levelFlag = self._metaManager:getLevelDifficultFlag(levelId, UBFLevelSceneType.kMain)
510        areaLocked = levelId == topPassedMainLevelId
511        levelScene = UBFLevelSceneType.kMain
512        buttonState = ButtonConst.buttonStateConfig.FMJ
513    elseif self._fullStarLevelLibManager:canPlay()  then
514        levelId = self._fullStarLevelLibManager:getNextLevelId()
515        levelFlag = LevelDiffcultFlag.kNormal
516        areaLocked = false
517        levelScene = UBFLevelSceneType.kFullLevelLibrary
518        buttonState = ButtonConst.buttonStateConfig.FSLB
519    else
520        levelScene = UBFLevelSceneType.kMain
521        buttonState = ButtonConst.buttonStateConfig.MJ
522        if self:isStarWandCanShow() then
523            showStarWand = true
524            local lvsMain = self:getStarWandData()
525            levelId = lvsMain[1]
526            buttonState = ButtonConst.buttonStateConfig.MJSW
527        end
528        if not levelId then
529            local RecommendLevelUtil = require "amf.feature.recommendlevel.RecommendLevelUtil"
530            levelId, levelScene = RecommendLevelUtil:getNotFullColorStarLowestLevel()
531            if not UBFLevelSceneFunc:isMainLevel(levelScene) then
532                return nil
533            end
534        end
535        levelFlag = self._metaManager:getLevelDifficultFlag(levelId, UBFLevelSceneType.kMain)
536        areaLocked = false
537    end
538
539    -- 取不到关卡，保底删除按钮
540    if not levelId then
541        return nil
542    end
543
544    return {
545        levelId = levelId,
546        levelFlag = levelFlag,
547        areaUnlocked = not areaLocked,
548        levelScene = levelScene,
549        showStarWand = showStarWand,
550        buttonState = buttonState,
551        fullLevelState = fullLevelState
552    }
553end

567function MainSceneButtonManager:refreshButton(onlyRefreshData)
568    if self.isLoadingRes then return end
569
570    local levelData = self:calLevelData()
571    if levelData == nil then
572        self:tryDeleteButton()
573        return
574    end
575
576    if not self.buttonVM then
577        return
578    end
579
580    local inputBuff, inputModule
581
582    if self:isActivityEnabled() then
583        local moduleConfig = self:getExtraModuleConfig() or {}
584        inputModule = moduleConfig
585
586        local nowButtonBuffData = self.buttonVM:getBuffData()
587        local nowBuffEnabled = nowButtonBuffData and not table.isEmpty(nowButtonBuffData)
588            and nowButtonBuffData.endTime > TimeKit:timeInSec() and nowButtonBuffData.uniqueId ~= inputModule.uniqueId
589        if nowBuffEnabled then
590            inputModule = nil
591        end
592        local config = self:getButtonConfig(inputModule and inputModule.uniqueId) or {}
593        inputBuff = config[1]
594        if nowBuffEnabled then
595            for k,v in ipairs(config) do
596                if v.uniqueId == nowButtonBuffData.uniqueId then
597                    inputBuff = v
598                    break
599                end
600            end
601        end
602
603        if inputModule and not table.isEmpty(inputModule) then
604            if inputBuff and not table.isEmpty(inputBuff) then
605                inputBuff.lightAnim = true
606            end
607        end
608    end
609
610    local function refreshFunc()
611        if not self.buttonVM or self.buttonVM.isDisposed then return end
612        self.isLoadingRes = false
613        self.buttonVM:refreshData(levelData, inputBuff, inputModule)
614        if not onlyRefreshData then
615            self.buttonVM:refreshView()
616        end
617    end
618
619    if inputBuff and inputBuff.needLoadRes then
620        self.isLoadingRes = true
621        self:loadInputRes(inputBuff.needLoadRes):next(refreshFunc, function()
622            self.isLoadingRes = false
623        end)
624    else
625        refreshFunc()
626    end
627end

718function MainSceneButtonManager:onButtonTapped(delegate)
719	self._autoPopupMgr:abortInterruptibleEntry(PASS_LEVEL_SKIP_ENTRY)
720	if not _G.amfBean("PopoutManager"):haveWindowOnScreen() then
721        -- 开始监控从响应点击到打开开始面板的耗时
722        UBFPerfMonitor.start("OpenSlp", UBFPerfMonitor.Profile.OPEN_SLP)
723
724        if self._userStateManager:isFMJ() then
725            local playLevelContext = PlayLevelContext.new(nil, PlayLevelContext.PlayType.UN_PASSED_MAIN)
726            playLevelContext:setAutoUnlock(true)
727            playLevelContext:setExtraData({
728                from = "MainSceneButton"
729            })
730            _G.amfTransaction(TransactionEvent.PLAY_LEVEL, playLevelContext)
731        -- 关卡库使用关卡库接口
732        elseif UBFLevelSceneFunc:isFullLevelLib(delegate.levelScene) then
733            self._fullStarLevelLibManager:tryOpenStartPanel(true)
734        else
735            local playLevelContext = PlayLevelContext.new(delegate.levelId, PlayLevelContext.PlayType.OPEN_SL_PANEL)
736            playLevelContext:setAutoUnlock(true)
737            playLevelContext:setExtraData({
738                from = "MainSceneButton"
739            })
740            playLevelContext:setTrunkInsight(true)
741            playLevelContext:setLevelScene(delegate.levelScene)
742            _G.amfTransaction(TransactionEvent.PLAY_LEVEL, playLevelContext)
743        end
744        self:playHaptic()
745        self:playEffect()
746
747        MainSceneButtonDC:sendDC("play_button_click", {
748            t1 = delegate.levelFlag,
749            t2 = delegate.actId1,
750            t3 = delegate.actId2,
751            t4 = delegate.buttonState,
752            current_stage = delegate.levelId
753        })
754    end
755end
