176function HonorRankManager:setWaitingRewardData(data)
177    if not data then return end
178    local localData = self.waitingRewardData
179    if localData and localData.rewards then --有未处理的信息，merge
180        if data.rewards then
181            for _,v in ipairs(data.rewards) do
182                table.insert(localData.rewards, v)
183            end
184            if localData.maxRewardLevel and data.maxRewardLevel then
185                localData.maxRewardLevel = math.max(localData.maxRewardLevel, data.maxRewardLevel)
186            end
187
188            if localData.beginRewardLevel and data.beginRewardLevel then
189                localData.beginRewardLevel = math.min(localData.beginRewardLevel, data.beginRewardLevel)
190            end
191
192            if localData.maxLevel and data.maxLevel then
193                localData.maxLevel = math.max(localData.maxLevel, data.maxLevel)
194            end
195        end
196    else
197        self.waitingRewardData = data

497function HonorRankManager:openRankPanel(rankType)
498    if self._honorRankAdapterPanelVM or self._isOpeningHonorRankAdapterPanel then
499        return
500    end
501    local sourceMainPanelVM = self._mainPanelVM
502    if not sourceMainPanelVM or sourceMainPanelVM.isDisposed then
503        return
504    end
505    if not self:getDataAdapter(rankType) then
506        return
507    end
508    self._isOpeningHonorRankAdapterPanel = true
509    self:requestRankData(rankType, function()
510        if sourceMainPanelVM ~= self._mainPanelVM or sourceMainPanelVM.isDisposed then
511            self._isOpeningHonorRankAdapterPanel = false
512            return
513        end
514        local HonorRankAdapterPanelVM = require "amf.feature.honorrank.view.HonorRankAdapterPanelVM"
515        local rankConfig = HonorRankConfig.RankConfig[rankType]
516        local backgrounds = rankConfig and rankConfig.backgrounds
517        local loadResList = backgrounds and {backgrounds.main, backgrounds.top} or nil
518        local vm = HonorRankAdapterPanelVM:create({rankType = rankType, loadResList = loadResList})
519        local didHideMainPanel = false
520        vm:setPanelReadyCallback(function()
521            self._isOpeningHonorRankAdapterPanel = false
522            if sourceMainPanelVM == self._mainPanelVM and not sourceMainPanelVM.isDisposed then
523                sourceMainPanelVM:callViewFunc("setVisible", false)
524                didHideMainPanel = true
525            end
526        end)
527        vm:setLoadResFailCallback(function()
528            self._isOpeningHonorRankAdapterPanel = false
529            if self._honorRankAdapterPanelVM == vm then
530                self._honorRankAdapterPanelVM = nil
531            end
532            if didHideMainPanel and sourceMainPanelVM == self._mainPanelVM and not sourceMainPanelVM.isDisposed then
533                sourceMainPanelVM:callViewFunc("setVisible", true)
534            end
535        end)
536        vm:setCloseCallback(function()
537            self._isOpeningHonorRankAdapterPanel = false
538            if self._honorRankAdapterPanelVM == vm then
539                self._honorRankAdapterPanelVM = nil
540            end
541            if didHideMainPanel and sourceMainPanelVM == self._mainPanelVM and not sourceMainPanelVM.isDisposed then
542                sourceMainPanelVM:callViewFunc("setVisible", true)
543            end
544        end)
545        self._honorRankAdapterPanelVM = vm
546        vm:popout()
547    end, function()
548        self._isOpeningHonorRankAdapterPanel = false
549    end)
550end

558function HonorRankManager:createPeakFlagByProfile(profile, parentNode, pos, canTouch, notFromRank, showLv0, scale, from)
559    if EnvUtil:isMiniGame() and not self.loadResSucceed then
560        return nil
561    end
562    if not self:canShowHonorRankIcon(true) then
563        return nil
564    end
565    local peakScore = nil
566    if profile and profile.peakLevelBean and profile.peakLevelBean.exp and (profile.peakLevelBean.exp > 0 or showLv0) then
567        peakScore = profile.peakLevelBean.exp
568    elseif profile and profile.peakScore then
569        peakScore = profile.peakScore
570    end
571    local needShowPeakFlag = peakScore and (peakScore > 0 or showLv0)
572    if parentNode then
573        if parentNode.peakFlag_ui then
574            parentNode.peakFlag_ui:updateSelf(profile.xxlId, peakScore, from)
575            parentNode.peakFlag_ui:setVisible(needShowPeakFlag)
576        elseif needShowPeakFlag then
577            parentNode.peakFlag_ui = self:createPeakFlag(profile.xxlId, peakScore, canTouch, notFromRank, from)
578            parentNode.peakFlag_ui:setVisible(needShowPeakFlag)
579            parentNode.peakFlag_ui:setScale(scale or 1)
580            if pos then
581                parentNode.peakFlag_ui:setPositionXY(pos.x, pos.y)
582            end
583            parentNode:addChild(parentNode.peakFlag_ui)
584            parentNode.peakFlag_ui:setName("peakFlag_ui")
585        end
586
587        return parentNode.peakFlag_ui, needShowPeakFlag
588    else
589        return self:createPeakFlag(profile.xxlid, peakScore, canTouch, notFromRank, from), true
590    end

614function HonorRankManager:addRewards(items, sp, fromType, doneCallback)
615
616    if table.isEmpty(items) then
617        safeRunCall(doneCallback)
618        return
619    end
620
621	local itemResPosInWorld = sp and sp:getPositionInWorldSpace() or ccp(360,640)
622	local normalItems = {}
623    local canPopNameCardPanel = true
624
625    for _,v in pairs(items) do
626        local isNormal = false
627        if self._propManager:isHeadImage(v.itemId) or
628                self._propManager:isNameCardBg(v.itemId) or
629                self._propManager:isFlowerSkin(v.itemId) or
630                self._propManager:isHeadFrame(v.itemId) then
631        else
632            table.insert(normalItems, v)
633            isNormal = true
634        end
635        if not isNormal then
636            local feature = "full_rank"
637            local source = "reward"
638            if fromType == 1 then
639                feature = "peak_level"
640                source = "level_reward"
641            end
642            HonorRankDC:dcGainSpecialItem(feature, source, v.itemId, v.num)
643        end
644    end
645    if #normalItems>0 then
646        local flyContext = FlyItemContext.new(normalItems, itemResPosInWorld)
647                                         :setFinishCallback(function ()
648        end)
649        _G.amfTransaction(TransactionEvent.FLY_ITEMS, flyContext)
650    end
651    _G.GainPropData:create()

850function HonorRankManager:getPeakLv(score)
851	score = score or 0
852	local config = self:getRewardConfig()
853	local lv = 0
854	local nextScore = 0
855	local lastScore = 0
856	for _,v in ipairs(config) do
857		if v.exp <= score then
858			lv = v.lv
859			nextScore = v.exp
860			lastScore = v.exp
861		else
862			nextScore = v.exp
863			break
864		end
865	end
866
867    local lvPercent = (score-lastScore)/(nextScore-lastScore)
868	if nextScore == lastScore then
869		lvPercent = 1
870	end
871
872
873	lvPercent = math.ceil(lvPercent*100)
874	nextScore = nextScore
875
876	return lv, nextScore, lvPercent
877end
