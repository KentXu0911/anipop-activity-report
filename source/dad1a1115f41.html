289function MusicMasterManager:isLimitEnabled()
290
291    --等级
292    if not self.leveIsEnabled then
293        local levelId = _G.amfBean("LevelAreaManager"):getTopPassedLevel()
294        if levelId < self:getConst().openPassLevel then 
295            return false 
296        else
297            self.leveIsEnabled = true
298        end
299    end
300
301    --actInfo
302    local actInfo = self:getActInfo()
303    if not actInfo then return false end
304    local extra = table.deserialize(actInfo.extra)
305
306    --客户端版本
307    if self:getConst().cliendVersion < extra.clientVersionControl then return false end
308
309    --版本
310    local limitver = self:getVersion( extra.minClientVersion )
311    local clientver = self:getVersion( _G.bundleVersion )
312    if clientver < limitver then return false end
313
314    return true
315end

583function MusicMasterManager:openActivity( bFromInLevel, closeCallback, userClick, extra, UBFPopDelegate, failCallback )
584
585    local tempCloseCallback = closeCallback
586    closeCallback = function()
587        if self:getData():isInMainTime() and self:getData():checkProgressIsAllGet() then
588            self.musicMasterIconMudule:removeIcon()
589        else
590            self.musicMasterIconMudule:updateIcon("panel_close")
591        end
592        safeRunCall(tempCloseCallback)
593        -- tempCloseCallback()
594    end
595
596    local function openFailFunc()
597        safeRunCall(failCallback)
598        safeRunCall(closeCallback)
599    end
600
601    if userClick == nil then userClick = false end
602    if extra == nil then extra = {} end
603
604    local function openMainPanel()
605        local MainPanelVM = require 'amf.feature.localactivity.musicmaster.view.MainPanelVM'
606        local mainPanel = MainPanelVM:create( { closeCallback =closeCallback, userClick = userClick, extra = extra, manager = self } )
607        if UBFPopDelegate then
608            mainPanel:setUBFPopDelegate({ popout = UBFPopDelegate.popoutFunc, removePopout = UBFPopDelegate.removeFunc })
609        end
610        mainPanel:popout()
611        self.panelPlug:managePanelVM(mainPanel)
612    end
613
614    if bFromInLevel  then
615        if self:isEnabledAndResReady() then
616            openMainPanel()
617        else
618            if not self:isResReady() then
619                self:downLoadRes()
620            end
621            safeRunCall(openFailFunc)
622        end
623    else
624        if self:isEnabledAndResReady() then
625            self:getInfoAsync( openMainPanel, openFailFunc, openFailFunc, userClick, false )
626        else
627            if not self:isResReady() then
628                self:downLoadRes()
629            end
630            safeRunCall(openFailFunc)
631        end
632    end
633end
