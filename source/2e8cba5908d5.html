55function NewbieGiftNewGiftpackage:_requestNewGiftData(actId, clientMd5, clientUserDataMd5, callback)
56
57    local callbackPromise = deferred.new()
58    callbackPromise:next(function (packages)
59        local newData = {}
60        if packages then
61            -- 3选一 todo:根据moudlename 判断
62            local giftType = 3
63            local  _newbieGiftOptimizeMgr = _G.amfBean("NewbieGiftOptimizeMgr")
64            local actInfo = _newbieGiftOptimizeMgr:getActInfo()
65            local actInfoExtra = table.deserialize(actInfo.extra or {})
66            if actInfoExtra.subInfos and #actInfoExtra.subInfos > 0 then
67                actInfoExtra = actInfoExtra.subInfos[1]
68            end
69
70            local moduleExtraMap = packages and #packages > 0 and packages[1]:getExtraValue("moduleExtra") or nil
71            local extraConfig = (moduleExtraMap) and table.deserialize(moduleExtraMap) or {}
72
73            local gotGiftPackTimesMap = {}
74            local giftPackMetaBeanList = (((extraConfig or {}).serverVersionedConfig or {}).config or {}).packageArr or {}
75            if giftPackMetaBeanList and #giftPackMetaBeanList > 0 then
76                for key, value in pairs(packages) do
77                    local obj = table.find(giftPackMetaBeanList,function ( t )
78                        return t.packageId == value:getGoodsId()
79                    end)
80                    if obj then
81                        local packageId = value:getGoodsId()
82                        obj.goodsId = packageId
83                        obj.limitBuyTimes = value.totalCanBuyTimes
84                        obj.permanentLimitBuyTimes = value.alreadyBuyTimes
85
86                        gotGiftPackTimesMap[tostring(packageId)] = value.totalCanBuyTimes
87                    end
88                end
89            end
90            if table.isEmpty(giftPackMetaBeanList) then
91                giftPackMetaBeanList =  _newbieGiftOptimizeMgr:getLocalDataServer():getDataByKey(giftType, "giftPackMetaBeanList")
92                --clear serverUserDataMd5
93                if table.isEmpty(giftPackMetaBeanList) then
94                    LocalDB:setString("NewbieGiftNew_serverUserDataMd5", "")
95                end
96            end
97            local strongPushConditionList = (((extraConfig or {}).serverVersionedConfig or {}).config or {}).strongPushConditionIds or {}
98            if table.isEmpty(strongPushConditionList) then
99                strongPushConditionList = _newbieGiftOptimizeMgr:getLocalDataServer():getDataByKey(giftType, "strongPushConditionList", {})
100            end
101
102            local giftInfo = {}
103            giftInfo.beginTime = actInfoExtra.roundBeginTime
104            giftInfo.buyEndTime = actInfoExtra.buyEndTime
105            giftInfo.canBuyRookieGoods = false
106            giftInfo.cdEndTime = actInfoExtra.cdEndTime
107            --check
108            giftInfo.clientControlVersion = 4
109            giftInfo.endTime = actInfoExtra.roundEndTime
110            giftInfo.giftPackMetaBeanList = giftPackMetaBeanList
111            giftInfo.gotGiftPackTimesMap = gotGiftPackTimesMap
112            giftInfo.gotRewardNodeIndex = 0
113            giftInfo.id = giftType
114            giftInfo.periodBeginTime = actInfoExtra.periodBeginTime
115            giftInfo.periodEndTime = actInfoExtra.periodEndTime
116            giftInfo.periodId = actInfoExtra.periodId
117            giftInfo.showRookieGoods = false
118            giftInfo.strongPushCdH = (((extraConfig or {}).serverVersionedConfig or {}).config or {}).strongPushCdH or 0
119            if giftInfo.strongPushCdH <= 0 then
120                giftInfo.strongPushCdH = _newbieGiftOptimizeMgr:getLocalDataServer():getDataByKey(giftType, "strongPushCdH", 999)
121            end
122            giftInfo.strongPushConditionList = strongPushConditionList
123            giftInfo.reward = actInfoExtra.reward
124
125            local giftInfoList = {giftInfo}
126            newData["giftPackActivityInfoList"] = giftInfoList
127
128            if moduleExtraMap then
129                newData["guideSet"] = (((extraConfig or {}).serverUserData or {}).userData or {}).finishedGuideIds or {}
130                if table.isEmpty(newData["guideSet"]) then
131                    newData["guideSet"] = _newbieGiftOptimizeMgr:getLocalDataServer():getDataByKey(giftType, "guideSet", {})
132                end
133                --获取礼包购买次数
134                newData["permanentGotGiftPackTimesMap"] = {}
135
136                --检查具体字段
137                local serverVersionedConfig = nil
138                local serverMd5 = ((extraConfig or {}).serverVersionedConfig or {}).serverMd5 or nil
139                if serverMd5 then
140                    serverVersionedConfig = { serverMd5 = serverMd5 }
141                    --json
142                    serverVersionedConfig.config = table.decodeToJson( ((extraConfig or {}).serverVersionedConfig or {}).config or {} )
143                end
144                --todo:多个礼包config合并数据
145                newData["serverVersionedConfig"] = serverVersionedConfig or {}
146
147                local serverUserDataMd5  = ((extraConfig or {}).serverUserData or {}).serverUserDataMd5 or ""
148                LocalDB:setString("NewbieGiftNew_serverUserDataMd5", serverUserDataMd5)
149            end
150            newData["errorCode"] = 0
151            newData["retCode"] = 200
152        end
153
154        safeRunCall(callback, newData)
155    end, function ()
156        CommonTip:showTip("网络异常，请稍后重试~")
157        safeRunCall(callback, false)
158    end)
159
160    if (clientMd5 and clientMd5 ~= "") and (not clientUserDataMd5 or clientUserDataMd5 == "") then
161        clientUserDataMd5 = LocalDB:getString("NewbieGiftNew_serverUserDataMd5", "")
162    end
163
164    -- clientMd5 = ""
165    -- clientUserDataMd5 = ""
166    GiftEventCenter:dispatch(GiftEvent.PULL_ACTIVITY_PACKAGE, {
167        actId = actId,
168        moduleName = "chooseOneFromThree2025",
169        callbackPromise = callbackPromise,
170        params = {
171            clientMd5 = clientMd5,
172            clientUserDataMd5 = clientUserDataMd5,
173        },
174    })
175end

179function NewbieGiftNewGiftpackage:requestMainInfo(giftType, onSuccessCallback, onFailCallback, onCancelCallback, bFromLightStart, userClick)
180    if not _G.amfBean("NetworkStatusManager"):isNetworkEnabled() then
181        CommonTip:showTip("网络异常，请检查网络~")
182        return
183    end
184
185
186    local function onSuccess(evt)
187        local data = evt.data or {}
188        _G.amfBean("NewbieGiftOptimizeMgr"):updateData(data)
189        -- _G.amfBean("NewbieGiftOptimizeMgr"):updateTipView()
190        -- printx(48, "[gk] request 3or1 onSuccess=", tts(data))
191        safeRunCall(onSuccessCallback)
192    end
193    local function onFail(evt)
194        if (not bFromLightStart) and userClick then
195            self:showErrorTip(evt)
196        end
197        local errCode = evt and evt.data or nil
198        if errCode and tonumber(errCode) == 733220 then
199            _G.amfBean("NewbieGiftOptimizeMgr"):clearActInfoAndUpdateIcon(giftType)
200        end
201        safeRunCall(onFailCallback)
202    end
203    local function onCancel()
204        safeRunCall(onCancelCallback)
205    end
206    local noLoading = bFromLightStart or (not userClick)
207
208    local needActivityInfoIdList = {}
209    if bFromLightStart then
210        for i, type in pairs(_G.amfBean("NewbieGiftOptimizeMgr").GIFT_TYPE) do
211            if _G.amfBean("NewbieGiftOptimizeMgr"):getLocalDataServer():checkTypeDataEnabled(type) then
212                table.insert(needActivityInfoIdList, type)
213            end
214        end
215    else
216        table.insert(needActivityInfoIdList, giftType)
217    end
218    if table.isEmpty(needActivityInfoIdList) then
219        safeRunCall(onFail)
220        return
221    end
222    local serverMd5 = giftType and _G.amfBean("NewbieGiftOptimizeMgr"):getLocalDataServer():getDataByKey(nil, "serverMd5", "") or ""
223
224    -- local clientVersionedConfig = {
225    --     clientMd5 = serverMd5
226    -- }
227    -- self:requestServerData("giftPackSystemInfo", {needActivityInfoIdList = needActivityInfoIdList, light = bFromLightStart, ios = EnvUtil:isIOS(), clientVersionedConfig = clientVersionedConfig}, onSuccess, onFail, onCancel, noLoading)
228
229    self:_requestNewGiftData(self.curActId, serverMd5, "", function (serverData)
230        if serverData then
231            local evt = {data=serverData}
232            onSuccess(evt)
233        else
234            safeRunCall(onFail)
235        end
236
237    end)
238
239    -- printx(48, "[gk] 新三选一 礼包请求", ";bFromLightStart=",bFromLightStart, ";userClick=",userClick)
240
241end
