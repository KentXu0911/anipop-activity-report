446	if not features then return nil end
447
448	local regTimeAvailable = self:getUserRegTime() ~= nil
449	local selected = nil
450	local selectedPriority = nil
451

453		local regTimeOk = true

455			regTimeOk = regTimeAvailable and self:checkRegTime(feature)
456		end
457
458		if regTimeOk then
459			local priority = self:getMaintenanceFeaturePriority(feature, regTimeAvailable)
460			if not selected or priority > selectedPriority or (priority == selectedPriority and tostring(feature.id) > tostring(selected.id)) then
461				selected = feature
462				selectedPriority = priority
463			end
464		end
465	end
466
467	return selected

1940function MaintenanceManager:getLevelRangesByExtra(key)
1941
1942	local cacheKey = tostring("LevelRangesByExtra") .. "_" .. tostring(key)
1943	if self.resultCache[cacheKey] then
1944		return table.clone( self.resultCache[cacheKey] )
1945	end
1946
1947	local extra = self:getExtra(key)
1948
1949	local resultList = nil
1950
1951	if extra then
1952		resultList = {}
1953		local ranges = string.split(extra, ",")
1954		local values = nil
1955		local startLevel, endLevel = 0, 0
1956		for i, range in ipairs(ranges) do
1957			values = string.split(range, "-")
1958			if #values == 1 then
1959				startLevel = tonumber(values[1]) or 0
1960				table.insert(resultList, {startLevel=startLevel, endLevel = startLevel})
1961			elseif #values == 2 then
1962				startLevel = tonumber(values[1]) or 0
1963				endLevel = tonumber(values[2]) or 0
1964				if startLevel > endLevel then startLevel, endLevel = endLevel, startLevel end
1965				table.insert(resultList, {startLevel=startLevel, endLevel = endLevel})
1966			end
1967		end
1968	end
1969
1970	self.resultCache[cacheKey] = resultList
1971	return resultList

2066	local maintenanceFeature = self:getMaintenanceByKey(maintenanceName)

2074		if log then log("getCurrentGroupName r not enable",maintenanceName) end
2075		return

2083			return
2084		end
2085	end
2086	local dummyData = self._dummyMaintenanceMaps and self._dummyMaintenanceMaps[maintenanceName]
2087	if dummyData then
2088		-- 测试数据
2089		for _ , key in ipairs( dummyData ) do
2090			if type(key) == "string" then
2091				return key
2092			end
2093		end
2094		-- 测试数据 未命中
2095		if log then log("getCurrentGroupName r not in dummyData",maintenanceName,ttsd(dummyData)) end
2096		return
2097	end
2098    local uid = getSafeUid()
2099	local childIndex , algorithmId, onlyOneGroupLeft = self:getChildIndexAndalgorithmId( maintenanceFeature , uid )
2100	if not childIndex then
2101		-- 分组未命中
2102		if log then log("getCurrentGroupName r NO childIndex",maintenanceName,ttsd(maintenanceFeature)) end
2103		return
2104	end

2106	if not groupData then
2107		if log then log("getCurrentGroupName r NO groupData",maintenanceName,childIndex) end
2108		return
2109	end
2110	if log then log("getCurrentGroupName result",maintenanceName,childIndex,ttsd(groupData)) end
2111	return groupData
