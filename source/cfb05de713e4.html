307function SuperMagicBird2024Manager:checkCanUseActInfoData( actInfoExtra )
308
309    log("checkCanUseActInfoData 是否使用后端actInfo数据检测开始", actInfoExtra)
310
311    if not actInfoExtra then return true end
312
313    local extra = actInfoExtra
314
315    local bCanChangeData = false
316    if extra.ignoreBreakVersion1 then
317        bCanChangeData = true
318        log("checkCanUseActInfoData 开关关闭，信任后端")
319    else
320        if self:isEnabled() then
321            log("checkCanUseActInfoData 本地有数据记录，校验本地数据")
322
323            local clientBreakVersion = self:getData():getDataByKey( 'breakVersion1' ) or 0
324            local serverBreakVersion = extra.breakVersion1 or 0
325
326            log("checkCanUseActInfoData 客户端断鸟版本 后端断鸟版本", clientBreakVersion, serverBreakVersion )
327
328            if clientBreakVersion <= serverBreakVersion then
329                bCanChangeData = true
330                log("checkCanUseActInfoData 当前版本小于等于后端，使用后端")
331            end
332        else
333            log("checkCanUseActInfoData 本地无数据记录，跳过校验，直接用后端的")
334            bCanChangeData = true
335        end
336    end
337
338    log("checkCanUseActInfoData 是否使用后端actInfo数据检测结束 bCanChangeData =", bCanChangeData)
339
340    return bCanChangeData
341end

500function SuperMagicBird2024Manager:curTimeTopLevelIsEnabled()
501
502    local isFMJ = _G.amfBean("UserStateManager"):isFMJ()
503    if isFMJ then return true end
504
505    --当前时间，头部关是否满足。 当前时间如果大于本地最高关开启时间7天，活动不开
506    local newLevelOpenTime, countDownAreaId = _G.amfBean("LevelUnLockManager"):getLastOpenNewLevelTime()
507    local diffTime = TimeKit:timeInSec() - newLevelOpenTime
508
509    if EnvUtil:isDevelopMode() then
510        log("检测新关解锁时间是否满足7日内", newLevelOpenTime, TimeKit:timeInSec(), diffTime, 60*60*24*7)
511    end
512
513    if diffTime >= 60*60*24*7 then
514        return false
515    end
516    return true
517end

605function SuperMagicBird2024Manager:isReviveCardFeatureEnabled()

608    if not reviveCardInfo or not reviveCardInfo.beginTs or not reviveCardInfo.endTs then
609        return false
610    end
611

613    local endTime = TimeKit:tryConvertMsToSec(reviveCardInfo.endTs)
614    local now = TimeKit:timeInSec()
615    return beginTime ~= nil and endTime ~= nil and now >= beginTime and now < endTime

673function SuperMagicBird2024Manager:getReviveCardSlpState(params)
674    params = params or {}
675    local reviveCardEnabled = self:isEnabled()
676    local canAccumulate = self:canAccumulateChargeProgress()
677        and self:isEnabledInLevel(params)
678    local hasUsableCard = self:hasReviveCardInBag()
679    local cardExpireAt = 0
680
681    if hasUsableCard then
682        cardExpireAt = TimeKit:getTodayStart() + SECONDS_PER_DAY
683        local reviveCardInfo = self:getReviveCardInfo() or {}
684        local activityExpireAt = tonumber(TimeKit:tryConvertMsToSec(reviveCardInfo.endTs)) or 0
685        if activityExpireAt > 0 then
686            cardExpireAt = math.min(cardExpireAt, activityExpireAt)
687        end
688        local now = TimeKit:timeInSec()
689        if now >= cardExpireAt then
690            hasUsableCard = false
691            cardExpireAt = 0
692        end
693    end
694
695    return {
696        reviveCardEnabled = reviveCardEnabled,
697        canAccumulate = canAccumulate,
698        hasUsableCard = hasUsableCard,
699        cardExpireAt = cardExpireAt,
700        isSuperBirdActive = self:getAvailable() == true,
701        currentProgress = self:getPassTopNums() or 0,
702        maxProgress = self:getNeedWinNum() or 8,
703    }
704end

706function SuperMagicBird2024Manager:onActDcStageCommon(event)
707    if not event then return end
708
709    local dcInfo = event:getData()
710    if not dcInfo
711        or dcInfo.category ~= "stage"
712        or dcInfo.sub_category ~= "start_show"
713    then
714        return
715    end
716
717    local reviveCardState = self:getReviveCardSlpState({
718        levelId = dcInfo.levelId or dcInfo.current_stage,
719        levelScene = dcInfo.levelScene,
720    })
721    local entryMaintenanceName = EnvUtil:isMiniGameProject() and "superbird_mini" or "superbird_app"
722    local entryEnabled = MaintenanceManager:getInstance():isEnabledInGroup(entryMaintenanceName, "ON")
723        and not (dcInfo.t1 == 1 and dcInfo.t2 == 0)
724    local hasUsableCard = reviveCardState and reviveCardState.hasUsableCard == true
725    local showReviveCard = entryEnabled
726        and hasUsableCard
727        and reviveCardState.reviveCardEnabled == true
728        and reviveCardState.canAccumulate == true
729        and reviveCardState.isSuperBirdActive ~= true
730
731    dcInfo.bird_status = showReviveCard and 1 or 0
732    dcInfo.bird_reborn_cnt = hasUsableCard and 1 or 0
733end

837function SuperMagicBird2024Manager:JumpFlyToAnimation(config)
838	if not config.sprites or #config.sprites <= 0 or not config.duration or config.duration <= 0 then return end
839	config.delayTime = config.delayTime or 0
840	config.dstPosition = config.dstPosition or ccp(0, 0)
841	config.height = config.height or 100
842	local counter = #config.sprites
843	local time = 0
844	for k, v in ipairs(config.sprites) do
845		local position = v:getPosition()
846		local parent = v:getParent()
847
848		local sequence = CCArray:create()
849		if config.easeIn and counter > 1 then
850			sequence:addObject(CCDelayTime:create(time))
851			time = time + math.cos((k - 1) / (counter - 1) * math.pi / 2) * config.delayTime
852		else
853			sequence:addObject(CCDelayTime:create(config.delayTime * (k - 1)))
854		end
855		local function onStart() if config.startCallback then config.startCallback(v) end end
856		sequence:addObject(CCCallFunc:create(onStart))
857		local spawn = CCArray:create()
858		local inPosition = parent:convertToNodeSpace(config.dstPosition)
859
860		spawn:addObject(CCEaseBackIn:create(CCMoveBy:create(config.duration, ccp(0, inPosition.y - position.y))))
861		spawn:addObject(CCMoveBy:create(config.duration, ccp(inPosition.x - position.x, 0)))
862		if config.dstSize then
863			local size = v:getGroupBounds().size
864            spawn:addObject(CCScaleBy:create(config.duration, config.dstSize.width / size.width, config.dstSize.height / size.height))
865		end
866		if config.extendAction then
867			local extendActionCopy = config.extendAction:copy()
868			spawn:addObject(extendActionCopy) 
869			extendActionCopy:release()
870		end
871		sequence:addObject(CCSpawn:create(spawn))
872		local function onReach()
873			counter = counter - 1
874			if config.reachCallback then config.reachCallback(v) end
875			if counter == 0 and config.finishCallback then config.finishCallback() end
876		end
877		sequence:addObject(CCCallFunc:create(onReach))
878		
879		v:runAction(CCSequence:create(sequence))
880	end
881end

977function SuperMagicBird2024Manager:getLocalValueByKey(key, needTime, needActStartTimeCheck)
978    if not key then return end
979    local saveJsonStr = self:getData():getValue(key)
980    if saveJsonStr == nil then return end
981    if type(saveJsonStr) ~= "string" then return end
982
983    local saveTable = table.deserialize(saveJsonStr)
984
985    local nowTime = getDayStartTimeByTS(TimeKit:timeInSec())
986    local actBeginTime = self:getRealBeginTime() or 0
987
988    local bTimeIsOk = true
989    if needTime then
990    	if saveTable.time ~= nowTime then
991    		bTimeIsOk = false
992    	end
993    end
994
995    local versionIsOk = true
996    if needActStartTimeCheck then
997    	if saveTable.actBeginTime ~= actBeginTime then
998    		versionIsOk = false
999    	end
1000    end
1001
1002    if bTimeIsOk and versionIsOk then
1003    	return saveTable.value
1004    end
1005end

1007function SuperMagicBird2024Manager:openMainActivity(closeCallback, extra, userClick, duanMianData)
1008    local function getActDataSuccess()
1009        local scene = Director:sharedDirector():run()
1010        if scene == nil or not scene:isScene("MainScene") then
1011            return
1012        end
1013
1014        if not self:curTimeTopLevelIsEnabled() then
1015            self:popNeedUpdateLevelTip(closeCallback)
1016            return
1017        end
1018
1019        if not extra then
1020            extra = {}
1021        end
1022
1023        local hasStartPanelContext = type(extra.closeSlpCallback) == "function"
1024            and extra.levelId ~= nil
1025            and extra.levelScene ~= nil
1026
1027        local delegate = {
1028            closeCallback = closeCallback,
1029            extra = extra,
1030            resumeData = duanMianData,
1031            userClick = userClick,
1032        }
1033
1034        if hasStartPanelContext and self:canActivateReviveCard() then
1035            local closeSlpCallback = extra.closeSlpCallback
1036            extra.closeSlpCallback = nil
1037            safeRunCall(closeSlpCallback)
1038            self:openReviveConfirmPanel({
1039                isRetryPanel = extra.isRetryPanel == true,
1040                closeCallback = function(activateSucceeded)
1041                    if activateSucceeded then
1042                        safeRunCall(delegate.closeCallback)
1043                        local playLevelContext = PlayLevelContext.new(extra.levelId, PlayLevelContext.PlayType.OPEN_SL_PANEL)
1044                        playLevelContext:setLevelScene(extra.levelScene)
1045                        _G.amfTransaction(TransactionEvent.PLAY_LEVEL, playLevelContext)
1046                    else
1047                        self:openMainPanel(delegate)
1048                    end
1049                end,
1050            })
1051            return
1052        end
1053
1054        -- 未激活但是已经进度条满了
1055        if not extra.closePopOut and not self:getAvailable() and self:getPassTopNums() >= self:getNeedWinNum() then
1056            local successCallback = function()
1057                self:openMainPanel(delegate, config)
1058            end
1059            local failCallback = function()
1060                self:openMainPanel(delegate, config)
1061            end
1062            self:requestActivate(successCallback, failCallback, failCallback)
1063            return
1064        end
1065        self:openMainPanel(delegate, config)
1066    end
1067
1068    getActDataSuccess()
1069    -- self:getInfoAsync(getActDataSuccess, function() safeRunCall(endCallback) end, function() safeRunCall(endCallback) end, false, userClick)
1070end
