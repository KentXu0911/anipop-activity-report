1local BaseFeatureModel = require "amf.middleware.activity.base.BaseFeatureModel"

5-- 服务端 aiAutoPopInfo 是「自然天编号 -> 当天预测」的 map，value 是一段 JSON 字符串：
6--   20705 = '{"target_date":"2026-09-08","prediction_id":"popuprec_20260908_191040_001",
7--             "strategy":"model","predictions":{"6074":0.0933,"4046":0.0001,...}}'
8-- key 是这条数据的归属天：服务端没取到当天预测时，会把前一天的 payload 挂在今天的 key 下
9-- 返回，此时按 key（今天）来用；payload 里的 target_date 原样保留，打点回传时不能改写。
10-- 活动范围和活动 ID 以服务端返回为准：返回的活动都参与排序，权重不合法的按约定兜底。
11
12-- 本地只保留最近几天，够当天用即可。
13local MAX_PREDICTION_DAYS = 3
14
15-- 排序权重：服务端给的值不在 [0,1] 内（含非数值）时按约定兜底。
16local DEFAULT_DEPTH_SCORE = 0.12

44local function copyDepthScores(predictions)
45    local result = {}
46    if type(predictions) ~= "table" then
47        return result
48    end
49    for actId, score in pairs(predictions) do
50        local numberScore = tonumber(score)
51        if not (numberScore and numberScore >= 0 and numberScore <= 1) then
52            numberScore = DEFAULT_DEPTH_SCORE
53        end
54        result[tostring(actId)] = numberScore
55    end
56    return result
57end

59local function decodePayload(rawInfo)
60    if type(rawInfo) ~= "string" or rawInfo == "" then return nil end
61    return jsonDecode(rawInfo)
62end

64-- 一条服务端记录 -> 本地一天的数据。aiPopupRecInfo 原样保留服务端 payload 供打点回传。
65local function buildPrediction(day, info)
66    return {
67        targetDate = tostring(day),
68        depthScores = copyDepthScores(info.predictions),
69        predictionId = info.prediction_id,
70        strategy = info.strategy,
71        aiPopupRecInfo = info,
72    }
73end

75local function collectPredictions(data)
76    local entries = {}
77    local responseMap = type(data) == "table" and data.aiAutoPopInfo or nil
78    if type(responseMap) ~= "table" then
79        return entries
80    end
81
82    for day, rawInfo in pairs(responseMap) do
83        local info = decodePayload(rawInfo)
84        if type(info) == "table" then
85            entries[#entries + 1] = buildPrediction(day, info)
86        end
87    end
88    return entries
89end

91local function findPrediction(predictions, day)
92    for _, prediction in ipairs(predictions or {}) do
93        if prediction.targetDate == day then return prediction end
94    end
95end
96
97local function trimPredictions(predictions)
98    table.sort(predictions, function(left, right)
99        return left.targetDate > right.targetDate
100    end)
101    while #predictions > MAX_PREDICTION_DAYS do
102        table.remove(predictions)
103    end
104end

106local function savePredictions(model, predictions)
107    trimPredictions(predictions)
108    model:setDataByKey("predictions", predictions, true)
109    local latest = predictions[1]
110    model:setDataByKey("predictionId", latest and latest.predictionId, false)
111    model:setDataByKey("strategy", latest and latest.strategy, false)
112end

114-- 用一次响应整体替换本地预测数据。
115function Model:assignPredictionData(data)
116    if type(data) ~= "table" then return false end
117    savePredictions(self, collectPredictions(data))
118    return true
119end

121-- 按自然天合并推荐数据：同一天已有本地记录时不覆盖。
122function Model:mergePredictionData(data)
123    if type(data) ~= "table" then return false, false end
124
125    local incoming = collectPredictions(data)
126    local predictions = self:getPredictions() or {}
127    local changed = false
128    for _, prediction in ipairs(incoming) do
129        if not findPrediction(predictions, prediction.targetDate) then
130            predictions[#predictions + 1] = prediction
131            changed = true
132        end
133    end
134
135    if changed then
136        savePredictions(self, predictions)
137    end
138    return true, changed
139end

165-- day 是归属天（服务端 key，如 20705），字符串或数字都可以。
166function Model:getDepthScoresByDate(day)
167    if day == nil then return nil end
168    local prediction = findPrediction(self:getPredictions(), tostring(day))
169    return prediction and (prediction.depthScores or {})
170end

172function Model:getAiPopupRecInfoByDate(day)
173    if day == nil then return nil end

175    return prediction and prediction.aiPopupRecInfo

178function Model:getActivityPriority(day, actId)
179    local depthScores = self:getDepthScoresByDate(day)
180    if not depthScores or actId == nil then
