14local function sortNodesByLayout(nodes)
15    table.sort(nodes, function(a, b)
16        local ay, by = a:getPositionY(), b:getPositionY()
17        if math.abs(ay - by) > 1 then
18            return ay > by
19        end
20        return a:getPositionX() < b:getPositionX()
21    end)
22end

109function LuckyDeliveryCardRewardPanel:initCardRewards()
110    local posNodes = {}
111    for _, child in ipairs(self.cardLayoutNode:getChildrenSafeList() or {}) do
112        if child.getName and string.starts(tostring(child:getName()), "cardPosNode_") then
113            table.insert(posNodes, child)
114        end
115    end
116    sortNodesByLayout(posNodes)
117
118    local cardDisplayInfos = self.cardDisplayInfos or self:buildCardDisplayInfos()
119    local cardScale = self:getCardScale(#cardDisplayInfos, self.extraReward ~= nil)
120    local appearAnimInfos = {}
121
122    for index, reward in ipairs(cardDisplayInfos) do
123        local card = self:addCardToPosNode(reward, posNodes[index], cardScale)
124        table.insert(appearAnimInfos, {
125            card = card,
126            posNode = posNodes[index],
127            reward = reward,
128            scale = cardScale,
129        })
130    end
131
132    self:playCardAppearAnim(appearAnimInfos)
133end

135function LuckyDeliveryCardRewardPanel:getCardScale(cardNum, hasExtraReward)
136    local displayScale = 1
137    if hasExtraReward then
138        if cardNum >= 7 then
139            displayScale = 0.8
140        elseif cardNum >= 3 then
141            displayScale = 0.9
142        end
143    elseif cardNum >= 7 then
144        displayScale = 0.9
145    end
146    return BASE_CARD_SCALE * displayScale

166function LuckyDeliveryCardRewardPanel:playCardAppearAnim(appearAnimInfos)
167    if table.isEmpty(appearAnimInfos) then return end
168
169    self.cardAppearAnimPlaying = true
170    local index = 1
171    local playNext
172    playNext = function()
173        if self.isDisposed then return end
174        local animInfo = appearAnimInfos[index]
175        if not animInfo then
176            self:onCardAppearAnimFinished()
177            return
178        end
179        index = index + 1
180        self:playSingleCardAppearAnim(animInfo, 0, playNext)
181    end
182    playNext()
183end

185function LuckyDeliveryCardRewardPanel:playSingleCardAppearAnim(animInfo, delay, callback)
186    local card = animInfo.card
187    local reward = animInfo.reward
188
189    local cardScale = animInfo.scale or BASE_CARD_SCALE
190    card:setScale(0)
191    card:runAction(ActionUtil:sequence{
192        CCDelayTime:create(delay or 0),
193        CCCallFunc:create(function()
194            if self.isDisposed or card.isDisposed then
195                safeRunCall(callback)
196                return
197            end
198            self:playCardAppearEffect(animInfo)
199            if reward.isRepeat then
200                card:runAction(ActionUtil:sequence{
201                    CCScaleTo:create(3 * OpenCardAnimConfig.FRAME_TIME, cardScale * 1.2),
202                    CCScaleTo:create(3 * OpenCardAnimConfig.FRAME_TIME, cardScale),
203                    CCCallFunc:create(function()
204                        safeRunCall(callback)
205                    end)
206                })
207            end
208        end),
209        CCDelayTime:create(reward.isRepeat and 0 or 15 * OpenCardAnimConfig.FRAME_TIME),
210        CCCallFunc:create(function()
211            if reward.isRepeat then return end
212            if self.isDisposed or card.isDisposed then
213                safeRunCall(callback)
214                return
215            end
216            card:setScaleX(0.6 * cardScale)
217            card:setScaleY(cardScale)
218            card:runAction(ActionUtil:sequence{
219                CCEaseSineInOut:create(CCScaleTo:create(2 * OpenCardAnimConfig.FRAME_TIME, cardScale, cardScale)),
220                CCCallFunc:create(function()
221                    safeRunCall(callback)
222                end)
223            })
224        end)
225    })
226end

272function LuckyDeliveryCardRewardPanel:buildCardDisplayInfos()
273    local cardIdList = {}
274    local getCardList = {}
275
276    for _, reward in ipairs(self.cardRewards or {}) do
277        local cardId = reward.cardId or self.collectCardManager:getCardIdByItemId(reward.itemId)
278        local cardData = cardId and self.collectCardManager:getCardDataById(cardId)
279        if cardData then
280            local num = math.max(math.floor(tonumber(reward.num) or 1), 1)
281            local cardIdKey = tostring(cardData.cardId)
282            getCardList[cardIdKey] = (getCardList[cardIdKey] or 0) + num
283            for _ = 1, num do
284                if #cardIdList >= MAX_CARD_NUM then break end
285                table.insert(cardIdList, cardData.cardId)
286            end
287        end
288    end
289
290    self.settleModel.getCardList = getCardList
291    return self.settleModel:buildCardInfoList(cardIdList)
292end

301function LuckyDeliveryCardRewardPanel:playSettleAnim(callback)
302    if self.cardAppearAnimPlaying then
303        self.pendingSettleCallback = callback
304        return
305    end
306
307    if table.isEmpty(self.shownCardList) then
308        safeRunCall(callback)
309        return
310    end
311
312    self.settleModel.localStarNum = self:getLocalStarNumBeforeSettle()
313    self.settleController:settleCards(self.shownCardList, nil):next(function()
314        safeRunCall(callback)
315    end, function()
316        safeRunCall(callback)
317    end)
318end

320function LuckyDeliveryCardRewardPanel:getLocalStarNumBeforeSettle()
321    local gotStarNum = 0
322    for _, cardInfo in ipairs(self.shownCardList or {}) do
323        if cardInfo.isRepeat and cardInfo.cardData then
324            gotStarNum = gotStarNum + (cardInfo.cardData.starLevel or 0)
325        end
326    end
327
328    return (self.collectCardManager:getStarNum() or 0) - gotStarNum
329end
