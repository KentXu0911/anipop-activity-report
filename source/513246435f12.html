127function ShopRewardGiftModule:markActivityPackageBought(packageId)

165function ShopRewardGiftModule:replaceActivityGiftWithOrigin(activityPackageId)
166    local manager = self:getManager()
167    local originPackageId = manager and manager:getOriginPackageIdByActivityPackageId(activityPackageId)
168    local originGiftData = self:getHiddenStoreGiftData(originPackageId)
169    if not originGiftData then return end
170
171    local originGiftView = self:createBaseGiftView(originGiftData)
172    local payload = {
173        targetPackageId = activityPackageId,
174        newGiftView = originGiftView,
175    }
176    StoreEventCenter:dispatch(_G.StoreEvent.STORE_REPLACE_GIFT_VIEW, payload)
177    if payload.replaced then
178        return originGiftView
179    end
180
181    disposeUnreplacedGiftView(originGiftView)
182end

197function ShopRewardGiftModule:replaceRemovingActivityGiftWithOrigin(data)
198    local giftView = data and data.giftView
199    local packageLayer = data and data.packageLayer
200    if not giftView or not packageLayer or packageLayer.isDisposed then return false end
201    if not giftView.getGiftData then return false end
202
203    local giftData = giftView:getGiftData()
204    if not self:isActivityGiftData(giftData) then return false end
205
206    local manager = self:getManager()
207    local originPackageId = manager and manager:getOriginPackageIdByActivityPackageId(giftData:getGoodsId())
208    local originGiftData = self:getHiddenStoreGiftData(originPackageId)

211    local originGiftView = self:createBaseGiftView(originGiftData)

213        targetGiftView = giftView,
214        newGiftView = originGiftView,

217    if payload.replaced == true then

237function ShopRewardGiftModule:onActivityGiftBought(giftData)
238    if not self:isActivityGiftData(giftData) then return end
239
240    local manager = self:getManager()
241    if not manager then return end
242
243    local buyHandledKey = tostring(manager:getPeriodId()) .. "_" .. tostring(giftData:getGoodsId())
244    self.handledActivityGiftBuyMap = self.handledActivityGiftBuyMap or {}
245    if self.handledActivityGiftBuyMap[buyHandledKey] then return end
246    self.handledActivityGiftBuyMap[buyHandledKey] = true
247
248    manager:dcJoin()
249    self:markActivityPackageBought(giftData:getGoodsId())
250    ShopRewardEventCenter:dispatch(ShopRewardEvent.BUY_COUNT_REFRESH, {
251        packageId = giftData:getGoodsId(),
252        remainBuyCount = manager:getRemainBuyCount(),
253    })
254
255    local activityEnded = manager:getRemainBuyCount() <= 0
256    if activityEnded then
257        self:onStoreActivityEndedDelay(manager)
258        return
259    end
260
261    self:replaceActivityGiftWithOriginDelay(giftData:getGoodsId())
262end
