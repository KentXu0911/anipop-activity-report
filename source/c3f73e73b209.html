38local function getClaimableBoxIdForTest(collectNum)
39    local remainCollectNum = math.max(tonumber(collectNum) or 0, 0)
40    local maxBoxId = 0
41    for boxId = 1, 10 do
42        remainCollectNum = remainCollectNum - TEST_BOX_CONFIG[boxId].target
43        if remainCollectNum < 0 then return maxBoxId end
44        maxBoxId = boxId
45    end
46    local cycleIndex = 11
47    while true do
48        local configIndex = 11 + (cycleIndex - 11) % 10
49        remainCollectNum = remainCollectNum - TEST_BOX_CONFIG[configIndex].target
50        if remainCollectNum < 0 then return maxBoxId end
51        maxBoxId = cycleIndex
52        cycleIndex = cycleIndex + 1
53    end
54end

849function TestUtils:testPackageTierPurchase()
850    local oldPackageMetaList = self._mgr:getConfigValue("packagePriceMetaList", {})
851    local oldBuyPackageMap = self._mgr:getDataValue("buyPackageMap", {})
852    local packageMetaList = {
853        {packageId = 1003, buyType = "TICKET", minBuyTimes = -1},
854        {packageId = 1002, buyType = "TICKET", minBuyTimes = 5},
855        {packageId = 1001, buyType = "TICKET", minBuyTimes = 2},
856        {packageId = 2002, buyType = "DOUBLE_COLLECT", minBuyTimes = -1},
857        {packageId = 2001, buyType = "DOUBLE_COLLECT", minBuyTimes = 1},
858    }
859    local cases = {
860        {name = "门票0次", buyType = "TICKET", buyPackageMap = {}, expected = 1001},
861        {name = "门票边界2次", buyType = "TICKET", buyPackageMap = {['1001'] = 2}, expected = 1001},
862        {name = "门票3次", buyType = "TICKET", buyPackageMap = {['1001'] = 3}, expected = 1002},
863        {name = "门票汇总6次", buyType = "TICKET", buyPackageMap = {['1001'] = 2, ['1002'] = 4}, expected = 1003},
864        {name = "翻倍0次", buyType = "DOUBLE_COLLECT", buyPackageMap = {}, expected = 2001},
865        {name = "翻倍2次", buyType = "DOUBLE_COLLECT", buyPackageMap = {['2001'] = 2}, expected = 2002},
866    }
867
868    self._mgr._model:setValue("localConfig.packagePriceMetaList", packageMetaList)
869    local failedCases = {}
870    for _, caseData in ipairs(cases) do
871        self._mgr:setDataValue("buyPackageMap", caseData.buyPackageMap)
872        local packageId = self._mgr:getPackageIdByBuyType(caseData.buyType)
873        if packageId ~= caseData.expected then
874            table.insert(failedCases, string.format("%s:%s", caseData.name, tostring(packageId)))
875        end
876    end
877
878    self._mgr:setDataValue("buyPackageMap", {})
879    self._mgr:addPackageBuyTimes(1001)
880    local incrementMap = self._mgr:getDataValue("buyPackageMap", {}) or {}
881    local incrementPassed = incrementMap["1001"] == 1
882    self._mgr._model:setValue("localConfig.packagePriceMetaList", oldPackageMetaList)
883    self._mgr:setDataValue("buyPackageMap", oldBuyPackageMap)
884
885    if #failedCases == 0 and incrementPassed then
886        CommonTip:showTip("梯度购买测试通过：7/7")
887    else
888        if not incrementPassed then table.insert(failedCases, "购买次数本地累加失败") end
889        CommonTip:showTip("梯度购买测试失败：" .. table.concat(failedCases, ","))
890    end
891end
