20function LevelUnlockConfigParser:loadLevelUnlockConfig()
21    local result = {}
22    MetaConfigUtil:traverseConfigThenRelease("level_unlock_config", function(k, v)
23        if k == "configVersion" then
24            result[k] = tonumber(v)
25        else
26            local cfgList = v[1].unlockData
27            local dataList = {}
28            for i, rawData in ipairs(cfgList) do
29                local dataPiece = {}
30                for key, val in pairs(rawData) do
31                    if key == "id" or key == "minVer" then
32                        dataPiece[key] = tonumber(val)
33                    elseif key == "startTime" then
34                        local startTimeStr = tostring(val)
35                        dataPiece[key] = tostring(val)
36                        if startTimeStr then
37                            dataPiece["startTimeStamp"] = os.time2(parseTime(startTimeStr))
38                        end
39                    elseif key == "areaIDs" then
40                        if type(val) == "number" then
41                            dataPiece[key] = {val}
42                        else
43                            local areaList = LevelUnlockConfigUtil:parseItemList(val)
44                            if areaList then
45                                table.sort(areaList, function(a,b) return a < b end)
46                                dataPiece[key] = areaList
47                            end
48                        end
49                    end
50                end
51                dataList[i] = dataPiece
52            end
53            result[k] = dataList
54        end
55    end)
56
57    self.levelUnlockConfig = result
58    self.levelUnlockConfigOri = table.clone(result)
59end

125-- 主线 & 隐藏关，当前时间段的下一周为top区域
126-- 原因：周五开新关前，会更新一版将top移到再下周的配置，本次重构将其之间节点提前，一开新关就移动top到下周
127-- 详情可以咨询庄sir，每周发level_unlock_config_config的时候，最后一条是保底配置，倒数第二条是预览配置，真正的配置是从倒数第三条开始
128function LevelUnlockConfigParser:setNextPeriodAreaAsTop(param)
129    local currTime = param.currTime
130    local targetConfig = param.targetConfig
131    local isMainLevel = param.isMainLevel
132
133    local start = #targetConfig
134    local isFound = false
135
136    if isMainLevel then
137        -- top不能是最后那条9999保底，也不能是倒数第二条是待发布仅预览
138        start = start + _G.levelUnlockConst.lastSafeConfigIndex
139        local data = targetConfig[start]
140        if data.startTimeStamp < currTime then
141            -- 最高合法配置的时间就已经过期，说明没发新关、或者本地没有拉取到新关配置，强行认为最高配置就是最新
142            data.lastOpenedArea = true
143            data.isTopArea = true
144            isFound = true
145        end
146    end
147
148    if not isFound then
149        for i = start, 2, -1 do
150            local data = targetConfig[i]
151            local formerData = targetConfig[i - 1]
152            if data and formerData then
153                --增加i==2, 兼容时间被调到很久之前，找不到配置的问题
154                if (data.startTimeStamp > currTime) and (formerData.startTimeStamp <= currTime) or (i == 2) then
155                    data.isTopArea = true
156                    formerData.lastOpenedArea = true
157                    break
158                end
159            end
160        end
161    end
162end

196-- 当前最低未开放区域 + 之后的一个区域（不包含9999条目）
197function LevelUnlockConfigParser:getMainAreaUnlockInfoOfTopAreas()
198    local infoTable = {}
199    local topAreaId = nil
200    local mainLevelConfig = self:getUnlockConfigListByConfigType(_G.levelUnlockConst.LevelUnlockConfigType.MAIN_LEVEL)
201    if mainLevelConfig then
202        for i = #mainLevelConfig, 1, -1 do
203            local v = mainLevelConfig[i]
204            if v and v.isTopArea then
205                table.insert(infoTable, v)
206                table.insert(infoTable, mainLevelConfig[i+1])
207                local topAreaIDs = v.areaIDs
208                if topAreaIDs and #topAreaIDs > 0 then
209                    topAreaId = topAreaIDs[#topAreaIDs]
210                end
211                break
212            end
213        end
214    end
215
216    table.sort(infoTable, function(a,b) return a.id < b.id end)
217
218    return infoTable, topAreaId
219end

288function LevelUnlockConfigParser:getUnlockDataByDataID(targetUnlockConfig, configID)
289    -- 一般来说应该是相等的，为了提高效率，先检查一波
290    if targetUnlockConfig[configID] and (targetUnlockConfig[configID].id == tonumber(configID)) then
291        return targetUnlockConfig[configID]
292    end
293    -- 未命中后的兜底
294    for _, unlockData in pairs(targetUnlockConfig) do
295        if unlockData.id == configID then
296            return unlockData
297        end
298    end
299    return nil
300end
