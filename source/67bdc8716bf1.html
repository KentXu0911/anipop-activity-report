166function Manager:checkCanPopMainPanelStage4()
167    if not self:isEnabled() then
168        return false
169    end

171    local launchOpportunity = tonumber(self:getModel():getLaunchOpportunity()) or 0
172    return launchOpportunity >= minLaunchOpportunity

175function Manager:popMainPanelStage4(onPopOver)
176    if not self:checkCanPopMainPanelStage4() then
177        safeRunCall(onPopOver)
178        return false
179    end
180
181    -- 同组内主面板优先；只在引流项实际入队时移除并记录三关延后。
182    local attractCheckPoint = _G.PopupCheckPoint.PENGPENG2026_ATTRACT_PANEL_STAGE4
183    local attractQueue = self.autoPopupMgr:getPopupQueueByCheckPoint(attractCheckPoint)
184    local isAttractPanelOverlapped = attractQueue and
185        attractQueue[attractCheckPoint] ~= nil and
186        self:checkCanPopAttractPanel()
187    self.autoPopupMgr:removePopupConfig(attractCheckPoint)
188    if isAttractPanelOverlapped then
189        self:markAttractPanelSuppressedByMainPanelAutoPop()
190    end
191    self:openActivity(onPopOver, {from = "auto_popup_stage4"}, false)
192    return true
193end

209-- 弹出前再次校验，避免旧队列项绕过三关延后状态；按钮点击进入活动主界面。
210function Manager:popAttractPanel(onPopOver, force)
211    if not self.downloadPlug:checkDownloadSuccess() then
212        safeRunCall(onPopOver)
213        return false
214    end
215    local canPop = force == true or self:checkCanPopAttractPanel()
216    if not canPop then
217        safeRunCall(onPopOver)
218        return false
219    end
220    return AttractPanelHelper.pop(self, onPopOver)
221end

305function Manager:_onResolveIcon(event)
306    if not event then return end
307    local data = event:getData()
308    if not data or not data.callback then return end
309
310    local itemId = tonumber(data.itemId)
311    local iconInfo = Meta.StaticItemIconInfo and Meta.StaticItemIconInfo[itemId]
312
313    if iconInfo then
314        data.callback({
315            Atlas = iconInfo.Atlas or "",
316            Frame = iconInfo.Frame,
317            IsPropPng = iconInfo.IsPropPng,
318        })
319        return
320    end

323function Manager:_onCustomGain(event)
324    if not event then return end
325    local reward = event:getData()
326    if not reward then return end
327
328    local id = tonumber(reward.itemId)
329    local num = tonumber(reward.num) or 0
330    local Const = self:getConst()
331    local model = self:getModel()
332
333    if id == Const.collectId then
334        model:addLaunchOpportunity(num)
335        self:updateTipView()
336    elseif id == Const.percent20Buff then
337        model:addScoreBuff20Time(num)
338    elseif id == Const.doubleBuff then
339        model:addDoubleCollectBuffTime(num)
340    end
341end
