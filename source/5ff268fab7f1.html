602function CrashResumeHandler:_interceptFeedTrialEnd(onLevelDone)
603	local origLaunchEndLevel = UBFProcessManager._launchEndLevelProcess
604	local hookActive = true
605	local hookFn
606	hookFn = function(mgr, processType, runtime)
607		if not hookActive then
608			return origLaunchEndLevel(mgr, processType, runtime)
609		end
610		hookActive = false
611		UBFProcessManager._launchEndLevelProcess = origLaunchEndLevel
612
613		local endProcess = origLaunchEndLevel(mgr, processType, runtime)
614		endProcess:afterEnd():next(function()
615			safeRunCall(onLevelDone, {
616				reason = "end_level_ended",
617				processType = processType,
618			})
619		end)
620		endProcess:afterAbort():next(function()
621			safeRunCall(onLevelDone, {
622				reason = "end_level_aborted",
623				processType = processType,
624			})
625		end)
626		return endProcess
627	end
628	UBFProcessManager._launchEndLevelProcess = hookFn
629
630	return function()
631		if hookActive then
632			hookActive = false
633			UBFProcessManager._launchEndLevelProcess = origLaunchEndLevel
634		end
635	end
636end

663	local levelScene = params.levelScene or UBFLevelSceneType.kTrial
664	local costType = params.costType or StartLevelCostType.kInfiniteEnergy
665	local logic = params.logic
666	local runtimeExtra = params.runtimeExtra or {}
667	local isTrialLevelScene = levelScene == UBFLevelSceneType.kTrial
668
669	require ("ubf.UBF.utils.UBFCommonRequire")("*")
670
671	local angry = UBFGameLoadingController:apply({
672		source = UBFGameLoadingSourceEnum.ORDINARY,
673		levelId = levelId,
674		levelScene = levelScene,
675		toleratedTime = 0,
676	}):retain()
677
678	local angryReleased = false
679	local function releaseAngry()
680		if angryReleased then return end
681		angryReleased = true
682		angry:release()
683	end
684
685	local handled = false
686	local restoreEndLevelHook
687	local function finishTrial(result)
688		if handled then return end
689		handled = true
690		if restoreEndLevelHook then restoreEndLevelHook() end
691		if isTrialLevelScene and logic and logic.onTrialFinished then
692			logic:onTrialFinished(result)
693		end
694		safeRunCall(onTrialFinished, result)
695	end
696
697	local started = false
698	local function startTrialLevel()
699		if started then return end
700		started = true
701
702		if isTrialLevelScene then
703			restoreEndLevelHook = self:_interceptFeedTrialEnd(finishTrial)
704		end
705
706		if isTrialLevelScene and logic and logic.onTrialAutoStart then
707			logic:onTrialAutoStart(params)
708		elseif logic and logic.onFeedAutoStart then
709			logic:onFeedAutoStart(params)
710		end
711
712		runtimeExtra.feedNewUserType = params.feedNewUserType
713		local process = UBFProcessManager:launch(UBFGameProcessEnum.MANUAL_START, {
714			levelId = levelId,
715			levelScene = levelScene,
716			source = params.processSource,
717			extra = runtimeExtra,
718		})
719		process:operate(UBFProcessNodeEnum.WAIT_MANUAL, {
720			energyCostType = costType,
721			finish = true,
722		})
723		process:afterEnd():next(function()
724			releaseAngry()
725			if not isTrialLevelScene then
726				self._autoPopupMgr:resetPopup()
727			end
728		end)
729		process:afterAbort():next(function()
730			releaseAngry()
731			finishTrial({
732				reason = "start_level_aborted",
733			})
734		end)
735	end
736	startTrialLevel()
737
738	return true
739end
