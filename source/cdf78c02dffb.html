572    if needTip == nil then

580            TimerUtil.removeAlarm(preloadTimeoutId)

582        end
583    end
584

587            return

599        finishPreload(successCallback)

608        end
609        finishPreload(errorCallback)
610    end
611
612    local function onLoadCancel()
613        finishPreload(cancelCallback)
614    end
615
616    if timeoutSeconds then
617        preloadTimeoutId = TimerUtil.addAlarm(onLoadTimeout, timeoutSeconds)
618    end
619    AssetsUtil:loadDynamicModule('meta/Memefox.xml', onLoadSuccess, onLoadError, onLoadCancel, needTip)
