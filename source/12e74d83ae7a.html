83function GiftPanelVM:onButton()
84    if self.isDisposed then return end
85    if self:isLocked() then return end
86
87    self.tracer:click()
88
89    local buyPowerNum = 0
90    local rewards = {}
91    for _, v in pairs(self.goodsObj:getItems()) do
92        if v.itemId == Meta.CollectionId then
93            buyPowerNum = v.num
94        end
95        local r = {itemId = v.itemId, num = v.num}
96        rewards[#rewards + 1] = r
97    end
98    local function successCallback(bBuy)
99        if bBuy then
100            self:lock()
101            local sucCall = function(...)
102                local source = self.panelType == PanelType.MAIN_POP and "main_gift" or "flow_gift"
103                self.mgr:getData():addBuyGoodsCount(self.goodsId, 1)
104                -- self.mgr:addRewards(rewards, {
105                --     dcFeature = "happy_merge",
106                --     dcSource = source,
107                --     dontNeedAddNormal = true,
108                -- })
109
110                self:callViewFunc("flyRewards", function()
111                    if self.isDisposed then return end
112                    self:unlock()
113                    self:onCloseBtnTapped()
114                end, self.customFlyFunc)
115
116                safeRunCall( self.buySucFunc ) 
117            end
118
119            local failCall = function(...)
120                self:unlock()
121            end
122
123            local cancelCall = function(...)
124                self:unlock()
125            end
126
127            self.buyLogic:requestBuyWithTracer(sucCall, failCall, cancelCall, self.tracer)
128        end
129    end
130
131    local totalPower = self.mgr:getData():getPowerTotal()
132    if buyPowerNum > 0 and totalPower + buyPowerNum > Meta.MaxPowerConfig then
133        self.mgr:openSecondConfirmPanel(successCallback, totalPower, buyPowerNum)
134    else
135        successCallback(true)
136    end
137end
