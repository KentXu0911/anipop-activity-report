122-- 获取当前进度新达到且尚未弹过抽条的档位 cfg
123-- 返回 nil 表示没有新达到的档，不需要弹
124function Manager:_getCurrentTierCfg(value)
125    local bpModel = self:getModel()
126    local cfg = bpModel:getConfigByKey("battlePass", {})
127    local shownIndex = bpModel:getFlagFromLocalByKey("iconTipShownTierIndex", false) or 0
128    -- 找所有 value 达到的档中，index > shownIndex 的最大那档（即最新解锁的档）
129    local newTierCfg = nil
130    local newTierIndex = 0
131    for index = 1, #cfg do
132        local v = cfg[index]
133        if value >= (v.need or 0) and index > shownIndex then
134            if index > newTierIndex then
135                newTierCfg = v
136                newTierIndex = index
137            end
138        end
139    end
140    if newTierCfg then
141        -- 记录已弹到的最大档
142        bpModel:saveFlagToLocalByKey("iconTipShownTierIndex", false, newTierIndex)
143    end
144    return newTierCfg
145end

147function Manager:onActCheckIconTip(evt)
148    if not self:isEnabled() then return false end
149    local data = evt:getData()
150    if not data or not data.stage then return false end
151    -- 仅在场景2（过关后）和场景4（子页面返回）触发
152    if data.stage ~= "STAGE_TWO" and data.stage ~= "STAGE_FOUR" then return false end
153    local model = self:getModel()
154    local value = model:getBattlePassValue()
155    -- 尚未有进度则不触发
156    if value <= 0 then return false end
157    local tierCfg = self:_getCurrentTierCfg(value)
158    if not tierCfg then return false end
159    local need = tierCfg.need or 1
160    local taskFlag = model:getTaskFlag()
161    local TASK_LABEL_PREFIX = {
162        FMJ   = "过最高关",   -- 非满级
163        MJFMC = "补星",       -- 满级非满星
164        MJMC  = "随机挑战",   -- 满级满星
165    }
166    local labelPrefix = TASK_LABEL_PREFIX[taskFlag] or TASK_LABEL_PREFIX.FMJ
167
168    local delegate = {
169        iconKey  = tostring(self:getConst().actId),
170        iconType = IconTipType.MESSAGE,
171        typeIndex = 1,
172        creatorDelegate = {
173            tipsType           = HomeIconTipsType.PROGRESS,
174            min                = 0,
175            max                = need,
176            value              = 0,          -- 动画从0开始
177            targetValue        = math.min(value, need),
178            progressChangeTime = 0.8,
179            textMode           = 0,          -- n/m 格式
180            labelPrefixes      = labelPrefix,
181            templateId         = 1,          -- 满格时右侧展示单宝箱
182        },
183        config = { priority = "A" },
184    }
185
186    IconTipEventCenter:dispatch(IconTipEvent.ADD_ICON_TIP, delegate)
187end

333function Manager:_isAllTaskCompleted()
334    local value = self:getModel():getBattlePassValue()
335    local cfg   = self:getModel():getConfigByKey("battlePass", {})
336    if #cfg == 0 then return false end
337    local maxNeed = 0
338    for _, v in ipairs(cfg) do
339        if (v.need or 0) > maxNeed then
340            maxNeed = v.need
341        end
342    end
343    return value >= maxNeed
344end
