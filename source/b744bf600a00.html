312function ShopRewardStoreView:initTimeLimit()
313    local endTime = self.giftData:getEndTime()
314    if endTime and endTime > 0 then
315        local remain = endTime - TimeKit:timeInSec()
316        if remain > 0 then
317            self._tickId = TimerUtil.addAlarm(function()
318                if self.isDisposed then
319                    self:removeTimeLimitTimer()
320                    return
321                end
322                remain = endTime - TimeKit:timeInSec()
323                self:updateByTime(remain)
324                if remain <= 0 then
325                    self:removeTimeLimitTimer()
326                    self:onCountDownEnd()
327                end
328            end, 1, 0)
329            self:updateByTime(remain)
330        else
331            self:updateByTime(0)
332            self:onCountDownEnd()
333        end
334    end
335end
336
337function ShopRewardStoreView:removeTimeLimitTimer()
338    if self._tickId then
339        TimerUtil.removeAlarm(self._tickId)
340        self._tickId = nil
341    end
342end
