3local MAIN_PUSH_TYPE = 71
4local AWARD_PUSH_TYPE = 72
5local PUSH_HOUR = 18
6local SECONDS_PER_HOUR = 3600
7local SECONDS_PER_DAY = 86400
8
9-- 注册一条指定时间的本地推送，文案由 notification_v2 配置提供。
10local function addPush(pushType, timeStamp)
11    _G.amfTransaction(TransactionEvent.LOCALPUSH, {
12        action = "add",
13        type = pushType,
14        timeStamp = timeStamp,
15    })
16end
17
18-- 两个推送类型由每日夺宝独占，重新计算前先删除旧推送。
19function LocalPush:cancel()
20    if not LocalPushConst or EnvUtil:isMiniGame() then return end
21    for _, pushType in ipairs({MAIN_PUSH_TYPE, AWARD_PUSH_TYPE}) do
22        _G.amfTransaction(TransactionEvent.LOCALPUSH, {
23            action = "remove",
24            type = pushType,
25        })
26    end
27end
28
29function LocalPush:register(mgr)
30    if not LocalPushConst or EnvUtil:isMiniGame() then return end
31    -- 每次进入后台都以最新活动数据重建，避免保留已经失效或已领取的推送。
32    self:cancel()
33
34    -- 活动时间由服务端以毫秒下发，本地推送接口使用秒。
35    local now = TimeKit:timeInSec()
36    local beginTime = TimeKit:tryConvertMsToSec(mgr:getDataValue("beginTime", 0))
37    local mainEndTime = TimeKit:tryConvertMsToSec(mgr:getDataValue("mainEndTime", 0))
38    local endTime = TimeKit:tryConvertMsToSec(mgr:getDataValue("endTime", 0))
39    if not beginTime or not mainEndTime or not endTime or now >= endTime then return end
40
41    -- 遍历正式期内从今天开始的每一天，只在周六、周日的 18:00 推送。
42    local todayStart = TimeKit:getDayStartTimeByTS(now)
43    local dayStart = TimeKit:getDayStartTimeByTS(math.max(now, beginTime))
44    while dayStart < mainEndTime do
45        local pushTime = dayStart + PUSH_HOUR * SECONDS_PER_HOUR
46        -- 当天使用当前剩余次数判断；未来日期按跨天会刷新一次免费机会进行预测。
47        -- now < beginTime 表示活动尚未开始，此时首日也应按未来日期处理。
48        local hasPlayCount = dayStart > todayStart or now < beginTime or mgr:getMsgNum() > 0
49        -- 只添加尚未到达、且完整落在正式期内的周末 18:00 推送；18:00 后不补发当天推送。
50        if pushTime > now and pushTime >= beginTime and pushTime < mainEndTime
51            and isWeekend(pushTime) and hasPlayCount then
52            addPush(MAIN_PUSH_TYPE, pushTime)
53        end
54        dayStart = dayStart + SECONDS_PER_DAY
55    end
56
57    -- 不要求当前已经进入领奖期，按现有宝箱和奖券数据预测领奖期是否有奖可领。
58    if not mgr:canClaimAwardReward() then return end
59    -- 有奖可领时，为领奖期内每一天的 18:00 注册推送，已过 18:00 的当天不再补发。
60    dayStart = TimeKit:getDayStartTimeByTS(math.max(now, mainEndTime))
61    while dayStart < endTime do
62        local pushTime = dayStart + PUSH_HOUR * SECONDS_PER_HOUR
63        if pushTime > now and pushTime >= mainEndTime and pushTime < endTime then
64            addPush(AWARD_PUSH_TYPE, pushTime)
65        end
66        dayStart = dayStart + SECONDS_PER_DAY
67    end
68end
69
70return LocalPush
