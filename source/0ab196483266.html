344            self._isResLoadReady = true
345            d:resolve()
346        end, function()
347            d:reject()
348        end)
349    else
350        local function realLoad()
351             for i,v in ipairs(resList) do
352                if string.find(v, ".plist") then
353                    UICreateUtil:loadPlist(v)
354                end
355            end
356            self._isResLoadReady = true
357            d:resolve()
358        end
359
360        if not self._isResDownloadReady and not ignoreDownloadAll then
361            local function _onResReady()
362                self._isResDownloadReady = true
363                realLoad()
364            end
365
366            local failCallback = function()
367                d:reject()
368            end
369
370            self._assetsUtil:loadDynamicModule("meta/PromotionEvent2025Res.xml", _onResReady, failCallback, failCallback, not silence)
371        else
372            realLoad()
373        end
374    end
375
376    return d

460function PromotionEvent2025Manager:openMainAct(isUserClick, jumpInfo, fromBanner)
461    local promise = deferred.new()
462    local openParams = normalizeOpenParams(jumpInfo, fromBanner)
463
464    local function resolveOpen()
465        if isUserClick ~= true and self:isV2IconActive() then
466            local consumed, consumeError = consumeAdapterNew(self._iconAdapter, "auto_popup")
467            if consumed == false then
468                logIconV2Error("consume auto popup New failed: " .. tostring(consumeError))
469            end
470        end
471        promise:resolve()
472    end
473
474    local function rejectOpen()
475        promise:reject()
476    end
477
478    local function doOpenAct()
479        if not self:isEnabled() then
480            promise:reject()
481            return
482        end
483
484        if EnvUtil:isDevelopMode() then
485            local periodConfig = self:getCurPeriodConfig()
486            local taskType = periodConfig.taskType
487            if taskType ~= PromotionEvent2025Const.TaskType.kAnimal and taskType ~= PromotionEvent2025Const.TaskType.kLine then
488                CommonTip:showTip("拿不到当前期配置，是不是调时间了！")
489                promise:reject()
490                return
491            end
492        end
493
494        if self:isPreHeat() then
495            self:showBeforeOpenPanel(isUserClick, PromotionEvent2025Const.BeforeOpenPanelType.kNewOpen):next(resolveOpen, rejectOpen)
496        elseif self:isRest() then
497            self:showBeforeOpenPanel(isUserClick, PromotionEvent2025Const.BeforeOpenPanelType.kRestTip):next(resolveOpen, rejectOpen)
498        elseif not self:isResultConfirmed() then
499            self:showLastPeriodRankPanel(openParams):next(resolveOpen, rejectOpen)
500        elseif not self:getPreInFlag() then
501            if self:getCompleteTaskIndex() <= 0 then
502                self:showDanChangePanel(nil, nil, openParams):next(resolveOpen, rejectOpen)
503            else
504                self:showPreInRankPanel(isUserClick, openParams):next(resolveOpen, rejectOpen)
505            end
506        elseif self:getCompleteTaskIndex() >= 1 then
507            self:showMainPanel(isUserClick, openParams, nil, nil, nil, not self:getMainPanelFlag()):next(resolveOpen, rejectOpen)
508        else
509            self:showPreInRankPanel(isUserClick, openParams):next(resolveOpen, rejectOpen)
510        end
511    end
512
513    local completeCount = 0
514    local completeProcess = function()
515        completeCount = completeCount + 1
516        if completeCount >= 2 then
517            doOpenAct()
518        end
519    end
520
521    self:loadGlobalRes(false):next(completeProcess, function()
522        promise:reject()
523    end)
524
525    self:requestGetMainInfo():next(completeProcess, function()
526        promise:reject()
527    end)
528
529    return promise
530end

532-- 自动领奖流程(依次全领)
533function PromotionEvent2025Manager:tryToAutoGetRewards()
534    self._model:resetAndCalTaskDataForGetReward()
535
536    local playId = "no_play_id"
537    local prevSceneMgr = UBFPrevSceneManager:getInstance()
538    if prevSceneMgr then
539        local levelInfo = prevSceneMgr:getLevelInfo()
540        if levelInfo then playId = levelInfo.playId end
541    else
542        playId = "no_prev_scene"
543    end
544
545    local lastTaskProgressInfo = self:getDataTempValue("lastTaskProgressInfo")
546    local reward_level = ""
547    for i = 1, #lastTaskProgressInfo - 1 do
548        local tm = lastTaskProgressInfo[i]
549        if i == #lastTaskProgressInfo - 1 then
550            reward_level = reward_level .. tm.taskInfo.curTaskIndex
551        else
552            reward_level = reward_level .. tm.taskInfo.curTaskIndex .. "_"
553        end
554    end
555
556    local beforeTaskId = self:getCompleteTaskIndex()
557
558    local function autoAcquireRewards()
559        local isOverPeriod = self:isOverPeriod()
560        local taskInfo = self:getCurTaskInfo()
561        local canGetReward = self:canGetReward()
562        if not canGetReward or not taskInfo or isOverPeriod then
563            PromotionEvent2025DC:reward_progress(
564                1,
565                #lastTaskProgressInfo - 1,
566                reward_level == "" and 0 or reward_level,
567                playId
568            )
569            local curTaskId = self:getCompleteTaskIndex()
570            return
571        end
572
573        local requestReward = function()
574            self:requestGetRewardsAuto(taskInfo, false):next(function()
575                self._model:setTaskDataStateByTaskIndex(taskInfo.curTaskIndex, PromotionEvent2025Const.TaskState.kRewarded)
576                self._model:makeTaskBaseId(true)
577                self:updateMainSceneRewardBanner()
578                if taskInfo.curTaskIndex == 1 then
579                    self:requestGetCurGroupInfo()
580                end
581                autoAcquireRewards()
582            end, function()
583                self._model:setRewardFailFlag(true)
584                -- 领奖失败（断网/请求异常）：在本次过关的进度数据上追加 rewardFailed 字段（不动 taskState 语义），
585                -- banner 通过已有的 RefreshData 轮询读到后结束等待；超时兜底见 UBFPromotionTopUI:finishRewardWait
586                self._model:setTaskRewardFailedByTaskIndex(taskInfo.curTaskIndex, true)
587            end)
588        end
589
590        requestReward()
591    end
592
593    autoAcquireRewards()
594end

597function PromotionEvent2025Manager:handleRankData(rankData, needCheckMe)
598    local ret = table.clone(rankData.rankList or {})
599
600    if needCheckMe then
601        local uid = self._userManager:getUid()
602        local find = false
603        for index, rankInfo in ipairs(ret) do
604            rankInfo.rank = index
605            if tostring(rankInfo.uid) == tostring(uid) then
606                rankInfo.isMe = true
607                find = true
608            end
609        end
610
611        if not find then
612            table.insert(ret,{
613                rank = self:getRank(),
614                score = self:getScore(),
615                profile = self._userManager.profile,
616                isMe = true,
617                updateTime = 0,
618                uid = uid
619            })
620        end
621    end
622
623    for index, rankInfo in ipairs(ret) do
624		if not rankInfo.profile then
625            local profile = table.find(rankData.profiles or {}, function(v)
626                return tostring(v.uid) == tostring(rankInfo.uid)
627            end)
628
629            if not profile then
630                profile = {}
631                profile.headUrl = "[Unknown]"
632                profile.headFrame = HeadFrameType.kNormal
633            end
634
635            rankInfo.profile = profile
636		end
637	end
638
639    table.sort(ret, function(a, b)
640        if a.score > b.score then
641            return true
642        elseif a.score == b.score then
643            if tonumber(a.updateTime) ==  tonumber(b.updateTime ) then
644                return tonumber(a.uid ) <  tonumber(b.uid )
645            else
646                return tonumber(a.updateTime ) < tonumber(b.updateTime )
647            end
648        else
649            return false
650        end
651    end)
652
653    return ret
654end

1931function PromotionEvent2025Manager:_adoptIconAdapter(adapter, provider, recoveryPending)

1949function PromotionEvent2025Manager:_disposeIconAdapter(reason)
1950    local adapter = self._iconAdapter
1951    if not adapter then
1952        return self._iconAdapterRecoveryPending ~= true
1953    end
1954    local disposed, disposeError = disposeAdapter(adapter)
1955    if not disposed then
1956        self._iconAdapterRecoveryPending = true
1957        logIconV2Error(
1958            "dispose failed reason=" .. tostring(reason) .. " error=" .. tostring(disposeError)
1959        )
1960        return false, disposeError
1961    end
1962    self._iconAdapter = nil
1963    self._iconDataProvider = nil
1964    self._iconAdapterRecoveryPending = false
1965    return true
1966end

1968function PromotionEvent2025Manager:_recoverPendingIconAdapter()
1969    if not self._iconAdapterRecoveryPending then
1970        return true, self._iconAdapter ~= nil
1971    end
1972    if self._iconAdapter then
1973        local disposed, disposeError = self:_disposeIconAdapter("recovery")
1974        return disposed, false, disposeError
1975    end
1976
1977    local createResult = self:createPackageIconAdapter()
1978    local ownerAdapter = createResult.adapter or createResult.cleanupOwner
1979    if ownerAdapter then
1980        self:_adoptIconAdapter(
1981            ownerAdapter,
1982            createResult.provider,
1983            createResult.state == ADAPTER_CREATE_WAIT_OWNER
1984        )
1985        if createResult.state == ADAPTER_CREATE_READY and not self._iconAdapterRecoveryPending then
1986            return true, true
1987        end
1988        self._iconAdapterRecoveryPending = true
1989        logIconV2Error("owner recovery pending cleanup: " .. tostring(createResult.error))
1990        return false, false, createResult.error
1991    end
1992
1993    self._iconAdapterRecoveryPending = true
1994    logIconV2Error("owner recovery still pending: " .. tostring(createResult.error))
1995    return false, false, createResult.error
1996end

1998function PromotionEvent2025Manager:_releaseIconAdapter(reason)

2000        return self:_disposeIconAdapter(reason)
2001    end
2002    if not self._iconAdapterRecoveryPending then
2003        return true
2004    end
2005
2006    local recovered, hasAdapter, recoveryError = self:_recoverPendingIconAdapter()
2007    if not recovered then
2008        return false, recoveryError
2009    end
2010    if hasAdapter and self._iconAdapter then
2011        return self:_disposeIconAdapter(reason)
2012    end
2013    return true
