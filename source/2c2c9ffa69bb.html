32-- 判断当前缓存候选是否满足自动弹出条件，并输出具体卡点日志。
33function ActivityCenterBannerModule:canShowAutoPop()
34    -- 新关期间头部玩家屏蔽自动强弹，保持与资讯福利站入口迁移引导一致。
35    if _G.amfBean("AutoPopupMgr"):ignoreForTopLevelPlayer() then
36        return false
37    end
38
39    local actId = self.bannerCandidateActId
40    if not actId then
41        return false
42    end
43    if not self.bannerCandidateNewActKey then
44        return false
45    end
46    if not self:canShowNewActFlagByKey(self.bannerCandidateNewActKey, actId) then
47        return false
48    end
49    if not self:isCandidateResReady(actId) then
50        return false
51    end
52    if not self:getBannerData() then
53        return false
54    end
55    if self:hasShownBanner() then
56        return false
57    end
58    return true
59end

61-- 条件满足时注册 AutoPopup creator；真正弹出由强弹框架调度。
62function ActivityCenterBannerModule:tryAddAutoPop()
63    --检测强弹banner
64    if not self:canShowAutoPop() then return false end
65
66    local function creator()
67        self:showAutoPop(function()
68            AutoPopEventCenter:dispatch(AutoPopEvent.SINGLE_POP_OVER, PopupCheckPoint.ACTIVITY_CENTER_BANNER)
69        end)
70    end
71    _G.amfBean("AutoPopupMgr"):addPopupConfig(PopupCheckPoint.ACTIVITY_CENTER_BANNER, creator)
72    return true
73end

75-- 强弹执行入口：重新取 banner 数据，加载 UI 资源后交给左下角入口池展示。

126-- 通过 IconManager 间接展示 banner，避免 activitycenter middleware 依赖 icon feature 实现。
127function ActivityCenterBannerModule:showBannerInLeftPool(bannerData, onFinish)
128    local iconManager = _G.amfBean("IconManager")
129    if not iconManager or not iconManager.showActivityCenterBanner then
130        safeRunCall(onFinish)
131        return false
132    end
133
134    local function finishOnce()
135        if self._autoPopFinishCalled then return end
136        self._autoPopFinishCalled = true
137        safeRunCall(onFinish)
138    end
139
140    self._autoPopFinishCalled = false
141    return iconManager:showActivityCenterBanner({
142        bannerData = bannerData,
143        onShown = function()
144            self:recordBannerShown(bannerData and bannerData.recordKey)
145        end,
146        onClose = finishOnce,
147        onClick = function()
148            self.manager:showActivityCenterPanel(finishOnce)
149        end,
150    })
151end

204-- 从当前可见福利站活动中挑最新的、带 banner 图且新字未展示过的候选。
205function ActivityCenterBannerModule:getAutoPopCandidateInfo()
206    local candidateInfo
207    for _, data in pairs(self.manager:getVisibleDatas() or {}) do
208        if self:hasBannerConfig(data) then
209            local info = self:buildNewActInfo(data)
210            if self:canShowNewActFlag(info) then
211                if not candidateInfo or self:getDataBeginTime(data) > self:getDataBeginTime(candidateInfo.data) then
212                    candidateInfo = info
213                end
214            end
215        end
216    end
217    return candidateInfo
218end

220-- 刷新 banner 候选；主动从缓存挑选，避免已展示过的新活动覆盖可弹候选。
221function ActivityCenterBannerModule:scheduleUpdateCandidate()
222    local candidateInfo = self:getAutoPopCandidateInfo()
223    if not candidateInfo then
224        self._updateCandidateSeq = (self._updateCandidateSeq or 0) + 1
225        self:updateCandidate(nil, false)
226        return
227    end
228
229    self._updateCandidateSeq = (self._updateCandidateSeq or 0) + 1
230
231    local seq = self._updateCandidateSeq
232    local pendingInfo = self:cloneNewActInfo(candidateInfo)
233
234    setTimeOut(function()
235        if seq ~= self._updateCandidateSeq then return end
236        self:updateCandidate(pendingInfo, true)
237    end, ActivityCenterBannerModule.UPDATE_CANDIDATE_DELAY)
238end

240-- 复制候选关键字段，防止延迟期间原 info 被外部复用或改写。
241function ActivityCenterBannerModule:cloneNewActInfo(info)

246        newActKey = info.newActKey,
247    }

250-- 写入或清空当前 banner 候选，并在活动资源未加载时触发加载。
251function ActivityCenterBannerModule:updateCandidate(info, canShowNewActFlag)
252    if not canShowNewActFlag or not info or not info.hasNew or not info.data then
253        self.bannerCandidateActId = nil
254        self.bannerCandidateNewActKey = nil
255        self.bannerCandidateData = nil
256        self.loadingCandidateActId = nil
257        return
258    end
259
260    local oldActId = self.bannerCandidateActId
261    self.bannerCandidateActId = info.data.actId
262    self.bannerCandidateNewActKey = info.newActKey
263    self.bannerCandidateData = info.data
264    if oldActId ~= self.bannerCandidateActId then
265        self.loadingCandidateActId = nil
266    end
267
268    self:loadCandidateResIfNeeded()
269    self:tryAddAutoPop()
270end

272-- 候选活动资源缺失时发起加载；若本轮场景检查已过，则等待下一次强弹检查。
273function ActivityCenterBannerModule:loadCandidateResIfNeeded()
274    local actId = self.bannerCandidateActId
275    if not actId then return end
276    if self:isCandidateResReady(actId) then
277        self.loadingCandidateActId = nil
278        return
279    end
280    if self.loadingCandidateActId == actId then
281        return
282    end
283
284    self.loadingCandidateActId = actId
285    self.manager:loadActivityRes(actId, function()
286        self.loadedCandidateActIdMap[tostring(actId)] = true
287        if self.loadingCandidateActId == actId then
288            self.loadingCandidateActId = nil
289        end
290    end, function()
291        if self.loadingCandidateActId == actId then
292            self.loadingCandidateActId = nil
293        end
294    end)
295end

297-- 判断候选资源是否已可用；福利站加载成功不会回写 ActivityNode 状态，这里用本模块缓存兜底。
298function ActivityCenterBannerModule:isCandidateResReady(actId)
299    if not actId then return false end
300    if self.manager._activityMgr:isActResLoaded(actId) then
301        return true
302    end
303
304    local data = self:getCandidateData()
305    if data and data.panelCls then
306        return true
307    end
308
309    return self.loadedCandidateActIdMap[tostring(actId)] == true
310end

312-- 按候选 actId 读取最新活动数据；找不到时使用延迟前缓存的数据。
313function ActivityCenterBannerModule:getCandidateData()

315    if not actId then return nil end
316

318    return datas and datas[tostring(actId)] or self.bannerCandidateData

321-- 组装 banner 展示数据，并过滤缺少图片的活动。
322function ActivityCenterBannerModule:getBannerData()
323    local data = self:getCandidateData()
324    local bannerData = data and data.activityCenterBanner
325    if not bannerData or not bannerData.image or bannerData.image == "" then return nil end
326    return self:buildBannerData(data, bannerData, self:getBannerRecordKey())

329-- 统一组装 banner 展示数据；recordKey 固化本次展示记录，避免异步期间候选切换后记错活动。
330function ActivityCenterBannerModule:buildBannerData(data, bannerData, recordKey)
331    return {
332        image = self:normalizeBannerImagePath(bannerData.image),
333        name = bannerData.name or "",
334        actId = data.actId,
335        hasReward = self.manager:hasRewardByActId(data.actId),
336        recordKey = recordKey,
337    }

340-- 兼容配置里省略 activity/ 前缀的图片路径。
341function ActivityCenterBannerModule:normalizeBannerImagePath(image)
342    if string.starts(image, "activity/") then return image end
343    return "activity/" .. image
344end

346-- 生成单活动 banner 已弹记录 key，默认使用当前候选的新字 key。
347function ActivityCenterBannerModule:getBannerRecordKey(newActKey)
348    local key = newActKey or self.bannerCandidateNewActKey
349    if not key then return nil end
350    return tostring(key) .. ActivityCenterBannerModule.BANNER_RECORD_SUFFIX
351end

353-- 判断当前候选 banner 是否已经真实展示过。
354function ActivityCenterBannerModule:hasShownBanner()
355    local key = self:getBannerRecordKey()
356    if not key then return true end
357    local shownMap = self:getShownBannerMap()
358    return shownMap[key] == true
359end

361-- 记录本次 banner 已展示；recordKey 来自展示前冻结的 bannerData，不能再回读当前候选。
362function ActivityCenterBannerModule:recordBannerShown(recordKey)
363    if recordKey then
364        local shownMap = self:getShownBannerMap()
365        shownMap[recordKey] = true
366        self.manager:setLocalCache(ActivityCenterBannerModule.BANNER_RECORD_CACHE_KEY, shownMap)
367    end
368end

370-- 读取本地已弹记录表，缓存异常时兜底为空表。
371function ActivityCenterBannerModule:getShownBannerMap()
372    local shownMap = self.manager:getLocalCache(ActivityCenterBannerModule.BANNER_RECORD_CACHE_KEY)
373    if type(shownMap) ~= "table" then
374        shownMap = {}
375    end
376    return shownMap
377end
