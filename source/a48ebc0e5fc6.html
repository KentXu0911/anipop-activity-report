61function Manager:parseParams(method, params)

71function Manager:syncTimer()
72    local shouldRun = self.launch and self.foreground and self.elapsed < Const.ONLINE_SECONDS
73    if shouldRun and not self.timerId then
74        self.timerId = TimerUtil.addAlarm(function()

76            if self.elapsed >= Const.ONLINE_SECONDS then
77                self:requestCloudMusicTaskReport2026(Const.REPORT_TYPE_ONLINE)
78                self:syncTimer()
79            end
80        end, 1, 0)
81    elseif not shouldRun and self.timerId then
82        TimerUtil.removeAlarm(self.timerId)
