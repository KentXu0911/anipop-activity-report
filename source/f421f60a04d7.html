43function MusicMasterPopMudule:checkCanPopStage4()
44	--场景4强弹主面板，每期一次
45	if not self.mgr:isEnabledAndResReady() then return end
46	if not self:checkCurTimeCanPop() then return end
47
48    local model = self.mgr:getModel()
49    local mainEndTime = model:getMainEndTime()
50    local hasEntered = (tonumber(mainEndTime) or 0) > 0
51        and model:getValue("inLevelCardParticipatedEndTime") == mainEndTime
52    if not hasEntered
53        and _G.amfBean("ActivityParticipationMgr"):shouldReduceAutoPopup(self.mgr:getConst().ACT_ID) then
54        return false
55    end
56
57	local bHadGuide = self.mgr:getModel():checkGuide( self.mgr:getConst().GuideIds.kPopPrePanel )
58	if not bHadGuide then
59		return true
60	end
61	return false
62end

225function MusicMasterPopMudule:checkPopProgressTip(popInfo,data)

227    if self.mgr:getModel():checkInPre() then---在预选赛阶段
228        local nowLevelIsPassed = data.levelPassedInfo
229        local levelState = self.mgr:getModel():getLocalValueByKey("play_level_state")
230        if not levelState then
231            return false
232        end

234            popInfo.popProgressTip = {
235                canPop = true,
236                max = self.mgr:getModel():getStreakDoorsillConfig(),
237                value = self.mgr:getModel():getPreWinNum(),

239            return true
240        end
241    else
242        return false
243    end

246function MusicMasterPopMudule:checkPopHasNewMatchTip(popInfo,data)
247    popInfo.popHasNewMatchTip = {}
248    local showType, groupPanelShowType = self.mgr:getModel():getPanelShowType()
249    local popTimes = self.mgr:getModel():getLocalValueByKey("new_match_pop_times",true) or 0
250    if popTimes >= 0 and popTimes < 5 and showType == self.mgr:getConst().MainPanelShowType.kGroup and groupPanelShowType == self.mgr:getConst().GroupPanelShowType.kWaitJoin then
251        popTimes = popTimes + 1
252        self.mgr:getModel():setLocalValueByKey("new_match_pop_times",true,popTimes)
253        popInfo.popHasNewMatchTip = {
254            canPop = true,
255        }
256        return true
257    else
258        return false
259    end

263function MusicMasterPopMudule:popProgressTip(popInfo,done)

271            min = 0,
272            textMode = 0,
273            labelPrefixes = "连胜",
274            -- templateId = 1,
275            progressChangeTime = 0.3,
276        }
277        
278        delegate.tipsType = HomeIconTipsType.PROGRESS
279
280        local IconTipEventCenter = require("amf.feature.icontip.event.IconTipEventCenter")
281        IconTipEventCenter:dispatch(IconTipEvent.ADD_ICON_TIP_DIRECTLY, {
282            iconKey = IconKey.kMusicMasterIcon,
283            iconType = IconTipType.MESSAGE,
284            creatorDelegate = delegate,
285            typeIndex = 1,
