646--当前任务信息
647function Model:getCurTaskInfo()
648	local curTaskIndex = self:getCurTaskIndex()
649	local tasks = self:getConfigValue("tasks")
650
651	self:makeTaskBaseId()
652
653    local taskBaseId = self:getValue("taskBaseId")
654    local taskAreaConfig = tasks[tostring(taskBaseId)]
655
656    for k,v in pairs(taskAreaConfig) do
657        if curTaskIndex == v.taskIndex then
658            local normalRewards, useSpecialRewards = self:getTaskBaseRewards(v, taskBaseId)
659            local oriRewards = table.clone(normalRewards)
660            normalRewards = self:getReplaceRewards(normalRewards,v.taskIndex)
661            local replaceRewards = table.clone(normalRewards)
662            local extraReward = self:getExtraRewards({
663                total = v.target,
664                taskBaseId = taskBaseId,
665                rewards = normalRewards,
666                taskIndex = v.taskIndex,
667            })
668            local rewards = table.union(normalRewards, extraReward)
669
670            return {
671                total = v.target,
672                rewards = rewards,
673                online = v.forceConnect,
674                curTaskIndex = curTaskIndex,
675                taskBaseId = taskBaseId,
676                isLast = curTaskIndex == #taskAreaConfig,
677                curBoxIcon = (v.boxIcon ~= "") and v.boxIcon,
678                extraReward = extraReward,
679                replaceRewards = replaceRewards,
680                oriRewards = oriRewards,
681                useSpecialRewards = useSpecialRewards,
682            }
683        end
684    end

841function Model:CalTaskDataForGetReward()
842
843    local taskIndex = self:getCompleteTaskIndex()
844    local curTaskIndex = self:getCurTaskIndex()
845
846    local tasks = self:getConfigValue("tasks")
847    local collectionNum = self:getCollectNum()
848    local retList = {}
849
850    -- 先不刷新塞当前的进去
851    self:makeTaskBaseId()
852
853    local taskBaseId = self:getValue("taskBaseId")
854
855    local taskAreaConfig = tasks[tostring(taskBaseId)]
856    if not taskAreaConfig then return end -- 所有任务已经完成了
857    local taskConfig = taskAreaConfig[tonumber(curTaskIndex)]
858    if not taskConfig then return end -- 所有任务已经完成了
859
860    local taskState = PromotionEvent2025Const.TaskState.kUnfinish
861    if collectionNum >= taskConfig.target then
862        taskState = PromotionEvent2025Const.TaskState.kFinished
863    end
864
865
866    local normalRewards, useSpecialRewards = self:getTaskBaseRewards(taskConfig, taskBaseId)
867    normalRewards = self:getReplaceRewards(normalRewards,taskConfig.taskIndex)
868    local extraReward = self:getExtraRewards({
869        total = taskConfig.target,
870        taskBaseId = taskBaseId,
871        rewards = normalRewards,
872        taskIndex = taskConfig.taskIndex,
873    })
874
875
876    local rewards = table.union(normalRewards, extraReward)
877
878    table.insert(retList, {
879        taskInfo ={
880            total = taskConfig.target,
881            rewards = rewards,
882            online = taskConfig.forceConnect,
883            curTaskIndex = curTaskIndex,
884            taskBaseId = taskBaseId,
885            isLast = curTaskIndex == #taskAreaConfig,
886            curBoxIcon = (taskConfig.boxIcon ~= "") and taskConfig.boxIcon,
887            useSpecialRewards = useSpecialRewards,
888        },
889        taskState = taskState,
890        collectionNum = collectionNum,
891    })
892    curTaskIndex = curTaskIndex + 1
893    if collectionNum >= taskConfig.target then
894        collectionNum = collectionNum - taskConfig.target
895    end
896
897    -- 如果还有再塞剩下的，注意从下一个开始要算等级跃迁，要先刷新
898    -- 2025.5.12 修改，不能无脑重置，当前奖也没有时会把当前也刷掉。
899    -- 当前之后的奖励应当永远是现算的
900    local newTaskBaseId = self:getTaskBaseId()
901    local taskAreaConfig = tasks[tostring(newTaskBaseId)]
902    local totalNum = #taskAreaConfig
903
904    for k,v in pairs(taskAreaConfig) do
905        if curTaskIndex == v.taskIndex then
906
907            local normalRewards, useSpecialRewards = self:getTaskBaseRewards(v, newTaskBaseId)
908            normalRewards = self:getReplaceRewards(normalRewards,v.taskIndex)
909
910            local extraReward = self:getExtraRewards({
911                total = v.target,
912                taskBaseId = taskBaseId,
913                rewards = normalRewards,
914                taskIndex = v.taskIndex,
915            })
916
917            local rewards = table.union(normalRewards, extraReward)
918
919            local taskInfo = {
920                total = v.target,
921                rewards = rewards,
922                online = v.forceConnect,
923                curTaskIndex = curTaskIndex,
924                taskBaseId = newTaskBaseId,
925                isLast = curTaskIndex == totalNum,
926                curBoxIcon = (v.boxIcon ~= "") and v.boxIcon,
927                useSpecialRewards = useSpecialRewards,
928            }
929
930            local taskState = PromotionEvent2025Const.TaskState.kUnfinish
931            if collectionNum >= v.target then
932                taskState = PromotionEvent2025Const.TaskState.kFinished
933            end
934
935            table.insert(retList, {
936                taskInfo = taskInfo,
937                taskState = taskState,
938                collectionNum = collectionNum,
939            })
940
941            if collectionNum >= v.target then
942                collectionNum = collectionNum - v.target
943                curTaskIndex = curTaskIndex + 1
944            else
945                break
946            end
947        end
948    end
949
950    return retList
951end
