118function WinRepeatRankManager:_getIconV2PeriodTimes(extra)
119    local data = self:getData()
120    local stageOneBeginTime = firstNumber(

122        millisToSeconds(data:getStageOneBeginTime()),
123        self:getStartTime()
124    )
125    local stageOneEndTime = firstNumber(
126        millisToSeconds(extra.stageOneEndTime),
127        millisToSeconds(data:getStageOneEndTime())
128    )

130        millisToSeconds(extra.stageTwoBeginTime),
131        millisToSeconds(data:getStageTwoBeginTime()),

134    local stageTwoEndTime = firstNumber(
135        millisToSeconds(extra.stageTwoEndTime),
136        millisToSeconds(data:getStageTwoEndTime()),
137        self:getEndTime()
138    )
139    local rewardEndTime = firstNumber(
140        millisToSeconds(extra.rewardEndTime),
141        self:getRealEndTime()
142    )
143
144    if not stageOneBeginTime or not stageOneEndTime
145        or not stageTwoBeginTime or not stageTwoEndTime or not rewardEndTime
146        or stageOneEndTime <= stageOneBeginTime
147        or stageTwoEndTime <= stageTwoBeginTime
148        or rewardEndTime < stageTwoEndTime then
149        return nil, "WIN_REPEAT_RANK_ICON_V2_PERIOD_INVALID"
150    end
151
152    return {
153        stageOneBeginTime = stageOneBeginTime,
154        stageOneEndTime = stageOneEndTime,
155        stageTwoBeginTime = stageTwoBeginTime,
156        stageTwoEndTime = stageTwoEndTime,
157        rewardEndTime = rewardEndTime,
158    }

190function WinRepeatRankManager:getIconV2Snapshot(now, overrides)
191    local actInfo = self:getActInfo()
192    local extra, extraError = self:_getIconV2ActExtra(actInfo)
193    if not extra then
194        return nil, extraError
195    end
196    local times, periodError = self:_getIconV2PeriodTimes(extra)
197    if not times then
198        return nil, periodError
199    end
200
201    now = tonumber(now) or TimeKit:timeInSec()
202    local data = self:getData()
203    overrides = type(overrides) == "table" and overrides or nil
204    local modelRankType = data:getCurMainTimeShowType()
205    local enterGroup = firstBoolean(
206        overrides and overrides.enterGroup,
207        data:checkIsEnterGroup(),
208        extra.entryGroup
209    )
210    if modelRankType == Meta.mainPanelShowType.kMainPanel0 then
211        enterGroup = false
212    end
213    local rankType = firstValidRaceType(
214        overrides and overrides.rankType,
215        modelRankType,
216        extra.raceType
217    ) or Meta.mainPanelShowType.kMainPanel1
218    local state = self:_resolveIconV2DisplayState(now, times, enterGroup)
219    local label, phaseStartTime, phaseEndTime, timedLabelMode, reachedText =
220        self:_resolveIconV2TimedLabel(now, times, state, rankType)
221    local available = now >= times.stageOneBeginTime and now <= times.rewardEndTime
222    local newActivityPermitted = available
223        and state ~= self.IconV2State.PENDING_SETTLEMENT
224    local hasReward = self:hasRewards() == true
225    local rankValue = actInfo.rank
226    if overrides and overrides.rank ~= nil then
227        rankValue = overrides.rank
228    end
229    local rank = enterGroup and resolvePositiveRank(rankValue) or 0
230    local assetAnimation = enterGroup
231        and rankType == Meta.mainPanelShowType.kMainPanel3
232        and "lv2"
233        or "lv1"
234
235    return {
236        enabled = available,
237        newActivityPermitted = newActivityPermitted == true,
238        activityState = state,
239        periodKey = table.concat({
240            "winrepeatrank2024",
241            tostring(self.ACT_ID),
242            tostring(times.stageOneBeginTime),
243            tostring(times.rewardEndTime),
244        }, ":"),
245        phaseStartTime = phaseStartTime,
246        phaseEndTime = phaseEndTime,
247        timedLabelMode = timedLabelMode,
248        reachedText = reachedText,
249        label = label,
250        rank = rank,
251        rankType = rankType,
252        enterGroup = enterGroup,
253        hasReward = hasReward,
254        assetAnimation = assetAnimation,
255    }
256end

520function WinRepeatRankManager:updateIconActInfo(actInfoPatch, extraPatch)
521    local actInfo = self:getActInfo()
522    if type(actInfo) ~= "table" then
523        return false, "WIN_REPEAT_RANK_ACT_INFO_REQUIRED"
524    end
525
526    local nextActInfo = table.clone(actInfo)
527    if type(extraPatch) == "table" then
528        local extra, extraError = cloneActInfoExtra(actInfo)
529        if not extra then
530            return false, extraError
531        end
532        applyPatch(extra, extraPatch)
533        nextActInfo.extra = table.serialize(extra)
534    end
535    applyPatch(nextActInfo, actInfoPatch)
536    nextActInfo.fromLocal = true
537
538    ActivityEventCenter:dispatch(ActivityEvent.SET_ACT_INFO, nextActInfo)
539    return true
540end

543function WinRepeatRankManager:onActActInfoUpdate(event)
544    local data = event:getData()
545    local actInfos = data.actInfos or {}
546    local fromServer = data.fromServer
547    local actInfo = actInfos[tostring(WinRepeatRankManager.ACT_ID)]
548
549    -- 没有actInfo，且是从服务器发来的，清空数据
550    if not actInfo and fromServer then
551        self:getData():reset()
552        local ActivityEventCenter = require "amf.middleware.activity.event.ActivityEventCenter"
553        ActivityEventCenter:dispatch(ActivityEvent.REMOVE_ACTIVITY,{actId = WinRepeatRankManager.ACT_ID})
554        return
555    end
556
557    if actInfo then
558        if self:isEnabled() and (not actInfo.fromLocal) then
559            -- 更新最新的领奖数据
560            self:getData():updateDataFromActInfo(actInfo)
561        end
562
563        local extra = table.deserialize(actInfo.extra or "")
564        -- 更新icon
565        local enterGroup = extra.entryGroup
566        local rankType = extra.raceType
567        local rank = actInfo.rank or 0
568
569        self:refreshIcon(rank, rankType, enterGroup)
570    end

655        if data.json then

858        bIsUp = true

862        --下降

864        bCanShowIconTip = true
865    end
866
867    return bCanShowIconTip, bIsUp, newRank, bOldIsDown
868end

1088function WinRepeatRankManager:processServerRankData(eventData)
1089    -- 把数据存起来
1090    self.eventDataInfo = table.clone(eventData.data)
1091
1092    local meUid = _G.amfBean("ProfileManager"):getXxlId()
1093
1094    local newScore = 0
1095    local oldScore = 0
1096
1097    local oldShowScore = self:getFlagFromLocalByKey(Meta.SaveOldRankScoreKey) or 0
1098    local oldShowRank = self:getFlagFromLocalByKey(Meta.SaveOldRankKey) or 0
1099
1100    -- 把profile整理进去
1101    local profilesByUid = {}
1102    local profiles = self.eventDataInfo.activityGroupCommonList.profiles
1103    local numProfiles = profiles and #profiles or 0
1104
1105    for i = 1, numProfiles do
1106        local profileItem = profiles[i]
1107        profilesByUid[profileItem.uid] = profileItem
1108    end
1109
1110    for i, v in ipairs(self.eventDataInfo.activityGroupCommonList.rankList) do
1111        if v.uid == meUid or (tonumber(v.uid) == tonumber(_G.amfBean("ProfileManager"):getUid())) then
1112            v.isMe = true
1113            -- v.score = 100001
1114        end
1115
1116        v.profile = v.isMe and _G.amfBean("ProfileManager"):getProfile() or profilesByUid[v.uid]
1117    end
1118
1119    -- 插入无人上榜
1120    local allNum = 15
1121    if self.eventDataInfo.raceType == Meta.mainPanelShowType.kMainPanel1 or self.eventDataInfo.raceType == Meta.mainPanelShowType.kMainPanel2 then
1122        allNum = 15
1123    elseif self.eventDataInfo.raceType == Meta.mainPanelShowType.kMainPanel3 then
1124        allNum = 5
1125    end
1126
1127    local curRankNum = #self.eventDataInfo.activityGroupCommonList.rankList
1128    if curRankNum < allNum then
1129        local addNum = allNum - curRankNum
1130        for i = 1, addNum do
1131            local info = {}
1132            info.score = -1
1133            info.uid = 12345 + i
1134            info.isEmpty = true
1135            info.updateTime = 0
1136            info.profile = {
1137                uid = info.uid
1138            }
1139            table.insert(self.eventDataInfo.activityGroupCommonList.rankList, info)
1140        end
1141    end
1142
1143    -- 排序
1144    local function sortRankList(rankList)
1145        -- 按分数排序
1146        local meRank = 0
1147        table.sort(rankList, function(a, b)
1148            if a and b and a.score and b.score then
1149                if a.score == b.score then
1150                    if a.updateTime == b.updateTime then
1151                        return a.uid > b.uid
1152                    else
1153                        return tonumber(a.updateTime) < tonumber(b.updateTime)
1154                    end
1155                else
1156                    return a.score > b.score
1157                end
1158            end
1159
1160            return false
1161        end)
1162
1163        for i, v in ipairs(rankList) do
1164            v.rank = i
1165
1166            if v.isMe then
1167                meRank = v.rank
1168            end
1169        end
1170
1171        return meRank
1172    end
1173
1174    local newRank = sortRankList(self.eventDataInfo.activityGroupCommonList.rankList)
1175
1176    -- 把本次获得的扣除, 记录下newScore。 使用记录的旧分数
1177    for i, v in ipairs(self.eventDataInfo.activityGroupCommonList.rankList) do
1178        if v.isMe then
1179            newScore = v.score
1180            v.score = v.score - self.eventDataInfo.addCollections
1181            oldScore = v.score
1182            break
1183        end
1184    end
1185
1186    -- 排序2
1187    local oldRank = sortRankList(self.eventDataInfo.activityGroupCommonList.rankList)
1188
1189    self.eventDataInfo.newRank = newRank
1190    self.eventDataInfo.newScore = newScore
1191    self.eventDataInfo.oldRank = oldRank
1192    self.eventDataInfo.oldScore = oldScore
1193
1194    -- 旧的展示排名， 播动画用
1195    self.eventDataInfo.oldShowScore = oldShowScore
1196    self.eventDataInfo.oldShowRank = oldShowRank
1197
1198    self:saveFlagToLocalByKey(Meta.SaveOldRankScoreKey, false, newScore)
1199    self:saveFlagToLocalByKey(Meta.SaveOldRankKey, false, newRank)
1200
1201    --
1202    self:setEventDataUpdated(true)
1203    return newRank
1204end

1332-- 加载服务器排行数据
1333function WinRepeatRankManager:requestServerRankData(successCallback, failCallback)
1334    local raceType = self:getData():getCurMainTimeShowType()
1335    if raceType == Meta.mainPanelShowType.kMainPanel0 then
1336        safeRunCall(failCallback)
1337        return
1338    end
1339    self._rankDataGeneration = (self._rankDataGeneration or 0) + 1
1340    local requestGeneration = self._rankDataGeneration
1341    local function onSuccess(evt)
1342        local data = evt.data
1343        local newProfileList = {}
1344        if data.profiles then
1345            for i,v in ipairs(data.profiles or {}) do
1346                local newProfile = {
1347                    headUrl = v.headUrl,
1348                    uid = v.uid,
1349                }
1350                table.insert(newProfileList, newProfile)
1351            end
1352            data.profiles = newProfileList
1353            data.groupSize = nil
1354            data.rankRewardAcquired = nil
1355            data.rewards = nil
1356        end
1357        local eventData = {
1358            activityGroupCommonList = data,
1359            addCollections = 0,
1360            raceType = raceType,
1361        }
1362        if self._rankDataGeneration == requestGeneration
1363            and self:getData():getCurMainTimeShowType() == raceType then
1364            local newRank = self:processServerRankData({data=eventData})
1365            self:updateIconActInfo({rank = newRank})
1366        end
1367        safeRunCall(successCallback)
1368    end
1369
1370    local function onFail(evt)
1371        safeRunCall(failCallback, evt)
1372    end
1373
1374    local request = require("amf.feature.localactivity.winrepeatrank.http.RequestRankList")
1375    local reqParams = {activityId = raceType,}
1376    request:post(reqParams):next(onSuccess, onFail)
1377end
