59function Handler:isShareEnabled(query)
60    if not tonumber(query.actId) or (tonumber(query.actId) ~= tonumber(self:getActId())) then return end
61
62    local shareType = tostring(query.shareType)
63    if shareType ~= tostring(self:getShareType()) and shareType ~= tostring(OpenShareConst.ShareType.HELP_ACTIVITY) then

83    end

88    end
89
90    local config = self:getConfig()

92    local respHelpContext = self:makeResHelpContext(query)
93    if isFunc(config.isVersionSupport) and not config.isVersionSupport() then
94        self:dcAfterHelpFail(respHelpContext)
95        self:actCheckAlertPanel(736431)
96        return
97    end
98
99    if isFunc(config.hasUpdateFlag) and not config.hasUpdateFlag() then
100        self:dcAfterHelpFail(respHelpContext)
101        self:actCheckAlertPanel(736431)
102        return
103    end
104
105    if isFunc(config.isActBegin) and not config.isActBegin() then
106        self:dcAfterHelpFail(respHelpContext)
107        self:actCheckAlertPanel(localize("error.tip." .. tostring(736433), {
108            actName = self:getActName()
109        }))
110        return
111    end
112
113    if isFunc(config.isActEnd) and config.isActEnd() then
114        self:dcAfterHelpFail(respHelpContext)
115        self:actCheckAlertPanel(localize("error.tip." .. tostring(736433), {
116            actName = self:getActName()
117        }))
118        return
119    end
120
121    if isFunc(config.isUnSupportPkg) and config.isUnSupportPkg() then
122        self:dcAfterHelpFail(respHelpContext)
123        self:actCheckAlertPanel(localize("error.tip." .. tostring(736432), {
124            actName = self:getActName()
125        }))
126        return
127    end
