87function BaseFeatureModel:setLocalValueByKey(key, needTime, value)
88    if not key then return end
89    -- if bSkip == nil then bSkip = true end
90
91    local nowTime = getDayStartTimeByTS(TimeKit:timeInSec())
92    local actBeginTime = self:getMainBeginTime() or 0
93
94    local saveValue = {
95        time = nowTime,
96        value = value,
97        actBeginTime = actBeginTime,
98    }
99    local saveStr = table.serialize(saveValue)
100
101    self:setValue(key, saveStr, true)
102end

104function BaseFeatureModel:getLocalValueByKey(key, needTime, needActStartTimeCheck)
105    if not key then return end
106    local saveJsonStr = self:getValue(key)
107    if saveJsonStr == nil then return end
108    if type(saveJsonStr) ~= "string" then return end
109
110    local saveTable = table.deserialize(saveJsonStr)
111
112    local nowTime = getDayStartTimeByTS(TimeKit:timeInSec())
113    local actBeginTime = self:getMainBeginTime() or 0
114
115    local bTimeIsOk = true
116    if needTime then
117        if saveTable.time ~= nowTime then
118            bTimeIsOk = false
119        end
120    end
121
122    local versionIsOk = true
123    if needActStartTimeCheck then
124        if saveTable.actBeginTime ~= actBeginTime then
125            versionIsOk = false
126        end
127    end
128
129    if bTimeIsOk and versionIsOk then
130        return saveTable.value
131    end
132end
