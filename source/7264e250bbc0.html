190function ServerData:getENum(bShowLog, levelState)
191	--获取E值
192	local eInfo = self:getDataByKey('eInfo')
193	if eInfo == nil then 
194		return false, {e=0, x=0, y=0}
195	end
196
197	local function checkHasTimeKey( costLogs )
198		local bHasTimeKey = true
199		for i,v in ipairs(costLogs) do
200			if v.time == nil then
201				bHasTimeKey = false
202				break
203			end
204		end
205		return bHasTimeKey
206	end
207
208	if not checkHasTimeKey(eInfo.costLogs) or not checkHasTimeKey(eInfo.extraRewardLogs) then 
209		return false, {e=0, x=0, y=0}
210	end
211
212	local getPlayLevelNum = 100 --取100次打关数据
213
214	local function sortFunc( a,b )
215		return tonumber(a.time) > tonumber(b.time)
216	end
217
218	local copyCostLogs = {}
219	if levelState == Meta.LevelStateList.kLevelLibrary then
220		copyCostLogs = table.clone(eInfo.fullColorStarLogs)
221	elseif levelState == Meta.LevelStateList.kRefreshStar then
222		copyCostLogs = table.clone(eInfo.farmStarLogs)
223	else
224		copyCostLogs = table.clone(eInfo.costLogs)
225	end
226	table.sort( copyCostLogs, sortFunc )
227
228	local newCostLogs = {}
229	for i,v in ipairs(copyCostLogs) do
230		if #newCostLogs < getPlayLevelNum then
231			table.insert( newCostLogs, table.clone(v) )
232		end
233	end
234
235	--找出第一关的时间，么么狐领取时间只要大于startTime 就加入判断
236	local startTime = TimeKit:timeInSec()*1000
237	if #newCostLogs > 0 then
238 		startTime = tonumber(newCostLogs[#newCostLogs].time)
239	end
240
241	local newExtraRewardLogs = {}
242	for i,v in ipairs(eInfo.extraRewardLogs) do
243		if tonumber(v.time) >= startTime then
244			table.insert( newExtraRewardLogs, table.clone(v) )
245		end
246	end
247
248	--区分有鸟无鸟
249	local birdCostLogs = {}
250	local noBirdCostLogds = {}
251	local allCost = 0
252	local birdCost = 0
253	local noBirdCost = 0
254
255	for i, v in ipairs(newCostLogs) do
256		if v.bird then
257			table.insert(birdCostLogs, v)
258
259			birdCost = birdCost + v.cost
260		else
261			table.insert(noBirdCostLogds, v)
262
263			noBirdCost = noBirdCost + v.cost
264		end
265
266		allCost = allCost + v.cost
267	end
268
269	local actGetNum = 0 
270	for i, v in ipairs(newExtraRewardLogs) do
271		actGetNum = actGetNum + v.num
272	end
273
274	local birdLevelNum = #birdCostLogs
275	local noBirdLevelNum = #noBirdCostLogds
276	if noBirdLevelNum == 0 then noBirdLevelNum = 1 end
277	if birdLevelNum == 0 then birdLevelNum = 1 end
278
279	local x = noBirdCost / noBirdLevelNum
280	local y = ( birdCost-actGetNum ) / birdLevelNum
281	local e = y - x
282
283	local bEnough = false
284
285	if e > 0 then
286		bEnough = true
287	else
288		bEnough = false
289	end
290	
291	if bShowLog then
292		log("*******getENum***********")
293		
294		log("eInfo", eInfo)
295		log("allCost", allCost)
296		log("birdCost", birdCost)
297		log("noBirdCost", noBirdCost)
298		log("noBirdLevelNum", noBirdLevelNum, noBirdCostLogds)
299		log("birdLevelNum", birdLevelNum, birdCostLogs)
300		log("actGetNum", actGetNum)
301		log("x", x)
302		log("y", y)
303		log("e", e)
304
305		log("*******getENum end***********", bEnough)
306	end
307
308	return bEnough, { e=e, x=x, y=y }
309end

382function ServerData:get80PercentNum( bShowLog, topLevelId )
383	--取出最近的100关消耗，并排序，找出80%位置，向上取整
384	local getLevelCostNum = 100
385	local usePropInfos = self:getDataByKey('usePropInfos')
386
387	if bShowLog then
388		log("***超鸟连胜fuuu判断V2*** *******get80PercentNum before***********")
389		log("usePropInfos",usePropInfos)
390		log("topLevelId",topLevelId)
391		log("***超鸟连胜fuuu判断V2*** *******get80PercentNum before end***********")
392	end
393	
394	if usePropInfos == nil then
395		return 0
396	end
397
398	local curTime = TimeKit:timeInSec()
399
400	--排序，取出最近100个来
401	local function sortTimeFunc( a,b )
402		return tonumber(a.time) > tonumber(b.time)
403	end
404	table.sort( usePropInfos, sortTimeFunc )
405
406	local costList = {}
407	for i=1, getLevelCostNum do
408		if usePropInfos[i] and usePropInfos[i].levelId < topLevelId then
409			table.insert( costList, table.clone(usePropInfos[i]))
410		end
411	end
412
413	--取出来的付费排序
414	local function sortTimeFunc( a,b )
415		return tonumber(a.cost) < tonumber(b.cost)
416	end
417	table.sort( costList, sortTimeFunc )
418
419	local listLength = #costList
420	local percent80Index = math.ceil( listLength*0.8 )
421
422	local percent80Cost = 0
423	if costList[percent80Index] then
424		percent80Cost = costList[percent80Index].cost
425	end
426
427	if bShowLog then
428		log("***超鸟连胜fuuu判断V2*** *******get80PercentNum***********")
429		
430		log("***超鸟连胜fuuu判断V2*** usePropInfos", usePropInfos)
431		log("***超鸟连胜fuuu判断V2*** costList", costList)
432		log("***超鸟连胜fuuu判断V2*** listLength", listLength)
433		log("***超鸟连胜fuuu判断V2*** percent80Index", percent80Index)
434		log("***超鸟连胜fuuu判断V2*** percent80Cost", percent80Cost)
435
436		log("***超鸟连胜fuuu判断V2*** *******get80PercentNum end***********", percent80Cost)

468function ServerData:getCurCost( topLevelId, levelStep, bShowLog )
469	--获取当前花费的钱, 以topLevelId往前数levelStep -1关
470	local srcLevelStep = levelStep
471	local costList = self:getDataByKey('curCost') or {}
472
473	local costNum = 0
474	
475	local maxLevelId = topLevelId
476	local hasCanUseLength = 0
477
478	for i,v in ipairs(costList) do
479		if v.levelId <= maxLevelId then
480			hasCanUseLength = hasCanUseLength + 1
481		end 
482	end
483
484	--如果现有数据长度不满足levelStep, 重置levelStep
485	if hasCanUseLength < levelStep then
486		levelStep = hasCanUseLength
487	end
488
489	--获取最小关, 如果是45关，那么获取44关的
490	local minLevelId = maxLevelId - levelStep
491
492	for i,v in ipairs(costList) do
493		if v.levelId >= minLevelId and v.levelId <= maxLevelId then
494			costNum = costNum + v.cost
495		end 
496	end
497
498	if bShowLog then
499		log("***超鸟连胜fuuu判断V2*** *******getCurCost***********")
500		log("***超鸟连胜fuuu判断V2*** costList", costList)
501		log("***超鸟连胜fuuu判断V2*** minLevelId", minLevelId)
502		log("***超鸟连胜fuuu判断V2*** maxLevelId", maxLevelId)
503		log("***超鸟连胜fuuu判断V2*** srcLevelStep", srcLevelStep)
504		log("***超鸟连胜fuuu判断V2*** costNum", costNum)
505		log("***超鸟连胜fuuu判断V2*** levelStep", levelStep)
506		log("***超鸟连胜fuuu判断V2*** *******getCurCost end***********")
507	end
508
509	return costNum, levelStep
510end
