218function LocalServer:getAndRefreshPKState()
219    local pkStatusInfo = self:getDataByKey("pkStatusInfo",{})
220    local status = pkStatusInfo.pkStatus
221    --1转移到2
222    if status == Meta.PKState.kInGame then
223        local matchInfo = self:getDataByKey("matchInfo",{})
224        local endTime = tonumber(matchInfo.endTime)
225        local nowTime = TimeKit:timeInMillis()
226        if nowTime > endTime then
227            pkStatusInfo.pkStatus = Meta.PKState.kToBeSettled 
228            self:setDataByKey("pkStatusInfo",pkStatusInfo,true)
229            status = Meta.PKState.kToBeSettled

231    end

236--获取分数状态
237function LocalServer:getScoreData(matchInfo)
238    local matchInfo = matchInfo or self:getDataByKey("matchInfo")
239    if not matchInfo then
240        return {
241            myScore = 0,
242            enemyScore = 0,
243        }
244    end
245
246    local members = matchInfo.members
247    local myScore,enemyScore = 0,0
248    local myUid = _G.amfBean("UserManager"):getUid()
249    for k,v in ipairs(members) do
250        if v.uid == myUid then
251            myScore = v.score
252        else
253            enemyScore = v.score
254        end
255    end
256
257    return {
258        myScore = myScore or 0,
259        enemyScore = enemyScore or 0
260    }
261end

284function LocalServer:checkWinState(myScore,enemyScore)
285    if myScore == enemyScore then
286        if myScore == 0 then
287            return Meta.MatchState.kLose
288        else
289            return Meta.MatchState.kWin
290        end
291    elseif myScore > enemyScore then
292        return Meta.MatchState.kWin
293    else
294        return Meta.MatchState.kLose
295    end

423    local total = 0
424    local days = 1
425
426    local curDayStartTime = getDayStartTimeByTS(TimeKit:timeInSec())
427    local todayIndex = math.ceil(curDayStartTime/86400)
428
429    for k, v in pairs(map) do

462    local ret = string.split(dayJoinLimitStr, ',')
463    local today = self:getDayFromActOpen()
464    local time = 0
