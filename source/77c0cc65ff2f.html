77function CollectCardIcon:onWindowStateChanged(event)
78    if self.isDisposed then return end
79    local data = event and event:getData()
80    if not data then return end
81
82    if data.hasFullScreenWindow == true then
83        self._wasUnderFullScreenWindow = true
84        return
85    end
86    if data.hasFullScreenWindow ~= false or not self._wasUnderFullScreenWindow then
87        return
88    end
89
90    self._wasUnderFullScreenWindow = false
91    if self._refreshIconSpineScheduled then return end
92    self._refreshIconSpineScheduled = true
93    setNextFrame(function()
94        self._refreshIconSpineScheduled = false
95        if self.isDisposed then return end
96        self:refreshIconSpine()
97    end)
98end

100function CollectCardIcon:refreshIconSpine()

114        oldIconSp:removeFromParentAndCleanup(true)
115    end
116    self.iconSp = newIconSp

118    self._isPlayingIconIdleAnim = false
119    if self.playType == PLAY_TYPE.LOCK then
120        self.iconSp:setScale(0)
121    else

123    end

182function CollectCardIcon:updateIconSpine()

238-- 等级到达后播解锁动画
239function CollectCardIcon:playUnlockAnim(callback)
240    if self.playType ~= PLAY_TYPE.LOCK then safeRunCall(callback) return end
241    self.playType = PLAY_TYPE.IDLE
242    self.needShowLockedState = false
243    if not _G.amfBean("IconManager"):useUIBottomBarTest() then
244        safeRunCall(callback)
245        return
246    end
247
248    if self.lockedIcon and not self.lockedIcon.isDisposed then
249        self.lockedIcon:runAction(ActionUtil:sequence{
250            CCDelayTime:create(18*FRAMETIME),
251            CCCallFunc:create(function()
252                if self.isDisposed then return end
253                self.lockedIcon:setVisible(false)
254            end),
255        })
256    end
257    if self.lockAnim and not self.lockAnim.isDisposed then
258        self.lockAnim:playOnceAndDispose("Unlock")
259    end
260    if self.iconSp and not self.iconSp.isDisposed then
261        self.iconSp:runAction(ActionUtil:sequence{
262            CCDelayTime:create(18*FRAMETIME),
263            CCCallFunc:create(function()
264                if self.isDisposed then return end
265                self.iconSp:setScale(0.5)
266            end),
267            CCEaseSineOut:create(CCScaleTo:create(4*FRAMETIME, 1.3)),
268            CCDelayTime:create(3*FRAMETIME),
269            CCEaseSineInOut:create(CCScaleTo:create(4*FRAMETIME, 0.95)),
270            CCEaseSineInOut:create(CCScaleTo:create(2*FRAMETIME, 1)),
271            CCCallFunc:create(function()
272                if self.isDisposed then return end
273                self:showIdleIcon()
274                safeRunCall(callback)
275            end),
276        })
277    end
278end

280function CollectCardIcon:hasHint()

310function CollectCardIcon:showNoOpenTimeTip()
311    local beginTime = self.mgr:getModel():getBeginTime()
312    local mainBeginTime = self.mgr:getModel():getMainBeginTime()
313    local now = TimeKit:timeInSec()
314    if now >= beginTime and now < mainBeginTime then
315        local TimeUtil = require "amf.middleware.ui.util.TimeUtil"
316        local timeTb = TimeUtil:convert2DHMS(mainBeginTime - now)
317        local str = ""
318        if timeTb.d > 0 then
319            str = string.format("%d天%d小时后活动开启", timeTb.d, timeTb.h)
320        else
321            str = string.format("%02d:%02d:%02d后活动开启", timeTb.h, timeTb.m, timeTb.s)
322        end
323        local tip = UITip.Builder.new()
324        :setTarget(self)
325        :setTitle(str)
326        :build()
327        LayoutUtil:move(tip, 0, 25)
328    else
329        CommonTip:showTip("敬请期待新的一期活动～")
330    end

364        labels = {
365            {text = "已有", fontSize = 22, color = normalColor},
366            {text = tostring(progressInfo.threshold) .. "套", fontSize = 22, color = highlightColor},
367            {text = "卡册快集齐啦，继续冲！", fontSize = 22, color = normalColor},
368        }
369    elseif progressInfo.tier == "c" then
370        labels = {
371            {text = "厉害！快集齐的卡册已超过一半！", fontSize = 22, color = normalColor},
372        }
373    else
374        labels = {
375            {text = "仅剩", fontSize = 22, color = normalColor},
376            {text = tostring(progressInfo.incompleteAlbumNum) .. "套", fontSize = 22, color = highlightColor},
377            {text = "卡册就集齐啦，加油！", fontSize = 22, color = normalColor},
378        }
379    end
380    local label = MultiLabel:createEasyMultiLabel(CCSizeMake(200, 0), labels)
381    local size = label:getContentSize()
382    local container = Layer:create()
383    container:setName("collect-tip")
384    local hei = 50
385    container:changeWidthAndHeight(size.width, hei)
386    container:setAnchorPoint(ccp(0.5, 1))
387    container:addChild(label)
388    local tip_builder = UITip.Builder.new()
389    :setTarget(self)
390    :setAutoCloseTime(5)
391    :setContent(container, false)
392    :setCloseCallback(function ()
393        self._tip = nil
394    end)
395    -- tip_builder.outsideTouchClose = false
396    local tip = tip_builder:build()
397    LayoutUtil:move(tip, 0, 25)
398    container:setPositionXY(size.width, hei)
399
400    UIEventCenter:unregister(UIEvent.Remove.AUTO_TIP, tip.removeSelfByEvent)
401    UIEventCenter:unregisterGlobal(NormalEvent.KEY_BACK_TAPPED, tip.removeSelfByEvent, nil, NormalEventPriority.KeyBack.REMOVE_TIP)

403    return true
