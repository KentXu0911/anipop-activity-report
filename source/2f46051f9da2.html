1-- 活动说明通用面板 ViewModel
2-- 基于 UeAMFBasePanelVM 体系，采用 Builder 模式构建
3-- 支持自定义 UI、标题、"点击任意处继续"功能、多页分页、动画
4
5local ActivityCommonNotesVM = class("ActivityCommonNotesVM", UeAMFBasePanelVM)
6
7------------------------------------------------------------
8-- Builder 定义
9------------------------------------------------------------
10
11local Builder = newBuilder(ActivityCommonNotesVM)
12
13function Builder:ctor()
14    self.uiSource = nil           -- string: 传入的 _UI 文件 require 路径（必填）
15                                  -- 示例: "amf.feature.xxx.uisource.XxxPanel_UI"
16    self.title = nil              -- string|string[]|nil: 标题文本，数组下标与页码一致
17    self.skipText = nil           -- string|bool|nil: "点击任意处继续"配置
18                                  --   string: 设置 uESkipText 的文本并显示 clickNode，添加点击任意处关闭功能
19                                  --   true:   显示 clickNode（使用默认文本"点击任意处继续"），添加点击任意处关闭功能
20                                  --   nil:    不显示 clickNode，不添加点击关闭功能
21    self.lockDuringAnim = false   -- bool: 入场动画期间是否锁定 UI（默认不锁，可点击关闭按钮）
22    self.ignoreAnim = false       -- bool: 是否忽略入场动画并直接展示 UI（默认播放动画）
23    self.bottomRichText = nil     -- string|nil: 底部富文本内容（RichText 格式字符串）
24                                  --   设置后显示 bottomRichText 节点，nil 时不显示
25    self.startPage = nil          -- number|nil: 入场动画起始页，nil 或非法值时从第 1 页开始
26    -- 以下继承自 AMFBasePanelVM.Builder 基类，无需重复定义：
27    -- self.closeCallback        -- func: 关闭页面的回调
28    -- self.panelReadyCallback   -- func: 面板初始化完成回调
29end
30
31-- 设置 _UI 文件路径
32function Builder:setUiSource(uiSource)
33    self.uiSource = uiSource
34    return self
35end

37-- 设置标题文本：字符串作为公共标题，数组按页码设置标题

39function Builder:setTitle(title)

42end

44-- 设置"点击任意处继续"配置
45function Builder:setSkipText(skipText)
46    self.skipText = skipText

50-- 设置入场动画期间是否锁定 UI（默认 false，不锁定）
51function Builder:setLockDuringAnim(lock)

56-- 设置是否忽略入场动画并直接展示 UI（默认 false，播放动画）
57function Builder:setIgnoreAnim(ignoreAnim)

59    return self
60end
61
62-- 设置底部富文本内容（RichText 格式字符串）
63function Builder:setBottomRichText(richText)
64    self.bottomRichText = richText
65    return self
66end
67
68-- 设置入场动画起始页，用于跳过前置页并从指定页开始展示和播放动画
69function Builder:setStartPage(startPage)

96------------------------------------------------------------
97-- 面板进入时的动画流程
98-- 动画逻辑封装在 Panel:_playEnterAnimation 中
99------------------------------------------------------------
100
101function ActivityCommonNotesVM:onEnterPanelRun()
102    -- 忽略动画时，Panel 已在初始化阶段直接展示 UI，无需再触发入场动画
103    if self.delegate.ignoreAnim then return end
104    self:callViewFunc("_playEnterAnimation")
105end
