106function RecruitHighPayExpandManager:canShowPaymentGuide()

108    if self:_getState(Const.StateKey.PAYMENT_STOPPED, false) then return false end
109    if self._paymentShownThisLogin then return false end
110    if self._paymentPopupScheduled then return false end
111    if self._paymentPanelVM and not self._paymentPanelVM.isDisposed then return false end
112
113    local totalCount = tonumber(self:_getState(Const.StateKey.PAYMENT_COUNT, 0)) or 0
114    if totalCount >= Const.PAYMENT_PERIOD_LIMIT then return false end
115
116    local today = TimeKit:getTodayStart()

118    local dayCount = tonumber(self:_getState(Const.StateKey.PAYMENT_DAY_COUNT, 0)) or 0
119    if storedDay == today and dayCount >= Const.PAYMENT_DAY_LIMIT then return false end
120    return true

265function RecruitHighPayExpandManager:_refreshActivityEndTime(force)
266    local endTime = tonumber(self:_callDelegate("getEndTime")) or 0
267    if endTime <= 0 then return end
268
269    -- ActivityMgr may drop expired actInfo, whose delegate fallback is a later
270    -- package end time. Keep the last reliable expand end time in that case.
271    local isReliable = force
272        or endTime <= TimeKit:timeInSec()
273        or self:isEnabled()
274    if isReliable then self._activityEndTime = endTime end
275end

298function RecruitHighPayExpandManager:openRecruitLink(scene)
299    if self._openingLink then return self:_makeRejectedDeferred() end
300    if self:_isActivityExpired() then
301        self:_showActivityEndedTip()
302        return self:_makeRejectedDeferred()
303    end
304    if not self:isEnabled() then
305        self:_showOpenFailedTip()
306        return self:_makeRejectedDeferred()
307    end
308
309    local url = self:_callDelegate("getUrl")
310    if not self:_isValidUrl(url) then
311        self:_showOpenFailedTip()
312        return self:_makeRejectedDeferred()
313    end
314
315    local isPad = self:_isPad()
316    if not isPad then
317        local canOpenOk, canOpen = pcall(function()
318            return OpenUrlUtil:canOpenUrl("weixin://")
319        end)
320        if not canOpenOk then
321            self:_showOpenFailedTip()
322            return self:_makeRejectedDeferred()
323        end
324        if not canOpen then
325            CommonTip:showTip("当前未安装微信，请先安装微信~", "negative")
326            return self:_makeRejectedDeferred()
327        end
328    end
329
330    local result = deferred.new()
331    local settled = false
332    self._openingLink = true
333    local timeoutId
334
335    local function finish(success)
336        if settled then return end
337        settled = true
338        self._openingLink = false
339        cancelTimeOut(timeoutId)
340        if success then
341            self:_onOpenUrlSuccess(scene)
342            result:resolve()
343        else
344            self:_showOpenFailedTip()
345            result:reject()
346        end
347    end
348
349    timeoutId = setTimeOut(function()
350        finish(false)
351    end, Const.OPEN_URL_TIMEOUT)
352
353    local openOk, openResult = pcall(function()
354        return OpenUrlUtil:openUrl(url, isPad)
355    end)
356    if not openOk or not openResult or type(openResult.next) ~= "function" then
357        finish(false)
358        return result
359    end
360    openResult:next(function()
361        finish(true)
362    end, function()
363        finish(false)
364    end)
365    return result
366end

409function RecruitHighPayExpandManager:onActivityInfoUpdate(evt)
410    if not self._delegate then return end

419function RecruitHighPayExpandManager:_scheduleActivityEnd()
420    if self._endTimer then
421        TimerUtil.removeAlarm(self._endTimer)
422        self._endTimer = nil
423    end
424    local endTime = tonumber(self._activityEndTime) or 0
425    local delay = endTime - TimeKit:timeInSec()
426    if delay <= 0 then return end
427    self._endTimer = TimerUtil.addAlarm(function()
428        self._endTimer = nil
429        self:_refreshSurfaces(nil, true)
430    end, delay, 1)
431end
