6local function requestActivityConfig(configName, successCallback, failCallback, tryCount)
7    tryCount = tryCount or 1
8    local url = AMFNetworkConfig:getMaintenanceURL()
9    local uid = _G.amfBean("UserManager"):getUid()
10
11    local deviceIdParam = ""
12    if EnvUtil:isDevelopMode() then
13        deviceIdParam = "&deviceid=" .. tostring(MetaInfo.getInstance():getUdid())
14    end
15    local params = string.format("?name=%s&uid=%s&_v=%s%s", configName, uid, _G.bundleVersion, deviceIdParam)
16    url = url .. params
17
18    local function onCallback(response) -- 整个Activity的大配置文件加载完毕
19        if response.httpCode ~= 200 then
20            tryCount = tryCount - 1
21            if tryCount > 0 then
22                requestActivityConfig(configName, successCallback, failCallback, tryCount)
23                return
24            else
25                safeRunCall(failCallback)
26            end
27        else
28            safeRunCall(successCallback, response.body)
29        end
30    end
31    AMFSimpleRequest.new(url, onCallback):setTimeoutMs(30 * 1000):setConnectionTimeoutMs(2000):setGet(true):execute()
32end

34function ActivityConfigLoader:parseConfig(xmlData)
35    local d = deferred.new()
36    local activitiesXML
37    pcall(function()
38        local activityXML = xml.eval(xmlData)
39        activitiesXML = xml.find(activityXML, "activitys")
40    end)
41    if not activitiesXML then
42        d:reject(Result.error(ActivityConst.ConfigError.PARSE_ERROR), "parse activity_config error")
43        return d
44    end
45
46    local ret = {}
47    for _, v in pairs(activitiesXML) do
48        if type(v) == "table" then
49            v[0] = nil
50            table.insert(ret, v)
51        end
52    end
53    
54    d:resolve(Result.success({
55        activities = ret,
56        rootUrl = activitiesXML.url_root or "",
57    }))
58    return d
59end

66function ActivityConfigLoader:getConfig(configName)
67    local d = deferred.new()
68    if self._isGetting then
69        d:reject()
70        return d
71    end
72    self._isGetting = true
73    local function successCallback(xmlData)
74        self._isGetting = false
75        d:resolve(Result.success(xmlData))
76    end
77
78    local function failCallback()
79        self._isGetting = false
80        d:reject(Result.error(ActivityConst.ConfigError.DOWNLOAD_ERROR), "download activity_config fail")
81    end
82
83    self:loadRemote(configName, successCallback, failCallback, 1)
84
85    return d
86end
