487        local winState = self:getLocalDataServer():getWinState()
488
489        --双0，播idle
490        if scoreData.myScore == scoreData.enemyScore and scoreData.myScore == 0 then
491            return self.meta.IconState.kIdle
492        end
493        if winState == self.meta.MatchState.kInvalid then
494            return self.meta.IconState.kIdle
495        elseif winState == self.meta.MatchState.kWin then
496            return self.meta.IconState.kWin
497        else

499        end

501        not self:getLocalDataServer():checkHasLeftTimes() then
502            return self.meta.IconState.kIdle2
503    else
504        return self.meta.IconState.kIdle

718                    label:setString(score)

720                    label:setString("999.99万")
721                elseif score >= 100000 then
722                    label:setString(NumberUtil:formatBigNumber(score,2,100000))
723                end
724            end

727            setLabelNum(label2,enemyScore)

729        end

742            end

744        end

752                    if self.needUpdateLightPos then

754                        light:setPositionXY(pos.x,pos.y - 7)
755                        
756                        self.needUpdateLightPos = false
757                    end
758                end
759            end)
760        end
761
762        if animationTs and animationTs > 0 then
763            self:runAction(ActionUtil:sequence({
764                CCDelayTime:create(animationTs),
765                CCCallFunc:create(playProgressAnim)
766            }))
767        else
768            safeRunCall(playProgressAnim)
769        end
770    end

843function Manager:ignoreTopPlayer(checkLevel)

852		return false
853	end
854
855	if not _G.amfBean("LevelUnLockManager"):isInNewLevelTime() then

859	return true

969function Manager:getStartGamePanelPriority()

971    local delta = endTime - TimeKit:timeInSec()
972
973    if delta > 0 and delta <= 10 * 60 then
974        return "GladiatorHighPriority"
975    else
976        local lastScore = self:getLastStartPanelAndTipScore()
977        local nowScore = self:getLocalDataServer():getScoreData()
978

981        end
982    end
983
984    return "GladiatorLowPriority"
