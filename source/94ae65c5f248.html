41function DailySpecialAIModule:_startIconV2ExpiryWatch()
42    self:_stopIconV2ExpiryWatch()
43    local packageData = self.packageData
44    local endTime = packageData and packageData:getEndTime() or 0
45    if endTime <= 0 or not TimerUtil or type(TimerUtil.addAlarm) ~= "function" then
46        return
47    end
48    local function onTick()
49        if not self._iconV2Adapter then
50            self:_stopIconV2ExpiryWatch()
51            return
52        end
53        if TimeKit:timeInSec() >= endTime then
54            self:_stopIconV2ExpiryWatch()
55            self:onDeprecated()
56        end
57    end
58    self._iconV2ExpiryTimer = TimerUtil.addAlarm(onTick, 1, 0)
59    onTick()
60end

62function DailySpecialAIModule:_disposeIconV2Adapter()
63    self:_stopIconV2ExpiryWatch()
64    local adapter = self._iconV2Adapter
65    if not adapter then
66        return true
67    end
68    local ok, disposed, disposeError = pcall(adapter.dispose, adapter)
69    if not ok or disposed ~= true then
70        logIconV2Error("dispose failed: " .. tostring(ok and disposeError or disposed))
71        return false, ok and disposeError or disposed
72    end
73    self._iconV2Adapter = nil
74    return true
75end

77function DailySpecialAIModule:_createIconV2Adapter()
78    local IconEventCenter = require "amf.feature.mainscene.module.icon.event.IconEventCenter"
79    local adapterLoaded, PackageIconV2Adapter = pcall(require, ICON_V2_ADAPTER_MODULE)
80    if not adapterLoaded or type(PackageIconV2Adapter.create) ~= "function" then
81        logIconV2Error("adapter unavailable: " .. tostring(PackageIconV2Adapter))
82        local mountedOk, mounted = pcall(
83            IconEventCenter.addIconIfNotExist,
84            IconEventCenter,
85            self.iconKey,
86            function()
87                return self:createLegacyIcon()
88            end
89        )
90        return mountedOk and mounted ~= false
91    end
92
93    local configOk, options = pcall(function()
94        local IconV2Config = require "amf.feature.giftpackage.module.singleicon.dailyspecial.IconV2Config"
95        return IconV2Config.create(self, IconEventCenter)
96    end)
97    if not configOk or type(options) ~= "table" then
98        local configError = configOk and "DAILY_SPECIAL_ICON_V2_CONFIG_REQUIRED" or options
99        options = {
100            ownerIconKey = self.iconKey,
101            iconEventCenter = IconEventCenter,
102            legacyFactory = function()
103                return self:createLegacyIcon()
104            end,
105            v2PreflightError = configError,
106        }
107    end
108
109    local createOk, adapter, modeOrError, cleanupOwner = pcall(
110        PackageIconV2Adapter.create,
111        PackageIconV2Adapter,
112        options
113    )
114    if not createOk or not adapter then
115        self._iconV2Adapter = cleanupOwner
116        logIconV2Error("create failed: " .. tostring(createOk and modeOrError or adapter))
117        return false
118    end
119    self._iconV2Adapter = adapter
120    if modeOrError == "v2" then
121        self:_startIconV2ExpiryWatch()
122    end
123    return true
124end

153function DailySpecialAIModule:canAutoPopup(commonGiftData)
154    --@haocong.liang新增修改 支付宝无付费档位限制
155    if not EnvUtil:isALIPAYMiniGame() then
156        local last60Pay = UserTagManager:getLast60DayPayAmount()
157        if last60Pay <= 0 then return end
158    end
159
160    local endTime = commonGiftData:getEndTime()
161    if endTime < TimeKit:timeInSec() - 60 then return false end
162    local last_popped_gift_endTime = LocalDB:getString("last_popped_gift_endTime")
163    return endTime > (tonumber(last_popped_gift_endTime) or 0)

166function DailySpecialAIModule:onActEnterStage4()

171        end
172
173        local creator = function()
174            -- 如果礼包已购买或者已失效，不弹
175            if not self.packageData then onHandleOver() return end
176
177            local vm = self:doCreateGiftView(true)
178            vm:setCloseCallback(onHandleOver)
179            vm:setLoadResFailCallback(onHandleOver)
180            vm:popout()
181
182            local endTime = self.packageData:getEndTime()
183            LocalDB:setString("last_popped_gift_endTime", tostring(endTime))
184        end
185
186        _G.amfBean("AutoPopupMgr"):addPopupConfig(PopupCheckPoint.DAILY_SPECIAL_AI_GIFT_PACKAGE, creator)
187    end
