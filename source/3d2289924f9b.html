6function ReturnGiftActInfo:ensureRegisterActInfoUpdate()
7    if self._actInfoUpdateRegistered then return end

12function ReturnGiftActInfo:onActInfosUpdate(event)
13    local data = event and event:getData() or {}
14    if not data.fromServer then return end
15
16    local actInfos = data.actInfos or {}
17    local actInfo = actInfos[tostring(Const.ActId)] or actInfos[Const.ActId]
18    self._actInfo = actInfo
19end
20
21function ReturnGiftActInfo:getActInfo()
22    self:ensureRegisterActInfoUpdate()
23    if self._actInfo then return self._actInfo end
24
25    local activityMgr = _G.amfBean("ActivityMgr")
26    if not activityMgr or not activityMgr.getActInfo then return nil end
27    return activityMgr:getActInfo(Const.ActId)
28end
29
30function ReturnGiftActInfo:getExtra(actInfo)
31    actInfo = actInfo or self:getActInfo()
32    if not actInfo or not actInfo.extra then return nil end
33    if type(actInfo.extra) == "table" then return actInfo.extra end
34    return table.deserialize(actInfo.extra or "")
35end

37function ReturnGiftActInfo:isShowableStatus(extra)
38    -- do return true end --debug
39    local status = tonumber((extra or {}).status) or 0
40    return status == 1 or status == 2
41end
42
43function ReturnGiftActInfo:getNextPackagePullTs(extra)
44    return tonumber((extra or {}).nextPackagePullTs) or -1
45end
46
47---活动数据，礼包类型，1、3认为是一个类型1.
48function ReturnGiftActInfo:getPackageType()
49    local extra = self:getExtra()
50    local type = tonumber((extra or {}).packageType) or 1
51    if type == 2 then return 2 end
52    return 1
53end
54
55function ReturnGiftActInfo:isActInfoEnabled()
56    local extra = self:getExtra()

58    return self:getNextPackagePullTs(extra) >= 0
59end
60
61function ReturnGiftActInfo:canRequestPackagesPull()
62    -- do return true end --debug
63    if EnvUtil:isOhos() then return false end
64    local extra = self:getExtra()

66
67    local nextPackagePullTs = self:getNextPackagePullTs(extra)
68    if nextPackagePullTs < 0 then return true end
