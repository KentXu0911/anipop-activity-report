30function ActivityModel:updateConfig(configName)
31    local d = deferred.new()
32
33    local saveValue = configName
34    local tempFlag = "useRemoteConfig" .. configName
35
36    -- 已经拉到过数据了，这次登录不再拉取（win32除外）
37    if self:getTempData(tempFlag) and not (EnvUtil:isWindows() or EnvUtil:isUnityEditor() or EnvUtil:isMiniGameDev()) then
38        d:resolve(self:getValue(saveValue))
39        return d
40    end
41    local Loader 
42    if EnvUtil:isUnityEditor() or ActivityConst.shouldUseLocalActivityConfig() then
43        Loader = require("amf.middleware.activity.system.unity.ActivityConfigLoader")
44    elseif EnvUtil:isDesktop() and not os.getenv("ACTIVITY_CFG_OL") then
45        Loader = require("amf.middleware.activity.system.windows.ActivityConfigLoader")
46    else
47        Loader = require("amf.middleware.activity.loader.ActivityConfigLoader")
48    end
49        
50    Loader:load(configName):next(function(res)
51        ALog.log("activity", "load activity config success")
52        self:setTempData(tempFlag, true)
53        local data = self:parseActivityConfigData(res.data.activities)
54        self:setValue(saveValue, data)
55        self:setValue("rootUrl", res.data.rootUrl)
56        d:resolve(data)
57    end, function()
58        ALog.log("activity", "load activity config fail")
59        d:resolve(self:getValue(saveValue))
60    end)
61    return d
62end

64-- 过滤掉空的活动
65function ActivityModel:parseActivityConfigData(configData)
66    local newConfigData = {}
67    for _, v in ipairs(configData or {}) do
68        local isEmptyActivity = v.isEmptyActivity == "true" or v.isEmptyActivity == true
69        if not isEmptyActivity then
70            table.insert(newConfigData, v)
71        end
72    end
73    return newConfigData
74end
