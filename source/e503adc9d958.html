107function ShopRewardManager:onActInfoUpdate(event)
108    local data = event and event:getData() or {}
109    local actInfos = data.actInfos or {}
110    local fromServer = data.fromServer
111    local actInfo = actInfos[tostring(self:getConst().actId)]
112
113    if fromServer and not actInfo then
114        if self:shouldKeepLocalDataWhenServerActInfoMissing() then
115            self:refreshStoreExtensions()
116            self:refreshDisplayByState()
117            return
118        end
119
120        self.moduleExtra = nil
121        self.packagePullModuleExtra = nil
122        self.packagePullSyncedByStage = nil
123        self.hiddenStoreGiftDataMap = nil
124        self:getModel():resetActInfoData()
125        self:removeActMainEndTimer()
126        self:removeGoldTag()
127        self:getLocalPushLogic():cancelPush()
128        self:refreshShopRewardIconModule()
129    elseif fromServer and actInfo then
130        local actInfoExtraData = self:getSyncActInfoExtraData(actInfo)
131        if self:isModuleExtraBuyProgressOutdated(actInfoExtraData) then
132            return
133        end
134
135        if self.moduleExtra then
136            if self:isCachedModuleExtraOlderThanActInfo(actInfo) then
137                self.moduleExtra = nil
138            elseif self:isModuleExtraBuyProgressOutdated(self.moduleExtra) then
139                self.moduleExtra = nil
140            else
141                actInfo = self:updateActInfoByModuleExtra(self.moduleExtra, actInfo)
142                actInfos[tostring(self:getConst().actId)] = actInfo
143                return
144            end
145        end
146
147        self:getModel():syncActInfo(actInfo)
148        self:registerActMainEndTimer()
149        if self:isEnabled() then
150            self:dcQualify()
151        end
152        self:refreshStoreExtensions()
153        self:refreshDisplayByState()
154    end

171function ShopRewardManager:isModuleExtraBuyProgressOutdated(extraData)
172    if not extraData then return false end
173
174    local localPeriodId = tonumber(self:getModel():getDataByKey("periodId", 0)) or 0
175    local syncPeriodId = tonumber(extraData.periodId or 0) or 0
176    if syncPeriodId ~= localPeriodId then return false end
177
178    -- 和剩余次数保持同一口径：只用 boughtPackageIdMap 合计判断购买进度新旧。
179    -- periodTotalBuyCount 可能因为支付回调和 actInfo/moduleExtra 乱序到达而重复叠加，不能作为唯一事实来源。
180    local localBuyCount = self:getModel():getBoughtPackageCount()
181    if localBuyCount <= 0 then return false end
182
183    local syncBuyCount = self:getBoughtPackageCountFromExtra(extraData)
184    local outdated = syncBuyCount < localBuyCount
185    -- 真实支付回到游戏时，ACT_INFOS_UPDATE 可能晚于 STORE_AFTER_BUY 到达，
186    -- 且携带的是购买前的旧 actInfo/moduleExtra。此时如果直接 syncActInfo，
187    -- 会把本地已处理的购买次数回滚为 0，导致后续购买无法触发活动结束刷新。
188    -- 因此仅在同一期内，待同步购买次数小于本地购买次数时，认为这次同步是旧数据，丢弃等待下一次更新。
189    -- 如果 periodId 已经变化，说明进入新一期，必须放行同步，避免上一期内存态阻塞新一期活动。
190    return outdated
191end

193function ShopRewardManager:isCachedModuleExtraOlderThanActInfo(actInfo)
194    if not self.moduleExtra then return false end
195
196    local actInfoExtraData = self:getSyncActInfoExtraData(actInfo)
197    if not actInfoExtraData then return false end
198
199    local cachedPeriodId = tonumber(self.moduleExtra.periodId or 0) or 0
200    local syncPeriodId = tonumber(actInfoExtraData.periodId or 0) or 0
201    -- actInfo 已进入新一期时，缓存的 moduleExtra 不能再覆盖本次 actInfo，
202    -- 否则会把新一期数据写回旧 periodId，导致新周期被旧内存态阻塞。
203    if cachedPeriodId ~= syncPeriodId then return true end
204
205    local cachedBuyCount = self:getBoughtPackageCountFromExtra(self.moduleExtra)
206    local syncBuyCount = self:getBoughtPackageCountFromExtra(actInfoExtraData)
207    local older = cachedBuyCount < syncBuyCount
208    return older
209end

211function ShopRewardManager:getBoughtPackageCountFromExtra(extraData)
212    local boughtPackageIdMap = extraData and extraData.boughtPackageIdMap or {}

216    end

281function ShopRewardManager:cacheHiddenStoreGiftDataByPackages(packages)
282    self.hiddenStoreGiftDataMap = nil
283    if not packages then return end
284
285    local replaces = self:getModel():getDataByKey("replaces", {})
286    local originPackageIdMap = {}
287
288    for _, replaceInfo in ipairs(replaces) do
289        if replaceInfo.originPackageId then
290            originPackageIdMap[tostring(replaceInfo.originPackageId)] = true
291        end
292    end
293
294    if not next(originPackageIdMap) then return end
295
296    for _, giftData in ipairs(packages) do
297        if giftData
298            and giftData.getGoodsId
299            and giftData.getExtraValue
300            and giftData:getExtraValue("hidden")
301            and originPackageIdMap[tostring(giftData:getGoodsId())]
302        then
303            self:cacheHiddenStoreGiftData(giftData)
304        end
305    end

667function ShopRewardManager:getPackageGoldNum(packageId)
668    if not packageId then return 0 end
669
670    local goods = GoodsManager:getGoods(packageId)
671    if not goods or not goods.getItems then return 0 end
672
673    local items = goods:getItems() or {}
674    for _, item in ipairs(items) do
675        if item.itemId == AMFPropId.GOLD then
676            return tonumber(item.num) or 0
677        end
678    end
679    return 0
680end

682function ShopRewardManager:isActivityPackageMetaReady(replaceInfo, stageType)
683    if not replaceInfo then return false end
684    local packageSynced = stageType and self:isPackagePullSynced(stageType) or self:isStorePackagePullSynced()
685    if not packageSynced then return false end
686
687    local activityPackageId = replaceInfo.activityPackageId
688    if not activityPackageId then return false end
689
690    local goods = GoodsManager:getGoods(activityPackageId)
691    return goods ~= nil
692end

694function ShopRewardManager:getReplaceInfoByActivityPackageId(activityPackageId)
695    if not activityPackageId then return end
696
697    local packageIdNum = tonumber(activityPackageId or -1) or -1
698    local replaces = self:getModel():getDataByKey("replaces", {})
699    for _, replaceInfo in pairs(replaces) do

702        end
703    end

706function ShopRewardManager:isMainPopupGoodsReady()
707    local replaces = self:getModel():getDataByKey("replaces", {})
708    if table.isEmpty(replaces) then
709        return false
710    end
711
712    for _, replaceInfo in pairs(replaces) do
713        local activityPackageId = replaceInfo.activityPackageId
714        local isRemainBuyable = self:isActivityPackageRemainBuyable(activityPackageId)
715        local replaceReady = self:isActivityPackageMetaReady(replaceInfo)
716
717        if isRemainBuyable and replaceReady then
718            return true
719        end
720    end
721
722    return false
723end

725function ShopRewardManager:getExtraGoldNumByActivityPackageId(activityPackageId, stageType)
726    local replaceInfo = self:getReplaceInfoByActivityPackageId(activityPackageId)
727    if not self:isActivityPackageMetaReady(replaceInfo, stageType) then return 0 end
728
729    local activityGoldNum = self:getPackageGoldNum(activityPackageId)
730    local originPackageId = replaceInfo and replaceInfo.originPackageId
731    local originGoldNum = self:getPackageGoldNum(originPackageId)
732    return math.max(activityGoldNum - originGoldNum, 0)
733end

735function ShopRewardManager:getMaxExtraGoldNum(stageType)
736    local maxExtraGoldNum = 0
737    local replaces = self:getModel():getDataByKey("replaces", {})
738    for _, replaceInfo in pairs(replaces) do
739        local packageId = replaceInfo.activityPackageId
740        if self:isActivityPackageRemainBuyable(packageId) and self:isActivityPackageMetaReady(replaceInfo, stageType) then
741            maxExtraGoldNum = math.max(maxExtraGoldNum, self:getExtraGoldNumByActivityPackageId(packageId, stageType) or 0)
742        end
743    end
744    return maxExtraGoldNum
745end

772function ShopRewardManager:getAddStepCandidatePackageIds(addFivePrice)
773    addFivePrice = tonumber(addFivePrice or 0) or 0
774    if addFivePrice <= 0 then return {} end
775
776    local packageIds = self:getModel():getDataByKey("uiPriorityPackageIds", {})
777    local candidates = {}
778    for _, packageId in ipairs(packageIds) do
779        local replaceInfo = self:getReplaceInfoByActivityPackageId(packageId)
780        if self:isActivityPackageRemainBuyable(packageId)
781            and self:isActivityPackageMetaReady(replaceInfo, GiftPackageConst.StageType.ADD_STEP)
782            and self:getPackageGoldNum(packageId) >= addFivePrice
783        then
784            table.insert(candidates, packageId)
785        end
786    end
787    return candidates
788end

790function ShopRewardManager:getAddStepTargetPackageId(addFivePrice)
791    local candidatePackageIds = self:getAddStepCandidatePackageIds(addFivePrice)
792    return candidatePackageIds[1]

812function ShopRewardManager:openActivity(closeCallback, params)
813    if not self:isEnabled() then
814        if closeCallback then closeCallback() end
815        return
816    end

820        if closeCallback then closeCallback() end
821        return
822    end
823
824    params = params or {}
825    local accurateSource = params.source or _G.StoreConfig.AccurateSource.Click_Act_ShopReward_Icon
826    if _G.StoreConfig and _G.StoreConfig.setAccurateSource and accurateSource then
827        -- Also sets happyStoreStageShopSource for StoreStage package pull.
828        _G.StoreConfig:setAccurateSource(accurateSource)
829    end
830
831    local storeManager = _G.amfBean("StoreManager")
832    if storeManager and storeManager.showStorePanel then
833        storeManager:showStorePanel({
834            source = accurateSource,
835            scrollPkgId = params.scrollPkgId or self:getTargetStorePackageId(),
836            scrollTime = 0,
837            scrollTopOffset = STORE_SCROLL_TOP_OFFSET,
838        })
839    end
840
841    if closeCallback then closeCallback() end
842end
