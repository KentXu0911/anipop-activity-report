383-- 注册并且检查领奖
384function SubMsgBase:registerSubMsgAndGetRewards(tmpInfoList, param)
385    local d = deferred.new()
386    local tmplIds = {}
387    for _, v in ipairs(tmpInfoList) do
388        table.insert(tmplIds, v.tmplId)
389    end
390
391    -- unity编辑器不弹， 直接失败
392    if EnvUtil:isUnityEditor() then
393        return d:reject(SubMessageConst.RegErrCode.PERMISSION)
394    end
395
396    self:tryShowPermissionGuide(tmplIds)
397
398    self._subMsgUtil:requestSubMsg(tmplIds):next(function(resTb)
399        self:tryClosePermissionGuide()
400        local failList = {}
401        local succList = {}
402        for _, v in ipairs(tmplIds) do
403            if resTb[v] == SubMessageConst.TmplState.ACCEPT then
404                table.insert(succList, v)
405            else
406                table.insert(failList, v)
407            end
408        end
409
410        -- 先检查一次状态,标记之前有没有选过总是保持
411        local lastMap = {}
412        for _, v in ipairs(failList) do
413            lastMap[v] = self._subMsgUtil:isAlwaysReject(v) or self._subMsgUtil:isAlwaysAccept(v)
414        end
415        -- 请求setting去对比检查
416        self._subMsgUtil:autoUpdateSettingInfo():next(function()
417            if #succList == 0 then
418                local isAlwaysReject = true
419                for _, v in ipairs(failList) do
420                    if not self._subMsgUtil:isAlwaysReject(v) or not lastMap[v] then
421                        isAlwaysReject = false
422                        break
423                    end
424                end
425                -- 转换成定义的id
426                local idList = {}
427                for _, tmplId in ipairs(failList) do
428                    table.insert(idList, self:getIdByTmplId(tmplId))
429                end
430                -- 打点
431                self:dc("subscription", "main_panel_refuse", {
432                    panel_id = param.panelId,
433                    t1 = table.concat(idList, "_")
434                })
435                -- 全部拒绝了，但是有不是永久拒绝的
436                d:reject(isAlwaysReject and SubMessageConst.RegErrCode.PERMISSION or SubMessageConst.RegErrCode.NONE)
437            else
438                self:doAfterReged(succList, failList, param)
439                self:checkGetRewards(d)
440            end
441        end)
442    end, function(errCode)
443        self:tryClosePermissionGuide()
444        d:reject(errCode)
445    end)
446    return d
447end

449function SubMsgBase:doAfterReged(succList, failList, param)
450    -- 只要有成功的就刷新一次icon
451    self:updateIcon()
452    self:tryClosePermissionGuide()
453
454    if self:getSubTempValue("canRecordRegTag") then
455        -- 记录有订阅成功的
456        self:setSubTempValue("hasRegSuccess", not table.isEmpty(succList))
457    end
458    -- 订阅完成后通知后端我订阅了
459    local notiIdList = {}
460    local historyList = self:getSubValue("historyRegList") or {}
461    for _, v in ipairs(succList) do
462        local id = self:getIdByTmplId(v)
463        table.insert(notiIdList, id)
464        -- 添加到订阅过列表中
465        if not table.exist(historyList, id) then
466            table.insert(historyList, id)
467            self:setSubValue("historyRegList", historyList)
468
469            -- 增加每日领奖的结束时间, 只有微信会变，但是微信没开，其他都不变
470            if self:hasDailyReward() then
471                local dailyRewardEndTime = self:getSubValue("dailyRewardEndTime")
472                local oneRegAddDay = self:getSubValue("oneRegAddDay")
473                if dailyRewardEndTime <= TimeKit:timeInSec() then
474                    dailyRewardEndTime = TimeKit:getTodayStart()
475                end
476                dailyRewardEndTime = dailyRewardEndTime + 24 * 3600 * oneRegAddDay
477                self:setSubValue("dailyRewardEndTime", dailyRewardEndTime)
478            end
479        end
480
481        -- 添加到当前订阅列表中
482        local regList = self:getSubValue("regList") or {}
483        if not table.exist(regList, id) then
484            table.insert(regList, id)
485            self:setSubValue("regList", regList)
486        end
487    end
488    -- print("historyList ---", tts(self:getSubValue("historyRegList")))
489    -- print("regList ---", tts(self:getSubValue("regList")))
490    local tb = {}
491    if not table.isEmpty(notiIdList) then
492        for _, id in ipairs(notiIdList) do
493            local sceneType = self:getSceneTypeById(id)
494            if sceneType == SubMessageConst.SceneType.PROMOTEDAILY then
495                local medalCfg = self._promoteDailyMgr:getCurMonthNearestKeyMedal()
496                local TimeUtil = require "amf.middleware.ui.util.TimeUtil"
497                if medalCfg then
498                    table.insert(tb, {
499                        id = id, 
500                        data = table.serialize({
501                            holiday = medalCfg.des,   -- 节日
502                            reward = self._nameMedalManager:getNameMedalName(medalCfg.medalId), -- 勋章名
503                            day = TimeUtil:getDayCountByYMD(medalCfg.year, medalCfg.month, medalCfg.day)
504                        })
505                    })
506                end
507            end
508        end
509        self.model:requestNotifyServer({
510            idList = notiIdList,
511            miniprogram_state = self:getServerMiniGameState(),
512            openId = self._sdkLoginManager:getOpenId(),
513            extra = table.serialize(tb)
514        })
515        -- 打点
516        self:dc("subscription", "main_panel_success", {
517            panel_id = param.panelId,
518            t1 = table.concat(notiIdList, "_")
519        })
520    end
521end

908function SubMsgBase:getSubNodeTextStruct(subMsgId)
909    -- 已经领过奖
910    local textStruct = LabelStruct.new()
911    local text = self:getTextById(subMsgId)
912    local reward = self:getRewardById(subMsgId)
913    if table.isEmpty(reward) then
914        reward = {itemId = 2, num = 1}
915    end
916    local state = SubMsgNode.State.NONE
917    local sceneType = self:getSceneTypeById(subMsgId)
918    local textCfg = SubMessageConst.SubMsgNodeText[sceneType]
919    if self:isReged(subMsgId) or self._subMsgUtil:isAlwaysAccept(self:getTmplIdById(subMsgId))
920    then
921        local str = self:getRegedTextById(subMsgId)
922        textStruct:setText({
923            "<<c="..textCfg.color..">"..str..">",
924        })
925        state = SubMsgNode.State.REWARDED_REG
926    elseif self:isRegRewarded(subMsgId) then
927        local str = self:getNoRewardTextById(subMsgId)
928        textStruct:setText({
929            "<<c="..textCfg.color..">"..str..">",
930        })
931        state = SubMsgNode.State.REWAREDED_NOREG
932    else
933        local icon = GameItemCreator:getPropIcon({itemId = reward.itemId, num = reward.num, needNum = false})
934        local NumberUtil = require "amf.middleware.ui.util.NumberUtil"
935        local extStr = NumberUtil:formatRewardNum(reward)
936        local txt = icon:getLeftTopText()
937        if txt and txt ~= "" then
938            icon:setLeftTopText("")
939            extStr = txt
940            icon:setItemType(_G.UEPropIcon_ItemType.None)
941        end
942        local firstActName = "开启"
943        textStruct:setText({
944            "<<c="..textCfg.color..">"..firstActName..text.."，"..(sceneType == SubMessageConst.SceneType.MARK and "得 >" or "\n得>"),
945            "<<node>>",
946            "<<c="..textCfg.color..">"..extStr .. ">",
947            nodes = {
948                icon
949            }
950        })
951        state = SubMsgNode.State.NONE
952    end
953    textStruct:setHorizontalAlignment(kCCTextAlignmentLeft)
954    if textCfg.outlineSize and textCfg.outlineColor then
955        textStruct:setOutline(textCfg.outlineColor, textCfg.outlineSize)
956    end
957    if textCfg.fontSize then
958        textStruct:setFontSize(textCfg.fontSize)
959    end
960    return textStruct, state
961end

963-- 自动注册
964function SubMsgBase:autoRegSubMsg(mustId, param)
965    param = param or {}
966    local subscribeMetaList = self:getSubValue("subscribeMetaList") or {}
967    local list = {}
968    local mustData = nil
969    for i, v in ipairs(subscribeMetaList) do
970        if self._subMsgUtil:isAlwaysAccept(v.tmplId) and not self:isReged(v.id) then
971            if mustId == v.id then
972                mustData = v
973            else
974                table.insert(list, v)
975            end
976        end
977    end
978    if mustData then
979        table.insert(list, 1, mustData)
980    end
981    list = table.headn(list, SubMessageConst.OneTimeRegCount)
982    if table.isEmpty(list) then return end
983    self:registerSubMsgAndGetRewards(list, {panelId = param.panelId})
984end

986-- 创建订阅节点
987function SubMsgBase:createSubMsgNode(sceneType, basePanelVM, param)
988    local d = deferred.new()
989    local node
990    local subMsgId = self:getIdBySceneType(sceneType)
991    local textStruct, state = self:getSubNodeTextStruct(subMsgId)
992    local params = {
993        state = state,
994        textStruct = textStruct,
995        disposeCall = function()
996            -- 自动注册
997            self:autoRegSubMsg(subMsgId, param)
998        end,
999        clickCallback = function()
1000            local info = self:getMetaInfoById(subMsgId)
1001            if not info then return end
1002            -- if not self._networkStatusManager:isNetworkEnabled() then
1003            --     CommonTip:showErrorTip(-2)
1004            --     return
1005            -- end
1006            if basePanelVM then basePanelVM:lock() end
1007            -- 打点
1008            self:dc("subscription", "main_panel_click", {
1009                panel_id = param.panelId,
1010                t1 = 2,
1011                t2 = info.id
1012            })
1013            -- 标记可以记录有订阅成功
1014            self:setSubTempValue("canRecordRegTag", true)
1015            self:registerSubMsgAndGetRewards({info}, param):next(function(rewardList)
1016                if basePanelVM then basePanelVM:unlock() end
1017                self:showRewardPanel(rewardList, function()
1018                    if node and not node.isDisposed then
1019                        local textSt, state = self:getSubNodeTextStruct(subMsgId)
1020                        node:setTextStruct(textSt)
1021                        node:setState(state)
1022                        node:update()
1023                    end
1024
1025                    -- 弹出每日登录奖励提醒界面
1026                    if self:getSubTempValue("hasRegSuccess") then
1027                        self:setSubTempValue("hasRegSuccess", false)
1028                        self:showDailyRewardNoticePanel()
1029                    end
1030                end)
1031            end, function(errCode)
1032                if basePanelVM then basePanelVM:unlock() end
1033                self:tryShowPermissionPanel(errCode)
1034            end)
1035        end
1036    }
1037    if param.nodeCreator then
1038        node = param.nodeCreator:create(params)
1039        d:resolve(node)
1040    else
1041        local files = Res.Files.EXTNODE
1042        self._assetsUtil:loadMiniFiles(files):next(function()
1043            node = SubMsgNode:create(params)
1044            d:resolve(node)
1045        end)
1046    end
1047    return d
1048end
