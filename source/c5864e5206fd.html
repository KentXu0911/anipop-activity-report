203local function areRewardsEqual(left, right)

213local function areRewardPoolsEqual(left, right)
214    if left == right then
215        return true
216    end
217    if not left or not right or #left ~= #right then
218        return false
219    end
220    for index, reward in ipairs(left) do
221        local other = right[index]
222        if not other
223            or reward.itemId ~= other.itemId
224            or reward.num ~= other.num
225            or reward.rate ~= other.rate
226        then
227            return false
228        end
229    end
230    return true
231end

324function Model:commitLotterySnapshot(previous)
325    previous = previous or {}
326    local current = self:getLotterySnapshot()
327    local changedKeys = {}
328    if not areRewardsEqual(previous.gachaResult, current.gachaResult) then
329        changedKeys[#changedKeys + 1] = "gachaResult"
330    end
331    if previous.registerState ~= current.registerState then
332        changedKeys[#changedKeys + 1] = "registerState"
333    end
334    if previous.gachaResultShow ~= current.gachaResultShow then
335        changedKeys[#changedKeys + 1] = "gachaResultShow"
336    end
337    if not areRewardPoolsEqual(previous.gachaRewardPool, current.gachaRewardPool) then
338        changedKeys[#changedKeys + 1] = "gachaRewardPool"
339    end
340    if previous.shipNoticeShown ~= current.shipNoticeShown then
341        changedKeys[#changedKeys + 1] = "shipNoticeShown"
342    end
343    if #changedKeys > 0 then
344        self:_increaseLotteryStateVersion()
345    end
346    return changedKeys
347end

437function Model:getLotteryEntryState(isMainTime, isFinished)
438    local registerState = self:getRegisterState()
439    local state = LotteryEntryState.UNAVAILABLE
440    if self:isDataEnabled() and self:hasGachaRewardPool() then
441        if registerState == LotteryRegisterState.PENDING then
442            state = LotteryEntryState.ADDRESS_PENDING
443        elseif registerState == LotteryRegisterState.COMPLETED then
444            state = LotteryEntryState.ADDRESS_COMPLETED
445        elseif self:getGachaResult() then
446            state = LotteryEntryState.DRAWN
447        else
448            state = LotteryEntryState.READY
449        end
450    end
451
452    isMainTime = self:isInMainTime()
453    isFinished = self:getGotFinalReward()
454    printx(44, "canDraw isFinished", isFinished)
455
456    local canDraw = state == LotteryEntryState.READY
457        and isMainTime
458        and isFinished
459    local redDotType
460    if canDraw then
461        redDotType = "reward"
462    elseif state == LotteryEntryState.ADDRESS_PENDING then
463        redDotType = "normal"
464    end
465    return {
466        state = state,
467        canDraw = canDraw,
468        redDotType = redDotType,
469        shouldShowRedDot = redDotType == "reward",
470        shouldShowNormalRedDot = redDotType == "normal",
471        hasUnreadResult = self:hasUnreadGachaResult(),
472        registerState = registerState,
473        addressMode = registerState == LotteryRegisterState.COMPLETED and "edit" or "create",
474        stateVersion = self:getLotteryStateVersion(),

845function Model:hasBattlePassReward()
846    local value = self:getBattlePassValue()
847    local hasVip = self:getBattlePassVip()
848    local vipGot = self:getBattlePassRewardsId(true)
849    local freeGot = self:getBattlePassRewardsId(false)
850    local cfg = self:getConfigByKey("battlePass", {})
851
852    local canGetFree = {}
853    local canGetVip = {}
854
855    for id, rewardCfg in ipairs(cfg) do
856        local need = rewardCfg.need or 0
857        local meetNeed = value >= need
858
859        if meetNeed then
860            if not freeGot[id] then
861                table.insert(canGetFree, id)
862            end
863
864            if hasVip and not vipGot[id] then
865                table.insert(canGetVip, id)
866            end
867        end
868    end
869
870    local hasReward = (#canGetFree > 0) or (#canGetVip > 0)
871    return hasReward, canGetFree, canGetVip
872end
873
874function Model:clearBattlePassReward()
875    local hasReward, canGetFree, canGetVip = self:hasBattlePassReward()
876
877    if not table.isEmpty(canGetFree) then
878        local ids = self:getDataByKey("battlePass.freeAlreadyGotIds", {})
879        for _, v in ipairs(canGetFree) do
880            table.insert(ids, v)
881        end
882        self:setDataByKey("battlePass.freeAlreadyGotIds", ids)
883    end
884
885    if not table.isEmpty(canGetVip) then
886        local ids = self:getDataByKey("battlePass.vipAlreadyGotIds", {})
887        for _, v in ipairs(canGetVip) do
888            table.insert(ids, v)
889        end
890        self:setDataByKey("battlePass.vipAlreadyGotIds", ids)
891    end
892end
