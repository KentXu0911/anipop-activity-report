88function GiftPanel:initGift()
89
90    local goodsMetaItems = self.goodsMetaItems
91
92    if self.panelType == PanelType.STAGE_4_POP or self.panelType == PanelType.TUJIAN_HALF then
93        local itemData = {}
94        local hasGold = false
95        for _, v in pairs(goodsMetaItems) do
96            if v.itemId ~= UPropId.GOLD then
97                table.insert(itemData, {
98                    itemId = v.itemId,
99                    num = v.num,
100                    hideNum = true,
101                })
102            else
103                local StoreGiftLayout = require "amf.feature.store.view.gift.layout.GiftLayout"
104                local icon = StoreGiftLayout:createGoldIcon(v.num)
105                icon:showGoldAni()
106                icon:setScale(0.9)
107                icon.reward = {itemId = UPropId.GOLD, num = v.num}
108                self.goldIcon = icon
109                self.goldNode:addChild(icon)
110                hasGold = true
111            end
112        end
113        self:createItemLayout(itemData, hasGold and GiftLayoutWithGold or GiftLayout)
114    else
115        -- 已经支持多种了。 看GiftLayout 这个的配置 124个的展示
116        local rewardNum = #goodsMetaItems
117        local giftLayoutConfig = GiftLayout[rewardNum] or GiftLayout[1]
118
119        for i,v in ipairs(goodsMetaItems) do
120            local reward = table.clone(goodsMetaItems[i])
121            reward = {itemId = reward.itemId, num = reward.num}
122            if reward.itemId == Meta.DoubleBuffId then
123                reward.needNum = false
124            else
125                reward.needNum = true
126            end
127
128            local posconfig = giftLayoutConfig[i] or giftLayoutConfig[1]
129            local icon
130            self:sendCommand("createIcon", reward, function(ret) icon = ret end)
131            icon:setPositionXY(posconfig.x, posconfig.y)
132
133            if reward.needNum then
134                local numWidget, diffX, diffY = AMFCompHelper:createPropNum(reward)
135                if numWidget then
136                    icon:addChild(numWidget)
137                    local sz = icon:getContentSize()
138                    numWidget:setPositionXY(sz.width+diffX,diffY)
139                    icon.numWidget = numWidget
140                end
141            end
142            self.grpGifts:addChild(icon)
143        end
144
145        
146    end
147end

162function GiftPanel:createItem(reward, scale)
163    local itemNode = self.builder:createChild("itemNode")
164    itemNode:setScale(scale)

168    end
169    itemNode.reward = reward
170
171    local icon
172    self:sendCommand("createIcon", reward, function(ret) icon = ret end)
173    LayoutUtil:centerAdd(itemNode, icon)
174    itemNode.mbMap.itemNum:setString(YH_REF_MAP.x .. reward.num)
175    if _G.amfBean("PropManager"):isMergableProp(reward.itemId) or reward.itemId == Meta.DoubleBuffId then
176        itemNode.mbMap.itemNum:setVisible(false)
177    end
178    LayoutUtil:topUI(itemNode.mbMap.itemNum)
179    return itemNode

233function GiftPanel:flyRewards(callback, customFlyFunc)
234    if self.isDisposed then return end
235
236    if customFlyFunc then
237        local items = self.goodsObj:getItems()
238        local rewardNum = #items
239        local giftLayoutConfig = GiftLayout[rewardNum] or GiftLayout[1]
240
241        for i,v in ipairs(items) do
242            local flyEndCall = nil
243            if i==#items then
244                flyEndCall = callback
245            end
246            local posconfig = giftLayoutConfig[i] or giftLayoutConfig[1]
247            local scrPos = self.grpGifts:convertToWorldSpace(ccp(posconfig.x, posconfig.y))
248            if v.itemId == Meta.CollectionId  then
249                customFlyFunc(scrPos, flyEndCall, v.itemId )
250            else
251                local reward = table.clone(v)
252                RewardFlyHelper:rewardFlyFunc(false, reward, scrPos, nil, nil, flyEndCall)
253            end
254        end
255    else
256        local srcWorldPos = self.goldIcon:getPositionInWorldSpace()
257        local reward = table.clone(self.goldIcon.reward)
258        RewardFlyHelper:rewardFlyFunc(false, reward, srcWorldPos, nil, nil, callback)
259
260        local icon
261        self:sendCommand("getActIcon", function(ret) icon = ret end)
262
263        local endWorldPos = icon and icon:getPositionInWorldSpace()
264        local function createFakeIcon()
265            local icon
266            self:sendCommand("getFlyActIcon", function(ret) icon = ret end)
267            return icon
268        end
269        for k, v in pairs(self.items) do
270            local reward = table.clone(v.reward)
271            local srcWorldPos = v:getPositionInWorldSpace()
272            RewardFlyHelper:rewardFlyFunc(true, reward, srcWorldPos, endWorldPos, function()
273                local icon
274                self:sendCommand("createIcon", reward, function(ret) icon = ret end)
275                return icon
276            end, nil, CCSizeMake(30, 30), createFakeIcon)
277        end
278    end
