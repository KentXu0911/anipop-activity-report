22function Manager:_registerAutoPopup()
23    if not self.autoPopupMgr then return end
24
25    require "amf.middleware.autopop.config.AutoPopupConst"
26    local popStage = AutoPopupConst.PopStage
27
28    -- 场景2放在 Activity 类型第一，场景4放在 Condition 类型第一；
29    -- 两个 checkPoint 共用同一个 GuideFlag，确保任一场景成功展示后另一场景不再弹。
30    self.autoPopupMgr:registerPopup({
31        checkPoint = PopupCheckPoint.ELIMINATE_FESTIVAL_2026_COMIC_STAGE2,
32        stages = { popStage.STAGE_TWO },
33        check = function()
34            if self:isDebugForceStage2Popup() then
35                return true
36            end
37            return self:canAutoPopup()
38        end,
39        creator = function(_, onPopOver)
40            return function()
41                -- 需求口径：展示过程中闪退也算成功展示，因此面板打开前立即写标记。
42                self:_markDisplayed()
43                self:openComicPanel(onPopOver)
44            end
45        end,
46    })
47
48    self.autoPopupMgr:registerPopup({
49        checkPoint = PopupCheckPoint.ELIMINATE_FESTIVAL_2026_COMIC_STAGE4,
50        stages = { popStage.STAGE_FOUR },
51        check = function()
52            if self:isDebugForceStage2Popup() then
53                return true
54            end
55            return self:canAutoPopup()
56        end,
57        creator = function(_, onPopOver)
58            return function()
59                -- 需求口径：展示过程中闪退也算成功展示，因此面板打开前立即写标记。
60                self:_markDisplayed()
61                self:openComicPanel(onPopOver)
62            end
63        end,
64    })
65end

67function Manager:canAutoPopup()
68    local inTime = self:isInActivityTime()
69    
70    local passLevel = _G.amfBean("LevelAreaManager"):getTopPassedLevel()
71    local levelEnough = not (passLevel < Meta.OPEN_LEVEL)
72    local displayed = self:hasDisplayed()
73    local ignoredForTopLevelPlayer = self:isIgnoredForTopLevelPlayer()
74
75    if not inTime then return false end
76    if not levelEnough then return false end
77    if displayed then return false end
78    if ignoredForTopLevelPlayer then return false end
79    return true
80end

120function Manager:_markDisplayed()
