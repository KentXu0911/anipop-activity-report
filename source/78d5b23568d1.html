18local function formatNumber(value)
19    if value == nil then
20        return "-"
21    end
22    if math.floor(value) == value then
23        return tostring(math.floor(value))
24    end
25    return tostring(value)
26end

32local function parseYYBRewardText(text)
33    local rewards = {}
34    text = text or ""
35    for itemIdText, numText in string.gmatch(text, "(%d+)%s*[:=,xX*]%s*(%d+)") do
36        table.insert(rewards, {
37            itemId = tonumber(itemIdText),
38            num = tonumber(numText),
39        })
40    end
41    return rewards
42end

167local function mergeGoodsItems(items)
168    local result = {}
169    local orderedIds = {}
170    for _, item in ipairs(items or {}) do
171        local itemId = tonumber(item.itemId)
172        local num = tonumber(item.num) or 0
173        if itemId then
174            if result[itemId] == nil then
175                result[itemId] = 0
176                table.insert(orderedIds, itemId)
177            end
178            result[itemId] = result[itemId] + num
179        end
180    end
181    table.sort(orderedIds)
182    return result, orderedIds
183end

185local function getGoodsSellPrice(goodsObj)
186    if not goodsObj or not goodsObj.getPrice then
187        return nil
188    end
189    local ok, price = pcall(function()
190        return goodsObj:getPrice()
191    end)
192    if ok then
193        return tonumber(price)
194    end
195    return nil
196end

198local function buildGoodsPriceResult(goodsId)
199    local goodsObj = _G.GoodsManager and _G.GoodsManager:getGoods(goodsId)
200    if not goodsObj then
201        return nil, "GoodsId不存在->" .. tostring(goodsId)
202    end
203
204    local propManager = _G.amfBean("PropManager")
205    local PriceCalculator = force_require("amf.middleware.purchase.model.PriceCalculator")
206    local sellPrice = getGoodsSellPrice(goodsObj)
207    local calculator = sellPrice and PriceCalculator:create(goodsId, sellPrice) or nil
208    local mergedItems, orderedIds = mergeGoodsItems(goodsObj:getItems())
209    if #orderedIds == 0 then
210        return nil, "商品没有配置道具->" .. tostring(goodsId)
211    end
212
213    local itemRows = {}
214    local totalBasePrice = 0
215    for _, itemId in ipairs(orderedIds) do
216        local num = mergedItems[itemId]
217        local basePrice = propManager:getPropTotalPriceBase(itemId, num)
218        totalBasePrice = totalBasePrice + basePrice
219        table.insert(itemRows, {
220            itemId = itemId,
221            num = num,
222            basePrice = basePrice,
223        })
224    end
225
226    for _, row in ipairs(itemRows) do
227        local ratio = 0
228        if totalBasePrice > 0 then
229            ratio = row.basePrice / totalBasePrice * 100
230        end
231        row.ratio = ratio
232        row.sellPrice = calculator and calculator:getItemSellPrice(row.itemId) or nil
233    end
234    return {
235        goodsId = goodsId,
236        goodsName = goodsObj.getName and goodsObj:getName() or "",
237        sellPrice = sellPrice,
238        totalBasePrice = totalBasePrice,
239        rows = itemRows,
240        hasCalculator = calculator ~= nil,
241    }
242end

572        Director:sharedDirector():popScene()
573    end, "FF6666")
574    backBtn:setPositionXY(10, vs.height - 50)
575    self:addChild(backBtn, 10)
576
577    -- 标题
578    local title = TTFLabel:createAnimalYH("道具系统调试", 36)
579    title:setAnchorPoint(ccp(0.5, 0.5))
580    title:setTextColor(hex2ccc4("FFFFFFFF"))
581    title:setPositionXY(vs.width / 2, vs.height - 30)
582    self:addChild(title, 10)
583
584    -- 按钮列表
585    local buttons = _buttons
586    local count = #buttons
587    local contentTop = vs.height - LIST_TOP_MARGIN
588    local listH = math.max(0, contentTop - LIST_BOTTOM_MARGIN)
589    local totalH = count > 0 and (count * BTN_HEIGHT + (count - 1) * BTN_GAP) or 0
590    local scroll = VerticalScrollable:create(vs.width, listH, true, false)
591    scroll:setPositionXY(0, contentTop)
592    self:addChild(scroll)
593
594    local contentH = math.max(totalH, listH)
595    local content = Layer:create()
596    content:changeWidthAndHeight(vs.width, contentH)
597    content.getHeight = function() return contentH end
598
599    for i, cfg in ipairs(buttons) do
600        local btn = self:_createActionBtn(cfg.label, cfg.action)
601        btn:setPositionXY(vs.width / 2, -(i - 1) * (BTN_HEIGHT + BTN_GAP) - BTN_HEIGHT / 2)
602        content:addChild(btn)
603    end
604    scroll:setContent(content)
605end
