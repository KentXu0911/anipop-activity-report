137function ColorfulStarWinRepeatManager:isLevelSupport(levelId, levelScene, startLevelType)
138    if not self:isEnabled() then
139        return false
140    end
141
142    if not self:curLevelTypeIsSupport() then
143        return false
144    end
145    startLevelType = startLevelType or UBFStartLevelType.kNormal
146    if startLevelType == UBFStartLevelType.kAskForHelp then
147        return false
148    end
149
150    local isMainLevel = UBFLevelSceneFunc:isMainLevel(levelScene)
151    local levelAreaManager = _G.amfBean("LevelAreaManager")
152    local scoreRef = levelAreaManager:getUserScore(levelId, levelScene)
153    local oldStar = scoreRef and scoreRef.star or 0
154    return isMainLevel and oldStar < 4
155end

237function ColorfulStarWinRepeatManager:onActCreateResumePanel(evt)
238    local data = evt:getData()
239    if not self:isEnabled(false) then return end
240    if not UBFLevelSceneFunc:isMainLevel(data.levelScene) then return false end
241    if not self:isLevelSupport( data.levelId, data.levelScene ) then return false end
242
243    local light = self:getWinningCount()
244    if light and light >= 1 then 
245        local actInfo = {}
246        actInfo.actId = self.actId
247        actInfo.winNum = light
248        actInfo.showStr = "当前" .. tostring(light) .. "连胜"
249        actInfo.priority = 1
250        table.insert(data.outData.winState, actInfo)
251    end
252end

254function ColorfulStarWinRepeatManager:onActCrashResumeFailedPopPanel(evt)
255
256    local data = evt.data
257    if not self:isEnabled(false) then return end
258    if not data or not data.json then return end
259
260    local breakWinNames = data.breakWinNames
261    local actIds = data.actIds
262
263    --闪退恢复失败弹板
264    local jsonInfo = data.json[''..self.actId]
265    if jsonInfo and jsonInfo.curWinNum and jsonInfo.curWinNum > 0  then 
266        table.insert(breakWinNames, self.showName )
267        table.insert(actIds, self.actId )
268    end
269end
