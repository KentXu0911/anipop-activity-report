106function MusicMasterConfig:getSafeAnimLevel( score, maxScore )
107	--根据分数 最大连胜数， 获取帘子打开的等级
108	local level = 0
109
110	local level1Num = math.ceil( maxScore * 1/3 )
111	local level2Num = math.ceil( maxScore * 2/3 )
112	local level3Num = maxScore
113
114	if score >= level3Num then
115		level = 3
116	elseif score >= level2Num then
117		level = 2
118	elseif score >= level1Num then
119		level = 1
120	else
121		level = 0
122	end
123
124	return level
125end
