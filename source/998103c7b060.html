158function DesktopShortcutManager:ctor()
159    self.iconHost = PackageIconV2Host:create({
160        owner = self,
161        ownerIconKey = OWNER_ICON_KEY,
162        iconEventCenter = IconEventCenter,
163        isEligible = function()
164            return self:shouldShowIcon()
165        end,
166        isV2Enabled = function()
167            return self:isV2IconExperimentEnabled()
168        end,
169        prepareLegacyAssets = function()
170            return _G.amfBean("AssetsUtil"):loadMiniLabels({LEGACY_LABEL})
171        end,
172        prepareV2Assets = function()
173            return _G.amfBean("AssetsUtil"):loadMiniFiles({ICON_V2_PATH})
174        end,
175        createV2Options = function()
176            local IconV2Config = require "amf.feature.minihub.adddesktopshortcut.icon.IconV2Config"
177            return IconV2Config:create(self, IconEventCenter)
178        end,
179        createLegacyIcon = function()
180            return self:createLegacyIcon()
181        end,
182        refreshLegacy = function(icon)
183            icon:updateState()
184            return true
185        end,
186        preflightErrorPrefix = "DESKTOP_SHORTCUT_V2_ASSET_LOAD_FAILED:",
187        logError = function(stage, detail)
188            if ALog and type(ALog.strict_log_error) == "function" then
189                ALog.strict_log_error(
190                    "[DesktopShortcutIconV2] " .. tostring(stage) .. ":" .. tostring(detail)
191                )
192            end
193        end,
194    })
195end

393function DesktopShortcutManager:checkStage4CanPop()
394    if not self:shouldShowIcon() then return false end
395    if self._autoPopupMgr:getPermanentShowNum("DesktopShortcutPop") > 0 then return false end
396    return true
397end

399function DesktopShortcutManager:onActEnterStage4()
400        printx(44,"onActEnterStage4 desktop")
401    if not self:checkStage4CanPop() then
402        return
403    end
404    local function onHandleOver()
405        AutoPopEventCenter:dispatch(AutoPopEvent.SINGLE_POP_OVER, PopupCheckPoint.KS_DESKTOP_SHORTCUT)
406    end
407
408    local function creator()
409        self:clickIcon(onHandleOver, 2)
410    end
411
412    self._autoPopupMgr:addPopupConfig(PopupCheckPoint.KS_DESKTOP_SHORTCUT, creator)
413end
