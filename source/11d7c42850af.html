122function KSPromotionManager:onActPassDay()
123    if not self:isActEnabled() then return end
124    local isClosePanel =  self:tryClosePanel()
125    if isClosePanel then
126        if self:getPeriodEndTime() < TimeKit:timeInSec() then
127        CommonTip:showTip(ACT_NAME..'活动已结束！')
128        else
129            CommonTip:showTip('新的一天开始了，数据已更新，\n请尝试重进活动~')
130        end
131    end
132    self:tryUpdateActIcon()
133end

257function KSPromotionManager:loadRes(delegate)
258    local successCallback = delegate.successCallback
259    local failCallback = delegate.failCallback
260    local resList = delegate.resList
261    if EnvUtil:isMiniGame() then
262        return self._assetsUtil:loadMiniFiles(resList):next(successCallback, failCallback)
263    else
264        safeRunCall(successCallback)
265    end
266end
267
268function KSPromotionManager:unloadRes(resList)
269    if not EnvUtil:isMiniGame() then
270        return
271    end
272    for k, v in pairs(resList) do
273        if string.find(v, ".png") then
274            CCTextureCache:sharedTextureCache():removeTextureForKey(
275                    CCFileUtils:sharedFileUtils():fullPathForFilename(SpriteUtil:getRealResourceName(v), true)
276            )
277        elseif string.find(v, ".skel") then
278            UICreateUtil:unloadSpineRes(v, true)
279        elseif string.find(v, ".plist") then
280            UICreateUtil:unloadPlist(v)
281        end
282    end
283end
