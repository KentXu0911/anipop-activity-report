39-- 进游戏打点,依靠scene字符串后4位确定进入游戏的类型（链接、扫码、点击私聊卡片）
40function Handler:onDcShareEnterGame(evt)
41    if not EnvUtil:isDYMiniGame() then return end
42    local data = evt:getData() or {}
43    if table.isEmpty(data) then return end
44    if (not data.query) or table.isEmpty(data.query) then return end
45    local query = data.query
46    if not tonumber(query.actId) or (tonumber(query.actId) ~= tonumber(self:getActId())) then return end
47    
48    local sceneStr = data.query.scene
49    local last4 = string.sub(sceneStr, -4)
50    if not last4 then return end
51    local key = 'k'..last4
52    if EnvUtil:isDevelopMode() then
53        UError("last4key:", key, Handler.t1Table[key])
54    end
55    data.dcParam.t1 = Handler.t1Table[key]
56end
