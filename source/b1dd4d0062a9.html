642--延迟关闭策略生效时，播放视频之后增加一定的随机cd，cd之后才能再次观看
643--这个cd是针对整个激励视频的，不区分场景，cd是纯前端的限制，清数据之后会丢失
644function InciteManager:setDelayCloseData()
645    if not self.delayCloseInfo then return end
646
647    local diffTime = TimeKit:timeInSec() - self.delayCloseInfo.bufferEndTime
648    if diffTime <= 0 then return end
649
650    local diffHour = math.floor(diffTime / 3600)
651    for _, v in ipairs(self.delayCloseInfo.randomList) do
652        if diffHour >= v.minHour and diffHour < v.maxHour then
653            local cd = math.random(v.minCd + 1, v.maxCd) --左开右闭区间
654            self.model:setValue("delayCdEndTime", TimeKit:timeInSec() + (cd * 60))
655            break
656        end
657    end
658end

771-- 视频没加载好时，等待n秒，n秒内加载好了就自动播放，没好才走失败回调
772-- waitDuration: 秒
773function InciteManager:_waitUntilVideoIsReady(waitDuration, entranceType)
774    local vm
775    local passedDuration = 0
776
777    local function stopWaitingTimer()
778        if self.waitForSDKReadyTimer then
779            TimerUtil.removeAlarm(self.waitForSDKReadyTimer)
780            self.waitForSDKReadyTimer = nil
781        end
782    end
783
784    local function removeLoadingAnimation()
785        if vm then
786            vm:close()
787        end
788    end
789
790    local function onGiveUpWaiting(canceledByPlayer)
791        stopWaitingTimer()
792        removeLoadingAnimation()
793
794        if canceledByPlayer then
795            CommonTip:showTip("视频加载已取消", "negative")
796        else
797            CommonTip:showTip("视频加载失败，请检查网络设置或稍后重试~", "negative")
798        end
799
800        self:onAdsError(nil, InciteConst.AdsError.kVideoNotReady, "no ad", entranceType, nil, true)
801
802        self:tryLoadAdIfNotReady("no_ad")
803    end
804
805    local function checkSDKReadyStatus()
806        local readySdkId = self:getLoadSucSdk(true, self.entranceType)
807        if readySdkId then
808            stopWaitingTimer()
809            removeLoadingAnimation()
810
811            if self.curPlayInfo then
812                self.curPlayInfo["sdk"] = readySdkId
813            end
814            self.readyAd = readySdkId
815
816            self:_playReadyVideo(readySdkId, entranceType, true)
817        else
818            if passedDuration >= waitDuration then
819                onGiveUpWaiting()
820            end
821        end
822    end
823
824    -- 每秒检测一次sdk状态
825    local function onTimerTick()
826        passedDuration = passedDuration + 1
827        checkSDKReadyStatus()
828    end
829    stopWaitingTimer()
830    self.waitForSDKReadyTimer = TimerUtil.addAlarm(onTimerTick, 1, waitDuration)
831
832    -- 增加一个等待中的动画
833    local function closeCall(click)
834        if click then
835            onGiveUpWaiting(true)
836        end
837    end
838
839    local LoadingPanelVM = require "amf.middleware.ui.kit.panel.loading.LoadingPanelVM"
840    vm = LoadingPanelVM.Builder.new()
841        :setLoadingText("视频加载中")
842        :setShowBtnClose(true)
843        :setCloseCallback(closeCall)
844        :build()
845end
