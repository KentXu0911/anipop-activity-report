414function Manager:buildPassLevelProgressAnimationPlan(oldProgress, newProgress, oldTaskId, newTaskId)
415    oldProgress = tonumber(oldProgress)
416    newProgress = tonumber(newProgress)
417    oldTaskId = tonumber(oldTaskId)
418    newTaskId = tonumber(newTaskId)
419    if not oldProgress or not newProgress or newProgress <= oldProgress
420        or not oldTaskId or not newTaskId or newTaskId < oldTaskId
421    then
422        return nil
423    end
424
425    local finalTotal = tonumber(self:getTaskTotalByIndex(newTaskId))
426    if not finalTotal or finalTotal <= 0 then return nil end
427    local steps = {}
428    for taskId = oldTaskId, newTaskId - 1 do
429        local completedTotal = tonumber(self:getTaskTotalByIndex(taskId))
430        local nextTotal = tonumber(self:getTaskTotalByIndex(taskId + 1))
431        if not completedTotal or completedTotal <= 0
432            or not nextTotal or nextTotal <= completedTotal
433        then
434            return nil
435        end
436        table.insert(steps, {
437            type = "animate",
438            current = completedTotal,
439            total = completedTotal,
440        })
441        table.insert(steps, {
442            type = "reset",
443            current = 0,
444            total = nextTotal,
445        })
446    end
447    table.insert(steps, {
448        type = "animate",
449        current = newProgress,
450        total = finalTotal,
451    })
452    return {
453        final = {current = newProgress, total = finalTotal},
454        steps = steps,
455    }
456end

625--打关结束时更新数据
626function Manager:updatePassLevelData(data, collectionCount, collectionMap, passLevelData)
627    if not self:isEnabled() then return end
628
629    local levelId = data and data.levelInfo and data.levelInfo.levelId
630    local originPassedLevelId = _G.getTopPassedLevelId()
631    if not originPassedLevelId then return end
632    -- 过关应该+1
633    local topPassedLevelId = originPassedLevelId+1
634    -- 限制不能超过topLevel
635    local topLevelId =  _G.getTopLevelId()
636    local levelStatus = self:getLocalLevelStatus(topLevelId)
637
638    --当前能打关卡库，判断最简单，直接判是否关卡库
639    if (levelStatus == 3) and data and data.levelInfo then
640        if LevelSceneFunc:isFullLevelLib(data.levelInfo.levelScene) then
641            -- printx(42, "随机挑战成功")
642            self:refreshLocalActInfo()
643        end
644        return
645    end
646    -- 下一主线关等于当前打关id，过主线成功
647    -- 不用处理top这一关，因为top全满的时候打最高关是top，topPassLevelId=top+1
648    if levelId == topPassedLevelId then
649        -- printx(42, "通过了topLevel关! levelId:", levelId)
650        self:refreshLocalActInfo()
651        return
652    end
653
654    -- 补星，判主线且补星成功
655    if (levelStatus == 2) and data and data.levelInfo then
656        local oldStar = data.levelInfo.oldStar or 0
657        local newStar = data.curStar or 0
658        local bRefreshStarSucess = newStar > oldStar
659        -- printx(42, "补星成功", bRefreshStarSucess)
660        if bRefreshStarSucess and LevelSceneFunc:isMainLevel(data.levelInfo.levelScene) then
661            self:refreshLocalActInfo()
662        end
663    end
664end
