216function Model:hasBattlePassReward()
217    local value = self:getBattlePassValue()
218    local hasVip = self:getBattlePassVip()
219    local vipGotSet = arrayToSet(self:getBattlePassRewardsId(true))
220    local freeGotSet = arrayToSet(self:getBattlePassRewardsId(false))
221    local cfg = self:getConfigByKey("battlePass", {})
222
223    local canGetFree = {}
224    local canGetVip = {}
225
226    for _, rewardCfg in ipairs(cfg) do
227        local id = rewardCfg.id
228        local need = rewardCfg.need or 0
229        local meetNeed = value >= need
230
231        if meetNeed then
232            if not freeGotSet[id] then
233                table.insert(canGetFree, id)
234            end
235            if hasVip and not vipGotSet[id] then
236                table.insert(canGetVip, id)
237            end
238        end
239    end
240
241    local hasReward = (#canGetFree > 0) or (#canGetVip > 0)
242    return hasReward, canGetFree, canGetVip
243end

245function Model:clearBattlePassReward()
246    local hasReward, canGetFree, canGetVip = self:hasBattlePassReward()
247
248    if not table.isEmpty(canGetFree) then
249        local ids = self:getDataByKey("battlePass.freeAlreadyGotIds", {})
250        for _, v in ipairs(canGetFree) do
251            table.insert(ids, v)
252        end
253        self:setDataByKey("battlePass.freeAlreadyGotIds", ids)
254    end
255
256    if not table.isEmpty(canGetVip) then
257        local ids = self:getDataByKey("battlePass.vipAlreadyGotIds", {})
258        for _, v in ipairs(canGetVip) do
259            table.insert(ids, v)
260        end
261        self:setDataByKey("battlePass.vipAlreadyGotIds", ids)
262    end
263end
