312--bubble飞行(开宝箱用)
313function RankRewardPanel:flyBubble(done)
314
315    local rewardsNum = #self.vm.rewards
316    local flyDuration = 0.05
317    local flyIndex = 1
318
319    local flyToTable = {}
320    local posNum = rewardsNum >= 10 and 9 or rewardsNum
321
322    local toScale = self.bubbleLayer:getBubbleScale()
323    for i = 1,posNum do
324        local worldPos = self.bubbleLayer:getBubbleWorldPos(i)
325        local newPos = self.ui:convertToNodeSpace(worldPos)
326        table.insert(flyToTable,newPos)
327    end
328
329    for i = 1,rewardsNum do
330        --先验证，再实际处理，以免内存泄漏
331        Tween.to(self.bubbleLayer:getBubble(i),(i-1)*flyDuration,{onComplete = function()
332            if self.isDisposed then return end
333
334            local node = self.bubbleLayer:createRealBubble(i)
335            node:pretreat()
336
337            local startPos = self.getBtn:getPosition()
338            local endPos
339            if i <= 9 then
340                endPos = flyToTable[i]
341            else
342                endPos = flyToTable[(i-1) % 3 + 7]
343                endPos = ccp(endPos.x,endPos.y)
344            end
345
346            node:setPositionXY(startPos.x,startPos.y)
347            node:setScale(0)
348            ActionUtil:setOpacity(node,255)
349            self.ui:addChild(node)
350
351            --如果是第十个，开启挪动
352            if i == 9 and rewardsNum >= 10 then
353                local num = 0
354                local index = 0
355                local lineNum = math.ceil((rewardsNum-9)/3)
356                local coeffcient = flyDuration*30*3
357                local function onTick()
358                    if self.isDisposed then return end
359
360                    if num >= coeffcient*lineNum then
361                        if self.schedId then
362                            CCDirector:sharedDirector():getScheduler():unscheduleScriptEntry(self.schedId)
363                            self.schedId = nil
364                        end
365                        return
366                    end
367                    num = num + 1
368                    index = index + 1
369
370                    --因为scroll中间默认空5像素，所以移动必须+5 像素
371                    self.bubbleLayer:gotoPosY((self.bubbleLayer:getBubbleHeight() + 5)/coeffcient*index,1/30)
372                end
373                self.schedId = CCDirector:sharedDirector():getScheduler():scheduleScriptFunc(onTick, 1/30, false)
374            end
375
376            --通过检测，开启飞行
377            flyIndex = flyIndex + 1
378            local offsetY = 0
379            --最后一行最后两个不改位置
380            local function needChangePos(index)
381                local lastNum = rewardsNum - index
382                if lastNum >= 2 then
383                    return true
384                elseif lastNum < 2 and rewardsNum % 3 ~= 1 then
385                    return false
386                else
387                    return true
388                end
389            end
390
391            local function isLastLine(index)
392                local lastNum = rewardsNum % 3
393                if lastNum == 0 then lastNum = 3 end
394                return index > (rewardsNum - lastNum)
395            end
396
397            if rewardsNum >= 10 and i >= 8 and needChangePos(i) then
398                if i % 3 == 1 then offsetY = 0
399                elseif i % 3 == 2 then offsetY = -self.bubbleLayer:getBubbleHeight() / 3
400                else offsetY = -self.bubbleLayer:getBubbleHeight() / 3 * 2 end
401            end
402
403            --因为最后一行拉不下去，导致额外处理
404            if isLastLine(i) then
405                offsetY = offsetY + 15
406            end
407
408            Tween.to(node,flyDuration * 3,{x = endPos.x,y = endPos.y - offsetY,scale = {toScale,toScale},ease = CCEaseSineInOut,onComplete = function()
409                if self.isDisposed then return end
410
411                node:removeFromParentAndCleanup(false)
412
413                self.bubbleLayer:replaceRealBubble(i,node)
414
415                if i == rewardsNum then
416                    safeRunCall(done)
417                end
418
419                node:playExtraShowAnim()
420            end})
421        end})
422    end
423end
