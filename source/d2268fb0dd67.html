31--小游戏进大藤蔓后从第一关滚动到当前关并震动两次
32function FunctionHandler:handleScrollToCurrentLevel(params)
33    if EnvUtil:isMiniGame() and EnvUtil:isDevelopMode() then
34        local datas = {
35            udid = MetaInfo:getInstance():getUdid(),
36            deviceModel = MetaInfo:getInstance():getDeviceModel(),
37            gameVersion = _G.bundleVersion,
38        }
39        CS.CallJS.CallS("globalThis.sendQADcData", table.serialize(datas))
40    end
41
42	local function isVibrateEnabled()
43		return true
44	end
45
46	local function vibrateShort(count, callback)
47		if count > 0 then
48			TimerUtil.addAlarm(function()
49				local vibrateType = "light"
50				if count == 1 then
51					vibrateType = "heavy"
52				elseif count == 2 then
53					vibrateType = "medium"
54				end
55				local config = {
56					miniGame = {
57						level = vibrateType
58					}
59				}
60				_G.HapticsContext.new()
61					:setConfig(config)
62					:execute()
63				vibrateShort(count - 1)
64			end, 0.20)
65		else
66			if callback then callback() end
67		end
68	end
69
70	--先只震动不滚动，刚进大藤蔓性能有问题
71	do
72		if isVibrateEnabled() then
73			setTimeOut(function()
74				vibrateShort(2)
75			end, 0.3)
76		end
77		return
78	end
79
80	local function onHandleOver()
81		AutoPopEventCenter:dispatch(AutoPopEvent.SINGLE_POP_OVER, PopupCheckPoint.SCROLL_TO_CURRENT_LEVEL)
82	end
83
84	local function creator()
85		--UWarning("DEVICE_BENCHMARK_LEVEL", DEVICE_BENCHMARK_LEVEL)
86		setTimeOut(function()
87			local topLevelId = getTopLevelId(true)
88			local dt = math.clamp(0.5 + topLevelId / 10000, 0.5, 1.0)
89			if topLevelId < 10 then
90				dt = 0.1
91			end
92			local MainSceneEventCenter = require 'amf.feature.mainscene.core.event.MainSceneEventCenter'
93	
94            MainSceneEventCenter:dispatch(MainSceneEvent.Trunk.MOVE_TRUNK_TO, {
95				levelId = topLevelId,
96                levelScene = UBFLevelSceneType.kMain,
97				duration = 0.5,
98				callback = onHandleOver
99			})
100			if isVibrateEnabled() then
101				vibrateShort(math.ceil(dt / 0.25))
102			end
103		end, 0.8)
104	end
105	self._autoPopupMgr:addPopupConfig(PopupCheckPoint.SCROLL_TO_CURRENT_LEVEL, creator)
106end
