58function Handler:requestGetShareReward(resHelpContext)
59    Manager = _G.amfBean("EnergyActV2Mgr")
60    --- 解析奖励数据
61    local data = resHelpContext or {}
62    if not data.extraParam or not data.extraParam.actId or tonumber(data.extraParam.actId) ~= config.actId then return end
63    local shareData = {
64        shareKey = data.extraParam.shareKey,
65        inviteCode = data.extraParam.invitecode or data.extraParam.uid,
66        openId = data.extraParam.openId,
67        source = data.extraParam.WXToken and 1 or 2, -- 1-微信 2-微博
68    }
69    self:RequestRewardAndShow(shareData)
70end

74function Handler:RequestRewardAndShow(shareData, done)
75    if not shareData then safeRunCall(done) return end
76
77    if shareData.shareKey then
78        self:handleCopyCodeShare(shareData, done) ---- 微信、微博、小游戏卡片
79    elseif shareData.inviteCode then
80        if shareData.openId then
81            self:handleMiniShare(shareData, done) ---- 原小程序的处理，兼容(可能废弃了，检查代码覆盖率后，可以删掉)
82        else
83            self:handleLinkShare(shareData, done) ---- Feed（朋友圈/QQ空间）
84        end
85    else
86        safeRunCall(done)
87    end
88end

137--- 显示奖励 或 错误提示
138function Handler:callApiWithHandlers(apiCall, done)
139    local successCallback = function(rewards, params)
140        self:handleHelpSelf(rewards, params)
141        local rewardItem, energy = self:extractRewardAndEnergy(rewards)
142        if rewardItem then
143            self:showRewardPanel(rewardItem, energy, done)
144        else
145            safeRunCall(done)
146        end
147    end
148
149    local errorCallback = function(evt)
150        self:handleShareError(evt, done)
151    end
152
153    apiCall(successCallback, errorCallback, done)
154end
155
156function Handler:handleHelpSelf(rewards, params)
157    local me = tostring(_G.amfBean("ProfileManager"):getXxlId())
158    if tostring(params.inviteCode) == me then
159        local energy
160        for _, v in ipairs(rewards or {}) do
161            if v.itemId == Meta.OTHER_ENERGY then
162                energy = v.num
163            end
164        end
165        if not energy then return end
166        EventCenter:dispatch(Events.kDailyBeHelpedChange, energy)
167    end
168end
169
170function Handler:extractRewardAndEnergy(rewards)
171    local rewardItem, energy
172    for _, v in ipairs(rewards or {}) do
173        if v.itemId == Meta.OTHER_ENERGY then
174            energy = v.num
175        else
176            rewardItem = v
177        end
178    end
179    return rewardItem, energy
180end

200-- 进游戏打点,依靠scene字符串后4位确定进入游戏的类型（链接、扫码、点击私聊卡片）
201function Handler:onDcShareEnterGame(evt)
202    local data = evt:getData() or {}
203    if table.isEmpty(data) then return end
204    if (not data.query) or table.isEmpty(data.query) then return end
205    local query = data.query
206    if not tonumber(query.actId) or (tonumber(query.actId) ~= tonumber(self:getActId())) then return end
207    
208    data.dcParam.t1 = self:getShareType()
209    data.dcParam.t4 = nil
210end
