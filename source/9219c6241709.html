170function Manager:onActPlayLevelEnd(evt)
171    if not self:isEnabled() then return end
172    local data = evt:getData()
173    if not data.isWin then return end
174    if data.isSubMode then return end
175
176    local runtimeRefer = data.runtimeRefer
177    local runtime = runtimeRefer and runtimeRefer.value
178    if not runtime then return end
179
180    -- 1.判断本次打关是否有效
181    local modEndLevelData = runtime:getEndLevelModData(UBFActivityIdConst.kSprintRealPrizeV2)
182    local passLevelJson = modEndLevelData and modEndLevelData.passLevelJson
183    if not passLevelJson or not passLevelJson.clientMaxLevel then return end
184
185-- TODO 给后端发送协议，确认是否中奖
186    -- 2.判断本次打关是否达到N值，达到才可以抽奖
187    if self:isCanRewardLottery() then
188        -- 3.向后端请求抽奖
189        self:requestRewardLottery():next(function ()
190            local rewards = self:getDataByKey(self._model.F.rewards)
191            if rewards then self:_rewardLotteryCheck(runtime, table.clone(rewards)) end
192        end, function ()
193            printx(48, "requestRewardLottery error.")
194
195        end)
196
197    end
198end
199function Manager:_rewardLotteryCheck(runtime, rewards)

201    if result then
202        -- local rewards = {}

231function Manager:isCanRewardLottery()
232    -- propIntervalNormal\propIntervalMaxLevel\realRewardInterval
233    local ret = true
234
235    --道具的
236    local dailyPropWinLimit = self:getDataByKey(self._model.F.dailyPropWinLimit, 0)
237    local todayPropWinCount = self:getDataByKey(self._model.F.todayPropWinCount, 0)
238    if todayPropWinCount >= dailyPropWinLimit then ret = false end
239    if ret then
240        local totalPropWinLimit = self:getDataByKey(self._model.F.totalPropWinLimit, 0)
241        local totalPropWinCount = self:getDataByKey(self._model.F.totalPropWinCount, 0)
242        if totalPropWinCount >= totalPropWinLimit then ret = false end
243    end
244
245    if ret then
246        local stage = self:getUserLevelState()
247        local propMaxLv = 0
248        if stage == 0 then
249            propMaxLv = self:getDataByKey(self._model.F.propIntervalNormal, 0)
250        else
251            propMaxLv = self:getDataByKey(self._model.F.propIntervalMaxLevel, 0)
252        end
253        local todayPropPassCount = self:getDataByKey(self._model.F.todayPropPassCount, 0)
254        if todayPropPassCount < propMaxLv then
255            ret = false
256        end
257    end
258
259    --实物
260    if not ret then
261        ret = not self:isGotRealReward()
262
263        if ret then
264            local realRewardInterval = self:getDataByKey(self._model.F.realRewardInterval, 0)
265            local todayRealPassCount = self:getDataByKey(self._model.F.todayRealPassCount, 0)
266            if todayRealPassCount < realRewardInterval then
267                ret = false
268            end
269        end
270    end
271
272    if EnvUtil:isDevelopMode() then printx(48, "4103 isCanRewardLottery ret=", ret) end
273    return ret
274end

331---是否截单，不允许修改地址
332function Manager:isContactCutoff()
333    -- do return true end --debug
334    local addressFilledTime = tonumber(self:getDataByKey(self._model.F.addressFilledTime, 0))
335    local isFilled = addressFilledTime ~= 0
336    if not isFilled then return false end
337
338    local cutoffTimes = self:getDataByKey(self._model.F.cutoffTimes) or {}
339    local firstCutoffTime = 0
340    for index, value in ipairs(cutoffTimes) do
341        firstCutoffTime = tonumber(value)
342        if addressFilledTime < firstCutoffTime then
343            break
344        end
345    end
346
347    local curTime = TimeKit:timeInMillis()
348    return curTime >= firstCutoffTime
349end

459function Manager:tryAutoPopup()
460
461    local popupKey = PopupCheckPoint.SPRINT_REAL_PRIZE_T2
462    local function onHandleOver()
463        AutoPopEventCenter:dispatch(AutoPopEvent.SINGLE_POP_OVER, popupKey)
464    end
465
466    local creator = function ()
467        if not self:_isUserNotOpened() then
468            safeRunCall(onHandleOver)
469            return
470        end
471
472        self:openActivity(onHandleOver, nil, false)
473    end
474
475    -- _G.amfBean("AutoPopupMgr"):addPopupConfig(popupKey, creator)
476    local autoPopupMgr = _G.amfBean("AutoPopupMgr")
477    autoPopupMgr:addPopupConfig(popupKey, creator)
