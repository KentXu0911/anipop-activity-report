371-- 过关更新数据
372function Manager:onActPlayLevelEnd(event)
373    local data = event:getData() or {}
374
375    if not data.isWin then return end
376
377    local saveData = {}
378    saveData.levelId = data.levelId
379    saveData.levelScene = data.levelScene
380    saveData.oldStar = data.oldStar
381    saveData.star = data.star
382    saveData.areaId = self:getAreaIdByLevelId(data.levelId)
383
384    -- 存储过关数据，用于记录数据变更
385    if UBFLevelSceneFunc:isMainLevel(saveData.levelScene) and saveData.star > saveData.oldStar then
386        self:getModel():setLocalCache("lastPassLevelData", saveData)
387        self:getModel():setTempData("tenAreaStarData", nil)
388    end
389
390    self:getModel():setTempData("lastPassLevelDataForIcon", saveData)
391
392    local currentYellowStarNum, currentColorStarNum = _G.amfBean("LevelAreaManager"):getYellowStarAndColorStarNum(
393        saveData.areaId)
394    local ts = TimeKit:timeInMillis()
395
396    local areaLevelNum = self:getConst().AREA_LEVEL_NUM
397    if _G.getTopPassedLevelId() == saveData.levelId and saveData.oldStar == 0 then
398        if saveData.star > 0 then
399            self:getModel():updateTenAreaPassTime(saveData.areaId, ts)
400        end
401    end
402
403    if currentYellowStarNum == areaLevelNum * 3 then
404        if saveData.oldStar < 3 then
405            self:getModel():updateAreaFullYellowTime(saveData.areaId, ts)
406        end
407    end
408
409    if currentColorStarNum == areaLevelNum * 3 then
410        if saveData.oldStar < 4 then
411            self:getModel():updateAreaFullColorTime(saveData.areaId, ts)
412        end
413    end
414
415    self:setFlyCollectContext()
416
417    self:tryloadRes()
418end

527function Manager:getCurrentLevelForIcon()
528    local curLevel = self:getModel():getDataByKey("initStarLevel", 0) or 0
529    if curLevel > 0 then
530        curLevel = math.max(self:getModel():getDataByKey("simpleFullLevelRankBean.nowStarLevel", 0) or 0, curLevel)
531    end
532    
533    -- 校验：实时计算当前星级，如果缓存值比实际值高，说明星级数据被修改了
534    -- 只返回正确的图标等级，不更新缓存（缓存在 validateAndFixCurRankType 中统一处理）
535    local realLevel = self:calCurRankType()
536    if curLevel ~= realLevel then
537        printx(50, "[StellarConvergenceManager] getCurrentLevelForIcon: cached level", curLevel, "is different from real level", realLevel, ", return real level for icon")
538        curLevel = realLevel
539    end
540    
541    return curLevel
542end

544-- 校验并修复 curRankType 缓存（在进入游戏时调用，确保缓存值不会比实际值高）
545function Manager:validateAndFixCurRankType(forceSync)
546    local cachedLevel = self:getCurRankType()
547    local realLevel = self:calCurRankType()
548    
549    if forceSync then
550        if cachedLevel ~= realLevel then
551            printx(50, "[StellarConvergenceManager] validateAndFixCurRankType: force sync cached level from", cachedLevel, "to", realLevel)
552            self:setCurRankType(realLevel)
553        end
554        return
555    end
556
557    if cachedLevel > realLevel then
558        printx(50, "[StellarConvergenceManager] validateAndFixCurRankType: cached level", cachedLevel, "is different from real level", realLevel, ", fix cache")
559        self:setCurRankType(realLevel)
560    end
561end

708function Manager:calCurRankType()
709    local userManager = _G.amfBean("UserManager")
710    local serverMaxLevel = self:getServerMaxLevel()
711    
712    local userTopLevel = _G.getTopPassedLevelId()
713    if userTopLevel < serverMaxLevel then
714        return 0
715    end
716
717    local curYellow = userManager:getStar()
718    local totalYellow = serverMaxLevel * 3
719
720    if curYellow < totalYellow then
721        local per = math.floor(curYellow / totalYellow * 100)
722        for i = 1, 4 do
723            local next = self:getNextPerRatioByRankType(i)
724            if per < next then
725                return i
726            end
727        end
728    end
729
730    local curColor = userManager:getColorfulStar()
731    local totalColor = serverMaxLevel * 3
732    if curColor < totalColor then
733        return 5
734    end
735
736    local curHide = userManager:getHiddenStar()
737    local totalHide = (serverMaxLevel / 15 - 1) * 9
738    if curHide < totalHide then
739        return 6
740    end
741
742    return 7
743end

1236-------------------------------------- 资源预加载 -------------------------------------------

1238-- only app
1239local ResType = {
1240    PLIST = "plist",
1241    SPINE = "spine",
1242    SPRITE = "sprite",
1243}
1244
1245function Manager:loadResListAndRetainAsync(resList, onCompleted)
1246    if not resList or #resList == 0 then
1247        if onCompleted then
1248            onCompleted()
1249        end
1250        return
1251    end
1252
1253    if EnvUtil:isMiniGame() then
1254        _G.amfBean("AssetsUtil"):loadMiniFiles(resList, true):next(onCompleted)
1255        return
1256    end
1257
1258    if self.allLoadPromise and self.curLoadPromise then
1259        -- 已经请求加载过了，直接接promise
1260        self.curLoadPromise = self.curLoadPromise:next(onCompleted)
1261        return
1262    end
1263
1264    local autoReferenceMap = {}
1265    self.autoReferenceMap = autoReferenceMap
1266    self.allLoadPromise = deferred.new()
1267    self.curLoadPromise = self.allLoadPromise:next(onCompleted)
1268
1269    local cond = Condition.new(function()
1270        printx(31, "allResLoaded", TimeKit:timeInMillis())
1271        self.allLoadPromise:resolve()
1272    end, #resList)
1273
1274    local function onSingleResLoaded()
1275        cond:notify()
1276    end
1277
1278    printx(31, "start load res list async", TimeKit:timeInMillis())
1279    for _, file in ipairs(resList) do
1280        if not autoReferenceMap[file] then
1281            if file:ends(".plist") then
1282                UICreateUtil:loadPlistAsync(file, onSingleResLoaded)
1283                autoReferenceMap[file] = ResType.PLIST
1284            elseif file:ends(".skel") then
1285                UICreateUtil:loadSpineResAsync(file, nil, onSingleResLoaded)
1286                autoReferenceMap[file] = ResType.SPINE
1287            elseif file:ends(".png") or file:ends(".jpg") then
1288                UICreateUtil:loadImageAsync(file, onSingleResLoaded)
1289                autoReferenceMap[file] = ResType.SPRITE
1290            end
1291        end
1292    end
1293end
1294
1295function Manager:releaseResList()
1296    if not self.autoReferenceMap then return end
1297
1298    if EnvUtil:isMiniGame() then
1299        local files = table.keys(self.autoReferenceMap)
1300        _G.amfBean("AssetsUtil"):unloadRes(files)
1301    else
1302        for file, resType in pairs(self.autoReferenceMap) do
1303            if resType == ResType.PLIST then
1304                UICreateUtil:unloadPlist(file)
1305            elseif resType == ResType.SPINE then
1306                UICreateUtil:unloadSpineRes(file, true, true)
1307            elseif resType == ResType.SPRITE then
1308                UICreateUtil:unloadImage(file)
1309            end
1310        end
1311    end
1312    
1313    self.autoReferenceMap = nil
1314    self.allLoadPromise = nil
1315    self.curLoadPromise = nil
1316end
