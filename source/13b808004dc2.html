18	local currentTime = TimeKit:timeInSec()

21
22	if date.hour>=22 then return end
23	if date.hour < 18 then
24		date.hour = 18
25		date.min = 0
26		nextTime = os.time2(date)
27	end
28	_G.amfTransaction(TransactionEvent.LOCALPUSH, {
29		action = "add",
30		type = pushType,
31		timeStamp = nextTime,
32		groupKey = groupKey,
