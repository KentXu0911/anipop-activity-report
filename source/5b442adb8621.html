331function FruitTreeManager:openFruitTreePanel(param)
332    param = param or {}
333    _G.__CollectWordTestMode = param.__collectWordTestMode == true
334    -- 各个活动监听，在这里把需要的各种方法设置到金银果树中
335    local preOpenContextList = {}
336    FruitTreeEventCenter:dispatch(FruitTreeEvent.PRE_OPEN_CHECK, preOpenContextList)
337    table.sort(preOpenContextList, function(a,b)
338        return a:getPriority() > b:getPriority()
339    end)
340    local context = preOpenContextList[1]
341
342    local function openMainPanel()
343        local function _realOpen()
344            if self._mainVm and not self._mainVm.isDisposed then
345                print("已经打开金银果树了")
346                return
347            end
348            local panelVM
349            if context and context:getPanelVM() and context:isEnabled() then
350                panelVM = context:getPanelVM()
351            else
352                panelVM = require_debug "amf.feature.fruittree.view.FruitTreeMainPanelVM"
353            end
354
355            local vm = panelVM:create(param)
356            -- 复访等外部流程需要感知面板“真正展示/关闭”的时机：钩子由 AMFBasePanelVM 提供，
357            -- 活动替换面板（如集字活动）继承同一套基类，无需单独适配。
358            if param.onOpened then
359                vm:setPanelReadyCallback(function()
360                    safeRunCall(param.onOpened)
361                end)
362            end
363            if param.onClosed then
364                vm:setCloseCallback(function(isClickCloseBtn)
365                    safeRunCall(param.onClosed, isClickCloseBtn)
366                end)
367            end
368            vm:setLoadResFailCallback(function()
369                safeRunCall(param.onLoadFailed)
370                safeRunCall(param.closeCallback)
371            end)
372            vm:popout(nil, true)
373            self:setMainVM(vm)
374        end
375        if context then
376            context:callFunction(function()
377                _realOpen()
378            end)
379        else
380            _realOpen()
381        end
382    end
383    self:tryloadRes(true):next(function()
384        openMainPanel()
385    end, function()
386        CommonTip:showTip("资源下载失败~")
387    end)
388
389end

392function FruitTreeManager:_updatePickFruitCount()
393    local level = self:getFruitValue("fruitTreeLevel")
394    local limitCount = self:getFruitTreeCanPickCount(level)
395    if not limitCount then return end
396
397    limitCount = limitCount + self._achievementManager:getExtraCount(AchievementConst.AchiRightType.FRUIT)
398    local picked = self._dailyDataManager:getDailyDataValue("pickFruitCount")
399    if picked >= limitCount then
400        self._dailyDataManager:setDailyDataValue("pickFruitCount", limitCount)
401        return
402    end
403    self._dailyDataManager:setDailyDataValue("pickFruitCount", picked+1)
404end

712function FruitTreeManager:isEnoughSpeedPropNumber(id)
713	local info = self:getFruitDataById(id)
714    if not info then return nil end
715    local growCycle = self:getFruitGrowCycle()
716	if info.growCount >= self:getCrowCountNum() then return nil end
717	local elapse = info.updateTime + growCycle - TimeKit:timeInMillis()
718	if elapse > growCycle then elapse = growCycle end
719	local minute = self._propManager:getPropEffectValue(AMFPropId.HOURGLASS)
720	if minute <= 0 then return nil end
721	local need = math.ceil(elapse / (minute * 60000))
722	local have = self._userPropsManager:getUserPropNumber(AMFPropId.HOURGLASS)
723	if need > have then return false, need end
724	return true, need
725end

923-- 获得果子的奖励
924function FruitTreeManager:getRewardById(id)
925    local info = self:getFruitDataById(id)
926    if not info then return {} end
927    if info.level == 0 then return {} end
928    local cfg = self:getFruitTreeUpgradeMetaByLevel()
929    local buff = 1
930    if cfg.plus then
931        buff = (cfg.plus + 100)/100
932    end
933    -- 只有银币才有加成
934    if info.type ~= FruitTreeConst.FruitType.COIN then
935        buff = 1
936    end
937    local fruitMeta = self:getFruitsMetaByLevel(info.level)
938    if not fruitMeta then return {} end
939    local itemId = self:getRewardItemIdById(info.id)
940    if not itemId then return {} end
941    for _, v in ipairs(fruitMeta.reward) do
942        if v.itemId == itemId then
943            return {itemId = itemId, num = v.num * buff}
944        end
945    end
946    return {}
947end

1003-- 升级条件是否满足
1004function FruitTreeManager:getFinishUpgradeCondition(level)
1005	level = level or self:getFruitValue("fruitTreeLevel")
1006    local upgradeCondition = self:getUpgredeCondition(level)
1007	if not upgradeCondition then return 0, false, 0, 0 end
1008	local condition = upgradeCondition[1]
1009	if type(condition) ~= "table" then return 0, false, 0, 0 end
1010	if condition.itemId == FruitTreeConst.TreeUpgradeCondition.STAR then
1011		local star = self._userManager:getStar()
1012		local finish = (star >= condition.num)
1013		return condition.itemId, finish, star, condition.num
1014	elseif condition.itemId == FruitTreeConst.TreeUpgradeCondition.PASSLEVEL then
1015		local level = _G.getTopPassedLevelId()
1016		return condition.itemId, level >= condition.num, level, condition.num
1017	elseif condition.itemId == FruitTreeConst.TreeUpgradeCondition.BIND then --绑定sns账号qq&手机
1018		local finish = false
1019        local bindCfg = _G.amfBean("SnsBindLogic"):getFruitTreeLevelUpNodeCfg()
1020		return condition.itemId,bindCfg.hasBind,0,condition.num
1021	elseif condition.itemId == FruitTreeConst.TreeUpgradeCondition.INVALID then --没有这个条件
1022		return condition.itemId, true, 0, 0
1023	else
1024		return condition.itemId, false, 0, condition.num
1025	end
1026end
