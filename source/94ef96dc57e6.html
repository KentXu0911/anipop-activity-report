64function AddMineManager:checkIsCompleteLocal()
65    local collected = self:getDataValue("collected")
66    local collectRewardFetchDay = self:getDataValue("collectRewardFetchDay")
67    local continueRewardList = self:getDataValue("continueRewardList")
68    local canGetDayNum = #continueRewardList
69    local curDay = TimeUtil:getDayIdxSince1970()
70    local diff = curDay - collectRewardFetchDay
71    local hasCanGet = self._model:canGetReward()

73    return collected and collectRewardFetchDay > 0 and canGetDayNum > 0 and diff >= canGetDayNum and not hasCanGet

285function AddMineManager:isUserTagCanPop()
286    local activationTag = UserTagManager:getUserTag(UserTagNameKeyFullMap.kLiveness)
287    if not activationTag then return false end
288
289    local activationTagValue = UserTagValueMap[UserTagNameKeyFullMap.kLiveness]
290    if activationTag == activationTagValue.kNewUser then return false end
291
292    local serverUserTag, serverUserTagUpdateTime = self._model:getLivenessInfo()
293    if serverUserTag ~= activationTagValue.kReturnUser then return true end
294
295    local beginDay = TimeUtil:getDayIdxSince1970(math.floor(serverUserTagUpdateTime/1000))
296    local endDay = beginDay + 7 - 1
297    local curDay = TimeUtil:getDayIdxSince1970()
298
299    if curDay >= beginDay and curDay <= endDay then return false end
300
301    return true
302end

658function AddMineManager:showItemTip(reward, target)

663    local propName = self._propManager:getPropName(reward)
664    if levelId < unlock then
665        propName = propName.."，第"..unlock.."关解锁"
666    end
667
668    builder:setTitle(propName)
669    builder:setTarget(target)
670    if self._propManager:isFlowerSkin(reward.itemId) then
671        builder:setFlowerId(reward.itemId)
672    elseif self._propManager:isNameCardBg(reward.itemId) then
673        builder:setFlowerId(reward.itemId)
674    elseif self._propManager:isHeadFrame(reward.itemId) then
675        builder:setHeadFrameId(reward.itemId)
676    elseif self._propManager:isHeadImage(reward.itemId) then
677        builder:setHeadImageId(reward.itemId)
678    elseif self._propManager:isPlayerStatusProp(reward.itemId) then
679        builder:setPlayerStatusId(reward.itemId)
680    end
681    self._rewardTip = builder:build()
