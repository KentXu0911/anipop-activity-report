78local function asTimeInSec(value, defaultValue)
79    local time = tonumber(value)
80    if not time then
81        return defaultValue
82    end
83    if TimeKit and TimeKit.tryConvertMsToSec then
84        return TimeKit:tryConvertMsToSec(time) or defaultValue
85    end
86    if time > 1e11 then
87        return math.floor(time / 1000)
88    end
89    return time
90end

284function AliPayDesktopShortcutManager:addShortcut(callback)
285    if self._addingShortcut then
286        safeRunCall(callback, {
287            ok = false,
288            reason = ErrorCode.ADD_SHORTCUT_BUSY,
289            code = ErrorCode.ADD_SHORTCUT_BUSY,
290            supportState = self:checkPlatformSupport(),
291        })
292        return
293    end
294
295    self._addingShortcut = true
296    local callbackDone = false
297    local function finish(result)
298        if callbackDone then
299            return
300        end
301        callbackDone = true
302        self._addingShortcut = false
303        result = self:_normalizeAddShortcutResult(result)
304        if result.action == "auth_guide" then
305            self:notifyPanelRefresh(result.ok and "shortcut_auth_guide_ok" or "shortcut_auth_guide_fail")
306        else
307            self:notifyPanelRefresh(result.ok and "add_shortcut_ok" or "add_shortcut_fail")
308        end
309        safeRunCall(callback, result)
310    end
311
312    if setTimeOut then
313        setTimeOut(function()
314            finish({
315                ok = false,
316                reason = "timeout",
317                code = "timeout",
318                supportState = self:checkPlatformSupport(),
319            })
320        end, AliPayDesktopShortcutManager.ADD_SHORTCUT_TIMEOUT_SECONDS)
321    end
322
323    local proxy = self:getPlatformProxy()
324    if proxy and proxy.addShortcut then
325        proxy:addShortcut(finish)
326    else
327        finish({
328            ok = false,
329            reason = ErrorCode.UNSUPPORTED,
330            code = ErrorCode.UNSUPPORTED,
331            supportState = self:checkPlatformSupport(),
332        })
333    end
334end

547function AliPayDesktopShortcutManager:requestAsyncMainInfo(reason, successCallback, failCallback, options)
548    reason = reason or MainInfoReason.ASYNC_SYNC
549    options = type(options) == "table" and options or {}
550    local source = self:consumeLaunchSource()
551    local requestOptions = table.clone(options)
552    requestOptions.loading = false
553    requestOptions.syncFrom = reason
554    if source then
555        return self:_requestMainInfoFromDesktopSource(source, reason, successCallback, failCallback, requestOptions)
556    end
557
558    local proxy = self:getPlatformProxy()
559    if proxy and proxy.refreshLaunchSource then
560        proxy:refreshLaunchSource(function(refreshSource)
561            if refreshSource then
562                self:_requestMainInfoFromDesktopSource(refreshSource, reason, successCallback, failCallback, requestOptions)
563                return
564            end
565            self:_setDesktopShortcutEntry(nil)
566            self:notifyPanelRefresh("desktop_entry_changed")
567            self:requestMainInfo(reason, successCallback, failCallback, requestOptions)
568        end, reason == "enter_main_scene")
569        return nil
570    end
571    self:_setDesktopShortcutEntry(nil)
572    self:notifyPanelRefresh("desktop_entry_changed")
573    return self:requestMainInfo(reason, successCallback, failCallback, requestOptions)
574end

597function AliPayDesktopShortcutManager:requestMainInfo(reason, successCallback, failCallback, options)
598    options = type(options) == "table" and options or {}
599    options.clientVersion = AliPayDesktopShortcutManager.CLIENT_VERSION
600    if self._model and self._model.requestMainInfo then
601        local request = self._model:requestMainInfo(reason, options)
602        if request and request.next then
603            request:next(function(response)
604                self._mainInfoError = nil
605                self:clearUnavailableState(reason)
606                self:refreshIcon(reason)
607                self:notifyPanelRefresh(reason)
608                safeRunCall(successCallback, response)
609            end, function(response)
610                self:handleMainInfoError(response, reason)
611                safeRunCall(failCallback, response)
612            end)
613            return request
614        end
615    end
616
617    if options.failOnUnavailable then
618        local errorResult = {
619            errorCode = ErrorCode.REQUEST_UNAVAILABLE,
620            errorMsg = ErrorCode.REQUEST_UNAVAILABLE,
621            needRefresh = true,
622        }
623        if options.hideOnMainInfoError then
624            self:handleMainInfoError(errorResult, reason)
625        else
626            self:refreshIcon(reason)
627            self:notifyPanelRefresh(reason)
628        end
629        safeRunCall(failCallback, errorResult)
630        return nil
631    end
632    self:refreshIcon(reason)
633    self:notifyPanelRefresh(reason)
634    safeRunCall(successCallback)
635    return nil
636end

660        safeRunCall(failCallback, response)
661        return nil
662    end
663
664    local rewardItem = callIfExists(self._model, "getClaimableRewardItem", params)
665    if not rewardItem then
666        local response = {
667            errorCode = ErrorCode.NOT_CAN_REWARD,
668            errorMsg = ErrorCode.NOT_CAN_REWARD,
669            needRefresh = false,
670        }
671        safeRunCall(failCallback, response)
672        return nil
673    end
674
675    local request = self._model:requestReward(rewardItem)
676    if request and request.next then
677        request:next(function(response)
678            local result = self:_buildRewardResult(response, rewardItem)
679            callIfExists(self._model, "setRewardClaimed", rewardItem)
680            self:_showReward(result.rewards)
681            self:refreshIcon("request_reward")
682            self:notifyPanelRefresh("request_reward")
683            safeRunCall(successCallback, result)
684        end, function(response)
685            safeRunCall(failCallback, self:_buildErrorResult(response, true))
686        end)
687    else
688        safeRunCall(failCallback, {
689            errorCode = ErrorCode.REQUEST_UNAVAILABLE,
690            errorMsg = ErrorCode.REQUEST_UNAVAILABLE,
691            needRefresh = false,
692        })
693    end
694    return request
695end

871        syncFrom = source or MainInfoReason.ICON_OPEN,
872        failOnUnavailable = true,
873    })

876function AliPayDesktopShortcutManager:_openPanelWithoutResourceCheck(source, closeCallback)
877    local gate = self:canShowEntry()
878    if not gate.enabled then
879        self:_showOpenPanelFailTip(gate.reason)
880        safeRunCall(closeCallback, gate.reason)
881        return nil, gate.reason
882    end
883    if self._mainPanelVM and not self._mainPanelVM.isDisposed then
884        return self._mainPanelVM
885    end
886
887    local delegate = { manager = self, source = source, closeCallback = closeCallback }
888    local ok, vm = pcall(function()
889        if self._panelFactory then
890            local injectedVM = self._panelFactory:create(delegate)
891            if injectedVM then
892                return injectedVM
893            end
894        end
895
896        local GiftOnTheDeskPanelVM = require "amf.feature.minihub.alipaydesktopshortcut.view.GiftOnTheDeskPanelVM"
897        return GiftOnTheDeskPanelVM:create(delegate)
898    end)
899    if not ok then
900        self:_showOpenPanelFailTip("panel_create_failed", vm)
901        safeRunCall(closeCallback, vm or "panel_create_failed")
902        return nil, "panel_create_failed"
903    end
904    if not vm then
905        self:_showOpenPanelFailTip("panel_factory_empty")
906        safeRunCall(closeCallback, "panel_factory_empty")
907        return nil, "panel_factory_empty"
908    end
909
910    self._mainPanelVM = vm
911    self._mainPanelReady = false
912    local openFinished = false
913    local openTimerId
914    local function cancelOpenTimer()
915        if openTimerId and cancelTimeOut then
916            cancelTimeOut(openTimerId)
917        end
918        openTimerId = nil
919        vm._alipayDesktopShortcutAbortOpen = nil
920    end
921    local function abortOpen()
922        if openFinished then
923            return
924        end
925        openFinished = true
926        cancelOpenTimer()
927    end
928    local function finishOpen(ok, reason, detail)
929        if openFinished then
930            return
931        end
932        openFinished = true
933        cancelOpenTimer()
934        if ok then
935            self._mainPanelReady = true
936            return
937        end
938        self:unbindPanelVM(vm)
939        self:_releasePanelVM(vm, reason)
940        self:_showOpenPanelFailTip(reason, detail)
941        safeRunCall(closeCallback, detail or reason)
942    end
943
944    self:bindPanelVM(vm, source)
945    if vm.setPanelReadyCallback then
946        vm:setPanelReadyCallback(function()
947            finishOpen(true)
948        end)
949    end
950    if vm.setLoadResFailCallback then
951        vm:setLoadResFailCallback(function(error)
952            finishOpen(false, "panel_res_load_failed", error)
953        end)
954    end
955    if setTimeOut then
956        openTimerId = setTimeOut(function()
957            finishOpen(false, ErrorCode.OPEN_PANEL_TIMEOUT)
958        end, AliPayDesktopShortcutManager.OPEN_PANEL_TIMEOUT_SECONDS)
959        vm._alipayDesktopShortcutAbortOpen = abortOpen
960    end
961
962    if vm.popout then
963        local popoutOk, popoutError = pcall(function()
964            vm:popout()
965        end)
966        if not popoutOk then
967            finishOpen(false, "panel_popout_failed", popoutError)
968            return nil, "panel_popout_failed"
969        end
970    else
971        finishOpen(false, "panel_popout_failed")
972        return nil, "panel_popout_failed"
973    end
974    return vm
975end
