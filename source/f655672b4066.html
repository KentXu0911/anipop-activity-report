15local function cloneTable(source)
16    local result = {}
17    for k, v in pairs(source or {}) do
18        if type(v) == "table" then
19            result[k] = cloneTable(v)
20        else
21            result[k] = v
22        end
23    end
24    return result
25end

84function ResConfig:getLayoutConfig()
85    local layoutConfig = cloneTable(self.Layout)
86    local theme = self.getTheme()
87    local themeConfig = self.ThemeLayout[theme]
88    if themeConfig then
89        for k, v in pairs(themeConfig) do
90            if type(v) == "table" then
91                layoutConfig[k] = layoutConfig[k] or {}
92                for key, value in pairs(v) do
93                    if value ~= nil then
94                        layoutConfig[k][key] = value
95                    end
96                end
97            elseif v ~= nil then
98                layoutConfig[k] = v
99            end
100        end
101    end
102    return layoutConfig
103end
