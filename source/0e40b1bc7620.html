7function PromoteDailyAutoPopLogic:init(mgr)

12function PromoteDailyAutoPopLogic:registerEvent()
13    -- 连过重燃的活动数据来自 mainInfo（首次进主场景时常常还在路上），
14    -- 所以数据 / 公共资源就绪后再补一次场景2强弹
15    PromoteDailyEventCenter:register(PromoteDailyEvent.MAIN_INFO_LOAD_SUCCESS, self.tryPopRepairStage2, self)
16    PromoteDailyEventCenter:register(PromoteDailyEvent.PUB_RES_READAY, self.tryPopRepairStage2, self)
17end

21-- 本期标识：repairPeriodId 优先，缺失时退回结束时间（与加入重燃打点的周期去重口径一致）
22function PromoteDailyAutoPopLogic:_getRepairPeriodKey()
23    local act = self._promoteDailyMgr:getRepairActData()
24    if not act then return nil end
25    local periodId = tonumber(act.repairPeriodId) or 0
26    if periodId > 0 then
27        return tostring(periodId)
28    end
29    local endTime = tonumber(act.endTime) or 0
30    if endTime > 0 then
31        return tostring(endTime)
32    end
33    return nil
34end

36function PromoteDailyAutoPopLogic:isRepairStage2Poped()
37    local periodKey = self:_getRepairPeriodKey()
38    if not periodKey then return false end
39    return self._promoteDailyMgr:getLocalValue(REPAIR_STAGE2_POP_PERIOD_KEY) == periodKey
40end

42function PromoteDailyAutoPopLogic:markRepairStage2Poped()
43    local periodKey = self:_getRepairPeriodKey()
44    if not periodKey then return end
45    self._promoteDailyMgr:setLocalValue(REPAIR_STAGE2_POP_PERIOD_KEY, periodKey)
46end

48-- 场景2强弹条件：功能可用（服务端数据 + 公共资源就绪）+ 活动进行中 + 进度还是 0 + 本期没弹过
49-- 进度不为 0 说明已经打过至少一关，介绍界面（「任过 n 关，救回…」）没意义，不再弹
50function PromoteDailyAutoPopLogic:canPopRepairStage2()
51    if not self._promoteDailyMgr:isEnabled() then return false end
52    if not self._promoteDailyMgr:isRepairActInProgress() then return false end
53    if self._promoteDailyMgr:getRepairProgress() ~= 0 then return false end
54    return not self:isRepairStage2Poped()
55end

57-- 场景2通知时入队（入队后由强弹框架在本场景统一消费，不用自己触发）
58function PromoteDailyAutoPopLogic:doActEnterStage2()
59    if not self:canPopRepairStage2() then return end
60    self:_addRepairStage2Popup()
61end

63-- 场景2通知过程（含 tryInitIcon 里同步派发的 PUB_RES_READAY）：
64-- 这段时间只入队，交给框架在本场景通知结束后统一消费，避免通知链里抢跑强弹状态机
65function PromoteDailyAutoPopLogic:beginEnterStage2()
66    self._inEnterStage2 = true
67end
68
69function PromoteDailyAutoPopLogic:endEnterStage2()
70    self._inEnterStage2 = nil
71end
72
73-- 数据 / 资源就绪后补一次：此前入队条件不满足时（首次进主场景数据未到）不漏掉这次强弹
74function PromoteDailyAutoPopLogic:tryPopRepairStage2()
75    if not self:canPopRepairStage2() then return end
76    -- 数据正好在场景2通知链里就绪：紧随其后的 doActEnterStage2 会入队，这里不重复处理
77    if self._inEnterStage2 then return end
78    local autoPopupMgr = _G.amfBean("AutoPopupMgr")
79    if not autoPopupMgr or not autoPopupMgr:isInMainScene() then return end
80    self:_addRepairStage2Popup(true)
81end

83function PromoteDailyAutoPopLogic:_addRepairStage2Popup(needTrigger)
84    local function onHandleOver()
85        AutoPopEventCenter:dispatch(AutoPopEvent.SINGLE_POP_OVER, PopupCheckPoint.PROMOTE_DAILY_REPAIR_STAGE2)
86    end
87
88    local function creator()
89        -- 真正弹之前再判一次：排队期间活动可能已经结束 / 已达标
90        if not self:canPopRepairStage2() then
91            safeRunCall(onHandleOver)
92            return
93        end
94        self:markRepairStage2Poped()
95        -- strongPop：介绍文案「任过 n 关，救回…」，不按进度分段
96        self._promoteDailyMgr:showRepairPanel({
97            strongPop = true,
98        }, onHandleOver)
99    end
100
101    local autoPopupMgr = _G.amfBean("AutoPopupMgr")
102    autoPopupMgr:addPopupConfig(PopupCheckPoint.PROMOTE_DAILY_REPAIR_STAGE2, creator)
103    if needTrigger and not autoPopupMgr:isRunnning() then
104        autoPopupMgr:triggerStage2AutoPopup()
105    end
106end

117function PromoteDailyAutoPopLogic:isFullSignPoped(year, month)
