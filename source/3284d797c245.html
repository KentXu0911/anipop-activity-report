251function MainPanelVM:tryRequestReward(done)
252    if self.isDisposed then return end

261        local reward = self.mgr.model:getCanGetHelpGift(i)

263            if i < curProcess or
264                    (i == curProcess and
265                            (panelState == _G.MeMeFoxConst.STATUS.TIMEOUT_FAIL or
266                                    panelState == _G.MeMeFoxConst.STATUS.PASS_FAIL)) then
267                hasMissHelpReward = true
268                table.insert(missHelpRewardIndex, i)
269            end
270        end
271    end
272    local hasCurrentHelpReward = false
273    local currentHelpRewardIndex = {}
274    self.currentHelpRewards = {}
275    if panelState ~= _G.MeMeFoxConst.STATUS.TIMEOUT_FAIL and panelState ~= _G.MeMeFoxConst.STATUS.PASS_FAIL then
276        local underAnim = self.view:getUnderAnim(curProcess)
277        if isSafeCocosObj(underAnim) and underAnim.rewardBubble and underAnim.rewardBubbleInfo then
278            hasCurrentHelpReward = true
279            table.insert(currentHelpRewardIndex, curProcess)

281    end

283    self.finalRewardRealGold = nil
284    if panelState == _G.MeMeFoxConst.STATUS.SUCCESS_NOT_REWARD then
285        hasFinalReward = true
286    end
287    if hasMissHelpReward or hasCurrentHelpReward or hasFinalReward then

293                        if table.exist(missHelpRewardIndex, tonumber(v.index)) then
294                            table.insert(self.missHelpRewards, v.exchangeRewardResult)
295                        elseif table.exist(currentHelpRewardIndex, tonumber(v.index)) then
296                            table.insert(self.currentHelpRewards, v.exchangeRewardResult)
297                        end
298                    end
299                end
300                if serverData and serverData.activityRewards and serverData.activityRewards.windmillCoinReward then
301                    self.finalRewardRealGold = serverData.activityRewards.windmillCoinReward
302                    if self.finalRewardRealGold and self.finalRewardRealGold.num > 0 then
303                        local Meta = require 'amf.feature.localactivity.supermagicbird.config.SuperMagicBirdConst'
304                        _G.amfEC:notifyNormal("SuperMagicBirdActGetGold", 
305                        {
306                            from = Meta.extraRewardLogForm.kMemeHu,
307                            gold = self.finalRewardRealGold.num 
308                        })
309                    end
310                end
311                self:callViewFunc("addSkipLayer")
312                safeRunCall(done)
313            end,
314            failCallback = function()
315                if self.asyncRunner then self.asyncRunner:clean() end
316                self:callViewFunc("addSkipLayer")
317            end,
318            cancelCallback = function()
319                if self.asyncRunner then self.asyncRunner:clean() end
320                self:callViewFunc("addSkipLayer")
321            end,
322        })
323    else
324        self.skipResponseOptimizeEligible = self.mgr:isSkipResponseOptimizeEnabled() and
325            panelState == _G.MeMeFoxConst.STATUS.RUNNING and
326            self.view.inLevelShowLabel:isVisible()
327        self:callViewFunc("setSkipResponseOptimizeEligible", self.skipResponseOptimizeEligible)
328        self:callViewFunc("addSkipLayer")
329        safeRunCall(done)
330    end

334function MainPanelVM:tryGetMissHelpReward(done)
335    if self.isDisposed then return end
336    local temp = {
337        repeatList = {},
338        rewardDirectList = {},
339        rewardList = {},
340        yuanBaoList = {},
341    }
342    if self.missHelpRewards and not table.isEmpty(self.missHelpRewards) then
343        for _, reward in pairs(self.missHelpRewards) do
344            for _, v in pairs(reward.rewardList) do
345                table.insert(temp.rewardList, v)
346            end
347            for _, v in pairs(reward.rewardDirectList) do
348                if self.mgr:isBuffItem(v.itemId) then
349                    v.hideNum = true
350                end
351                table.insert(temp.rewardDirectList, v)
352            end
353            for _, v in pairs(reward.repeatList) do
354                table.insert(temp.repeatList, v)
355            end
356            for _, v in pairs(reward.yuanBaoList) do
357                table.insert(temp.yuanBaoList, v)
358            end
359        end
360    else
361        safeRunCall(done)
362        return
363    end
364
365    temp.rewardDirectList = table.mergeRewards(temp.rewardDirectList)
366    if (not table.isEmpty(temp.repeatList)) or
367            (not table.isEmpty(temp.rewardDirectList)) or
368            (not table.isEmpty(temp.rewardList)) or
369            (not table.isEmpty(temp.yuanBaoList)) then
370        if self.gotMissHelpReward then
371            safeRunCall(done)
372            return
373        end
374        self.gotMissHelpReward = true
375        MeMeFoxViewHelper:delayTimeCall(self.view.ui, 0.1, function()
376            self.mgr:addMultiRewardsExt({
377                closeCallback = function ()
378                    if self.isDisposed then return end
379                    safeRunCall(done)
380                end,
381                subTitle = "么么狐送上随机奖励，继续加油！",
382                rewardList = temp.rewardList,
383                repeatList = temp.repeatList,
384                rewardDirectList = temp.rewardDirectList,
385                yuanBaoList = temp.yuanBaoList,
386            })
387        end)
388    else
389        safeRunCall(done)
390        return
391    end
392end

666function MainPanelVM:onBeforeClosePanel()
667    local latestStatus = self.mgr.model:getMeMeFoxStatus() --最新的状态
668    local panelState = self.mgr.uiUseData.panelState --进入面板时的状态
669    local curTime = TimeKit:timeInSec()
670    local configEndTime = self.mgr.model:getConfigMainEndTime()
671
672    if panelState == _G.MeMeFoxConst.STATUS.FAILED_CD and
673            configEndTime ~= nil and not self._clickCloseBtn then
674        local tipText = "你的活动数据已更新，请重新进入活动~"
675        if curTime >= configEndTime then
676            tipText = "未完成挑战，活动已结束~"
677        end
678        MeMeFoxViewHelper:delayTimeTip(0.5, tipText)
679    end
680
681    if (latestStatus == _G.MeMeFoxConst.STATUS.NOT_START or
682            latestStatus == _G.MeMeFoxConst.STATUS.END)
683            and configEndTime ~= nil and
684            not self._clickCloseBtn then
685        local tipText = ""
686        if curTime >= configEndTime then
687            tipText = "活动已结束~"
688        elseif latestStatus == _G.MeMeFoxConst.STATUS.NOT_START then
689            tipText ="你的活动数据已更新，请重新进入活动~"
690        end
691        MeMeFoxViewHelper:delayTimeTip(0.5, tipText)
692    end
693
694    if panelState == _G.MeMeFoxConst.STATUS.TIMEOUT_FAIL or
695            panelState == _G.MeMeFoxConst.STATUS.PASS_FAIL then
696        if configEndTime ~= nil then
697            if curTime >= configEndTime then
698                MeMeFoxViewHelper:delayTimeTip(0.5, "未完成挑战，活动已结束~")
699                self.mgr.model:setMeMeFoxStatus(_G.MeMeFoxConst.STATUS.NOT_START)
700            else
701                local tipText = "本次挑战失败~"
702                if panelState == _G.MeMeFoxConst.STATUS.TIMEOUT_FAIL then
703                    tipText = "未在限定时间内完成，本次挑战失败~"
704                end
705                MeMeFoxViewHelper:delayTimeTip(0.5, tipText)
706            end
707        end
708        self:sendFailedStateChange()
709    end
710
711    if latestStatus ~= _G.MeMeFoxConst.STATUS.NOT_START then
712        --TODO 这里把所有的奖字都取消
713    end
714
715    if panelState == _G.MeMeFoxConst.STATUS.RUNNING then
716        self.mgr.model:cleanRobotInfoEliminateList()
717    end
718end

720function MainPanelVM:sendFailedStateChange()
721    local roundEndTime = self.mgr.model:getActMainEndTime()
722    local function refreshFunc()
723        local curTime = TimeKit:time()
724        local triggerTime = self.mgr.model:getTriggerTime()
725        local lastDayEndTime = self.mgr.model:getConfigMainEndTime()
726
727        local curDay =  TimeKit:getDayStartTimeByTS(curTime/1000)
728        local triggerDay =  TimeKit:getDayStartTimeByTS(triggerTime)
729        local lastDay =  TimeKit:getDayStartTimeByTS(lastDayEndTime)
730
731        if curDay < lastDay and triggerDay < curDay then
732            self.mgr.model:setMeMeFoxStatus(_G.MeMeFoxConst.STATUS.NOT_START)
733        else
734            self.mgr.model:setActMainEndTime(roundEndTime*1000)
735            self.mgr.model:setFailCdTime(tostring(curTime + 30*60*1000))
736            self.mgr.model:setMeMeFoxStatus(_G.MeMeFoxConst.STATUS.FAILED_CD)
737        end
738    end
739
740    local mode = self.mgr.model:getMeMeFoxMode()
741    if mode ~= _G.MeMeFoxConst.MODE.UNKNOWN then
742        local Request = require "amf.feature.winningstreak.module.memefox.http.StreakGoalInfo"
743        Request:post({
744            clientState = mode,
745            light = false,
746            versionControl = 1,
747            { clientMd5 = nil }
748        }, nil, true):next(function(response)
749            local data = response:getData()
750            self.mgr.model:cleanRobotInfoEliminateList()
751            self.mgr.model:updateServerData(data)
752            refreshFunc()
753            self.mgr:tryRemoveActIcon()
754        end, function()
755            refreshFunc()
756            self.mgr:tryRemoveActIcon()
757        end)
758    end
759end

761function MainPanelVM:immediateSkipAnim()
762    if self.isDisposed then return end
763    local panelState = self.mgr.uiUseData.panelState
764    local isJoinGroup = self.mgr.uiUseData.isJoinGroup
765    local removeHeadList = self.view:getJumpToWaterHeadList()
766    if self.asyncRunner then
767        self.asyncRunner:clean()
768    end
769    self:callViewFunc("immediateFinishUnderAnim")
770    self:callViewFunc("immediateFinishHeadAnim")
771    self:callViewFunc("refreshGoBtn")
772    if isJoinGroup then
773        self:callViewFunc("scrollToBottom")
774    end
775    local asyncRunner = AsyncRunner:create()
776    asyncRunner:add(function (done)
777        if self.isDisposed then return end
778        self:tryGetMissHelpReward(done)
779    end)
780
781    asyncRunner:add(function (done)
782        if self.isDisposed then return end
783        self:tryGetCurrentHelpReward(done)
784    end)
785
786    --失败面板
787    asyncRunner:add(function (done)
788        if self.isDisposed then return end
789        self:tryShowFailedPanel(done)
790    end)
791
792    --最终大奖
793    asyncRunner:add(function (done)
794        if self.isDisposed then return end
795        self:tryGetFinalReward(done)
796    end)
797
798    --1秒自动关闭
799    asyncRunner:add(function (done)
800        if self.isDisposed then return end
801        if panelState == _G.MeMeFoxConst.STATUS.RUNNING or
802                panelState == _G.MeMeFoxConst.STATUS.TIMEOUT_FAIL or
803                panelState == _G.MeMeFoxConst.STATUS.PASS_FAIL then
804            MeMeFoxViewHelper:delayTimeCall(self.view.ui, 1, function()
805                if self.view.inLevelShowLabel:isVisible() then
806                    self:onCloseBtnTapped()
807                end
808            end)
809        end
810        safeRunCall(done)
811    end)
812
813    asyncRunner:add(function (done)
814        if self.isDisposed then return end
815        self.canClickContinueClose = true
816        safeRunCall(done)
817    end)
818
819    asyncRunner:run()
820end
