112function PromoteDailyModelLogic:resetContinuseInfo()
113    local count = 0
114    -- 今天是第几天
115    local dayNum = self._todayNum
116    local endNum = 0
117    local startNum = 0
118    while true do
119        local date = Utils:getYMDByDayCount(dayNum)
120        local signState = self:getSignState(date.year, date.month, date.day)
121        -- 次数>0说明是有passCount的，可以算连过天数
122        if PromoteDailyConst.isContinousState(signState) then
123            if endNum == 0 then
124                endNum = dayNum
125            end
126            count = count + 1
127            dayNum = dayNum - 1
128        else
129            if dayNum == self._todayNum then
130                dayNum = dayNum - 1
131            else
132                startNum = dayNum + 1
133                break
134            end
135        end
136    end
137    -- 仅本地推导值，连过天数展示/比较以服务端 mainInfo.continuousPassingDay 为准；
138    -- 过关的本地 +1 走 PromoteDailyMgr:addContinousPassNum（活动期间改为累加 repairTaskInfo.progress）
139    self.continousPassNum = count
140    self.continousPassBeginNum = startNum
141    self.continousPassEndNum = endNum

344-- 出关七日条：今天落在第几格（1~7）= ((连过天数 - 1) % 7) + 1，连过天数含今天这一关
345--   新旧七天任务**统一**用「客户端自己算出来的当前连过天数」（2026-09-18 统一口径）：
346--   新七天任务（连过里程碑）本来就是这么算的（2026-09-17 产品口径：连过天数后端值可能是重燃
347--     「救回来」的旧值，位置要用自己算的，这样今天在第几格、前面补的格子、真正算连过的日子三者自洽）；
348--   旧七天任务原先用服务端 diaryTaskData.finishProgress 定位，但它和「算连过的日子」不是一个口径
349--     （状态 5 那天、被数据工具改过的那天都会算进任务进度、却不进七日数据），进度比连过天数大时
350--     今天左边就会空出格子，2026-09-18 修掉这个空日期后与上面收敛成同一条。
351--   注意：finishProgress 只用于七日任务自身的进度 / 领奖判断，不再参与「今天在第几格」。
352function PromoteDailyModelLogic:getSevenDaySlotIndex()
353    local progress = self:getLocalContinousPassDayCount()
354    if progress < 1 then
355        progress = 1
356    end
357    return (progress - 1) % PromoteDailyConst.TASK_REWARD_DAY + 1
358end

360-- 七日数据取数口（出关七日条 / 七天任务条专用）：取 7 个格子 = 今天所在的那个 7 天连过周期
361--   今天在第 getSevenDaySlotIndex() 格（新旧任务统一 = 自算连过天数）
362--   索引 1~7 = 七日条从左到右：
363--     今天之前那几格 = 今天之前刚连过的日子（日期从今天往前顺推：自算连过天数 ≥ 今天所在格，
364--       所以这几格一定都是算连过的日子，直接读日历就是点亮态，不需要再判断）；
365--     今天这格 = 读今天的签到态（状态 5 按未签兜底）；
366--     今天之后那几格 = 还没到的日子，一定未签，直接给 NONE（不读日历）。
367-- 口径：状态 0（没签到）与状态 5（重燃期间打关，算签到但不算连过）都不算连过，因此都不会出现在
368--   七日数据里（不占格子、不点亮），也就不会出现「空的日期」（2026-09-17 产品口径）。
369--   状态 5 的表达只留给主面板月历（Calendar / Card），主面板仍走 getSignState（保留 5）。
370function PromoteDailyModelLogic:getSevenDaySlots()
371    local today = self:getTodayNum()
372    local todaySlotIndex = self:getSevenDaySlotIndex()
373    local slots = {}
374    for i = 1, PromoteDailyConst.TASK_REWARD_DAY do
375        -- 相对今天的天数偏移：负 = 今天之前（刚连过的日子）、0 = 今天、正 = 还没到的日子
376        local offset = i - todaySlotIndex
377        local date = Utils:getYMDByDayCount(today + offset)
378        local signState
379        if offset > 0 then
380            signState = PromoteDailyConst.SignState.NONE
381        else
382            signState = self:getSignState(date.year, date.month, date.day)
383            -- 状态 5 不进七日数据：该天按未签处理（该格不点亮）
384            if signState == PromoteDailyConst.SignState.REPAIR then
385                signState = PromoteDailyConst.SignState.NONE
386            end
387        end
388        slots[i] = {
389            index = i,
390            date = date,
391            -- 星期几：1 = 周一 ... 7 = 周日
392            weekIndex = Utils:getWeekDayByTime(date),
393            signState = signState,
394            isToday = i == todaySlotIndex,
395        }
396    end
397    return slots
398end

400-- 上一段连过（连续过关）区间的开始 / 结束 dayNum：找不到返回 0, 0
401-- 口径与 resetContinuseInfo 一致（状态 SIGN / SHIELD 才算连过），区别是遇到「当前这段断签」不退出，
402-- 继续往前找第一段连过；找这段连过的「结束日」时最多回溯到 minNum（0 / 缺省表示不设下限）
403function PromoteDailyModelLogic:getLastContinousPassRange(fromNum, minNum)
404    fromNum = tonumber(fromNum) or self._todayNum
405    minNum = tonumber(minNum) or 0
406    local endNum = 0
407    local dayNum = fromNum
408    while dayNum > 0 and (minNum <= 0 or dayNum >= minNum) do
409        if self:_isContinousDayNum(dayNum) then
410            endNum = dayNum
411            break
412        end
413        dayNum = dayNum - 1
414    end
415    if endNum <= 0 then return 0, 0 end
416    local beginNum = endNum
417    while beginNum > 1 do
418        if not self:_isContinousDayNum(beginNum - 1) then break end
419        beginNum = beginNum - 1
420    end
421    return beginNum, endNum
422end

1123function PromoteDailyModelLogic:isCurTarget(year, month, day)
1124    if not self._promoteDailyMgr:getDataValue("chooseTaskOK") then return false end
1125    local dayNum = Utils:getDayNumByYMD(year, month, day)
1126    local curProgress = self._promoteDailyMgr:getTargetValue("nowProgress")
1127    local todaySigned = self:getTodaySignState() ~= PromoteDailyConst.SignState.NONE
1128    local todayNum = self:getTodayNum()
1129    local chooseData = self._promoteDailyMgr:getDataValue("chooseProgressTaskData") or {}
1130    local taskList = chooseData.perTaskDataList or {}
1131    local curTarget = nil

1133        if v.target > curProgress then
1134            curTarget = v.target
1135            break
1136        end
1137    end
1138    if curTarget then
1139        local targetDayNum = todayNum + curTarget - curProgress
1140        if not todaySigned then
1141            -- 没签到就减去一天
1142            targetDayNum = targetDayNum - 1
1143        end
1144        return targetDayNum == dayNum
1145    end
1146    return false
