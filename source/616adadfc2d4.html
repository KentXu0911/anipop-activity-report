1-- @ref 强弹框架: 场景调度顺序
2--
3-- checkPoint 是唯一身份（身份 = 业务规则 = 队列槽位），所以调度只回答一个问题：
4-- 本场景里所有候选 checkPoint 的消费顺序是什么。
5--
6-- 静态顺序由业务定义的 PopType 场景顺序 + 组内优先级决定；resolver（例如 AI
7-- 推荐权重）可以在同一批 checkPoint 上重排，但不能增删候选，也不能把候选搬到
8-- 别的槽位 —— 已经没有"别的槽位"这回事了。

15local function getStageKey(popStage)
16	return tostring(AutoPopupConst.PopStageKeyMapping[tostring(popStage)] or popStage)
17end

19local function getStageId(popStage)
20	local numericStage = tonumber(popStage)
21	if numericStage then return numericStage end
22	for stageId, mappedStage in pairs(AutoPopupConst.PopStageKeyMapping) do
23		if mappedStage == popStage then return tonumber(stageId) end
24	end
25	return nil
26end

28local function normalizePopStage(popStage)
29	return AutoPopupConst.PopStageKeyMapping[tostring(popStage)] or popStage
30end

55function PopupSchedule:setResolver(resolver)

62-- 队列内容变化后调用；下一次 prepare 会重新计算顺序。

64	for _, stageCache in pairs(self._stageCache) do

69function PopupSchedule:_getStageCache(popStage)
70	local stageKey = getStageKey(popStage)
71	local cache = self._stageCache[stageKey]
72	if not cache then
73		cache = {orderedCheckPoints = {}, dirty = true}
74		self._stageCache[stageKey] = cache
75	end
76	return cache
77end

79-- 静态顺序：场景的 PopType 顺序优先，其次组内优先级，最后用 checkPoint 名兜底，
80-- 保证同一批候选在任何设备上的顺序都稳定。
81function PopupSchedule:_sortStatically(popStage, candidates)
82	local normalizedPopStage = normalizePopStage(popStage)
83	table.sort(candidates, function(left, right)
84		local order = self:comparePriority(
85			normalizedPopStage, left.checkPoint, right.checkPoint
86		)
87		if order ~= 0 then return order < 0 end
88		return tostring(left.checkPoint) < tostring(right.checkPoint)
89	end)
90	return candidates
91end

93-- resolver 只能重排同一批 checkPoint；任何增删、重复或未知值都整体回退静态顺序。
94function PopupSchedule:_applyResolver(popStage, candidates)
95	if #candidates < 2 or not self._resolver then return candidates end
96
97	local byCheckPoint = {}
98	local resolverInput = {}
99	for index, candidate in ipairs(candidates) do
100		byCheckPoint[candidate.checkPoint] = candidate
101		resolverInput[index] = {
102			checkPoint = candidate.checkPoint,
103			source = candidate.source or "config",
104			actId = candidate.actId,
105			metaId = candidate.metaId,
106			popType = candidate.popType,
107		}
108	end
109
110	local context = {
111		popStageId = getStageId(popStage),
112		popStage = normalizePopStage(popStage),
113	}
114	local ok, resolved = pcall(self._resolver, resolverInput, context)
115	if not ok or type(resolved) ~= "table" or #resolved ~= #candidates then
116		if not ok then
117			AutoPopupDebugLog:log("PopupSchedule:resolver failed, fallback", tostring(resolved))
118		end
119		return candidates
120	end
121
122	local ordered = {}
123	local seen = {}
124	for index, value in ipairs(resolved) do
125		local checkPoint = value.checkPoint or value
126		local candidate = byCheckPoint[checkPoint]
127		if not candidate or seen[checkPoint] then
128			AutoPopupDebugLog:log("PopupSchedule:resolver returned invalid order, fallback")
129			return candidates
130		end
131		seen[checkPoint] = true
132		ordered[index] = candidate
133	end
134	return ordered
135end

137-- 场景候选顺序 = 注册式定义快照（含未通过 check 的候选）+ 本场景配置式候选。
138function PopupSchedule:prepareDefinitions(popStage, definitions)
139	local stageKey = getStageKey(popStage)
140	if self._candidateOrder[stageKey] then
141		return self._candidateOrder[stageKey]
142	end
143
144	local registration = self._autoPopupMgr and self._autoPopupMgr._popupRegistration
145	local source = definitions or
146		(registration and registration:getSceneLogDefinitions(popStage)) or {}
147
148	local candidates = {}
149	local seen = {}
150	for _, identity in ipairs(source) do
151		local checkPoint = identity.checkPoint
152		if checkPoint and not seen[checkPoint] then
153			seen[checkPoint] = true
154			candidates[#candidates + 1] = {
155				checkPoint = checkPoint,
156				source = identity.source or "registered",
157				actId = identity.actId,
158				metaId = identity.metaId,
159				popType = identity.popType,
160			}
161		end
162	end
163
164	-- 运行时候选只属于本次解析，不能回写注册中心的定义快照。
165	for checkPoint, identity in pairs(self._runtimeCandidates[stageKey] or {}) do
166		if not seen[checkPoint] then
167			seen[checkPoint] = true
168			candidates[#candidates + 1] = identity
169		end
170	end
171
172	local staticOrder = self:_sortStatically(popStage, candidates)
173	local ordered = self:_applyResolver(popStage, staticOrder)
174	local order = {byCheckPoint = {}, originalByCheckPoint = {}, count = #ordered}
175	for rank, candidate in ipairs(ordered) do
176		order.byCheckPoint[candidate.checkPoint] = rank
177		-- resolver 只是重排同一批候选，所以 staticOrder[rank] 就是被挪到这个顺序
178		-- 位置的候选所挤掉的原位置候选。
179		order.originalByCheckPoint[candidate.checkPoint] = staticOrder[rank]
180	end
181	self._candidateOrder[stageKey] = order
182	self:publishScheduleOrder(popStage)
183	return order
184end

186-- 把整份候选顺序（顺序位置 + 被挤掉的原位置候选）发布给日志。未入队的候选
187-- 同样有位置，日志才不用退回观察顺序；重复发布是幂等的。
188function PopupSchedule:publishScheduleOrder(popStage)
189	local order = self:getCandidateOrder(popStage)
190	AutoPopupDebugLog:recordPopupScheduleRanks(order.byCheckPoint, self:_collectInterventions(order))
191end

193-- 被 resolver 挪动位置的候选：本来排在这个位置上的是谁，那个候选的 actId 是多少。
194function PopupSchedule:_collectInterventions(order)
195	local interventions = {}
196	for checkPoint, original in pairs(order.originalByCheckPoint) do
197		if original.checkPoint ~= checkPoint then
198			interventions[checkPoint] = {
199				checkPoint = original.checkPoint,
200				actId = self:_getOriginalActId(original),
201			}
202		end
203	end
204	return interventions
205end

207-- 原位置候选的 actId 优先取队列里的真实 entry；那一轮没入队时退回它的注册/配置声明。
208function PopupSchedule:_getOriginalActId(original)
209	local queue = self._autoPopupMgr:getPopupQueueByCheckPoint(original.checkPoint)
210	local originalEntry = queue and queue[original.checkPoint]
211	return (originalEntry and originalEntry.actId) or original.actId
212end

214function PopupSchedule:getCandidateOrder(popStage)
215	local stageKey = getStageKey(popStage)

217end

219function PopupSchedule:getRank(popStage, checkPoint)

222	return order and order.byCheckPoint[checkPoint] or nil

225-- 登记本次场景出现过的配置式候选（含被限流挡在队列外的），让下一个解析周期
226-- 把它纳入排序；只有拿到调度位置的候选，日志才能按真实顺序而不是观察顺序排列。
227function PopupSchedule:recordRuntimeCandidate(popupEntry)
228	if not popupEntry or popupEntry.source ~= "config" then return end
229	local popStage = AutoPopupDebugLog:getPopupSceneStage()
230	if not popStage then return end
231	local checkPoint = popupEntry.checkPoint
232	if not checkPoint then return end
233
234	local stageKey = getStageKey(popStage)
235	local runtime = self._runtimeCandidates[stageKey] or {}
236	local registered = runtime[checkPoint]
237	if registered then
238		-- 同一候选可能先被挡住、之后才真正入队（或反过来）。活动归属要补齐，
239		-- 否则 resolver 会因为拿不到 actId 而漏算这个候选的优先级。
240		if registered.actId == nil and popupEntry.actId ~= nil then
241			registered.actId = popupEntry.actId
242			self._candidateOrder[stageKey] = nil
243			self:markDirty()
244		end
245		return
246	end
247	runtime[checkPoint] = {
248		checkPoint = checkPoint,
249		source = "config",
250		actId = popupEntry.actId,
251	}
252	self._runtimeCandidates[stageKey] = runtime
253	self._candidateOrder[stageKey] = nil
254	self:markDirty()
255end

257-- 把当前各 PopType 队列里的候选整理成本场景的消费顺序，并写入调度序号。
258function PopupSchedule:prepare(popStage)
259	local stageCache = self:_getStageCache(popStage)
260	if not stageCache.dirty then return stageCache end
261
262	local mgr = self._autoPopupMgr
263	local candidates = {}
264	for _, popType in ipairs(mgr:getPopupCheckList(popStage) or {}) do
265		for checkPoint, popupEntry in pairs(mgr:getPopupQueueByPopType(popType) or {}) do
266			candidates[#candidates + 1] = {
267				checkPoint = checkPoint,
268				source = popupEntry.source or "config",
269				actId = popupEntry.actId,
270				popType = popType,
271				popupEntry = popupEntry,
272			}
273		end
274	end
275
276	-- 声明式顺序是基线；不在定义快照里的配置式候选排在基线之后，彼此保持
277	-- 静态优先级顺序。
278	local candidateOrder = self:getCandidateOrder(popStage)
279	local rankBase = candidateOrder and candidateOrder.count or 0
280	local ranked = {}
281	for index, candidate in ipairs(candidates) do
282		ranked[#ranked + 1] = {
283			candidate = candidate,
284			rank = candidateOrder and candidateOrder.byCheckPoint[candidate.checkPoint] or nil,
285			index = index,
286		}
287	end
288	table.sort(ranked, function(left, right)
289		if left.rank and right.rank and left.rank ~= right.rank then
290			return left.rank < right.rank
291		end
292		if left.rank and not right.rank then return true end
293		if right.rank and not left.rank then return false end
294		return left.index < right.index
295	end)
296
297	stageCache.orderedCheckPoints = {}
298	for index, item in ipairs(ranked) do
299		local candidate = item.candidate
300		local checkPoint = candidate.checkPoint
301
302		stageCache.orderedCheckPoints[#stageCache.orderedCheckPoints + 1] = checkPoint
303
304		local rank = item.rank or (rankBase + index)
305		self:_recordIntervention(candidate.popupEntry, checkPoint, candidateOrder)
306		AutoPopupDebugLog:recordPopupScheduleOrder(candidate.popupEntry, rank, checkPoint)
307	end
308	stageCache.dirty = false
309	return stageCache
310end

312-- 把"被 resolver 挤出原顺序位置"的结果写在真实 popupEntry 上：原位置的候选身份，
313-- 以及该候选的 actId。actId 优先取队列里的真实 entry；原位置候选这轮没入队时退回
314-- 它的注册/配置声明，两者都取不到才留空。曝光打点与场景日志共用这一份数据。
315function PopupSchedule:_recordIntervention(popupEntry, checkPoint, candidateOrder)
316	local original = candidateOrder and candidateOrder.originalByCheckPoint[checkPoint]
317	if original and original.checkPoint ~= checkPoint then
318		popupEntry.originalCheckPoint = original.checkPoint
319		popupEntry.originalActId = self:_getOriginalActId(original)
320	else
321		popupEntry.originalCheckPoint = nil
322		popupEntry.originalActId = nil
323	end
324end

326function PopupSchedule:getOrderedCheckPoints(popStage)
327	return self:prepare(popStage).orderedCheckPoints
328end

330-- 与真实调度共用的纯比较逻辑：先比较当前场景的 PopType 顺序，再比较组内优先级。
331function PopupSchedule:comparePriority(popStage, leftCheckPoint, rightCheckPoint)
332	if leftCheckPoint == rightCheckPoint then return 0 end
333
334	local mgr = self._autoPopupMgr
335	local normalizedPopStage = normalizePopStage(popStage)
336	local popTypeOrder = mgr:getPopupCheckList(normalizedPopStage)
337	local leftPopType = mgr:getPopType(leftCheckPoint)
338	local rightPopType = mgr:getPopType(rightCheckPoint)
339	if leftPopType ~= rightPopType then
340		local leftOrder = #popTypeOrder + 1
341		local rightOrder = #popTypeOrder + 1
342		for index, popType in ipairs(popTypeOrder) do
343			if popType == leftPopType then leftOrder = index end
344			if popType == rightPopType then rightOrder = index end
345		end
346		if leftOrder ~= rightOrder then
347			return leftOrder < rightOrder and -1 or 1
348		end
349		return 0
350	end
351
352	local leftPriority = mgr:getPriority(leftCheckPoint)
353	local rightPriority = mgr:getPriority(rightCheckPoint)
354	if leftPriority == rightPriority then return 0 end
355	return leftPriority < rightPriority and -1 or 1
356end
