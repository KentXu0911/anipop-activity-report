20-- 三关延后状态统一经活动 Model 的 local flag 接口读写。
21local function getLocalFlag(mgr, key, needTime)
22    local model = mgr and mgr.getModel and mgr:getModel()
23    if not model or not model.getFlagFromLocalByKey then
24        return nil
25    end
26    return model:getFlagFromLocalByKey(key, needTime)
27end

29local function saveLocalFlag(mgr, key, needTime, value)
30    local model = mgr and mgr.getModel and mgr:getModel()
31    if not model or not model.saveFlagToLocalByKey then
32        log(mgr, "PengPeng2026 save local flag failed", key, "hasModel", model ~= nil)
33        return
34    end
35    model:saveFlagToLocalByKey(key, needTime, value)
36end
37
38local function getLocalIntFlag(mgr, key, needTime)
39    local value = tonumber(getLocalFlag(mgr, key, needTime)) or 0
40    if value < 0 then
41        return 0
42    end
43    return math.floor(value)
44end
45
46local function saveLocalIntFlag(mgr, key, needTime, value)
47    saveLocalFlag(mgr, key, needTime, math.max(0, math.floor(tonumber(value) or 0)))
48end

50-- 主面板同轮强弹后调用；只写三次过关抑制计数，不写每日已弹，避免误消耗引流曝光机会。
51function Helper.markSuppressedByMainPanelAutoPop(mgr)

58-- 每次玩家过关后消耗一次抑制计数；返回 true 表示本次确实扣减了一次。
59function Helper.consumeSuppressPassCount(mgr)
60    local panelConfig = getConfig()
61    local count = getLocalIntFlag(mgr, panelConfig.SUPPRESS_COUNT_KEY, true)
62    if count <= 0 then
63        return false
64    end
65
66    saveLocalIntFlag(mgr, panelConfig.SUPPRESS_COUNT_KEY, true, count - 1)
67    log(mgr, "PengPeng2026 attract panel suppressed", "left", count - 1)
68    return true
69end

71-- 检测当前是否允许弹引流面板；场景和每日次数由强弹框架保证。
72function Helper.checkCanPop(mgr)
73    local panelConfig = getConfig()
74    if not mgr:isEnabled() then
75        log(mgr, "PengPeng2026 attract panel check false", "not_main_time")
76        return false
77    end
78    local suppressCount = getLocalIntFlag(mgr, panelConfig.SUPPRESS_COUNT_KEY, true)
79    if suppressCount > 0 then
80        log(mgr, "PengPeng2026 attract panel check false", "suppressedPassLeft", suppressCount)
81        return false
82    end
83    local launchOpportunity = tonumber(mgr:getModel():getLaunchOpportunity()) or 0
84    if launchOpportunity < panelConfig.MIN_LAUNCH_OPPORTUNITY then
85        log(mgr, "PengPeng2026 attract panel check false", "launchOpportunity", launchOpportunity)
86        return false
87    end
88
89    log(mgr, "PengPeng2026 attract panel check true", "launchOpportunity", launchOpportunity)
90    return true
91end

103local function createRewardTouchHandler(reward)

113                :setOrdered()
114                :setTitle(name and name ~= "" and name or tostring(reward.itemId))
115                :setAutoCloseTime(3)
