141function DailyLevelMgr:onActInfoUpdate(event)
142    local data = event:getData()
143    local actInfos = data.actInfos or {}
144    local actInfo = actInfos[tostring(DailyLevelConst.ACT_ID)]
145    if not actInfo then
146        -- 服务端全量列表中没有本活动才表示活动已下线；其他活动的局部更新与本活动无关。
147        if not data.fromServer and not self._isClearingActInfo then return end
148        self:closeAllPanelVM()
149        ActInfo:syncActInfo(nil)
150        self:clearActAutoPopUniqueKey()
151        self:removeIcon()
152        return
153    end
154    ActInfo:syncActInfo(actInfo)
155    if not self:isIconEnabled() then
156        self:removeIcon()
157        return
158    end
159    if not self:isActEnabled() then
160        self:tryAddIcon()
161        return
162    end
163    -- 没请求过主info就请求一次
164    if data.fromServer and not self:getDataValue("fromServer") then
165        self:requestMainInfo(false):next(function()
166            self:tryAddIcon()
167        end)
168    else
169        self:tryAddIcon()
170    end
171end

292function DailyLevelMgr:openMainPanel(param)
293    param = param or {}
294    if self:hasAliveMainPanelVM() then
295        if param.fromLevel then
296            self:finishExitLevelLoading()
297        end
298        safeRunCall(param.closeCallback)
299        return
300    end
301
302    local isAutoPopup = param.autoPopup == true
303    local lightStart = param.userClick ~= true
304    -- 用户点击或强弹进入领奖期且没有可展示奖励时，在打开主界面前直接结算并刷新下一期。
305    -- 其他入口仍由主界面统一处理宝箱、奖券和换期表现。
306    local skipAutoFinalReward = param.userClick ~= true and not isAutoPopup
307    self:requestMainInfo(param.loading, skipAutoFinalReward, lightStart, isAutoPopup):next(function()
308        if param.fromLevel then
309            self:finishExitLevelLoading()
310        end
311        if self:hasAliveMainPanelVM() then
312            safeRunCall(param.closeCallback)
313            return
314        end
315        if isAutoPopup and param.autoPopupCheck and not param.autoPopupCheck() then
316            safeRunCall(param.closeCallback)
317            return
318        end
319        if not self:isV2IconActive() then
320            self:clearNewActFlag()
321        end
322        self:markOnceGuide(DailyLevelConst.GuideId.STAGE2_MAIN_PANEL)
323        self:_markStage4PopupRecord()
324        local DailyLevelMainPanelVM = require "amf.feature.localactivity.dailylevel.view.DailyLevelMainPanelVM"
325        param.consumeV2NewOnOpen = isAutoPopup and self:isV2IconActive()
326        local vm = DailyLevelMainPanelVM:create(param)
327        vm:setLoadResFailCallback(function()
328            safeRunCall(param.closeCallback)
329        end)
330        vm:popout()
331        self:managePanelVM(vm)
332    end, function()
333        if param.fromLevel then
334            self:finishExitLevelLoading()
335        end
336        safeRunCall(param and param.closeCallback)
337    end)

340function DailyLevelMgr:prepareLevelLoadingRes()
341    if not EnvUtil:isMiniGame() or self._levelLoadingResRetained then
342        return deferred.success()
343    end
344    if self._levelLoadingResDeferred then
345        return self._levelLoadingResDeferred
346    end
347
348    self._levelLoadingResReleasePending = false
349    local loading = deferred.new()
350    self._levelLoadingResDeferred = loading
351    ResLoader:loadAndRetainRes(Res:getLoadingResList()):next(function()
352        if self._levelLoadingResDeferred ~= loading then return end
353        self._levelLoadingResDeferred = nil
354        self._levelLoadingResRetained = true
355        if self._levelLoadingResReleasePending then
356            self:releaseLevelLoadingRes()
357        end
358        loading:resolve()
359    end, function(event)
360        if self._levelLoadingResDeferred ~= loading then return end
361        self._levelLoadingResDeferred = nil
362        self._levelLoadingResReleasePending = false
363        loading:reject(event)
364    end)
365    return loading
366end

374    self._levelLoadingResReleasePending = false
375    if not self._levelLoadingResRetained then return end
376    self._levelLoadingResRetained = false

632function DailyLevelMgr:handleV2IconLifecycleBoundary(payload)
633    if payload and payload.kind == "main_end" then
634        return self:refreshV2Icon("main_end_boundary", payload)
635    end
636    if not payload or payload.kind ~= "activity_end" then
637        return false, "DAILY_LEVEL_ICON_LIFECYCLE_BOUNDARY_INVALID"
638    end
639
640    local oldPeriodId = self:getDataValue("periodId", 0)
641    local requested, promise = pcall(self.requestNextPeriodMainInfo, self, oldPeriodId, false)
642    if not requested or not promise or type(promise.next) ~= "function" then
643        self:removeIcon()
644        return false, requested and "DAILY_LEVEL_ICON_PERIOD_REFRESH_PROMISE_REQUIRED" or promise
645    end
646    local chained, chainError = pcall(promise.next, promise, nil, function()
647        self:removeIcon()
648    end)
649    if not chained then
650        self:removeIcon()
651        return false, chainError
652    end
653    return true

1145function DailyLevelMgr:getMaxClaimableBoxIdByCollectNum(collectNum)
1146    collectNum = math.max(tonumber(collectNum) or 0, 0)
1147    local boxConfigList = self:getConfigValue("boxConfigList", {}) or {}
1148    local boxCycleBeginId = tonumber(self:getConfigValue("boxCycleBeginId"))

1150    local cycleConfigList = {}
1151
1152    for _, config in ipairs(boxConfigList) do
1153        local id = tonumber(config.id)
1154        if id then
1155            local targetList = fixedConfigList
1156            if boxCycleBeginId and id >= boxCycleBeginId then
1157                targetList = cycleConfigList
1158            end
1159            table.insert(targetList, {
1160                id = id,
1161                target = math.max(tonumber(config.target) or 0, 0),
1162            })
1163        end
1164    end
1165
1166    local function sortById(a, b)
1167        return a.id < b.id
1168    end
1169    table.sort(fixedConfigList, sortById)
1170    table.sort(cycleConfigList, sortById)
1171
1172    local usedCollectNum = 0
1173    local maxId = 0
1174    for _, config in ipairs(fixedConfigList) do
1175        usedCollectNum = usedCollectNum + config.target
1176        if collectNum < usedCollectNum then
1177            return maxId
1178        end
1179        maxId = config.id
1180    end
1181
1182    if not boxCycleBeginId or #cycleConfigList == 0 then
1183        return maxId
1184    end
1185
1186    local cycleTarget = 0
1187    for _, config in ipairs(cycleConfigList) do
1188        cycleTarget = cycleTarget + config.target
1189    end
1190    if cycleTarget <= 0 then
1191        return maxId
1192    end
1193
1194    local remainCollectNum = collectNum - usedCollectNum
1195    local completedCycleNum = math.floor(remainCollectNum / cycleTarget)
1196    maxId = boxCycleBeginId - 1 + completedCycleNum * #cycleConfigList
1197    remainCollectNum = remainCollectNum - completedCycleNum * cycleTarget
1198
1199    for _, config in ipairs(cycleConfigList) do
1200        if remainCollectNum < config.target then
1201            break
1202        end
1203        remainCollectNum = remainCollectNum - config.target
1204        maxId = maxId + 1
1205    end
1206
1207    return maxId
1208end

1323function DailyLevelMgr:getPackageIdByBuyType(buyType)

1325    local buyPackageMap = self:getDataValue("buyPackageMap", {}) or {}
1326    local matchedMetaList = {}
1327    local buyTimes = 0
1328    for _, meta in ipairs(packageMetaList) do
1329        if meta.buyType == buyType and meta.packageId then
1330            table.insert(matchedMetaList, meta)
1331            buyTimes = buyTimes + (tonumber(buyPackageMap[tostring(meta.packageId)] or buyPackageMap[meta.packageId]) or 0)
1332        end
1333    end
1334    table.sort(matchedMetaList, function(a, b)
1335        local aMinBuyTimes = tonumber(a.minBuyTimes) or -1
1336        local bMinBuyTimes = tonumber(b.minBuyTimes) or -1
1337        if aMinBuyTimes == -1 then return false end
1338        if bMinBuyTimes == -1 then return true end
1339        return aMinBuyTimes < bMinBuyTimes
1340    end)
1341
1342    for _, meta in ipairs(matchedMetaList) do
1343        local minBuyTimes = tonumber(meta.minBuyTimes) or -1
1344        if minBuyTimes == -1 or buyTimes <= minBuyTimes then
1345            return tonumber(meta.packageId) or meta.packageId
1346        end
1347    end
1348    return nil
1349end
