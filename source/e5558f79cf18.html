237-- 清屏雨触发阈值来自后端 boardPieceMinCount；投放数量从后端 bonusCupCountRange 内随机。
238function Model:getClearRainConfig()
239    local config = self:getConfig()
240    local versionMeta = self:getCurrentVersionMeta() or {}
241    local minCount = tonumber(versionMeta.boardPieceMinCount) or 0
242    local bonusCupCountRange = versionMeta.bonusCupCountRange or config.bonusCupCountRange
243    local cupCount = self:_getRandomClearRainCupCount(bonusCupCountRange)
244    local enabled = minCount > 0 and cupCount > 0
245    if self.printLog then
246        self:printLog("BONUS", true, "clear rain config", {
247            enabled = enabled,
248            boardPieceMinCount = versionMeta.boardPieceMinCount,
249            versionBonusCupCountRange = versionMeta.bonusCupCountRange,
250            rootBonusCupCountRange = config.bonusCupCountRange,
251            bonusCupCountRange = bonusCupCountRange,
252            triggerCupCount = minCount,
253            cupCount = cupCount,
254        })
255    end
256    return {
257        enabled = enabled,
258        triggerCupCount = minCount,
259        cupCount = cupCount,
260        bonusCupCountRange = bonusCupCountRange,

264-- 按后端 bonusCupCountRange 生成本次清屏雨投放数量；配置缺失或非法时不投放。
265function Model:_getRandomClearRainCupCount(countRange)
266    if type(countRange) ~= "table" then return 0 end
267
268    local minCount = tonumber(countRange[1])
269    local maxCount = tonumber(countRange[2])
270    if not minCount or not maxCount then return 0 end
271
272    minCount = math.floor(minCount)
273    maxCount = math.floor(maxCount)
274    if minCount > maxCount then
275        minCount, maxCount = maxCount, minCount
276    end
277    if maxCount <= 0 then return 0 end
278    minCount = math.max(minCount, 0)
279
280    return math.random(minCount, maxCount)
281end

522    local addMillis = (tonumber(minutes) or 0) * 60 * 1000

551-- 当前是否有可领取进度奖励；优先尊重后端聚合标记，缺失或未刷新时按本地分数与配置兜底计算。
552function Model:hasProgressRewardToClaim()
553    if self:hasClaimableReward() then
554        return true
555    end
556    return self:_hasNormalProgressRewardToClaim() or self:_hasEndlessProgressRewardToClaim()
557end
558
559-- 普通进度奖励按累计阈值和全局已领游标判断。
560function Model:_hasNormalProgressRewardToClaim()
561    local score = tonumber(self:getScore()) or 0
562    local lastClaimedIndex = tonumber(self:getDataByKey("lastClaimedRewardIndex", 0)) or 0
563    for index, nodeCfg in ipairs(self:getNormalProgressConfig() or {}) do
564        if index > lastClaimedIndex and score >= (tonumber(nodeCfg.incrementScore) or 0) then
565            return true
566        end
567    end
568    return false
569end
570
571-- 无尽进度奖励按普通进度结束后的循环节点数判断。
572function Model:_hasEndlessProgressRewardToClaim()
573    local normalCount = #(self:getNormalProgressConfig() or {})
574    local claimedEndlessCount = math.max(0, (tonumber(self:getDataByKey("lastClaimedRewardIndex", 0)) or 0) - normalCount)
575    local completedEndlessCount = self:_getCompletedEndlessRewardCount()
576    return completedEndlessCount > claimedEndlessCount
577end
578
579-- 根据当前总分计算已达成的无尽奖励节点数量。
580function Model:_getCompletedEndlessRewardCount()
581    local normalEndScore = self:_getNormalProgressEndScore()
582    local remainScore = (tonumber(self:getScore()) or 0) - normalEndScore
583    if remainScore <= 0 then return 0 end
584
585    local endlessList = self:getEndlessProgressConfig() or {}
586    if table.isEmpty(endlessList) then return 0 end
587
588    local count = 0
589    while remainScore > 0 do
590        local progressed = false
591        for _, nodeCfg in ipairs(endlessList) do
592            local needScore = tonumber(nodeCfg.incrementScore) or 0
593            if needScore > 0 then
594                if remainScore >= needScore then
595                    count = count + 1
596                    remainScore = remainScore - needScore
597                    progressed = true
598                else
599                    return count
600                end
601            end
602        end
603        if not progressed then return count end
604    end
605    return count
606end
607
608-- 普通进度的最终累计分数，用于无尽奖励起点换算。
609function Model:_getNormalProgressEndScore()
610    local score = 0
611    for _, nodeCfg in ipairs(self:getNormalProgressConfig() or {}) do
612        score = math.max(score, tonumber(nodeCfg.incrementScore) or 0)
613    end
614    return score
615end

627-- 获取普通进度奖励配置；requiredScore 是后端增量，这里只为现有进度 UI 计算累计展示坐标。
628function Model:getNormalProgressConfig()
629    local ret = {}
630    local accumulatedScore = 0
631    for _, rewardMeta in ipairs((self:getCurrentVersionMeta() or {}).progressRewardMetas or {}) do
632        accumulatedScore = accumulatedScore + (tonumber(rewardMeta.requiredScore) or 0)
633        ret[#ret + 1] = {
634            nodeId = tonumber(rewardMeta.rewardId) or #ret + 1,
635            incrementScore = accumulatedScore,
636            rewardList = table.clone(rewardMeta.rewardItems or {}),
637        }
638    end
639    return ret
640end

643function Model:getEndlessProgressConfig()

646        ret[#ret + 1] = {
647            cycleId = tonumber(rewardMeta.rewardId) or #ret + 1,
648            incrementScore = tonumber(rewardMeta.requiredScore) or 0,
649            rewardList = table.clone(rewardMeta.rewardItems or {}),
650        }
651    end
652    return ret
653end
