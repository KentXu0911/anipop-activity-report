108local function createFlow(onShow)

119test.describe("CollectCard over-process tip lifetime limit", function()
120    test.beforeEach(function()
121        mock.unload(FLOW_MODULE)
122        installDependencies()
123    end)
124
125    test.it("shows once by default and records before the panel closes", function()
126        local runtime = createRuntime()
127        local showCount = 0
128        local closePanel
129        local flow = createFlow(function(_, closeCallback)
130            showCount = showCount + 1
131            closePanel = closeCallback
132        end)
133
134        local promise = flow:showNextRoundPreview()
135
136        assert.equals(showCount, 1)
137        assert.equals(runtime.getPermanentCount(), 1)
138        assert.equals(runtime.getSentCount(), 1)
139        assert.equals(runtime.getRequestedConfigKey(), "overProcessTipPopLimit")
140        assert.falsy(promise.resolved)
141
142        closePanel()
143        assert.truthy(promise.resolved)
144
145        local secondShowCount = 0
146        local secondFlow = createFlow(function()
147            secondShowCount = secondShowCount + 1
148        end)
149        local secondPromise = secondFlow:showNextRoundPreview()
150
151        assert.equals(secondShowCount, 0)
152        assert.equals(runtime.getSentCount(), 1)
153        assert.truthy(secondPromise.resolved)
154    end)
