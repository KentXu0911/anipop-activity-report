175function ShopRewardSingleIconModule:doCreateIconAsync()
176    if not self:_isIconV2Enabled() then
177        if self._iconV2Adapter and not self:_disposeIconV2Adapter() then
178            return
179        end
180        if not self:isIconEnabled() then
181            self:removeIcon()
182            return
183        end
184
185        self:loadRes(self:getIconResMap()):next(function()
186            local icon = self:getIcon()
187            if icon then
188                self:refreshExistingIconByState()
189                return
190            end
191
192            local IconEventCenter = require "amf.feature.mainscene.module.icon.event.IconEventCenter"
193            IconEventCenter:addIconIfNotExist(self.iconKey, function()
194                return self:doCreateSingleIcon()
195            end)
196        end)
197        return
198    end
199
200    if not self:isIconEnabled() then
201        self:removeIcon()
202        return
203    end
204
205    self:loadRes(self:getIconResMap()):next(function()
206        if self._iconV2Adapter then
207            if self:_refreshIconV2("shop_reward_create_refresh") then
208                self:_markV2NewShownIfNeeded()
209                return
210            end
211            if not self:_disposeIconV2Adapter() then return end
212        end
213
214        local icon = self:getIcon()
215        if icon then
216            local IconEventCenter = require "amf.feature.mainscene.module.icon.event.IconEventCenter"
217            IconEventCenter:removeIconEvent(self.iconKey)
218        end
219        self:_createIconV2Adapter()
220    end)
221end
