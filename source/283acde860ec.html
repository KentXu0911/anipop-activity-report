93-- 记录触发结果
94function UBFAddFiveUpgradeBuffUtil:_recordTriggerList(actData, localData, contextData, triggerSucceed)
95	if not actData then actData = UBFAddFiveUpgradeDataUtil:getValidActData() end
96	local levelID = contextData.levelID
97	if not actData or not levelID then return end
98	
99	local triggerRecordList = localData[LocalBuffDataKey.kTriggerRecord]
100	if not triggerRecordList then
101		triggerRecordList = {}
102	else
103		-- 如果当前 levelID 向前推进了，清理老旧触发列表
104		-- List 里的应该都是同一关的触发记录，挑一个检测即可
105		local recordData = triggerRecordList[1]
106		if recordData and recordData.levelID then
107			if levelID > recordData.levelID then
108				triggerRecordList = {}
109			end
110		end
111	end
112
113	-- 太长的话，清理一下
114	if #triggerRecordList > 20 then
115		local trimmedList = {}
116		local lastRecord
117		for _, record in pairs(triggerRecordList) do
118			-- 因为要算CD，所以可以只保留成功触发的
119			if record.triggerFlag then
120				table.insert(trimmedList, record)
121			end
122			if not lastRecord or (record.failTimes > lastRecord.failTimes) then
123				lastRecord = record
124			end
125		end
126		-- 最后一条记录是触发失败的，也保留
127		if lastRecord and not lastRecord.triggerFlag then
128			table.insert(trimmedList, lastRecord)
129		end
130		triggerRecordList = trimmedList
131	end
132
133	local triggerFlag = false
134	if triggerSucceed then triggerFlag = true end
135
136	local recordData = {
137		levelID = levelID,
138		failTimes = actData.failTimes or 0,
139		triggerFlag = triggerFlag,
140		timeStamp = TimeKit:timeInSec(),
141	}
142	table.insert(triggerRecordList, recordData)
143	if logSwitchOpened() then self:log("=== _recordTriggerList:", tts(recordData)) end
144
145	localData[LocalBuffDataKey.kTriggerRecord] = triggerRecordList
146	UBFAddFiveUpgradeBuffUtil:setCustomLocalData(localData)
147end

234-- 检测是否能触发buff
235function UBFAddFiveUpgradeBuffUtil:tryTriggerBuff(contextData)
236	if logSwitchOpened() then self:log("=== --- tryTriggerBuff --- ===") end
237	local actData = UBFAddFiveUpgradeDataUtil:getValidActData()
238	local localData = UBFAddFiveUpgradeBuffUtil:getCustomLocalData() or {}
239	if not actData then return end
240
241	-- 被触发记录限制触发：每关的每个失败次数一次、CD期内限制次数
242	if UBFAddFiveUpgradeBuffUtil:_triggerBannedByTriggerRecord(actData, localData, contextData) then
243		return
244	end
245	-- 被领奖CD限制触发
246	if UBFAddFiveUpgradeBuffUtil:_triggerBannedByRewardCD(actData) then
247		return
248	end

251	local triggerResult = UBFAddFiveUpgradeBuffUtil:_executeTriggerLogic(actData, localData, contextData)
252	if triggerResult then
253		UBFAddFiveUpgradeBuffUtil:_recordTriggerList(actData, localData, contextData, true)

256		UBFAddFiveUpgradeBuffUtil:_initTriggerData(actData, localData, contextData)
257	else
258		UBFAddFiveUpgradeBuffUtil:_recordTriggerList(actData, localData, contextData, false)
259	end
260	if logSwitchOpened() then self:log("=== triggerResult", triggerResult) end
261	return triggerResult

264-- 被触发记录限制触发：每关的每个失败次数一次、CD期内限制次数
265function UBFAddFiveUpgradeBuffUtil:_triggerBannedByTriggerRecord(actData, localData, contextData)
266	if logSwitchOpened() then self:log("+++ check _triggerBannedByTriggerRecord:") end
267	local levelID = contextData.levelID
268	local currFailTimes = actData.failTimes or 0	-- 在onBeforeSendPassLevel和闪退不恢复时已更新数值
269
270	local triggerRecord = localData[LocalBuffDataKey.kTriggerRecord]
271	if triggerRecord and (#triggerRecord > 0) then
272		-- CD：24h最多可触发2次
273		local triggerCount = 0
274		local CDDuration = 24 * 3600
275		local currTime = TimeKit:timeInSec()
276		
277		for _, record in pairs (triggerRecord) do
278			local recordLevel = record.levelID or -1
279			local recordFailTimes = record.failTimes or 0
280			-- 每关的每个失败次数只检测一次触发，无论检测结果是否成功
281			if (recordLevel == levelID) and (recordFailTimes == currFailTimes) then
282				if logSwitchOpened() then self:log("=== xxx === Trigger Banned By:", "Level has trigger data:", tts(record)) end
283				return true
284			else
285				-- 计算CD内成功触发的次数
286				if record.triggerFlag then
287					local recordTs = record.timeStamp
288					if currTime < recordTs + CDDuration then
289						triggerCount = triggerCount + 1
290					end
291				end
292			end
293		end
294
295		if triggerCount >= 2 then
296			if logSwitchOpened() then self:log("=== xxx === Trigger Banned By:", "Trigger-CD", triggerCount, tts(triggerRecord)) end
297			return true
298		end
299	end
300	return false
301end

323function UBFAddFiveUpgradeBuffUtil:_executeTriggerLogic(actData, localData, contextData)
324	if logSwitchOpened() then self:log(">>> _executeTriggerLogic:") end
325
326	local levelID = contextData.levelID
327	local currFailTimes = actData.failTimes or 0	-- 在onBeforeSendPassLevel和闪退不恢复时已更新数值
328	local currPayTag = UserTagManager:getUserPaymentTag2ServerVersion()

331	-- 本关上次成功触发时的失败次数
332	local function getFailTimesOfLastTrigger()
333		local lastFailTimes = 0
334		local triggerRecord = localData[LocalBuffDataKey.kTriggerRecord]
335		if triggerRecord and (#triggerRecord > 0) then
336			for _, record in pairs (triggerRecord) do
337				-- 只考虑成功触发的情况
338				if record.triggerFlag then
339					local recordLevel = record.levelID or -1
340					if recordLevel == levelID then
341						local failTimes = record.failTimes or 0
342						if failTimes > lastFailTimes then
343							lastFailTimes = failTimes
344						end
345					end
346				end
347			end
348		end
349		-- 因为失败次数是进关时的，是滞后的，当次打关若失败，数值上需要 +1 才是那次的实际失败次数
350		if lastFailTimes > 0 then
351			lastFailTimes = lastFailTimes + 1
352		end
353		return lastFailTimes
354	end

356	local function getCurrTriggerPercentage()
357		if currPayTag >= UsetPaymentTagServer.kNone and currPayTag <= UsetPaymentTagServer.kHigh_4 then
358			if logSwitchOpened() then self:log("=== getCurrTriggerPercentage, payment group 1:") end
359
360			-- 新分组（App: buff / 小程序: ONstepV2）使用新数值 startTriggerFailNum=3
361			local useNewBuffValue = false
362			if EnvUtil:isMiniGame() then

364			else

366			end

368			local triggerGap = 5 			-- 将来走配置
369			local lastTriggerFailNum = getFailTimesOfLastTrigger()
370			local gapNum = triggerGap
371			if lastTriggerFailNum == 0 then
372				gapNum = startTriggerFailNum
373			end
374			local triggerN = currFailTimes - lastTriggerFailNum - gapNum
375			local resultPercentage = math.max(20 + triggerN * 20, 0)
376			if logSwitchOpened() then self:log("=== lastTriggerFailNum, triggerGap, triggerN, resultPercentage:", startTriggerFailNum, triggerGap, lastTriggerFailNum, triggerN, resultPercentage) end
377			return resultPercentage
378
379		elseif currPayTag >= UsetPaymentTagServer.kHigh_5 and currPayTag <= UsetPaymentTagServer.kHigh_6 then
380			if logSwitchOpened() then self:log("=== getCurrTriggerPercentage, payment group 2:") end
381			-- 关卡难度限制：难关/超难关
382			local MetaManager = _G.amfBean("MetaManager")
383			local difficultyFlag = (MetaManager and MetaManager:getLevelDifficultFlag_ForStartPanel(levelID)) or LevelDiffcultFlag.kNormal
384			if not (difficultyFlag == LevelDiffcultFlag.kDiffcult or difficultyFlag == LevelDiffcultFlag.kExceedinglyDifficult) then
385				if logSwitchOpened() then self:log("=== xxx === Trigger Failed By:", "Difficulty", difficultyFlag) end
386				return 0
387			end
388
389			-- 每关只能成功触发一次
390			local triggerRecord = localData[LocalBuffDataKey.kTriggerRecord]
391			if triggerRecord and (#triggerRecord > 0) then
392				for _, record in pairs (triggerRecord) do
393					-- 只限制成功触发的情况
394					if record.triggerFlag then
395						local recordLevel = record.levelID or -1
396						if recordLevel == levelID then
397							if logSwitchOpened() then self:log("=== xxx === Trigger Failed By:", "Level Triggered", levelID) end
398							return 0
399						end
400					end
401				end
402			end
403
404			if (currFailTimes < 1) or (currFailTimes > 2) then
405				if logSwitchOpened() then self:log("=== xxx === Trigger Failed By:", "currFailTimes", currFailTimes) end
406				return 0
407			end
408
409			return 50
410		end
411	end
412	local triggerPercentage = getCurrTriggerPercentage()
413	if logSwitchOpened() then self:log("curr triggerPercentage", triggerPercentage) end
414	
415	if triggerPercentage and triggerPercentage > 0 then
416		local randVal = math.random(0, 100)
417		if logSwitchOpened() then self:log("randVal, triggerPercentage", randVal, triggerPercentage) end
418		if randVal <= triggerPercentage then
419			if logSwitchOpened() then self:log("=== trigger SUCCEED !") end
420			return true
421		end
422		if logSwitchOpened() then self:log("=== trigger FAILED...") end
423		return false
424	end
425	if logSwitchOpened() then self:log("=== trigger FAILED...") end
426	return false
427end

444	local isInfiniteEnergy = UserEnergyManager and (UserEnergyManager:getEnergyState() == UserEnergyState.INFINITE)

448	local function getRewardsFromStr(rewardsStr)
449		local rewardStrList = string.split(rewardsStr, ',')
450		local rewards = {}
451		for i,v in ipairs(rewardStrList) do
452			local rewardStr = string.split(v, ':')
453			local reward = {itemId = tonumber(rewardStr[1]), num = tonumber(rewardStr[2])}
454			table.insert(rewards, reward)
455		end
456		return rewards
457	end

459	local function getRewardsAndGroupTypeByConfig(energyConfig, fullEnergyGroup)
460		local rewardGroupType = RewardGroupType.kNone
461		local rewards = {}
462
463		local index
464		for i, v in ipairs(energyConfig) do
465			if table.exist(v.tags, currPayTag) then 
466				index = i
467				rewards = table.clone(getRewardsFromStr(v.rewards))
468				break
469			end
470		end
471
472		if fullEnergyGroup then
473			if index == 1 then
474				rewardGroupType = RewardGroupType.kGroupType2
475			elseif index == 2 then
476				rewardGroupType = RewardGroupType.kGroupType4

478		else
479			if index == 1 then
480				rewardGroupType = RewardGroupType.kGroupType2
481			elseif index == 2 then
482				rewardGroupType = RewardGroupType.kGroupType3
483			elseif index == 3 then
484				rewardGroupType = RewardGroupType.kGroupType5
485			end
486		end
487
488		return rewards, rewardGroupType
489	end
490
491	local rewards = {}
492	local rewardGroupType = RewardGroupType.kNone
493	if energyNum > 5 or isInfiniteEnergy then
494		local fullEnergyRewardsByTagsConfig = actData and actData.fullEnergyRewardsByTagsConfig or {}
495		rewards, rewardGroupType = getRewardsAndGroupTypeByConfig(fullEnergyRewardsByTagsConfig, true)
496	else
497		local notFullEnergyRewardsByTagsConfig = actData and actData.notFullEnergyRewardsByTagsConfig or {}
498		rewards, rewardGroupType = getRewardsAndGroupTypeByConfig(notFullEnergyRewardsByTagsConfig, false)
499
500	end
501	if logSwitchOpened() then self:log("rewardGroupType, reward", rewardGroupType, tts(rewards)) end
502
503	-- 确实随机到了buff内容
504	if rewardGroupType ~= RewardGroupType.kNone then
505		-- 关卡外面板要用的参数包
506		local panelParams = { 
507			levelId = contextData.levelID, 
508			failNum = actData.failTimes or 0, 
509			curPayTag = currPayTag, 
510			rewards = rewards, 
511			rewardGroupType = rewardGroupType,
512			energyNum = energyNum,
513			isInfiniteEnergy = isInfiniteEnergy,
514			playId = contextData.playID or '',
515		}
516		-- 当前触发的buff数据
517		local currBuffData = {
518			levelID = contextData.levelID, 
519			failTimes = actData.failTimes or 0, 
520			playID = contextData.playID or '',
521
522			panelParams = panelParams,
523		}
524		UBFAddFiveUpgradeBuffUtil:setCustomLocalData(currBuffData, LocalBuffDataKey.kCurrTriggeredBuffData)
525
526		return true
527	end

564	end
565	-- 跳关skip
566	local hasSkippedLevel = _G.amfBean("LevelAreaManager") and _G.amfBean("LevelAreaManager"):hasPassedByTrick(levelID)
567	if hasSkippedLevel then
568		if logSwitchOpened() then self:log("--- skip!", "jump level") end
569		onEnd()
570        return
571	end
572	-- 刚开新关一段时间内，打新关不弹
573	if UBFAddFiveUpgradeBuffUtil:_forbidPopForTopLevelPlayer() then
574		if logSwitchOpened() then self:log("--- skip!", "forbid pop for top level player") end
575		onEnd()
576        return
577	end
578
579	-- 没有长尾活动数据skip
580	if not modContext then
581		if logSwitchOpened() then self:log("--- skip!", "no modContext data") end
582		onEnd()
583        return
584	end
585	-- 当前不是长尾目标关卡(在初始化 modContext 时就已经判断了 actData 随最高关的重置)
586	if not modContext.levelID or (levelID ~= modContext.levelID) then
587		if logSwitchOpened() then self:log("--- skip!", "not target level") end
588		onEnd()
589        return
590	end
591
592	local currFailTimes = -1
593	if modContext.failTimes then
594		currFailTimes = modContext.failTimes
595	end
596
597	---- 当前已经有触发好的，判断是否直接显示结果面板（成功、失败）
598	-- 本地的当前的触发数据记录
599	local currBuffData = UBFAddFiveUpgradeBuffUtil:getCustomLocalData(LocalBuffDataKey.kCurrTriggeredBuffData)
600	if currBuffData then
601		if logSwitchOpened() then self:log(">=> CurrTriggeredBuffData:", tts(currBuffData)) end
602		local triggeredLevelID = currBuffData.levelID
603		local triggeredFailTimes = currBuffData.failTimes
604		local triggeredPlayID = currBuffData.playID
605
606		-- 首先，要是同一关……
607		if triggeredLevelID == levelID then
608			-- 失败重玩面板，尝试展示获得buff失败面板
609			if byFailPanel then
610				if logSwitchOpened() then self:log("=== check pop fail panel") end
611				-- 领奖有效性
612				if UBFAddFiveUpgradeBuffUtil:isEnabled2() then 
613					if logSwitchOpened() then self:log("=== enabled for get reward", prePlayID, triggeredPlayID, currBuffData.result) end
614					-- 是同一关，并且有打关结果，记录为失败，弹buff失败面板
615					if (prePlayID == triggeredPlayID) and currBuffData.result and (currBuffData.result == 0) then
616						if logSwitchOpened() then self:log("=== can pop fail panel") end
617						if outsideManagerData and outsideManagerData.popoutBuffFailPanel and currBuffData.panelParams then
618							if logSwitchOpened() then self:log("=== popoutBuffFailPanel !!") end
619							safeRunCall(outsideManagerData.popoutBuffFailPanel({
620								panelParam = currBuffData.panelParams,
621								endCallback = onEnd,
622							}))
623
624							-- 清除触发数据
625							UBFAddFiveUpgradeBuffUtil:_clearCurrTriggerData()
626							return
627						end
628					end
629				end
630			else
631				-- 非失败面板，尝试展示已触发的buff
632				if logSwitchOpened() then self:log("=== check pop old trigger panel") end
633				-- 活动有效性
634				if UBFAddFiveUpgradeBuffUtil:isEnabled() then 
635					if logSwitchOpened() then self:log("=== act enabled", triggeredFailTimes, currFailTimes, currBuffData.result) end
636					-- 是同一关，失败次数一致，没有打关结果，可以显示老的触发内容
637					if (triggeredFailTimes == currFailTimes) and (not currBuffData.result) then
638						if logSwitchOpened() then self:log("=== can pop old trigger panel") end
639						if outsideManagerData and outsideManagerData.popoutTriggerBuffPanel and currBuffData.panelParams then
640							if logSwitchOpened() then self:log("=== popoutTriggerBuffPanel !!") end
641							safeRunCall(outsideManagerData.popoutTriggerBuffPanel({
642								panelParam = currBuffData.panelParams,
643								endCallback = onEnd,
644							}))
645
646							-- playID可能是新生成的，更新一下本地数据中的playID
647							currBuffData["playID"] = playID
648							if currBuffData.panelParams then
649								currBuffData.panelParams["playId"] = playID
650							end
651							UBFAddFiveUpgradeBuffUtil:_updateCurrTriggerData(currBuffData)
652							return
653						end
654					end
655				end
656			end
657		else
658			if logSwitchOpened() then self:log("not the same level...") end
659		end
660	end
661
662	---- 没有触发好的，检测触发
663	-- 活动有效性
664	if not UBFAddFiveUpgradeBuffUtil:isEnabled() then 
665		if logSwitchOpened() then self:log("--- skip!", "act not enabled") end
666		onEnd()
667		return 
668	end
669
670	local checkTriggerContext = {
671		levelID = levelID,
672		playID = playID,
673	}
674	local triggerResult = UBFAddFiveUpgradeBuffUtil:tryTriggerBuff(checkTriggerContext)
675	if triggerResult then
676		-- 触发成功了！
677		local currBuffData = UBFAddFiveUpgradeBuffUtil:getCustomLocalData(LocalBuffDataKey.kCurrTriggeredBuffData)
678		if currBuffData and currBuffData.panelParams and outsideManagerData and outsideManagerData.popoutTriggerBuffPanel then
679			if logSwitchOpened() then self:log("=== popoutTriggerBuffPanel, new triggered buff !!") end
680			safeRunCall(outsideManagerData.popoutTriggerBuffPanel({
681				panelParam = currBuffData.panelParams,
682				endCallback = onEnd,
683			}))
684			return
685		end
686	end
687
688	onEnd()
689end

692function UBFAddFiveUpgradeBuffUtil:_forbidPopForTopLevelPlayer()
693	if logSwitchOpened() then self:log("+++ check _forbidPopForTopLevelPlayer:") end
694
695	-- 满星玩家不管
696    local userState = _G.amfBean("UserStateManager"):getUserSegmentState()
697    if userState == _G.UserStateConst.UserLevelState.MHZC then
698		if logSwitchOpened() then self:log("--- nevermind:", "UserLevelState.MHZC") end
699        return false
700    end
701	-- 不是新关区域不管
702    local topLevelId = _G.getTopLevelId()
703    if not _G.amfBean("LevelUnLockManager"):isTopNewMainLevel(topLevelId) then 
704		if logSwitchOpened() then self:log("--- nevermind:", "not top new MainLevel area") end
705        return false
706    end
707	-- 不是刚开新关的时间不管
708    if not _G.amfBean("LevelUnLockManager"):isInNewLevelTime() then 
709		if logSwitchOpened() then self:log("--- nevermind:", "not isInNewLevelTime") end
710        return false
711    end
712    
713	if logSwitchOpened() then self:log("xxx banned by:", "forbidPopForTopLevelPlayer") end
714    return true
715end
