667				end
668			)
669		else
670			if localLevelData.activityDays[#localLevelData.activityDays] < nowDayIndex then
671				table.insert( localLevelData.activityDays , nowDayIndex )
672			end
673		end
674		
675	end
676
677	if #localLevelData.activityDays > 30 then
678		table.remove( localLevelData.activityDays  , 1 )
679	end
680
681end

727--获取用户付费标签
728function UserTagManager:getUserPaymentTag( ... )
729	local userPayAmount = self:getLast60DayPayAmount() or 0
730	if userPayAmount >= 240 then
731		return UserPaymentTag.kHigh
732	elseif userPayAmount >= 47 then
733		return UserPaymentTag.kMidHigh
734	elseif userPayAmount >= 13 then
735		return UserPaymentTag.kMidLow
736	elseif userPayAmount > 0 then
737		return UserPaymentTag.kLow
738	else
739		return UserPaymentTag.kNone
740	end
741end

743local function _getPaymentTagByAmount(amount)
744	amount = amount or 0
745	if amount >= 3000 then
746		return UserPaymentTag.kHigh6
747	elseif amount >= 1000 then
748		return UserPaymentTag.kHigh5
749	elseif amount >= 240 then
750		return UserPaymentTag.kHigh4
751	elseif amount >= 47 then
752		return UserPaymentTag.kMidHigh
753	elseif amount >= 13 then
754		return UserPaymentTag.kMidLow
755	elseif amount > 0 then
756		return UserPaymentTag.kLow
757	else
758		return UserPaymentTag.kNone
759	end
760end
