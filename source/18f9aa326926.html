66--这里的状态 是本地的满级满星状态 （有没有更新到新关是不同的）
67function UserStateManager:updateUserState()
68    local oldUserState = self.userLevelState
69	self.userLevelState, self.mainStarFull, self.hideStarFull, self.colorfulStarFull = self:getUserLevelInfo()
70
71    if oldUserState ~= _G.UserStateConst.UserLevelState.NONE and oldUserState ~= self.userLevelState then
72        _G.amfEC:notifyFeature(FeatureEvent.USER_LEVEL_STATE_CHANGED, {
73            oldUserState = oldUserState,
74            newState = self.userLevelState,
75            segmentState = self:getUserSegmentState(),
76        })
77    end
78end

80function UserStateManager:getUserLevelInfo(specifiedFullStar, specifiedFullLevel)
81	local userState = _G.UserStateConst.UserLevelState.NONE
82    local _levelUnLockManager = _G.amfBean("LevelUnLockManager")
83    local configTopLevel = _levelUnLockManager:getLocalTopLevel(true)
84    local totalHiddenStar = _levelUnLockManager:getFullStarInHiddenRegion(true)
85    local totalTopStar = LevelConfigBridge:getTotalStar(configTopLevel)
86
87    local userStarMain = self._userManager:getStar()
88
89    local mainStarFull = false
90    local hideStarFull = false
91    local colorfulStarFull = false
92    if userStarMain >= totalTopStar then
93        mainStarFull = true
94    end
95
96    local userStarHide = self._userManager:getHiddenStar()
97    if userStarHide >= totalHiddenStar then
98        hideStarFull = true
99    end
100
101    local userColorfulStar = self._userManager:getColorfulStar()
102    local serverTopLevel = specifiedFullLevel or configTopLevel
103    local totalColorfulStar = LevelConfigBridge:getTotalMainColorStar(serverTopLevel)
104
105    if userColorfulStar >= totalColorfulStar then
106        colorfulStarFull = true
107    end
108
109    local topPassedLevelId = self._levelAreaManager:getTopPassedMainLevelId()
110    local configFullStar = specifiedFullStar or (totalTopStar + totalHiddenStar)
111    local userFullStar = userStarMain + userStarHide
112
113    if topPassedLevelId >= serverTopLevel then --满级
114        if userFullStar >= configFullStar and colorfulStarFull then
115	        userState = _G.UserStateConst.UserLevelState.TOP_STAR
116        else
117            userState = _G.UserStateConst.UserLevelState.TOP_LEVEL
118        end
119    else
120    	userState = _G.UserStateConst.UserLevelState.NORMAL
121    end
122
123    return userState, mainStarFull, hideStarFull, colorfulStarFull
124end

138-- 细分状态，关卡外一般用这个
139function UserStateManager:getUserSegmentState()
140    self:tryInit()
141    local state
142    if self:isFMJ() then
143        state = _G.UserStateConst.UserSegmentState.FMJ
144    elseif not self:isMainStarFull() then
145        state = _G.UserStateConst.UserSegmentState.FMH
146    elseif not self:isColorfulStarFull() then
147        state = _G.UserStateConst.UserSegmentState.FMC
148    elseif not self:isHideStarFull() then
149        state = _G.UserStateConst.UserSegmentState.FMZ
150    else
151        state = _G.UserStateConst.UserSegmentState.MHZC
152    end
153	return state
154end

180function UserStateManager:getGoToLevelStr(mainStr, fullLevelStr)
181    local mainLevelStr
182    local userState = self:getUserSegmentState()
183    if userState == _G.UserStateConst.UserSegmentState.FMJ then
184        mainLevelStr = mainStr or "去主线关"
185    elseif userState == _G.UserStateConst.UserSegmentState.MHZC then
186        if self._fullStarLevelLibManager:canPlay() then
187            mainLevelStr = "去闯关"
188        else
189            mainLevelStr = "去刷分"
190        end
191    else
192        if self._fullStarLevelLibManager:canPlay() then
193            mainLevelStr = "去闯关"
194        else
195            mainLevelStr = fullLevelStr or "去补星"
196        end
197    end
198    return mainLevelStr
199end
