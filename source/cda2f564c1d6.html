1-- 过关后炫耀（区域满星炫耀、全满星炫耀）

3-- a = 单区域强弹 (OneAreaFullStarPanel)

5-- c = 星座强弹 (ConstellationFullStarPanel) - 100个区域/1500关

8-- 强弹优先级逻辑：

31-- 统一的星光之域强弹处理入口
32function Module:onHandleStellarConvergenceShowOff(evt)
33    local data = evt:getData()
34    local passedLevelId = data.passedLevelId
35    local levelScene = data.levelScene
36    local oldStar = data.oldStar
37    local newStar = data.star
38    local areaId = _G.amfBean("LevelAreaManager"):getAreaIdByLevelId(passedLevelId)
39
40    local callback = function()
41        evt:redeliver()
42    end
43
44    -- 新关期间，屏蔽强弹
45    local ignore = _G.amfBean("AutoPopupMgr"):ignoreForTopLevelPlayer(passedLevelId, levelScene)
46    if ignore then
47        safeRunCall(callback)
48        return
49    end
50
51    -- 非主线关卡不处理
52    if not UBFLevelSceneFunc:isMainLevel(levelScene) then
53        safeRunCall(callback)
54        return
55    end
56
57    local StellarConvergenceManager = _G.amfBean("StellarConvergenceManager")
58
59    -- 如果星光之域未启用或资源未准备好，走旧逻辑
60    if not StellarConvergenceManager or not StellarConvergenceManager:isDataEnabled() or not StellarConvergenceManager:isNormalResReady() then
61        self:runLegacyShowOffLogic(data, callback)
62        return
63    end
64
65    -- 使用新的统一强弹逻辑
66    self:runUnifiedShowOffLogic(data, callback)
67end

69-- 新的统一强弹逻辑
70function Module:runUnifiedShowOffLogic(data, finalCallback)
71    local passedLevelId = data.passedLevelId
72    local levelScene = data.levelScene
73    local oldStar = data.oldStar
74    local newStar = data.star
75    local areaId = _G.amfBean("LevelAreaManager"):getAreaIdByLevelId(passedLevelId)
76
77    if areaId > 40000 then
78        areaId = areaId - 40000
79    end
80
81    local StellarConvergenceManager = _G.amfBean("StellarConvergenceManager")
82
83    -- 检查各类型面板的触发条件
84    local checkResults = self:checkAllShowOffConditions(data, areaId, oldStar, newStar)
85
86    local hasShowOff = (checkResults.a and checkResults.a.canShow)
87        or (checkResults.b and checkResults.b.canShow)
88        or (checkResults.c and checkResults.c.canShow)
89        or (checkResults.e and checkResults.e.canShow)
90    if hasShowOff then
91        _G.amfBean("AutoPopupMgr"):blockPassLevelSkip()
92    end
93
94    -- 记录本次流程中已强弹的面板类型
95    local shownPanels = {}
96
97    -- 最终回调：在所有新面板逻辑之后，调用旧的全满星检查
98    local afterAllPanelsCallback = function()
99        self:tryOpenFullStarShowOffPanel(finalCallback)
100    end
101
102    -- 步骤1: 首先检查a（单区域）是否有可领取的宝箱 → 弹 a
103    local step1Callback = function()
104        -- 步骤2: 按 e>c>b>a 选最高的弹一个
105        local priorityOrder = {"e", "c", "b", "a"}
106        if not StellarConvergenceManager:isFullLevel() then
107            priorityOrder = {"e", "b", "a"}
108        end
109
110        local selectedPanel = nil
111        for _, panelType in ipairs(priorityOrder) do
112            if not shownPanels[panelType] and checkResults[panelType] and checkResults[panelType].canShow then
113                selectedPanel = {type = panelType, data = checkResults[panelType]}
114                break
115            end
116        end
117
118        if selectedPanel then
119            self:showPanelByType(selectedPanel.type, selectedPanel.data, data, afterAllPanelsCallback)
120        else
121            afterAllPanelsCallback()
122        end
123    end
124
125    if checkResults.a and checkResults.a.canShow and checkResults.a.hasReward then
126        shownPanels["a"] = true
127        self:showAreaFullStarPanel(checkResults.a, data, step1Callback)
128    else
129        step1Callback()
130    end
131end

134function Module:showPanelByType(panelType, panelData, passLevelData, callback)
135    if panelType == "a" then
136        self:showAreaFullStarPanel(panelData, passLevelData, callback)
137    elseif panelType == "b" then
138        self:showSubsectionFullStarPanel(panelData, callback)
139    elseif panelType == "c" then
140        self:showConstellationFullStarPanel(panelData, callback)

144        safeRunCall(callback)
145    end
146end

148-- 检查所有强弹条件
149function Module:checkAllShowOffConditions(data, areaId, oldStar, newStar)
150    local results = {}
151    local StellarConvergenceManager = _G.amfBean("StellarConvergenceManager")
152
153    -- a: 单区域满星检查
154    results.a = self:checkAreaFullStar(areaId, oldStar, newStar)
155
156    -- b: 小节满星检查（10个区域/150关）
157    results.b = self:checkSubsectionFullStar(areaId, oldStar, newStar)
158
159    -- c: 星座满星检查（100个区域/1500关）
160    results.c = self:checkConstellationFullStar(areaId, oldStar, newStar)
161
162    -- e: 星级跃迁检查
163    results.e = self:checkStarLevelChange(data)
164
165    return results
166end

168-- a: 检查单区域满星
169function Module:checkAreaFullStar(areaId, oldStar, newStar)
170    local result = {canShow = false, hasReward = false, isFullColor = false, areaId = areaId}
171
172    local yellowNum, starNum = Meta:getYellowStarAndColorStarNum(areaId)
173
174    -- 检查是否满星
175    if (starNum == 45 and oldStar < 4) or (yellowNum == 45 and oldStar < 3) then
176        result.canShow = true
177        result.isFullColor = starNum == 45
178
179        -- 检查是否有可领取的宝箱
180        local StellarConvergenceManager = _G.amfBean("StellarConvergenceManager")
181        if StellarConvergenceManager then
182            local hasGotReward = StellarConvergenceManager:checkHasGotReward(areaId)
183            local canOpenBoxToday = StellarConvergenceManager:getModel():checkCanOpenBoxToday()
184
185            -- 有可领取的宝箱：未领过且今日次数未满
186            result.hasReward = not hasGotReward and canOpenBoxToday
187        end
188
189        -- 如果仅满黄星（未满彩星）且没有奖励可领取，则不弹窗
190        if not result.isFullColor and not result.hasReward then
191            result.canShow = false
192        end
193        local ratingMgr = _G.amfBean("OhosRatingManager")
194        if ratingMgr then
195            ratingMgr:onAreaFullStar()
196        end
197
198    end
199
200    return result
201end

203-- b: 检查小节满星（10个区域/150关）
204function Module:checkSubsectionFullStar(areaId, oldStar, newStar)
205    local result = {canShow = false, isFullColor = false, subsectionId = 0}
206    local StellarConvergenceManager = _G.amfBean("StellarConvergenceManager")
207
208    if not StellarConvergenceManager then return result end
209
210    -- 旧星级更高或者新的小于3都不触发
211    if (oldStar >= newStar) or (newStar < 3) then
212        return result
213    end
214
215    local ONE_SUBSECTION_AREA_NUM = StellarConvergenceManager:getConst().ONE_SUBSECTION_AREA_NUM
216    local AREA_LEVEL_NUM = StellarConvergenceManager:getConst().AREA_LEVEL_NUM
217
218    local subsectionId = math.ceil(areaId / ONE_SUBSECTION_AREA_NUM)
219    local areaStart = (subsectionId - 1) * ONE_SUBSECTION_AREA_NUM + 1
220    local areaEnd = subsectionId * ONE_SUBSECTION_AREA_NUM
221
222    local maxLevel = _G.amfBean("LevelUnLockManager"):getCanPlayTopLevel()
223    local maxArea = maxLevel / AREA_LEVEL_NUM
224
225    -- areaEnd大于目前最高区域，不触发
226    if areaEnd > maxArea then
227        return result
228    end
229
230    local allYellow = true
231    local allColor = true
232
233    for i = areaStart, areaEnd do
234        local yellowStar, colorStar = _G.amfBean("LevelAreaManager"):getYellowStarAndColorStarNum(i)
235
236        if yellowStar < AREA_LEVEL_NUM * 3 then
237            allYellow = false
238            allColor = false
239            break
240        end
241
242        if colorStar < AREA_LEVEL_NUM * 3 then
243            allColor = false
244        end
245    end
246
247    if allColor and oldStar < 4 then
248        result.canShow = true
249        result.isFullColor = true
250        result.subsectionId = subsectionId
251    elseif allYellow and oldStar < 3 then
252        result.canShow = true
253        result.isFullColor = false
254        result.subsectionId = subsectionId
255    end
256
257    return result
258end

315-- e: 检查星级跃迁
316function Module:checkStarLevelChange(data)
317    local result = {canShow = false}
318    local StellarConvergenceManager = _G.amfBean("StellarConvergenceManager")
319
320    if not StellarConvergenceManager then
321        return result
322    end
323
324    if not StellarConvergenceManager:isFullLevelResReady() then
325        return result
326    end
327
328    local lastEndTime = StellarConvergenceManager:getLastEndTimeSec()
329    local curTime = TimeKit:timeInSec()
330
331    if not StellarConvergenceManager:isFullLevel() then
332        return result
333    end
334
335    if lastEndTime <= 0 or curTime > lastEndTime then
336        return result
337    end
338
339    -- 获取实际当前等级（过关后）
340    local localCurLevel = StellarConvergenceManager:calCurRankType()
341    -- 获取缓存的等级（进入游戏时已经校验过，是过关前的真实等级）
342    local curRankType = StellarConvergenceManager:getCurRankType()
343
344    -- 检查是否已经弹出过这个等级的跃迁面板
345    local hasPopped = StellarConvergenceManager:getPopChangeType(localCurLevel)
346
347    if curRankType < 1 then
348        return result
349    end
350
351    -- 如果当前等级已经弹出过，不再弹
352    if hasPopped then
353        return result
354    end
355
356    -- curRankType < localCurLevel 说明升级了
357    if curRankType < localCurLevel then
358        result.canShow = true
359        result.oldRankType = curRankType
360        result.newRankType = localCurLevel
361        return result
362    end
363
364    return result
365end

464-- 全满星炫耀
465function Module:tryOpenFullStarShowOffPanel(callback)
466    local totalStar = _G.amfBean("UserManager"):getTotalStar()
467    local fullStarNum = _G.amfBean("LevelAreaManager"):getFullStarNum()
468    if totalStar < fullStarNum then
469        safeRunCall(callback)
470        return
471    end
472
473    local cacheKey = "fullStarShowOffed_"..fullStarNum
474    if self.mgr:getModel():getLocalCache(cacheKey) then
475        safeRunCall(callback)
476        return
477    end
478
479    local RequestOpNotifyOnline = require "amf.middleware.interconnection.remote.http.RequestOpNotifyOnline"
480    local OpNotifyType = RequestOpNotifyOnline.OpType
481
482    local function onSuccess(response)
483        self.mgr:getModel():setLocalCache(cacheKey, true)
484        local data = response:getData()
485        if data then
486            if _G.amfBean("HonorRankManager") and _G.amfBean("HonorRankManager"):checkAudit() and not data.audit then
487                safeRunCall(callback)
488                return
489            end
490
491            local fields = string.split(data.extra or '', ',') or {}
492            local rank, ts, count = tonumber(fields[1]), tonumber(fields[2]), tonumber(fields[3])
493            if rank and rank > 0 and ts then
494                self:openFullStarShowOffPanel({
495                    fullstar_rank = tonumber(fields[1]),
496                    fullstar_ts = tonumber(fields[2]),
497                    fullstar_count = tonumber(fields[3]),
498                }, callback)
499            else
500                safeRunCall(callback)
501            end
502        else
503            safeRunCall(callback)
504        end
505    end
506    local function onFail(response)
507        safeRunCall(callback)
508    end
509
510    RequestOpNotifyOnline:post(OpNotifyType.kFullStarCheck, nil, nil, true):next(onSuccess, onFail)
511end
