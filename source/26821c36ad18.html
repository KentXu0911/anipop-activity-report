56function Module:getProgressKey(taskInfo, hasReward)

68function Module:buildTargetState()
69    local model = self.mgr and self.mgr:getModel()
70    local taskInfo = model and model:getCurrentTask()
71    local isCompleted = false
72    if not taskInfo and model:isAllTasksFinished() then
73        local taskList = model:getTaskList()
74        taskInfo = taskList and taskList[#taskList]
75        isCompleted = true
76    end
77    if not taskInfo then return nil end
78
79    local hasReward = self.mgr:canSupplementReward()
80    local progressKey = self:getProgressKey(taskInfo, hasReward)
81    if progressKey == self.lastProgressKey then return nil end
82
83    if not self.lastTaskIndex and isCompleted and not hasReward then return nil end
84
85    local taskIndex = taskInfo.taskId
86    local kind
87    if not self.lastTaskIndex then
88        kind = "init"
89    elseif taskIndex ~= self.lastTaskIndex then
90        kind = "switch"
91    else
92        kind = "step"
93    end
94
95    return {
96        kind = kind,
97        taskIndex = taskIndex,
98        requiredWinCount = taskInfo.needStreakWin or 0,
99        curStepInTask = math.min(taskInfo.progress or 0, taskInfo.needStreakWin or 0),
100        boxTier = self.mgr:getBoxTier(taskInfo),
101        taskType = self.mgr:getTaskType(),
102        progressKey = progressKey,
103        isCompleted = isCompleted,
104        hasReward = hasReward,
105    }
106end

165function Module:refreshPassExtraModule(showOwnExtra)

216function Module:applyTargetState(target)
217    if not self:ensureExtraUI() then return end
218
219    self:refreshStarWandIconPos(target.requiredWinCount)
220
221    local token = self:startTransition()
222
223    if target.kind == "init" then
224        self.extraUI:update(target)
225        local skipAnim = self.mgr:hasInitAnimPlayedToday()
226        self.mgr:setInitAnimPlayedToday(true)
227        self.extraUI:playInitAnim(skipAnim):next(function()
228            if self.animToken ~= token then return end
229            return self.extraUI:playBubbleIn(skipAnim)
230        end):next(function()
231            if self.animToken ~= token then return end
232            self:finishTransition(target)
233            self:refreshPassExtraModule(true)
234        end)
235    elseif target.kind == "step" then
236        local lastStepInTask = self.lastStepInTask or 0
237        if target.curStepInTask > lastStepInTask then
238            self.extraUI:update(target, target.curStepInTask - 1)
239            self.extraUI:playStepAnim(target.curStepInTask):next(function()
240                if self.animToken ~= token then return end
241                self:finishTransition(target)
242            end)
243        elseif target.curStepInTask < lastStepInTask then
244            self.extraUI:update(target, lastStepInTask)
245            self.extraUI:playRollbackAnim(lastStepInTask, target.curStepInTask):next(function()
246                if self.animToken ~= token then return end
247                self:finishTransition(target)
248            end)
249        else
250            self.extraUI:update(target)
251            self:finishTransition(target)
252        end
253    elseif target.kind == "switch" then
254        self.extraUI:playStepAnim(self.lastRequiredWinCount, true):next(function()
255            if self.animToken ~= token then return end
256            return self.extraUI:playBubbleOut("vfx")
257        end):next(function()
258            if self.animToken ~= token then return end
259            self.extraUI:update(target)
260            return self.extraUI:playBubbleIn()
261        end):next(function()
262            if self.animToken ~= token then return end
263            self:finishTransition(target)
264        end)
265    end
266end
