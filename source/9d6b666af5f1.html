6local function isMiniGame()
7    if not EnvUtil or type(EnvUtil.isMiniGame) ~= "function" then return false end
8    local ok, result = pcall(EnvUtil.isMiniGame, EnvUtil)
9    return ok and result == true
10end

12local function currentWeekend()
13    if type(isWeekend) ~= "function" or not TimeKit or type(TimeKit.timeInSec) ~= "function" then
14        return false
15    end
16    local ok, result = pcall(isWeekend, TimeKit:timeInSec())
17    return ok and result == true
18end
19
20local function buildAssetVariants()
21    if isMiniGame() then
22        return {
23            weekday = { type = "spriteFile", path = Res.Texture.ICON_WEEKDAY_MINIGAME },
24            weekend = { type = "spriteFile", path = Res.Texture.ICON_WEEKEND_MINIGAME },
25        }
26    end
27    return {
28        weekday = { type = "spine", path = Res.Spine.ICON_WEEKDAY_V2 },
29        weekend = { type = "spine", path = Res.Spine.ICON_WEEKEND_V2, scale = 0.9 },
30    }
31end
