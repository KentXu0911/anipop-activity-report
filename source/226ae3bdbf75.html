18local function deepCopyTable(source, seen)
19    seen = seen or {}
20    if seen[source] then
21        return seen[source]
22    end
23    local copied = {}
24    seen[source] = copied
25    for key, value in pairs(source) do
26        copied[key] = type(value) == "table" and deepCopyTable(value, seen) or value
27    end
28    return copied
29end

31local function copyAsset(asset)
32    if type(asset) ~= "table" or next(asset) == nil then
33        return nil
34    end
35    return deepCopyTable(asset)
36end

38local function resolveMiniGameNotificationAsset(config, iconBtn)
39    local iconConfig = type(config) == "table" and config.icon or nil
40    if type(iconConfig) ~= "table" then
41        return nil
42    end
43
44    local asset = copyAsset(iconConfig.miniGameNotificationAsset)
45        or copyAsset(iconConfig.miniGameAsset)
46        or copyAsset(iconBtn and iconBtn.assetConfig)
47    if asset then
48        return asset
49    end
50
51    local assets = iconConfig.miniGameAssets
52    local defaultAssetKey = iconConfig.defaultAssetKey
53    return copyAsset(type(assets) == "table" and assets[defaultAssetKey] or nil)
54end

64local function callCleanupMethod(iconBtn, methodName)
65    if type(iconBtn[methodName]) ~= "function" then
66        return false, "missing " .. methodName
67    end
68
69    local success, result, detail = pcall(iconBtn[methodName], iconBtn)
70    if not success then
71        return false, result
72    end
73    if result == false then
74        return false, detail or (methodName .. " returned false")
75    end
76    return true

109function IconProcessor:process(actInfo)
110    if not self._iconSystemReady then return end
111    if self:checkNoIcon() then return end
112
113    local ownerReleaseCoordinator = self._ownerReleaseCoordinator
114    if ownerReleaseCoordinator and ownerReleaseCoordinator:isReleased() then
115        return true, "released"
116    end
117    if ownerReleaseCoordinator and ownerReleaseCoordinator:hasPendingRelease() then
118        return ownerReleaseCoordinator:retryRelease({
119            reason = "retry",
120        })
121    end
122
123    local config = require("activity/" .. self._source)
124    local result
125    if type(config) == "table" and config.isSupport then
126        pcall(function(...)
127            result = config.isSupport()
128        end)
129    end
130    if self:checkNoIcon() then return end
131    if not result then
132        return false
133    end
134
135    if self.iconBtn and self.iconBtn.isDisposed then
136        self.iconBtn = nil
137        self._isDataDrivenIcon = nil
138        self._explicitNewActivityCycleKey = nil
139    end
140
141    if self._failedCleanupIcon then
142        local cleaned, cleanupError = self:cleanupFailedDataDrivenIcon()
143        if not cleaned then
144            return false, cleanupError
145        end
146    end
147
148    if self.iconBtn
149        and self._isDataDrivenIcon == true
150        and self.iconBtn._structuralCleanupPending == true
151    then
152        local destroyed, destroyError = self:destroyDataDrivenIconBtn()
153        if not destroyed then
154            return false, destroyError
155        end
156    end
157
158    local explicitCycleKey = getExplicitNewActivityCycleKey(config)
159    if self.iconBtn
160        and self._isDataDrivenIcon == true
161        and self._explicitNewActivityCycleKey ~= explicitCycleKey
162    then
163        return self:rebuildIcon(actInfo)
164    end
165
166    if not self.iconBtn then
167        local iconBtn, isDataDrivenIcon, mountResult, notificationContract, buildError =
168            self:buildIconBtn(self._source, actInfo)
169        if mountResult == DATA_DRIVEN_CLEANUP_FAILED then
170            ActivityDebugLog:log(self._actId .. " Icon V2清理未完成，暂不创建")
171            return
172        end
173        if mountResult == DATA_DRIVEN_OWNER_RELEASE_FAILED then
174            return false, buildError
175        end
176        if mountResult == DATA_DRIVEN_NOT_PRESENT then
177            return true, mountResult
178        end
179        self.iconBtn = iconBtn
180        self._isDataDrivenIcon = isDataDrivenIcon == true
181        self._explicitNewActivityCycleKey = self._isDataDrivenIcon
182            and explicitCycleKey or nil
183
184        notificationContract = notificationContract
185            or (isDataDrivenIcon
186                and NEW_ACTIVITY_NOTIFICATION_CONTRACT.DATA_DRIVEN
187                or NEW_ACTIVITY_NOTIFICATION_CONTRACT.LEGACY)
188        if self._popTimeEnabled
189            and notificationContract ~= NEW_ACTIVITY_NOTIFICATION_CONTRACT.NONE
190        then
191            self:tryAddNewIconActAutoPopup(
192                self._source,
193                notificationContract == NEW_ACTIVITY_NOTIFICATION_CONTRACT.DATA_DRIVEN
194            )
195        end
196        if not isDataDrivenIcon then
197            self.iconBtn:addToUI()
198        end
199        ActivityDebugLog:log(self._actId .. "icon创建成功")
200    end
201
202    self:refreshActIconFlag(actInfo)
203end

297function IconProcessor:cleanupFailedDataDrivenIcon()
298    local iconBtn = self._failedCleanupIcon
299    if not iconBtn then
300        self._failedCleanupIcon = nil
301        self._failedCleanupNeedsRemove = false
302        self._failedCleanupNeedsDispose = false
303        self._failedCleanupError = nil
304        return true
305    end
306
307    if self._failedCleanupNeedsRemove then
308        local removed, removeError = callCleanupMethod(iconBtn, "removeFromUI")
309        if not removed then
310            self._failedCleanupError = removeError
311            ActivityDebugLog:logError(
312                "[IconV2] failed icon cleanup failed source=" .. tostring(self._source)
313                .. " actId=" .. tostring(self._actId)
314                .. " stage=removeFromUI reason=" .. tostring(removeError)
315            )
316            return false, removeError
317        end
318        self._failedCleanupNeedsRemove = false
319        if iconBtn.isDisposed then
320            self._failedCleanupNeedsDispose = false
321        end
322    end
323
324    if self._failedCleanupNeedsDispose then
325        local disposed, disposeError = callCleanupMethod(iconBtn, "dispose")
326        if not disposed then
327            self._failedCleanupError = disposeError
328            ActivityDebugLog:logError(
329                "[IconV2] failed icon cleanup failed source=" .. tostring(self._source)
330                .. " actId=" .. tostring(self._actId)
331                .. " stage=dispose reason=" .. tostring(disposeError)
332            )
333            return false, disposeError
334        end
335    end
336
337    self._failedCleanupIcon = nil
338    self._failedCleanupNeedsRemove = false
339    self._failedCleanupNeedsDispose = false
340    self._failedCleanupError = nil
341    return true
342end

344function IconProcessor:buildIconBtn(source, actInfo)
345    local config = require("activity/" .. source)
346
347    if self:shouldUseDataDrivenIcon(config) then
348        local iconBtn, mountResult, buildError = self:buildDataDrivenIconBtn(config, actInfo)
349        if mountResult == DATA_DRIVEN_CLEANUP_FAILED then
350            return nil, true, mountResult, NEW_ACTIVITY_NOTIFICATION_CONTRACT.NONE, buildError
351        end
352        if mountResult == DATA_DRIVEN_NOT_PRESENT
353            or mountResult == DATA_DRIVEN_OWNER_RELEASE_FAILED
354        then
355            return nil, true, mountResult, NEW_ACTIVITY_NOTIFICATION_CONTRACT.NONE, buildError
356        end
357        if iconBtn then
358            return iconBtn, true, mountResult,
359                NEW_ACTIVITY_NOTIFICATION_CONTRACT.DATA_DRIVEN
360        end
361        return self:buildLegacyIconBtn(config), false, nil,
362            NEW_ACTIVITY_NOTIFICATION_CONTRACT.NONE
363    end
364
365    return self:buildLegacyIconBtn(config), false, nil,
366        NEW_ACTIVITY_NOTIFICATION_CONTRACT.LEGACY
367end

369function IconProcessor:buildDataDrivenIconBtn(config, actInfo)
370    if self._failedCleanupIcon then
371        return nil, DATA_DRIVEN_CLEANUP_FAILED, self._failedCleanupError
372    end
373
374    local iconBtn = nil
375    local mountResult = nil
376    local notPresentResult = nil
377    local stage = "require"
378    local success, err = pcall(function()
379        local DataDrivenActivityIconButton = require(DATA_DRIVEN_ICON_CLASS)
380
381        stage = "instantiate"
382        assert(type(DataDrivenActivityIconButton.new) == "function", "missing new")
383        iconBtn = DataDrivenActivityIconButton.new()
384        assert(iconBtn, "new returned nil")
385        stage = "owner_release_coordinator"
386        self:_getOrCreateOwnerReleaseCoordinator()
387        local ownerReleaseRequester = function(requestIcon, providerResult, context)
388            return self:requestOwnerRelease(requestIcon, providerResult, context)
389        end
390        if type(iconBtn.setOwnerReleaseRequester) == "function" then
391            iconBtn:setOwnerReleaseRequester(ownerReleaseRequester)
392        else
393            iconBtn._ownerReleaseRequester = ownerReleaseRequester
394        end
395
396        stage = "initialize"
397        assert(type(iconBtn.init) == "function", "missing init")
398        local initialized, initError = iconBtn:init(config)
399        if initialized == false or initError ~= nil then
400            error(initError or "init returned false", 0)
401        end
402
403        if actInfo and type(iconBtn.onActivityStatusChange) == "function" then
404            stage = "initialize_act_info"
405            local refreshed, refreshError = iconBtn:onActivityStatusChange(actInfo)
406            if refreshed == false then
407                error(refreshError or "initial actInfo refresh returned false", 0)
408            end
409        end
410
411        stage = "activate_mount_state"
412        assert(type(iconBtn.activateMountState) == "function", "missing activateMountState")
413        local activated, activateResult, activateDetail = iconBtn:activateMountState()
414        if activated ~= true then
415            error(activateResult or "activateMountState returned false", 0)
416        end
417        if activateResult == "not_present" then
418            mountResult = DATA_DRIVEN_NOT_PRESENT
419            notPresentResult = activateDetail
420            return
421        end
422        assert(activateResult == "mounted", "invalid activateMountState result")
423        mountResult = activateResult
424    end)
425    if success then
426        if mountResult == DATA_DRIVEN_NOT_PRESENT then
427            local coordinator = self:_getOrCreateOwnerReleaseCoordinator()
428            coordinator:stageRelease(notPresentResult)
429            local cleaned, cleanupError = self:_disposeUnmountedDataDrivenIcon(iconBtn)
430            if not cleaned then
431                return nil, DATA_DRIVEN_CLEANUP_FAILED, cleanupError
432            end
433            local released, releaseError = coordinator:requestRelease(nil, {
434                reason = "provider_not_present_before_mount",
435            })
436            if not released then
437                return nil, DATA_DRIVEN_OWNER_RELEASE_FAILED, releaseError
438            end
439            return nil, DATA_DRIVEN_NOT_PRESENT, notPresentResult
440        end
441        self:notifyDataDrivenIconMounted(config)
442        return iconBtn, mountResult
443    end
444
445    self:logDataDrivenFallback(stage, err)
446    local cleaned, cleanupError = self:disposeFailedDataDrivenIcon(iconBtn)
447    if not cleaned then
448        return nil, DATA_DRIVEN_CLEANUP_FAILED, cleanupError
449    end
450end

469function IconProcessor:shouldUseDataDrivenIcon(config)
470    if type(config) ~= "table" or type(config.icon) ~= "table" then
471        return false
472    end
473    if config.icon.mode ~= "dataDriven" then
474        return false
475    end
476    if type(config.icon.dataProvider) ~= "string" or config.icon.dataProvider == "" then
477        self:logDataDrivenFallback("config", "invalid dataProvider")
478        return false
479    end
480
481    local success, enabled = pcall(function()
482        local IconV2Experiment = require(ICON_V2_EXPERIMENT)
483        return IconV2Experiment:isEnabled()
484    end)
485    if not success then
486        self:logDataDrivenFallback("experiment", enabled)
487        return false
488    end
489    if enabled ~= true then
490        self:logDataDrivenFallback("experiment", "NewVineicon_v2/ON disabled or missing")
491        return false
492    end
493
494    local useDataDrivenWhen = rawget(config.icon, "useDataDrivenWhen")
495    if useDataDrivenWhen == nil then
496        return true
497    end
498    if type(useDataDrivenWhen) ~= "function" then
499        self:logDataDrivenFallback("selection", "icon.useDataDrivenWhen must be a function")
500        return false
501    end
502
503    local selected, shouldUse = pcall(useDataDrivenWhen)
504    if not selected then
505        self:logDataDrivenFallback("selection", shouldUse)
506        return false
507    end
508    if type(shouldUse) ~= "boolean" then
509        self:logDataDrivenFallback(
510            "selection",
511            "icon.useDataDrivenWhen must return a boolean"
512        )
513        return false
514    end
515
516    return shouldUse
517end

550function IconProcessor:refreshActIconFlag(actInfo)
551
552    if actInfo then
553        if not actInfo or tonumber(actInfo.actId) ~= tonumber(self._actId) then
554            return
555        end
556
557        if self.iconBtn and not self.iconBtn.isDisposed and actInfo then
558            self.iconBtn:onActivityStatusChange({
559                actId = self._actId,
560                reward = actInfo.reward,
561                msgNum = actInfo.msgNum,
562                rank = actInfo.rank,
563                progress = actInfo.progress,
564                maxProgress = actInfo.maxProgress
565            })
566        end
567    else
568        --没有actInfo的活动icon尝试刷新
569        if self.iconBtn then
570            self.iconBtn:updateState()
571        end
572    end

576function IconProcessor:tryAddNewIconActAutoPopup(source, isDataDrivenIcon)
577    local config = require("activity/" .. source)
578    if isDataDrivenIcon then
579        if not self.iconBtn or type(self.iconBtn.buildNewActivityNotificationCandidate) ~= "function" then
580            return
581        end
582        local iconLabelStr = self.iconBtn.btnLabel
583        if type(iconLabelStr) ~= "string" or iconLabelStr == "" then
584            iconLabelStr = config.icon and (config.icon.btnLabel or config.icon.iconLabel)
585        end
586        local candidate = self.iconBtn:buildNewActivityNotificationCandidate({
587            source = source,
588            actId = config.actId,
589            iconKey = tostring(config.actId),
590            iconLabelStr = iconLabelStr,
591            miniGameStaticIconAsset = resolveMiniGameNotificationAsset(config, self.iconBtn),
592            v2Notification = true,
593        })
594        if candidate then
595            _G.amfBean("PopupNewActMgr"):addDataDrivenNewActivityCandidate(source, candidate)
596        end
597        return
598    end
599    --is new act
600	local uniqueKey = ActivityUtil:getActUniqueKey(config)
601	if uniqueKey then
602		local panelShowNum = _G.amfBean("AutoPopupMgr"):getPermanentShowNum(uniqueKey .. "_panel")
603        local tbl = {
604            uniqueKey = uniqueKey,
605            iconKey = tostring(config.actId),
606            iconLabel = config.icon and config.icon.iconLabel,
607        }
608
609        if panelShowNum < 1 then
610            _G.amfBean("PopupNewActMgr"):addNewAutoPopAct(source, tbl)
611        end
612	end
613end

615function IconProcessor:_disposeIconLocally()
616    local cleaned, cleanupError = self:cleanupFailedDataDrivenIcon()
617    if not cleaned then

619    end
620
621    if self._isDataDrivenIcon == true then
622        local destroyed, destroyError = self:destroyDataDrivenIconBtn()
623        if not destroyed then
624            return false, destroyError
625        end
626    else

628    end
629    return true
