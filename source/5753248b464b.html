73function ActivityMgr:reloadAll()

88function ActivityMgr:updateActivities(configs, configName)
89    PerformanceMonitor:traceBegin("Activity-UpdateAll")
90    local versionName = "_updateVersion" .. configName
91    if not self[versionName] then
92        self[versionName] = 0
93    end
94
95    self[versionName] = self[versionName] + 1
96
97    for _, v in pairs(configs) do
98        local activityNode = self._activityMap[tostring(v.actId)]
99        if not activityNode then
100            activityNode = ActivityNode:create(v, self[versionName], self._iconSystemReady,
101                self.actInfosModel:getActInfo(v.actId), self)
102            self._activityMap[tostring(v.actId)] = activityNode
103
104            ActivityDebugLog:log(v.actId,"activityNode创建")
105        else
106            activityNode:update(v, self[versionName], self._iconSystemReady, self.actInfosModel:getActInfo(v.actId))
107            ActivityDebugLog:log(v.actId,"activityNode更新")
108        end
109    end
110
111    -- 后端没有但前端节点有，代表node已过期，释放
112    for k, node in pairs(self._activityMap) do
113        node:setLatestConfigVersion(self[versionName])
114        if node:isInvalid() then
115            ActivityDebugLog:log("活动",k,"的activityNode释放")
116            local disposed, disposeError = node:dispose()
117            if disposed == false then
118                ActivityDebugLog:logError(
119                    "活动" .. tostring(k) .. "清理失败，保留Owner重试: "
120                        .. tostring(disposeError)
121                )
122            else
123                self._activityMap[k] = nil
124            end
125        end
126    end
127
128    self.activityNodeCount = 0
129
130    for k,v in pairs(self._activityMap) do
131        self.activityNodeCount = self.activityNodeCount + 1
132    end
133
134    PerformanceMonitor:traceEnd()
135end

214--节点处理完后整体处理强弹
215function ActivityMgr:onActivityResolveOver(event)
216    if not self:isEnabled() then
217        return
218    end
219
220    local data = event:getData()
221    
222    local actId = data.actId
223    local configVersion = data.configVersion
224    local result = data.result
225
226    local tmpTable = self._iconAutoPopHandleMap[tostring(configVersion)]
227
228    if not self._iconAutoPopHandleMap[tostring(configVersion)] then
229        self._iconAutoPopHandleMap[tostring(configVersion)] = {}
230        self._iconAutoPopHandleMap[tostring(configVersion)].countNum = 0
231        tmpTable = self._iconAutoPopHandleMap[tostring(configVersion)]
232    end
233
234    --有活动重新加载完成了，重新走触发
235    if tmpTable[tostring(actId)] == false and result == true then
236        tmpTable[tostring(actId)] = result
237        tmpTable.hadHandled = false
238    end
239
240    if tmpTable.hadHandled then
241        ActivityDebugLog:log("这一批节点处理完了" .. configVersion)
242        return
243    end
244
245    if not tmpTable[tostring(actId)] then
246        tmpTable[tostring(actId)] = result
247        tmpTable.countNum = tmpTable.countNum + 1
248        ActivityDebugLog:log("actId" .. actId .. "处理完毕" .. tostring(result))
249    end
250
251    if tmpTable.countNum == self.activityNodeCount and (self._isFirstEnter or self._isHome) then
252        for k,v in pairs(tmpTable) do
253            if v == true then
254                ActivityDebugLog:log("尝试触发强弹" .. configVersion)
255                _G.amfBean("PopupNewActMgr"):handleNewIconActAutoPopup(true)
256                break
257            end
258        end
259        
260        tmpTable.hadHandled = true
261        _G.amfEC:notifyFeature(FeatureEvent.ACT_INIT_OVER, {
262            isFirstEnter = self._isFirstEnter,
263            isHome = self._isHome
264        })
265
266        self._isFirstEnter = false
267        self._isHome = false
268    end
269end

426-- 配置式强弹是否已被该活动的声明式注册接管。
427-- 只按 actId 判定：主面板等 checkPoint 是多活动共用的槽位，按 checkPoint 判定会误伤其它活动。
428-- 拿不到强弹管理器（老包/单测环境）时按“未接管”处理，与 AutoPopupMgr:addPopupConfig 的拒绝逻辑保持一致。
429function ActivityMgr:isConfigAutoPopupTakenOverByRegistration(autoPopupMgr, actId)
430    if actId == nil or not autoPopupMgr then
431        return false
432    end
433
434    local isTakenOver = autoPopupMgr:hasRegisteredPopupForActivity(actId) == true
435    if isTakenOver then
436        ActivityDebugLog:log("配置式强弹已被注册式接管，不参与候选竞争", actId)
437    end
438    return isTakenOver
439end

441-- 获取强弹数据
442function ActivityMgr:getAutoPopupData(delegate)
443    if not self:isEnabled() then
444        return {}
445    end
446
447    local key = delegate.popStage .. "_".. (delegate.isMetaPop and "metaPop" or "noMetaPop")
448    if self._tmpAutoPopupData[key] then
449        return self._tmpAutoPopupData[key]
450    end
451
452    local autoPopupMgr = _G.amfBean("AutoPopupMgr")
453    local ret = {}
454    for k, v in pairs(self._activityMap) do
455        local datas = v:getAutoPopupData(delegate)
456        for _, data in ipairs(datas or {}) do
457            -- 剔除必须在下面的 limitNum 截断之前：被接管的候选若先占掉唯一名额，之后会在
458            -- AutoPopupMgr:addPopupConfig 被拒且没有补位，同场景其它活动已被截断出局，
459            -- 最终整场不弹（7026 周末无限精力曾因此不弹）。
460            if not self:isConfigAutoPopupTakenOverByRegistration(autoPopupMgr, data.actId) then
461                table.insert(ret, data)
462            end
463        end
464    end
465
466    table.sort(ret, function(a, b)
467        return a.priority < b.priority
468    end)
469
470    for i = #ret, delegate.limitNum + 1, -1 do
471        ret[i] = nil
472    end
473
474    if not table.isEmpty(ret) then
475        self._tmpAutoPopupData[key] = ret
476    end
477
478    -- ActivityDebugLog:logAutoPop("获取强弹数据".. key .. "数量".. #ret .. "内容" .. tts(ret))
479
480    return ret
481end
