17local function buildConfig(behaviorConfig)
18    -- 只保留线上业务需要的两项，性能参数由 Rust 默认配置统一管理。
19    if prophet and prophet.makeComposableProgressV2Config then
20        return prophet.makeComposableProgressV2Config(behaviorConfig)
21    end
22    return {
23        evaluator_preset_name = "composable_progress_v2",
24        behavior_config = behaviorConfig,
25    }
