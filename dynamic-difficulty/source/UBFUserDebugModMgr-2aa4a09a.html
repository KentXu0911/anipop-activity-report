269function UBFUserDebugModMgr:_tryUpdateAdjustDebugInfo()
270    local uiMgr =  UBFGlobal:getLogic(UBFUiLogicId.UI_VIEW_LOGIC_MANAGER)
271    if not uiMgr:existNode(UBFUserDebugNodeId.WIDGET) then UBFLogE("No UBFUserDebugNode") return end
272
273    local sceneMgr = UBFGameSceneManager:getInstance()
274    if not sceneMgr then UBFLogE("No UBFGameSceneManager") return end
275
276    local msgLogic = sceneMgr:getMessageLogic()
277    if not msgLogic then UBFLogE("No msgLogic") return end
278
279    local promise = msgLogic:sendDataRequire(UBFGameRequireDataKey.kAdjustDebugInfo)
280    promise:next(function(data)
281        -- UBFLogD(25, "AdjustDebugInfo from Logic", data)
282        local isDataValid, errMsg = self:_checkAdjustDebugInfoData(data)
283        if not isDataValid then
284            UBFLogE("UBFUserDebugModMgr AdjustDebugInfo data invalid", errMsg, data)
285        end
286        self:_doUpdateAdjustDebugInfo(data, errMsg)
287    end)
288end

290function UBFUserDebugModMgr:_checkAdjustDebugInfoData(data)
291    if type(data) ~= "table" then
292        return false, "response is not table: "..tostring(data)
293    end
294    if data.err then

296    end

300function UBFUserDebugModMgr:_doUpdateAdjustDebugInfo(data, errMsg)
301    data = type(data) == "table" and data or {}
302    local adjustData = data.adjustData or {}
303    local ubfuuCheckResult = adjustData.ubfuuCheckResult or {}
304    local calculateDiffIndexResult = ubfuuCheckResult.calculateDiffIndexResult or {}
305    local adjustStepData = data.adjustStepData or {}
306
307    local updateItemsData = {}
308    local function addInfo(groupKey, key, value, style)
309        updateItemsData[#updateItemsData + 1] = {
310            groupKey = groupKey,
311            key = key,
312            value = value,
313            style = style,
314        }
315    end
316
317    local function addDeltaInfo(groupKey, key, deltaVal, valFormatter)
318        local str
319        local style
320
321        if tonumber(deltaVal) then
322            deltaVal = tonumber(deltaVal)
323            style = {}
324            if valFormatter then
325                str = string.format(valFormatter, math.abs(deltaVal))
326                if tonumber(str) then
327                    str = tostring(tonumber(str))
328                end
329            else
330                str = tostring(math.abs(deltaVal))
331            end
332            if deltaVal > 0 then
333                str = "差"..str
334                style.fontColor = "FF5555"
335            else
336                str = "超"..str
337                style.fontColor = "6FCF97"
338            end
339        else
340            str = tostring(deltaVal)
341        end
342        addInfo(groupKey, key, str, style)
343    end
344
345    addInfo("userTag", "activityIndex", calculateDiffIndexResult.activityIndex)
346    addInfo("userTag", "paymentIndex", calculateDiffIndexResult.paymentIndex)
347    addInfo("userTag", "playerAbility", calculateDiffIndexResult.playerAbility)
348
349    addInfo("diffFix", "desiredDiffAvg", ubfuuCheckResult.desiredDiffAvg)
350    addInfo("diffFix", "diffIndex1", calculateDiffIndexResult.diffIndex1)
351    addInfo("diffFix", "diffIndex2", calculateDiffIndexResult.diffIndex2)
352    addInfo("diffFix", "diffIndex", calculateDiffIndexResult.diffIndex)
353    addInfo("diffFix", "desiredDiffAvgFix1", ubfuuCheckResult.desiredDiffAvgFix1)
354    addInfo("diffFix", "desiredDiffAvgFix2", ubfuuCheckResult.desiredDiffAvgFix2)
355    addInfo("diffFix", "desiredDiffAvgFix", ubfuuCheckResult.desiredDiffAvgFix)
356    addInfo("diffFix", "diffFix1", ubfuuCheckResult.diffFix1)
357    addInfo("diffFix", "diffFix", ubfuuCheckResult.diffFix)
358    addDeltaInfo("diffFix", "originDiffMove", adjustStepData.originDiffMove, "%.2f")
359    addDeltaInfo("diffFix", "fixedDiffMove1", adjustStepData.fixedDiffMove1, "%.2f")
360    addDeltaInfo("diffFix", "diffMove", adjustStepData.diffMove, "%.2f")
361    addDeltaInfo("diffFix", "adjustStepCurrRate", adjustStepData.currRate, "%d%%")

368    addInfo("details", "todayPassLevelCount", calculateDiffIndexResult.todayPassLevelCount)
369    addInfo("details", "paymentIndexCount", calculateDiffIndexResult.paymentIndexCount)
370    addInfo("details", "v250801CashValue", calculateDiffIndexResult.v250801CashValue)
371    addInfo("details", "v250801FarmCashValue", calculateDiffIndexResult.v250801FarmCashValue)

373    local uiMgr =  UBFGlobal:getLogic(UBFUiLogicId.UI_VIEW_LOGIC_MANAGER)
374    local opParams = {
375        nodeId = UBFUserDebugNodeId.WIDGET,
376        uiData = {
377            nodeConfig = {
378                data = {
379                    [UBFUserDebugNodeId.WIDGET] = {
