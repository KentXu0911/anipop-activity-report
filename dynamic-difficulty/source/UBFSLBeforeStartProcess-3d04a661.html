40function UBFSLBeforeStartProcess:supplyLevelInfo(runtime, levelInfo)

42	local energyMgr = runtime:getEnergyMgr()
43    local levelJsonData = runtime:getLevelJsonData()
44
45	--准备levelInfo

52	levelInfo.useInfiniteEnergy = energyMgr.useInfiniteEnergy

55		levelInfo.so_prophet = runtime.rustProphetKey
56		levelInfo.prophetLaunchInfo = runtime.prophetLaunchInfo
57	end

65		levelInfo.startLevelCostType = energyMgr:getStartLevelCostType()
66	end
67	levelInfo.playLevelMode = runtime.playLevelMode
68	local willingModAllowList = runtime:getExtra("willingModAllowList")
69	levelInfo.willingModAllowList = type(willingModAllowList) == "table"
70		and UBFHelper:shallowClone(willingModAllowList) or nil
71	levelInfo.playerStateSandbox = runtime:isPlayerStateSandbox()
72	levelInfo.dcData = runtime:getDCData()
73	levelInfo.outLevelAttachData = runtime.outLevelAttachData
74	levelInfo.diffFix = tryGetDiffFix(modMgr)
75	levelInfo.temp = {
76		selectedItemsData = selectedItemsData,
77	}
78
79	if #selectedItemsData > 0 then 
80		levelInfo.hasUsePreprop = true
81	end
82
83    levelInfo.stats = runtime:getAllStats()
84
