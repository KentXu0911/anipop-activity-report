11local function validateProcessCmd(cmd, params, currProcess)
12    if not currProcess then
13        UBFLogN("Ignore process cmd", cmd, "current process not found", params)
14        return false
15    end
16    if type(params) ~= "table" then
17        return false, buildBridgeError(UBFAmfBridge.ErrorCode.INVALID_PARAMS, "params must be table")
18    end
19    local processId = params.processId
20    if processId == nil then
21        return false, buildBridgeError(UBFAmfBridge.ErrorCode.INVALID_PARAMS, "params.processId is required")
22    end
23    local currentProcessId = currProcess:getProcessId()
24    if processId ~= currentProcessId then
25        UBFLogN("Ignore process cmd", cmd, "processId does not match, request", processId, "current", currentProcessId, params)
26        return false
27    end
28
29    return true
30end

145function UBFProcessManager:_doStartRun(launchData, processCreator, runtimeCreator)
146	-- 正常启动进关流程时需要检查当前是否存在未完成的出关流程的,skipEndProcessCheck为true时跳过该检查
147    if not launchData.skipEndProcessCheck and self.currEndProcess then
148        local msg = "[UBFProcessManager] Cannot start level process: unfinished end-level process exists!"
149        if EnvUtil:isDevelopMode() then
150            UBFLogE(msg) 
151            CommonTip:showTip(msg)
152        else
153            he_log_error(msg)
154        end
155    end
156    -- 同一时间只能有一个进关process运行，当已有process未完成时，拒绝开启新的process
157    local process = processCreator()
158    if self.currStartProcess then
159        if not launchData.forceLaunchNew then
160            -- UBFDevAlert("同一时间只能有一个进关process运行，当前有process未完成时，拒绝开启新的process")
161            process:abort()
162            return process
163        end
164    end
165
166    UBFCachedFile:clearCache()
167    self:_clearCurrStartProcess()
168    self:_setCurrStartProcess(process)
169
170    local levelId = launchData.levelId
171    local levelScene = launchData.levelScene
172
173    local function nextStep(data)
174        UBFPerfMonitor.mark("OpenSlp", "SuperBaseResEnd")
175        if process ~= self:getCurrStartProcess() then return end
176        if not data.success then 
177            UBFProcessManager:showResourceLoadFailMessage()
178            self:_clearCurrStartProcess()
179            process:abort()
180            return
181        end
182        local runtime = runtimeCreator()
183        if launchData.startSource then runtime.startSource = launchData.startSource end
184        process:setResult("runtime", runtime)
185        if launchData.hook then launchData.hook(process) end
186        process:start()
187    end
188
189    UBFPerfMonitor.mark("OpenSlp", "SuperBaseResStart")
190    if launchData.editorMode then self:_prepareSuperBaseResourceForEditor(launchData.levelCfg):next(nextStep)
191    else UBFGameSceneResourceMgr:prepareSuperBaseResource(levelId, levelScene):next(nextStep) end
192
193    return process
194end

461-- end level process
462function UBFProcessManager:_launchEndLevelProcess(processType, runtime)
463    local processCreateData = self:_getProcessCreateDataByType(runtime, processType)
464	local process = UBFELProcessFlowController:create(processCreateData)
465    -- 同一时间只能有一个出关process运行，当已有process未完成时，拒绝开启新的process
466    if self.currEndProcess then
467        UBFDevAlert("同一时间只能有一个出关process运行，当前有process未完成时，拒绝开启新的process")
468        process:abort()
469        return process
470    end
471    UBFCachedFile:clearCache()
472    self:_clearCurrEndProcess(process)
473    self:_setCurrEndProcess(process)
474	self:_pushEndLevelProcess(process)
475	setNextFrame(function() process:start() end)
476    return process
477end
