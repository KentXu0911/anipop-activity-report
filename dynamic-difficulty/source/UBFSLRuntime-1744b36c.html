160function UBFSLRuntime:_buildCreateDataBySectionData(createData, sectionData)
161    local levelInfo = sectionData.levelInfo
162    createData.levelId = levelInfo.levelId
163    createData.levelScene = levelInfo.levelScene
164    createData.levelGroupInfo = levelInfo.levelGroupInfo
165    -- dev 断面恢复：Editor 关卡不在关卡配置表里，用归档中保存的关卡配置兜底（见 UBFDebugSectionResume）
166    if sectionData.devSectionResume == true and type(sectionData.devEditorLevelCfg) == "table" then
167        createData.levelCfg = sectionData.devEditorLevelCfg
168        createData.editorMode = true
169    else
170        createData.levelCfg = UBFLevelConfigManager:getLevelConfigByUUID(levelInfo.uuid)
171    end
172end

174function UBFSLRuntime:_postInit()
175    --migrator
176    self.migrator = require 'ubf.UBFGamePlayScene.logics.processFlow.module.UBFSLMigrator'
177    self:registerListenerByMap(self.migrator.gListener)
178
179    --init other
180    if self.resumeData then
181        -- 闪退恢复
182        self:_initOtherByResumeData(self.resumeData)
183    elseif self.replayLog then
184        -- 回放
185        self:_initOtherByReplayData(self.replayLog)
186    elseif self.editorMode then
187        -- 编辑器 
188        self:_initOtherOnEditor()
189    elseif self:isAIMode() then
190        -- AI
191        self:_initOtherOnEditor()
192    else
193        -- 正常进关
194        self:_initOtherByEnv()
195    end
196end

301function UBFSLRuntime:initLevelFarmingType(data)
302    local levelId = self.levelId
303    local levelType = self.levelType
304    local topLevelId = self.topLevelId
305    local oldStar = self.oldStar or 0
306
307    if levelType == GameLevelType.kMainLevel then
308        if oldStar >= 4 then
309            self.levelFarmingType = UBFLevelFarmingType.FARM_SCORE
310        elseif oldStar > 0 then
311            self.levelFarmingType = UBFLevelFarmingType.FARM_STAR
312        else
313            if levelId == topLevelId then
314                self.levelFarmingType = UBFLevelFarmingType.NORMAL
315            else
316                self.levelFarmingType = UBFLevelFarmingType.FARM_STAR
317            end
318        end
319    elseif levelType == GameLevelType.kHiddenLevel then
320        if oldStar >= 3 then
321            self.levelFarmingType = UBFLevelFarmingType.FARM_SCORE
322        elseif oldStar > 0 then
323            self.levelFarmingType = UBFLevelFarmingType.FARM_STAR
324        else
325            self.levelFarmingType = UBFLevelFarmingType.NORMAL
326        end
327    else
328        self.levelFarmingType = UBFLevelFarmingType.NORMAL
329    end
330
331end

353function UBFSLRuntime:updateWinningStreakInfo(sourceName, winningStreakInfo)
354    if not self.enabledWinningStreak then
355        self.enabledWinningStreak = {}
356    end
357
358    if winningStreakInfo then
359        winningStreakInfo.sourceName = sourceName
360        table.insert( self.enabledWinningStreak, winningStreakInfo )
361    end
362end
