6-- linspace 函数，用于生成等间隔的数值
7local function linspace(start, stop, num)
8    local t = {}
9    if num <= 0 then
10        return t
11    end
12    local step = (stop - start) / (num - 1)
13    t[0] = 0
14    t[1] = 0
15    for i = 1, num - 1 do
16        -- table.insert(t, start + i * step)
17        t[i+1] = start + i * step
18    end
19    return t
20end

22-- 根据配置，输入收集物数量，计算权重，返回当前进度百分比
23local function make_predict_v1(meta)
24    -- print("make_predict meta:",tts(meta))
25    local model, weights = meta.model, meta.weights
26    if not meta.weights then
27        return function() return end
28    end
29    if not meta._initV1 then
30        meta._initV1 = true
31        -- print("model",tts(model),"weights",tts(weights))
32        local total_weight = 0
33        for _, w in pairs(weights) do
34            total_weight = total_weight + w
35        end
36        -- print("meta",table.serialize(meta))
37        local maxStep = 0
38        for k,v in pairs(model) do
39            local keys = {}
40            local map = {}
41            for kk, vv in pairs(v) do
42                map[tonumber(kk)] = vv
43                table.insert(keys,tonumber(kk))
44            end
45            table.sort(keys)
46            maxStep = math.max(maxStep,keys[#keys])
47            map._keys = keys
48            model[k] = map
49        end
50        meta.maxStep = maxStep
51        meta.linspace = linspace(0, 1, meta.maxStep + 1)
52    end
53    -- 创建一个预测函数，基于给定的模型和权重。
54
55    -- 获取模型的行数（假设所有列的长度相同）
56    local index_normalized =  meta.linspace
57    -- print('meta.maxStep',meta.maxStep,"index_normalized",tts(index_normalized))
58    local scaler = meta['scaler'] or 1
59
60    return function(x, step , static_step)
61        -- 根据输入值和步长计算进度值。
62        local progress = 0
63        local tMap = {}
64        for target_id, value in pairs(x) do
65            -- 按照步数增加 target_value 值, step 是 [0, 1] 之间的值
66            value = math.max(value + 1 + step / static_step, 0) * scaler
67            -- print("value",value)
68
69            -- 获取对应目标 ID 的列名
70            local column_name = tostring(target_id)
71
72            if not model[column_name] then
73                error("Target ID " .. target_id .. " not in model")
74            end
75
76            -- 获取目标列的值
77            local column_values = model[column_name]
78            -- print("column_values",target_id,tts(column_values))
79            local keys = column_values._keys
80            -- print("keys",tts(keys))
81            -- 找到给定值所在的区间，并计算归一化后的索引值
82            for i = 1,  #keys-1 do
83                if column_values[keys[i]] <= value and value <= column_values[keys[i + 1]] then
84                    -- 线性插值计算归一化索引值
85                    local t = (value - column_values[keys[i]]) / (column_values[keys[i + 1]] - column_values[keys[i]])
86                    -- print("t",t,i,"keys?:",keys[i],keys[i+1],"in:",index_normalized[keys[i]],index_normalized[keys[i+1]])
87                    local target_progress = index_normalized[keys[i]] + t * (index_normalized[keys[i+1]] - index_normalized[keys[i]])
88                    progress = progress + target_progress * weights[column_name]
89                    tMap[target_id] = "v:" .. value .. " in:(" .. column_values[keys[i]] .. "/" .. column_values[keys[i+1]] .. ") t:" .. target_progress .. " w:" .. weights[column_name] .. " p:" .. progress 
90                    break
91                end
92            end
93            -- 如果值超出范围，使用最近的端点
94            -- print("value < column_values[1]",value , column_values[1],column_name,tts(column_values))
95            if value < column_values[keys[1]] then
96                progress = progress + index_normalized[1] * weights[column_name]
97            elseif value > column_values[keys[#keys]] then
98                progress = progress + index_normalized[#index_normalized] * weights[column_name]
99            end
100        end
101
102        -- print("make_predict_v1 tMap",progress,tts(tMap))
103        return progress
104    end
105end

107--根据配置，输入收集物数量，返回每个收集物折算的步数
108local function make_predict_v2(config,errorlog)
109    local model = config['model']
110    local level_id = config['level_id'] or 0
111    local scaler = config['scaler'] or 1
112
113    local function predict(x, current_step, static_step)
114        if current_step > static_step then
115            if errorlog then
116                errorlog(string.format("Level %d Current step %d is greater than static step %d", level_id, current_step, static_step))
117            end
118            current_step = static_step
119        end
120        if current_step < 0 then
121            if errorlog then
122                errorlog(string.format("Level %d Current step %d is less than 0", level_id, current_step))
123            end
124            current_step = 0
125        end
126        if static_step < 0 then
127            if errorlog then
128                errorlog(string.format("Level %d Static step %d is less than 0", level_id, static_step))
129            end
130            static_step = math.min(current_step, 20)
131        end
132        if static_step == 0 then
133            if errorlog then
134                errorlog(string.format("Level %d Static step %d is 0 and cannot be used as divisor", level_id, static_step))
135            end
136            static_step = 20
137        end
138        if config['steps'] and config['steps'] ~= static_step then
139            if errorlog then
140                errorlog(string.format("Level %d Static step %d is not equal to config step %d", level_id, static_step, config['steps']))
141            end
142            static_step = config['steps']
143        end
144
145        local step_adjust = current_step / static_step
146        local result = {}
147        -- local tMap = {}
148
149        for target_id, target_values in pairs(model) do
150            local sorted_items = {}
151            for k, v in pairs(target_values) do
152                if tonumber(k) then
153                table.insert(sorted_items, {tonumber(k), math.max(v, 0)})
154                end
155            end
156            table.sort(sorted_items, function(a, b) return a[1] < b[1] end)
157
158            local indices = {}
159            local values = {}
160
161            for _, item in ipairs(sorted_items) do
162                table.insert(indices, item[1])
163                table.insert(values, item[2])
164            end
165
166            -- 将 target value 已经达到最大值的情况, 直接返回最大值的索引
167            -- print("x[tonumber(target_id)?",tts(x),target_id,scaler,tts(values))
168            if (x[tonumber(target_id)] + 1) * scaler >= values[#values] then
169                result[target_id] = indices[#indices]
170            else
171
172                if indices[1] ~= 0 then
173                    table.insert(indices, 1, 0)
174                    table.insert(values, 1, 0)
175                end
176
177                local value = math.max(x[tonumber(target_id)] + step_adjust, 0) * scaler
178
179                for i = 1, #values - 1 do
180                    if values[i] <= value and value <= values[i + 1] then
181                        if values[i] == values[i + 1] then
182                            result[target_id] = indices[i]
183                            break
184                        end
185                        local t = (value - values[i]) / (values[i + 1] - values[i])
186                        local target_progress = indices[i] + t * (indices[i + 1] - indices[i])
187                        result[target_id] = target_progress
188                        -- tMap[target_id] = "v:" .. value .. " in:(" .. values[i] .. "/" .. values[i+1] .. ") t:" .. target_progress 
189                        break
190                    end
191                end
192
193                if not result[target_id] then
194                    if value < values[1] then
195                        result[target_id] = indices[1]
196                    elseif value > values[#values] then
197                        result[target_id] = indices[#indices]
198                    end
199                end
200            end
201        end
202        -- print("tMap",tts(tMap))
203
204        return result
205    end
206
207    return predict
208end
