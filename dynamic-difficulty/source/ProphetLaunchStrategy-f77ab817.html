353local function safeCall(target, methodName, ...)
354    local method = type(target) == "table" and target[methodName] or nil
355    if type(method) ~= "function" then
356        return false, nil
357    end
358
359    local args = { ... }
360    local ok, result = pcall(function()
361        return method(target, unpackValues(args))
362    end)
363    if not ok then
364        return false, nil
365    end
366    return true, result
367end

369-- 版本号按整数段逐段比较，避免字符串比较把 "1.100" 判成小于 "1.99"。
370local function compareVersion(left, right)
371    local function toParts(value)
372        local parts = {}
373        for part in tostring(value or ""):gmatch("[^.]+") do
374            parts[#parts + 1] = tonumber(part) or 0
375        end
376        return parts
377    end
378
379    local leftParts = toParts(left)
380    local rightParts = toParts(right)
381    local count = math.max(#leftParts, #rightParts)
382    for index = 1, count do
383        local leftValue = leftParts[index] or 0
384        local rightValue = rightParts[index] or 0
385        if leftValue ~= rightValue then
386            return leftValue < rightValue and -1 or 1
387        end
388    end
389    return 0
390end

474local function isRuleEnabled(rule)
475    if type(rule) ~= "table" or rule.enabled == false then
476        return false
477    end
478
479    local mode = rule.mode
480    if mode == "allow" or mode == "exclude" then
481        return type(rule.anyOf) == "table" and #rule.anyOf > 0
482    end
483    if mode == "block" then
484        return type(rule.blocklist) == "table" and #rule.blocklist > 0
485    end
486    if mode == "equals" then
487        return rule.expected ~= nil
488    end
489    if mode == "value" then
490        return rule.op ~= nil
491            and rule.value ~= nil
492            and VALUE_COMPARATORS[rule.op] ~= nil
493    end
494    if mode == "min" or mode == "max" then
495        return rule.min ~= nil or rule.max ~= nil
496    end
497    return false
498end

500local function isGateEnabled(gate)
501    if type(gate) ~= "table" or gate.enabled == false or gate.id == nil then
502        return false

504    if type(gate.rules) ~= "table" then
505        return false

507    for _, rule in ipairs(gate.rules) do

527local function matchRuleValue(rule, actual)
528    local mode = rule.mode
529    if mode == "equals" then
530        -- 动更就绪规则可按版本放宽：高于 requireRustDyGoReadyBelow 的包视为已直接生效。
531        if rule.field == "isRustDyGoReady"
532            and not shouldCheckRustDyGoReady(rule.requireRustDyGoReadyBelow) then
533            return rule.expected == true
534        end
535        return actual == rule.expected
536    end
537    if mode == "value" then
538        -- 通用数值比较：op = eq/lt/lte/gt/gte，value 为期望值。取不到数值即未通过。
539        local comparator = VALUE_COMPARATORS[rule.op]
540        local numericActual = tonumber(actual)
541        local numericExpected = tonumber(rule.value)
542        if not comparator or not numericActual or not numericExpected then
543            return false
544        end
545        return comparator(numericActual, numericExpected)
546    end
547    if mode == "allow" then
548        return matchAnyOf(actual, rule.anyOf)
549    end
550    if mode == "exclude" then
551        return not matchAnyOf(actual, rule.anyOf)
552    end
553    if mode == "min" or mode == "max" then
554        -- min/max 是版本号门槛，阈值必须写成字符串："1.160" 才不会被当成浮点 1.16。
555        local threshold = mode == "min" and rule.min or rule.max
556        if type(threshold) ~= "string" then
557            return false
558        end
559        local compared = compareVersion(actual, threshold)
560        if mode == "min" then
561            return compared > 0 or (rule.minStrict ~= true and compared == 0)
562        end
563        return compared < 0 or (rule.maxStrict ~= true and compared == 0)
564    end
565    if mode == "block" then
566        -- blocklist 是「字段取值黑名单」：字段取不到值时不参与屏蔽。
567        if actual == nil or type(rule.blocklist) ~= "table" then
568            return true
569        end
570        return not matchAnyOf(actual, rule.blocklist)
571    end
572    return false
573end

575local function matchGateRule(rule, runtime)
576    if not isRuleEnabled(rule) then
577        return true
578    end
579    return matchRuleValue(rule, getFieldValue(runtime, rule.field))
580end

582-- when 只用于把某条门禁收窄到指定平台或指定环境，默认 all。
583local function matchGateCondition(condition, runtime)
584    if type(condition) ~= "table" then
585        return true
586    end
587
588    if condition.field == "isRustDyGoReady" then
589        -- 带 requireRustDyGoReadyBelow 时，高于门槛的版本视为「已直接生效」。
590        local ready = getRustDyGoReady()
591        if not shouldCheckRustDyGoReady(condition.requireRustDyGoReadyBelow) then
592            ready = true
593        end
594        return ready == condition.expected
595    end
596
597    local actual = getFieldValue(runtime, condition.field)
598    if type(condition.expected) == "table" then
599        return matchAnyOf(actual, condition.expected)
600    end
601    return actual == condition.expected
602end

604local function matchGateWhen(gate, runtime)
605    local when = gate.when
606    if type(when) ~= "table" then
607        return true
608    end
609
610    local conditions = when.conditions or when
611    local mode = when.cond or when.mode or "all"
612    local matchedAny = false
613
614    for _, condition in ipairs(conditions) do
615        local matched = matchGateCondition(condition, runtime)
616        if mode == "any" then
617            if matched then
618                matchedAny = true
619                break
620            end
621        elseif not matched then
622            return false
623        end
624    end
625
626    return mode ~= "any" or matchedAny
627end

805local function isUserLevelAllowed(runtime)
806    local userLevelConfig = USER_LEVEL_CONFIG
807    if type(userLevelConfig) ~= "table" or userLevelConfig.enabled == false then
808        return true
809    end
810
811    local minimum = tonumber(userLevelConfig.min)
812    local maximum = tonumber(userLevelConfig.max)
813    if not minimum and not maximum then
814        return true
815    end
816
817    local rawUserTopLevel = getFieldValue(
818        runtime,
819        userLevelConfig.field or "getUserTopLevel"
820    )
821    local userTopLevel = tonumber(rawUserTopLevel)
822
823    -- 取不到有效数值时按配置决定；默认视为未命中，避免口径不明的用户误入实验。
824    if not userTopLevel then
825        if userLevelConfig.missingAsReject == true then
826            return false, "user_top_level_missing", nil
827        end
828        return true
829    end
830
831    if minimum then
832        -- 注意不能用 "a and b or c" 表达包含/不包含：false and x or y 会回退到 y。
833        local passed
834        if userLevelConfig.minInclusive == false then
835            passed = userTopLevel > minimum
836        else
837            passed = userTopLevel >= minimum
838        end
839        if not passed then
840            return false, "user_top_level_below_threshold", userTopLevel
841        end
842    end
843
844    if maximum then
845        local passed
846        if userLevelConfig.maxInclusive == false then
847            passed = userTopLevel < maximum
848        else
849            passed = userTopLevel <= maximum
850        end
851        if not passed then
852            return false, "user_top_level_above_threshold", userTopLevel
853        end
854    end
855
856    return true, nil, userTopLevel
857end

1074local function matchesExpectedValue(actual, expected)
1075    if type(expected) ~= "table" then
1076        return actual == expected
1077    end
1078
1079    if expected.min ~= nil or expected.max ~= nil then
1080        local numericActual = tonumber(actual)
1081        if not numericActual then
1082            return false
1083        end
1084        if expected.min ~= nil and numericActual < expected.min then
1085            return false
1086        end
1087        if expected.max ~= nil and numericActual > expected.max then
1088            return false
1089        end
1090        return true
1091    end
1092
1093    if expected[actual] ~= nil then
1094        return expected[actual] == true
1095    end
1096
1097    for _, value in ipairs(expected) do
1098        if actual == value then
1099            return true
1100        end
1101    end
1102
1103    return false
1104end

1143function ProphetLaunchStrategy:evaluate(runtime, resultDatas)
1144    if not PROPHET_EXPERIMENT_ENABLED then
1145        return nil
1146    end
1147
1148    -- 新手期实验独立于旧体验矩阵：开关命中且满足条件时直接使用新配置；
1149    -- 其他情况继续走原有矩阵，避免新开关改变旧逻辑的门禁和回退行为。
1150    local newUserDecision = evaluateNewUserProphet(runtime, resultDatas)
1151    if newUserDecision then
1152        return newUserDecision
1153    end
1154
1155    local groupName, groupConfig, experimentConfig, switchConfig =
1156        getExperimentGroup()
1157    if not groupName or not experimentConfig or not switchConfig then
1158        return nil
1159    end
1160
1161    local checkResult = resultDatas and resultDatas.checkResult
1162    local decision = {
1163        available = groupConfig and groupConfig.enableProphet == true or false,
1164        enabled = false,
1165        group = groupName,
1166        maintenanceName = switchConfig.maintenanceName,
1167        experimentSwitch = switchConfig.id,
1168        configKey = nil,
1169        -- Prophet 启动后不与旧 diffFix 共存；该字段只用于结果记录和下游打点。
1170        removeDiffFix = groupConfig and groupConfig.enableProphet == true or false,
1171        originalDiffFix = resultDatas and resultDatas.diffFix,
1172        fixedDesiredDiffAvg = checkResult and checkResult.fixedDesiredDiffAvg,
1173        desiredDiffAvg = checkResult and checkResult.desiredDiffAvg,
1174        diffIndex = checkResult and checkResult.diffIndex,
1175        requireToplevelBaoDi =
1176            groupConfig and groupConfig.requireToplevelBaoDi == true or false,
1177        toplevelBaoDiFlag = getToplevelBaoDiFlag(checkResult),
1178        experienceType = "experience",
1179    }
1180
1181    if not groupConfig then
1182        decision.reason = "experience_config_missing"
1183        return decision
1184    end
1185
1186    local allowed, reason = isPlatformAllowed()
1187    if not allowed then
1188        decision.reason = reason
1189        return decision
1190    end
1191
1192    allowed, reason = isRuntimeAllowed(runtime)
1193    if not allowed then
1194        decision.reason = reason
1195        return decision
1196    end
1197
1198    if not decision.available then
1199        decision.reason = "control_group"
1200        return decision
1201    end
1202
1203    local userTopLevel
1204    allowed, reason, userTopLevel = isUserLevelAllowed(runtime)
1205    decision.userTopLevel = userTopLevel
1206    if not allowed then
1207        decision.reason = reason
1208        return decision
1209    end
1210
1211    allowed, reason = isDifficultyAllowed(checkResult)
1212    if not allowed then
1213        decision.reason = reason
1214        return decision
1215    end
1216
1217    allowed, reason = isToplevelBaoDiAllowed(checkResult, decision.requireToplevelBaoDi)
1218    if not allowed then
1219        decision.reason = reason
1220        return decision
1221    end
1222
1223    local experienceRouting = groupConfig.experienceRouting
1224    if experienceRouting and experienceRouting.enabled == true then
1225        decision.experienceContext = getExperienceContext(
1226            runtime,
1227            checkResult,
1228            userTopLevel
1229        )
1230        local configKey, experienceType, routeReason = selectExperienceConfig(
1231            decision.experienceContext,
1232            experienceRouting,
1233            groupConfig.experienceMode
1234        )
1235        decision.experienceRouteReason = routeReason
1236        if experienceType then
1237            decision.experienceType = experienceType
1238            decision.configKey = configKey
1239        elseif routeReason == "experience_route_not_matched" then
1240            decision.reason = routeReason
1241            return decision
1242        end
1243    end
1244
1245    if decision.removeDiffFix and type(runtime.adjustData) ~= "table" then
1246        decision.reason = "adjust_data_missing"
1247        return decision
1248    end
1249
1250    if not isProphetConfigAvailable(decision.configKey) then
1251        decision.reason = "experience_config_missing"
1252        return decision
1253    end
1254
1255    decision.enabled = true
1256    decision.reason = "enabled"
1257    return decision
1258end

1306function ProphetLaunchStrategy:apply(runtime, resultDatas)
1307    local decision = self:evaluate(runtime, resultDatas)
1308    if not decision then
1309        return nil
1310    end
1311
1312    runtime.prophetLaunchInfo = decision
1313    if not decision.enabled then
1314        return decision
1315    end
1316
1317    local adjustLogic = runtime.UBFuuuAdjustLogic
1318    if not adjustLogic or not adjustLogic.setSelfFeedbackEnableFlag then
1319        decision.enabled = false
1320        decision.reason = "self_feedback_control_missing"
1321        return decision
1322    end
1323
1324    local selfFeedbackSourceName = decision.maintenanceName .. "_" .. decision.group
1325    adjustLogic:setSelfFeedbackEnableFlag(
1326        selfFeedbackSourceName,
1327        SELF_FEEDBACK_ENABLED_WHEN_PROPHET_LAUNCHED
1328    )
1329
1330    local checkResult = resultDatas and resultDatas.checkResult
1331    if checkResult and adjustLogic.getUbfuuuAdjustFixInfo then
1332        checkResult.adjustFixInfo = adjustLogic:getUbfuuuAdjustFixInfo()
1333    end
1334
1335    runtime:setProphet(decision.configKey)
1336
1337    if decision.removeDiffFix and runtime.adjustData then
1338        runtime.adjustData.diffFix = nil
1339    end
1340
1341    return decision
1342end

1344function ProphetLaunchStrategy:applySafely(runtime, resultDatas)
1345    local ok, result = pcall(self.apply, self, runtime, resultDatas)
1346    if ok then
1347        return result
1348    end
1349
1350    if he_log_error then
1351        he_log_error("ProphetLaunchStrategy_apply_error_" .. tostring(result))
1352    elseif printx then
1353        printx(0, "ProphetLaunchStrategy:applySafely error", tostring(result))
1354    else
1355        print("ProphetLaunchStrategy:applySafely error", tostring(result))
1356    end
1357    return nil
1358end
