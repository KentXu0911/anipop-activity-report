45function UBFuuuSLRuntimeAdjustLogic:_updatUbfuuuAdjustFixInfo(dataKey, sourceName, fixValue, fixMode, datas)
46    if not self.ubfuuuAdjustFixInfo then self.ubfuuuAdjustFixInfo = {} end
47    if not self.ubfuuuAdjustFixInfo[dataKey] then self.ubfuuuAdjustFixInfo[dataKey] = {} end
48    if not datas then datas = {} end
49    local _data = {
50        sourceName = sourceName,
51        value = fixValue or 0,
52        fixMode = fixMode or UBFuuuFixMode.kaddVaule,
53        rangeMin = datas.rangeMin,         --原始值不小于rangeMin，才会按fixMode尝试修改
54        rangeMax = datas.rangeMax,         --原始值不大于rangeMax，才会按fixMode尝试修改
55        resultMin = datas.resultMin,       --原始值如果大于等于resultMin，被fixMode尝试修改后，结果值小于resultMin时，会被重置为resultMin；如果原始值小于resultMin，则不做处理
56        resultMax = datas.resultMax,       --原始值如果小于等于resultMax，被fixMode尝试修改后，结果值大于resultMax时，会被重置为resultMax；如果原始值大于resultMax，则不做处理
57        minValue = datas.minValue,         --结果值不能小于minValue，否则会被重置为minValue
58        maxValue = datas.maxValue,         --结果值不能大于maxValue，否则会被重置为maxValue
59        stringValue = datas.stringValue,
60    }
61
62    if fixValue then
63        if type(fixValue) == "number" then
64            _data.value = fixValue
65        else
66            _data.value = 0
67            _data.stringValue = tostring(fixValue)
68        end
69    end
70    
71    
72    table.insert(self.ubfuuuAdjustFixInfo[dataKey], _data)
73end

150--[[
151-- 更新难度调节的篡改逻辑，篡改 加步数时的干预逻辑
152-- sourceName，来源名称，标识谁在调用此接口，打点用。活动的话默认传actId
153-- addSteps， 增加的步数里（不含前置加三步），多少步之内，需要执行干预
154-- addStepAdjustMode，篡改类型，枚举值为: UBFuuuAddStepAdjustMode
155]]
156local function normalizeAddStepIntervalAvailableSteps(value)
157    if type(value) ~= "number" then
158        return nil
159    end
160
161    if value < 0 then
162        return 0
163    end
164
165    return math.floor(value)
166end

168local function normalizeAddStepIntervalAdjustMode(value, defaultValue)
169    if value == UBFuuuAddStepAdjustMode.kDoNothing
170        or value == UBFuuuAddStepAdjustMode.kLockLastDs
171        or value == UBFuuuAddStepAdjustMode.kAddMaxDiffMoveByStep
172        or value == UBFuuuAddStepAdjustMode.kAddMaxDiffMoveByFixedValue
173        or value == UBFuuuAddStepAdjustMode.kLockMaxDiffMove then
174        return value
175    end
176
177    return defaultValue
178end

180local function normalizeAddStepIntervalTargetMaxdiffMove(value)
181    if type(value) == "number" then
182        return value
183    end
184
185    return nil
186end

188local function buildAddStepIntervalFixRow(sourceName, configOrAvailableSteps, defaultAdjustMode, targetMaxdiffMove, stackingPreValue)
189    local intervalConfig = type(configOrAvailableSteps) == "table" and configOrAvailableSteps or nil
190    local availableSteps = intervalConfig and intervalConfig.availableSteps or configOrAvailableSteps
191    local addStepAdjustMode = defaultAdjustMode
192    if intervalConfig and intervalConfig.addStepAdjustMode ~= nil then
193        addStepAdjustMode = intervalConfig.addStepAdjustMode
194    end
195
196    if addStepAdjustMode == nil then
197        addStepAdjustMode = UBFuuuAddStepAdjustMode.kAddMaxDiffMoveByStep
198    end
199
200    local finalTargetMaxdiffMove = targetMaxdiffMove
201    if intervalConfig and intervalConfig.targetMaxdiffMove ~= nil then
202        finalTargetMaxdiffMove = intervalConfig.targetMaxdiffMove
203    end
204
205    local finalStackingPreValue = stackingPreValue
206    if intervalConfig and intervalConfig.stackingPreValue ~= nil then
207        finalStackingPreValue = intervalConfig.stackingPreValue
208    end
209
210    return {
211        sourceName = sourceName,
212        addStepAdjustMode = normalizeAddStepIntervalAdjustMode(addStepAdjustMode, defaultAdjustMode or UBFuuuAddStepAdjustMode.kAddMaxDiffMoveByStep),
213        targetMaxdiffMove = normalizeAddStepIntervalTargetMaxdiffMove(finalTargetMaxdiffMove),
214        availableSteps = normalizeAddStepIntervalAvailableSteps(availableSteps),
215        stackingPreValue = finalStackingPreValue == true,
216    }
217end

219function UBFuuuSLRuntimeAdjustLogic:updateUbfuuuAdjustFixInfoByAddStep(sourceName, addSteps, addStepAdjustMode)
220    if not self.ubfuuuAdjustFixInfo then
221        self.ubfuuuAdjustFixInfo = {}
222    end
223
224    local availableSteps = 1
225    if type(addSteps) == "number" then
226        -- 这里的 addSteps 已经是“需要覆盖多少步”的计数，新模型消费端使用 stepIndex < availableSteps，
227        -- 因此写入时不能再额外 +1，否则 legacy 迁移后的后置区间会多生效一轮。
228        availableSteps = math.max(0, math.floor(addSteps))
229    end
230
231    self.ubfuuuAdjustFixInfo[UBFuuuFixType.kAddStepFix] = nil
232    self.ubfuuuAdjustFixInfo[UBFuuuFixType.kPreAddStepIntervalFix] = buildAddStepIntervalFixRow(
233        sourceName,
234        nil,
235        UBFuuuAddStepAdjustMode.kAddMaxDiffMoveByStep,
236        nil,
237        false
238    )
239    self.ubfuuuAdjustFixInfo[UBFuuuFixType.kPostAddStepIntervalFix] = buildAddStepIntervalFixRow(
240        sourceName,
241        availableSteps,
242        addStepAdjustMode or UBFuuuAddStepAdjustMode.kDoNothing,
243        nil,
244        false
245    )
246
247    return self.ubfuuuAdjustFixInfo[UBFuuuFixType.kPostAddStepIntervalFix]
248end
