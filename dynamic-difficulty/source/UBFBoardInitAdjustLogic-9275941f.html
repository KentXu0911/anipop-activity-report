400function UBFBoardInitAdjustLogic:doChangeBoardByVirtualSeed(resultData)
401
402    local boardContext = self.levelContext:getBoardContext(1)
403    local availableColorList = boardContext:getUseColors()
404
405    local r = resultData.centerR
406	local c = resultData.centerC
407
408	local configList = patternConfig["mode" .. tostring(resultData.typeIndex)]
409	local config = configList[resultData.patternIndex]
410
411	local resultPosList = {}
412
413	-- local boardModel = self.mainLogic:getBoardModel()
414	-- local itemMap = self.mainLogic:getItemMap()
415	-- local boardMap = self.mainLogic:getBoardMap()
416
417    local gridRaw = boardContext:getGridRawDataByPos(r, c)
418	if gridRaw then
419
420		-- local color = self.mainLogic:randomColor()
421        
422		local cIndex1 = UBFRandom:calcValue( boardContext:nextSeed(seedKey) , 1 , #availableColorList )
423		local cIndex2 = nil
424
425		if cIndex1 == #availableColorList then
426			cIndex2 = UBFRandom:calcValue( boardContext:nextSeed(seedKey) , 1 , #availableColorList - 1 )
427		else
428			cIndex2 = UBFRandom:calcValue( boardContext:nextSeed(seedKey) , cIndex1 + 1 , #availableColorList )
429		end
430
431		local color1 =  availableColorList[cIndex1]
432		local color2 =  availableColorList[cIndex2]
433
434        
435		local itemRaw = gridRaw.itemMaster
436        if itemRaw then
437            itemRaw.color = color1
438        end
439		if _needLog then printx( 1 , "UBFBoardInitAdjustLogic  doChangeBoardByVirtualSeed 111") end
440
441		for k,v in ipairs(config) do
442
443			local fixr = v.dr
444			local fixc = v.dc
445
446			local ra = r + fixr - 4
447			local ca = c + fixc - 4
448
449			if _needLog then printx( 1 , "UBFBoardInitAdjustLogic  doChangeBoardByVirtualSeed 222" , ra , ca) end
450            local aroundGridRaw = boardContext:getGridRawDataByPos(ra, ca)
451
452			if aroundGridRaw then
453				local aroundItemRaw = aroundGridRaw.itemMaster
454				if aroundItemRaw then
455                    if v.ct == 1 then
456                        aroundItemRaw.color = color1
457                    else
458                        aroundItemRaw.color = color2
459                    end
460                end
461			end
462		end
463	end
