10local function findUpAndLow(checklist, desiredDiff)
11
12    local _upK = nil
13    local _upV = nil
14    local _upDV = nil
15    
16    local _lowK = nil
17    local _lowV = nil
18    local _lowDV = nil
19    
20
21    for k, _data in ipairs(checklist) do
22        local _diff = _data.v
23        if _diff > desiredDiff then
24            if not _upDV then 
25                _upDV = _diff - desiredDiff 
26                _upK = _data.k
27                _upV = _diff
28            elseif _diff - desiredDiff < _upDV then
29                _upDV = _diff - desiredDiff
30                _upK = _data.k
31                _upV = _diff
32            end
33        else
34            if not _lowDV then 
35                _lowDV = desiredDiff - _diff 
36                _lowK = _data.k
37                _lowV = _diff
38            elseif desiredDiff - _diff < _lowDV then
39                _lowDV = desiredDiff - _diff
40                _lowK = _data.k
41                _lowV = _diff
42            end
43        end
44    end
45    return { upK = _upK, upV = _upV, upDV = _upDV, lowK = _lowK, lowV = _lowV, lowDV = _lowDV }

52local function randomUpOrLow( updv, lowdv )
53    -- updv = 28.984
54    -- lowdv = 0.095
55
56    local _r = updv / (updv + lowdv) --差值越大，随机到的概率越低
57    -- 1 - ( 28.984 / ( 28.984 + 0.095 ) )
58    -- 1 - ( 28.984 / ( 29.079 ) )
59    -- 1 - 0.99673
60    -- 0.0033
61    
62    if UBFuuuSelfFeedback.debugMode then printx(1,"selfFeedback ---------- randomUpOrLow  updv:" , updv, "lowdv:", lowdv) end
63    local _random = math.random()
64    if UBFuuuSelfFeedback.debugMode then printx(1,"selfFeedback ---------- randomUpOrLow  _r = 1 - (_updv / (_updv + lowdv) = " , _r , "  _random =" , _random ) end
65
66    -- _random > 0.99673
67    if _random > _r then
68        return 1
69    else
70        return 2
71    end

74local function checkDiffFixByLowDiff(_diffFix, _checkmap)
75    --[[
76        从diffFix0档开始尝试（当前diffFix设置为0）
77            如果 查表后难度 > 1.05
78                diffFix降低一档，重新比对
79            如果 1.03 <= 查表后难度 <= 1.05
80                如果 当前diffFix >= -2
81                    diffFix降低一档，重新比对
82                否则
83                    使用当前diffFix档位
84            如果 查表后难度 < 1.03
85                使用当前diffFix档位
86    ]]
87    local UBFuuuSelectAdjust = require("ubf.UBFGamePlayScene.logics.sceneModules.gameMod.modFolder.diffAdjust.ubfuuu.UBFuuuSelectAdjust")
88    local _range = UBFuuuSelectAdjust:getDiffFixRange()
89    if _diffFix <= _range.min then
90        return _range.min
91    end
92
93    local _diff = _checkmap[_diffFix]
94    if not _diff then
95        return _diffFix
96    elseif _diff > 1.05 then
97        return checkDiffFixByLowDiff( _diffFix-1, _checkmap )
98    elseif _diff > 1.03 then
99        if _diffFix >= -2 then
100            return checkDiffFixByLowDiff( _diffFix-1, _checkmap )
101        else
102            return _diffFix
103        end
104    else
105        return _diffFix
106    end
107end
