707function UBFuuuAdjustManager:parseFarmStarFailCountMap(rawData)
708    local _raw = rawData
709    if type(_raw) == "string" then
710        if _raw == "" then return {} end
711        local _parsed = table.deserialize(_raw)
712        if type(_parsed) ~= "table" then
713            he_log_error("UBFuuuAdjustManager_parseFarmStarFailCountMap_deserialize_fail")
714            return {}
715        end
716        _raw = _parsed
717    end
718
719    if type(_raw) ~= "table" then return {} end
720
721    local _map = {}
722    for k, v in pairs(_raw) do
723        local _levelId = tonumber(k)
724        local _count = tonumber(v)
725        if _levelId and _levelId > 0 and _count and _count >= 0 and _count == math.floor(_count) then
726            _map[tostring(_levelId)] = _count
727        else
728            he_log_error("UBFuuuAdjustManager_parseFarmStarFailCountMap_invalid k=" .. tostring(k) .. " v=" .. tostring(v))
729        end
730    end
731    return _map

753-- 纯函数（可单测）：根据结算 runtime 的最终状态决定刷星计数动作
754-- state = { isWin, star, oldStar, failReason }
755-- 刷星口径与决策侧一致：只看星级数（首闯 0 星、满星刷分不参与），不依赖关卡类型字段
756-- 返回 "reset"（升星清零）/ "increment"（未升星正常完成）/ nil（不更新）
757function UBFuuuAdjustManager.resolveFarmStarResultAction(state)
758    if type(state) ~= "table" then return nil end
759
760    local _oldStar = tonumber(state.oldStar) or 0
761    -- 首闯（0 星）与满星刷分（>= maxLevelStar）不参与刷星计数
762    if _oldStar <= 0 or _oldStar >= _max_level_star then return nil end
763
764    local _failReason = state.failReason
765    local _failReasonConst = _G.UBFGameFailReason or {}
766    -- 中途退出 / 重玩不计入
767    if _failReason == _failReasonConst.kDropout or _failReason == _failReasonConst.kRetry then
768        return nil
769    end
770
771    local _isWin = state.isWin == true
772    local _isNormalFail = (not _isWin) and (_failReason == _failReasonConst.kNoLife
773        or _failReason == _failReasonConst.kDeadBoard
774        or _failReason == _failReasonConst.kStarLack)
775
776    -- 既不是正常过关也不是正常失败（含闪退恢复失败等）：fail-closed
777    if (not _isWin) and (not _isNormalFail) then return nil end
778
779    local _star = tonumber(state.star) or 0
780    if _isWin and _star > _oldStar then
781        return "reset"
782    end
783
784    return "increment"
785end

797    if _oldCount < 0 then _oldCount = 0 end

821-- 纯函数（可单测）：服务端下发与本地缓存合并，返回新表
822-- 服务端下发该 key 即覆盖；未下发的关卡仍保留本地乐观值
823function UBFuuuAdjustManager.mergeFarmStarFailCountMap(localMap, serverMap, nowTime)
824    local _result = {}

826        if type(v) == "table" then

828                count = v.count,
829                localRevision = v.localRevision,
830                updateTime = v.updateTime,
831            }
832        end
833    end
834
835    local _now = nowTime or 0
836    for k, _serverCount in pairs(serverMap or {}) do
837        local _local = _result[k]
838        if not _local then
839            _result[k] = { count = _serverCount, updateTime = _now }
840        else
841            _local.count = _serverCount
842            _local.updateTime = _now
843        end
844    end
845
846    return _result

849-- 冷数据审计：仅当 serverMap 存在（即最近一次服务端下发）时才按 30 天窗口清理
850function UBFuuuAdjustManager:auditFarmStarFailCountMap(_map, serverMap)
851    local _now = self:timeInSec()
852    local _changed = false
853
854    for k, v in pairs(_map or {}) do
855        if type(v) ~= "table" then
856            _map[k] = nil
857            _changed = true
858        else
859            if v.pending ~= nil or v.pendingAction ~= nil or v.pendingPlayId ~= nil then
860                v.pending, v.pendingAction, v.pendingPlayId = nil, nil, nil
861                _changed = true
862            end
863            if not v.updateTime then
864                v.updateTime = _now
865                _changed = true
866            elseif serverMap and (not serverMap[k]) and (_now - v.updateTime > _farm_star_fail_count_expireTime) then
867                _map[k] = nil
868                _changed = true
869            end
870        end
871    end
872
873    return _changed
874end

876function UBFuuuAdjustManager:trimFarmStarFailCountMap(_map)
877    local _total = 0
878    for _ in pairs(_map or {}) do _total = _total + 1 end
879    if _total <= _farm_star_fail_count_maxKeep then return false end
880
881    local _candidates = {}
882    for k, v in pairs(_map) do
883        table.insert(_candidates, { key = k, time = (type(v) == "table" and v.updateTime) or 0 })
884    end
885    table.sort(_candidates, function(a, b) return a.time < b.time end)
886
887    local _needRemove = _total - _farm_star_fail_count_maxKeep
888    for i = 1, math.min(_needRemove, #_candidates) do
889        _map[_candidates[i].key] = nil
890    end
891
892    return true
893end
