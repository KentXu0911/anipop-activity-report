815                    lowDiff: -0.7 - -1 = 0.3

1057function UBFuuuSelectAdjust:changeValueByAdjustFixInfo(adjustContext, adjustFixInfo, originValue)
1058
1059    local _returnValue = originValue
1060    local _inRange = true
1061    adjustFixInfo.input = originValue
1062
1063    if adjustFixInfo.rangeMin and originValue < adjustFixInfo.rangeMin then
1064        _inRange = false
1065    end
1066    if adjustFixInfo.rangeMax and originValue > adjustFixInfo.rangeMax then
1067        _inRange = false
1068    end
1069
1070    if _inRange then
1071
1072        local _newValue = originValue
1073
1074        if adjustFixInfo.fixMode == UBFuuuFixMode.kaddVaule then
1075            _newValue = _newValue + (adjustFixInfo.value or 0)
1076        elseif adjustFixInfo.fixMode == UBFuuuFixMode.kMultiplyVaule then
1077            _newValue = math.floor( _newValue * (adjustFixInfo.value or 0) * 1000 ) / 1000
1078        elseif adjustFixInfo.fixMode == UBFuuuFixMode.kSetVaule then
1079            _newValue = adjustFixInfo.value or 0
1080        elseif (
1081                    adjustFixInfo.fixMode == UBFuuuFixMode.kMapping
1082                    or adjustFixInfo.fixMode == UBFuuuFixMode.kMappingAddVaule
1083                    or adjustFixInfo.fixMode == UBFuuuFixMode.kMappingMultiplyValue
1084                ) 
1085                and adjustFixInfo.stringValue then
1086
1087            local _mappingConfig = require("ubf.metaConfig." .. tostring(adjustFixInfo.stringValue) .. "_config")
1088            if _mappingConfig then
1089                local configCAGroup = self:__getConfigGroupByFileName( adjustContext, adjustFixInfo.stringValue )
1090                local _mappingResult = self:_findValueInMap(_mappingConfig, originValue, configCAGroup or 0)
1091
1092                -- _mappingResult 为 nil 时保持 _newValue 不变，等价无调整
1093                if _mappingResult ~= nil then
1094                    if adjustFixInfo.fixMode == UBFuuuFixMode.kMapping then
1095                        _newValue = _mappingResult
1096                    elseif adjustFixInfo.fixMode == UBFuuuFixMode.kMappingAddVaule then
1097                        _newValue = _newValue + _mappingResult
1098                    elseif adjustFixInfo.fixMode == UBFuuuFixMode.kMappingMultiplyValue then
1099                        _newValue = math.floor( _newValue * _mappingResult * 1000 ) / 1000
1100                    else
1101                        _newValue = _mappingResult
1102                    end
1103                end
1104            end
1105        end
1106
1107        if adjustFixInfo.resultMin then
1108            if originValue >= adjustFixInfo.resultMin and _newValue < adjustFixInfo.resultMin then
1109                _newValue = adjustFixInfo.resultMin
1110            elseif originValue < adjustFixInfo.resultMin then
1111                _newValue = nil
1112            end
1113        end
1114
1115        if _newValue and adjustFixInfo.resultMax then
1116            if originValue <= adjustFixInfo.resultMax and _newValue > adjustFixInfo.resultMax then
1117                _newValue = adjustFixInfo.resultMax
1118            elseif originValue > adjustFixInfo.resultMax then
1119                _newValue = nil
1120            end
1121        end
1122
1123        if _newValue then _returnValue = _newValue end
1124
1125        if adjustFixInfo.minValue and _returnValue < adjustFixInfo.minValue then
1126            _returnValue = adjustFixInfo.minValue
1127        end
1128        if adjustFixInfo.maxValue and _returnValue > adjustFixInfo.maxValue then
1129            _returnValue = adjustFixInfo.maxValue
1130        end
1131    end

1133    adjustFixInfo.output = _returnValue
