210function UBFELSendPassLevelProcess:passLevel()
211    if FeedContext.shouldPlayTrialLevel() then -- 体验关不需要真实过关流程
212        self:success()
213        return
214    end
215
216    local runtime = self:get("runtime")
217	local win = runtime:isWin()
218
219	--云游戏失败成功不发请求，重玩退出发
220	if _G.isYunGame then
221		local giveUp = self:isFailByGiveUp()
222		local needRetry = self:isNeedRetry()
223		if giveUp then
224			self:passLevelGiveUp()
225		elseif needRetry then
226			self:passLevelRetry()
227		else
228			self:success()
229		end
230	else
231		if win then
232			self:passLevelWin()
233		else
234			local giveUp = self:isFailByGiveUp()
235			local needRetry = self:isNeedRetry()
236			-- printx(16,giveUp,needRetry,"-----------------------")
237			if giveUp then
238				self:passLevelGiveUp()
239			elseif needRetry then
240				self:passLevelRetry()
241			else
242				self:passLevelFail()
243			end
244		end
245	end
246end
