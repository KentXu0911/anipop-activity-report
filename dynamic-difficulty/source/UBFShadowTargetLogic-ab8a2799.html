298function UBFShadowTargetLogic:_calBoard( itemType,fieldsCheck,isGlobal )
299	local boardContext = self.levelContext:getBoardContext(self.boardId)
300
301	local filterData = {
302		isGlobal = isGlobal,
303        itemType = itemType,
304        fieldCond = {}
305
306	}
307	for k,check in ipairs(fieldsCheck or {}) do 
308		if check.value then
309			table.insert(filterData.fieldCond,{
310				field = check.field,
311				mode = UBFBoardFilterConst.kValue,
312				val = check.value
313			})
314		elseif check.range then
315			table.insert(filterData.fieldCond,{
316				field = check.field,
317				mode = UBFBoardFilterConst.kRange,
318				min = check.range[1] , 
319				max = check.range[2] ,
320			})
321		end
322	end
323    local baseItemRaws = boardContext:filter(filterData)
324
325
326    if baseItemRaws then
327    	return #baseItemRaws
328    else
329    	return 0
330    end
331
332    boardContext:release()

405function UBFShadowTargetLogic:checkBindingItemType( watchConfig,itemContextRaw )
406	if not watchConfig then return end
407	if not watchConfig.watch then return end
408	--没有绑定直接返回true
409	if not watchConfig.watch.bindingItemType then return true end
410
411	local boardContext = self.levelContext:getBoardContext(self.boardId)
412	local bindingItemType = watchConfig.watch.bindingItemType
413	local itemContext = boardContext:getItemContextByPosType( itemContextRaw.pos,bindingItemType)
414
415	if not itemContext then
416		local pos = itemContextRaw.pos
417		local gridContext = boardContext:getGridContextByPos( pos.r , pos.c , pos.d )
418		local bodyRows = gridContext:getItemContextListByItemType( UBFGameItemType.kBody )
419		for _,bodyRow in ipairs(bodyRows) do 
420			local headItemRaw = UBFFeatureHelper:getHomologousHead(boardContext, bodyRow)
421			if headItemRaw and headItemRaw.itemType == bindingItemType then
422				itemContext = headItemRaw
423				break
424			end
425		end
426
427		gridContext:release()
428	end
429
430	boardContext:release()
431
432	return not (not itemContext)
433end

435function UBFShadowTargetLogic:checkWatchConfigFields( watchConfig,itemContextRaw )
436	if watchConfig.watch.fieldsCheck then
437		local fieldsCheck = watchConfig.watch.fieldsCheck
438		for _,eachrule in ipairs(fieldsCheck) do 
439            if not eachrule.field then return false end
440            local value = itemContextRaw[eachrule.field]
441            if eachrule.range then 
442                local min = eachrule.range[1] or 0
443                local max = eachrule.range[2] or 0xffff
444                if value < min or value > max then return false end
445            else
446                if value ~= eachrule.value then return false end
447            end
448		end
449		return true
450	end
451	return true
452end

582function UBFShadowTargetLogic:doWatchAction( watchData,watchConfig,params )
583	local action = watchConfig.action
584	local coverage = action.coverage
585	local resultRange = action.resultRange
586	local resultOffset = action.resultOffset
587	local flip = action.flip
588
589	lua_switch(action.mode){
590		[UBFShadawTargetActionMode.FIELD_DIFF] = function ( ... )
591			local diff =  params.diff
592			local scale =  action.scale or 1
593			if coverage and (not checkCoverage(coverage,diff)) then return end
594			watchData.cur = watchData.cur + diff * scale
595		end,
596		[UBFShadawTargetActionMode.NUM_CHANGE] = function ( ... )
597			local value =  action.value
598			local scale =  action.scale or 1
599			watchData.cur = watchData.cur + value * scale
600		end,
601		[UBFShadawTargetActionMode.CUR_LEVEL] = function ( ... )
602			local scale =  action.scale or 1
603			if coverage and (not checkCoverage(coverage,params.level)) then return end
604			watchData.cur = watchData.cur + params.level * scale
605		end,
606		[UBFShadawTargetActionMode.PATH_POS] = function ( ... )
607			self:calBindingPath(watchData,watchConfig)
608		end,
609		[UBFShadawTargetActionMode.PATH_POS_TO_END] = function ( ... )
610			self:calPathPosToEnd(watchData, watchConfig)
611		end,
612		[UBFShadawTargetActionMode.INIT_PATH_LEN] = function ( ... )
613			self:calPathLength(watchData,watchConfig)
614		end,
615	}
616
617	-- printx(16,flip,watchData.initNum,watchConfig.watch.itemType)
618	if flip and watchData.initNum then
619		watchData.flipNum  = watchData.initNum - watchData.cur
620	end
621	if flip and watchData.initNumMap then
622		if not watchData.flipNumMap then
623		 	watchData.flipNumMap = {}
624		 end
625
626		for id,cur in pairs(watchData.curMap) do
627			local initNum = watchData.initNumMap[id]
628			if initNum then
629				watchData.flipNumMap[id] = initNum - cur
630			end
631		end
632		-- printx(16,tts(watchData))
633	end
634
635
636	if resultOffset then
637		self:calResultOffset(watchData,resultOffset)
638	end
639
640	if resultRange then
641		self:ensureResultRange(watchData,resultRange)
642	end

702			end
703		end
704	end
705
706
707	local shadowTargetsMap = self:getShadowTargetsMap()
708	for shadowTargetType,watchDataList in pairs(shadowTargetsMap) do 
709		local watchConfigs = UBFShadowTargetConfig[shadowTargetType]
710		for index,watchData in ipairs(watchDataList) do 
711			local watchConfig = watchConfigs[index]
712			-- printx(16,watchConfig.watch.recordInit,"watchConfig.watch.recordInit")
713			if watchConfig.watch.recordInit then
714				self:doWatchAction(watchData,watchConfig)
715				-- printx(16,tts(watchData))
716				reInitByCur(watchData)
717				-- printx(16,tts(watchData))
718			end
719		end
720	end
721
722end

724local function convert2fieldCond( fieldsCheck )
725	-- body
726	local fieldCond = {}
727	for _,rule in ipairs(fieldsCheck) do
728		if rule.value then
729			table.insert(fieldCond,{
730            	 val = rule.value , 
731            	 field = rule.field , 
732            	 mode = UBFBoardFilterConst.kValue,
733			})	
734		elseif rule.range then
735			table.insert(fieldCond,{ 
736				min = rule.range[1], 
737				max = rule.range[2] , 
738				field = rule.field , 
739				mode = UBFBoardFilterConst.kRange ,
740			})
741		end
742	end
743	return fieldCond
744end

818function UBFShadowTargetLogic:calPathLength( watchData,watchConfig )
819	local boardContext = self.levelContext:getBoardContext(self.boardId)
820
821	local itemType = watchConfig.watch.itemType
822    local baseItemRaws = boardContext:filter({
823        isGlobal = true,
824        itemType = itemType, 
825    })
826    local visited = {}
827	for _,itemRaw in ipairs(baseItemRaws) do 
828        local startItemRow =  UBFFeatureHelper:getRouteStartItemContextRaw(boardContext, itemRaw)
829        if not visited[startItemRow.id] then
830        	-- printx(16,tts(startItemRow.route))
831        	local currLength = startItemRow.route.routeRef.currLength
832        	local initLength = startItemRow.route.routeRef.initLength
833
834        	visited[startItemRow.id] = true
835        	watchData.curMap[startItemRow.id] = (initLength - currLength)
836        end
837    end
838	boardContext:release()
