73function UBFGameResumeLogic:checkCanResume(callback)
74	if log then log("checkCanResume") end
75	if not self:hasResumeData() then
76		--没有数据，本身就没法恢复
77		callback(false,nil)
78		return
79	end
80
81	-- 检查恢复条件失败时的处理逻辑
82	local function onCheckFail(errCode)
83		if log then log("onCheckFail",errCode) end
84		-- 数据处理，断连胜、发协议等
85		self:afterNotResume(false,errCode)
86		-- 回调，通知强弹框架
87		callback(false,errCode)
88	end
89
90	-- 检查基础恢复条件
91	local function checkBase(nextAction)
92		if log then log("checkBase") end
93
94		self._checkResult = self:_checkCanResume()
95		if self._checkResult.errCode ~= 0 then
96			onCheckFail(self._checkResult.errCode)
97			return
98		end
99		nextAction()
100	end
101
102	-- 发送成功回调
103	local function sendSuccess()
104		if log then log("checkCanResume sendSuccess") end
105		callback(true)
106	end
107
108	local list = {
109		checkBase,
110		-- checkResumeTimes,
111		sendSuccess
112	}
113	runSequence(list)
114end

305-- 检查mod数据是否允许闪退恢复的逻辑
306function UBFGameResumeLogic:_resumeModCheck(resumeData)
307	local levelInfo = resumeData:getLevelInfo()
308	local modList = levelInfo and levelInfo.modList or {}
309	for _, v in ipairs(modList) do
310		-- 只判断活动
311		if tonumber(v.sourceId) and tonumber(v.sourceId) > 0 then
312			local isActMod = v.mType and v.mType == UBFPlayModBuilder.BuffBuilderEnum.kMod
313			local actConfig = UBFActHelper:getActConfig(v.sourceId)
314			if isActMod and not actConfig then
315				local resumeCurMd5 = resumeData:getCurMd5()
316				local msg = string.format("actConfig has been deleted — game progress recovery not possible. sourceId: %s, beginTime: %s, resumeMd5: %s", v.sourceId, levelInfo.beginTime, resumeCurMd5)
317				if EnvUtil:isDevelopMode() then
318					UBFLogW(msg)
319					CommonTip:showTip(msg)
320				else
321					he_log_error(msg)
322				end
323				return true
324			end
325		end
326	end
327end

650	local function checkResumeTimes(nextAction)
651		if log then log("checkResumeTimes") end
652
653		local function onSuccess()
654			if log then log("checkResumeTimes success") end
655			nextAction()
656		end
657		local function onFail()
658			if log then log("checkResumeTimes onFail") end

662

670				})
671			else CommonTip:showTip(text.tip) end
672		end
673
674		local resumeData = UBFResumeData:create(self:getResumeData())
675		local function isResumeTimesValid()
676			local resumed = resumeData:isResumed()
677			if log then log("checkResumeTimes isResumeTimesValid resumed",resumed,resumeData,resumeData:getResumeCount()) end
678
679			if resumed then
680				local resumeCount = resumeData:getResumeCount()
681				local usePropCount = resumeData:getUsePropCount()
682				if log then log("checkResumeTimes isResumeTimesValid", resumeCount, usePropCount) end
683				if resumeCount <= 5 then return true
684				elseif usePropCount > 0 then return true end
685			end
686		end
687
688		if isResumeTimesValid() then
689			onSuccess()
690		else
691			if log then log("checkResumeTimes not valid") end
692			local levelId = resumeData:getLevelId()
693			local md5str = HeMathUtils:md5(tostring(resumeData:getPlayId()))
694			UBFProtocolMethod:send {
695				method = UBFGameMethodType.kResumeLevel,
696				body = {
697					md5 = md5str
698				},
699			} : next(function(response)
700				if log then log("checkResumeTimes not valid response",table.tostring(response)) end
701
702				if response.byCancel then return self:cancelResume() end
703				if response.err then return onFail() end
704				return onSuccess()
705			end)
706		end
707	end
708
709	checkResumeTimes(onSuccess)
710end

731function UBFGameResumeLogic:afterNotResume( bFromCancle, errCode )
732	if log then log("afterNotResume",bFromCancle,errCode) end
733
734	if bFromCancle == nil then bFromCancle = false end
735	local resumeData = UBFResumeData:create(self:getResumeData())
736	local uid = resumeData:getUid()
737	local playId = resumeData:getPlayId()
738	local levelId = resumeData:getLevelId()
739	local levelType = resumeData:getLevelType()
740	local levelScene = resumeData:getLevelScene()
741	local md5str = HeMathUtils:md5( tostring(playId) )
742	local udid = resumeData:getUdid()
743	local sceneData = resumeData:getSceneData()
744
745	if PreBuffLogic and PreBuffLogic.isInited then
746		PreBuffLogic:onPassLevel( levelId , 0 , GameGuide:isNoPreBuffLevel(levelId) , false , "crash" )
747	end
748
749	-- 失败 断连胜
750	local topLevelId = UserService:getInstance().user:getTopLevelId()
751	if levelId == topLevelId then
752		if UserManager:getInstance():isTopNoPassLevel(levelId) then
753			UserManager:getInstance().userExtend:resetTopLevelWinCount()
754			UserService:getInstance().userExtend:resetTopLevelWinCount()
755			UserManager:getInstance().userExtend:resetTopLevelLogicalWinCount()
756			UserService:getInstance().userExtend:resetTopLevelLogicalWinCount()
757		end
758		UserManager:getInstance().userExtend:resetTopLevelContinueWinCount()
759		UserService:getInstance().userExtend:resetTopLevelContinueWinCount()
760		if BarrageAchievementUtil then
761			BarrageAchievementUtil:checkAndUpdatePassDataOnPassLevel(false)
762		end
763	end
764
765	local logicalFail = false
766	local curMoves =  sceneData.moveCount or 0
767	local levelInfo = UBFLevelConfigManager:userConfig():getLevelBaseInfo(levelId, levelScene)
768	local totalMoves = levelInfo and levelInfo[UBFLevelBaseInfoKey.STATIC_STEP] or 0
769	logicalFail = curMoves >= totalMoves * 0.5
770
771	if logicalFail then
772		-- 只加逻辑失败，是因为普通失败已经在进关时加过了
773		UserManager:getInstance().userExtend:incrTopLevelLogicalFailCount()
774		UserService:getInstance().userExtend:incrTopLevelLogicalFailCount()
775	end
776	-- LocalBox:setData( "buffOnResumeFlag" , false )
777	local cancelCrashReason = errCode
778	local levelInfo = resumeData:getLevelInfo()
779	local usePropList = levelInfo and levelInfo.usePropList or {}
780	local usedProps = {}
781	for i, v in ipairs(usePropList) do
782		local found = false
783		for i2, v2 in ipairs(usedProps) do
784			if v2.itemId == v.propId then
785				found = true
786				v2.num = v2.num + 1
787				break
788			end
789		end
790		if not found then
791			table.insert(usedProps, {itemId = v.propId, num = 1})
792		end
793	end
794	local _data = {
795		levelId = levelId,
796		levelType = levelType,
797		levelScene = levelScene,
798		gameData = nil,
799		resumeData = resumeData,
800		checkFailed = false,
801		json = {},
802		--下面的CancelCrashResumePanel协议结束时的回调，活动可在此处同步最新的数据（连胜）
803		callbacks = {},
804		cancelCrashReason = cancelCrashReason,
805	}
806
807	self:sendResumeFail(_data, logicalFail, levelScene, playId, usedProps)
808
809	self:_onOverByNotResume()
810
811	-- 打点
812	self:dcFail(errCode)
813	-- 清除闪退数据
814	UBFSectionResumeHelper:removeSavedSection()
815
816	self._lastResumeFailData = _data
817end

820function UBFGameResumeLogic:getLevelFarmingType(levelInfo)
821	local levelId = levelInfo.levelId
822	local levelType = levelInfo.levelType
823	local topLevelId = levelInfo.topLevelId
824	local oldStar = levelInfo.oldStar or 0
825
826	local levelFarmingType = UBFLevelFarmingType.NORMAL
827	if levelType == GameLevelType.kMainLevel then
828		if oldStar >= 4 then
829			levelFarmingType = UBFLevelFarmingType.FARM_SCORE
830		elseif oldStar > 0 then
831			levelFarmingType = UBFLevelFarmingType.FARM_STAR
832		else
833			if levelId == topLevelId then
834				levelFarmingType = UBFLevelFarmingType.NORMAL
835			else
836				levelFarmingType = UBFLevelFarmingType.FARM_STAR
837			end
838		end
839	elseif levelType == GameLevelType.kHiddenLevel then
840		if oldStar >= 3 then
841			levelFarmingType = UBFLevelFarmingType.FARM_SCORE
842		elseif oldStar > 0 then
843			levelFarmingType = UBFLevelFarmingType.FARM_STAR
844		else
845			levelFarmingType = UBFLevelFarmingType.NORMAL
846		end
847	end
848	return levelFarmingType
849end
