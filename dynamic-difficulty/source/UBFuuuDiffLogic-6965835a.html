581-- 计算后置加步区间的基础贡献值。
582-- 后置区间的 kAddMaxDiffMoveByStep 从 1 开始递增；如果 stackingPreValue=true，
583-- 需要先加上前置区间已经贡献过的值，让后置加步接着前置加步继续累加。
584-- 例：无前置加步时基础贡献为 0，后置第 1 步为 +1。
585-- 例：旧逻辑前置 3 步贡献到 +3，后置第 1 步为 +4。
586-- 例：step_item_adjust 弱化前置 3 步贡献到 +1，后置第 1 步为 +2。
587local function calculatePostBaseContribution(preAddStepIntervalFix, preAddStepCount)
588    if type(preAddStepCount) ~= "number" or preAddStepCount <= 0 then
589        return 0
590    end
591
592    local preAddStepCountInt = math.floor(preAddStepCount)
593    if not preAddStepIntervalFix then
594        return preAddStepCountInt
595    end
596
597    if preAddStepIntervalFix.addStepAdjustMode == UBFuuuAddStepAdjustMode.kAddMaxDiffMoveByStep then
598        local availableSteps = preAddStepIntervalFix.availableSteps
599        if type(availableSteps) == "number" then
600            return math.min(preAddStepCountInt, availableSteps)
601        end
602    elseif preAddStepIntervalFix.addStepAdjustMode == UBFuuuAddStepAdjustMode.kAddMaxDiffMoveByFixedValue then
603        local targetMaxdiffMove = preAddStepIntervalFix.targetMaxdiffMove
604        if type(targetMaxdiffMove) == "number" then
605            return targetMaxdiffMove
606        end
607    elseif preAddStepIntervalFix.addStepAdjustMode == UBFuuuAddStepAdjustMode.kLockMaxDiffMove then
608        local targetMaxdiffMove = preAddStepIntervalFix.targetMaxdiffMove
609        if type(targetMaxdiffMove) == "number" then
610            return targetMaxdiffMove
611        end
612    end
613
614    return 0
615end

617-- 判断当前区间配置是否覆盖当前步。
618local function isAddStepIntervalFixActive(intervalFixConfig, stepIndexInInterval)
619    if not intervalFixConfig then
620        return false
621    end
622
623    local availableSteps = intervalFixConfig.availableSteps
624    if type(availableSteps) == "number" and (stepIndexInInterval < 0 or stepIndexInInterval >= availableSteps) then
625        return false
626    end
627
628    return true
629end

632function UBFuuuDiffLogic:calculateDeviation(gameBoardLogic, adJustRawData, lineFix, moveCount, staticMoveCount, preAddSteps)

634    local resultData = {}
635
636    local _fixedMoveCount = math.min( moveCount, staticMoveCount )
637
638    local fixedLineFix = lineFix
639    local _lineFixUp = math.ceil(lineFix)

650

652    local fixedMaxDiffMove = nil
653
654    if _debug then
655        table.insert( _G.UBFuuuDebugCore.showPrintInfo, tostring(moveCount) .. "," )
656    end
657
658    maxDiffMove = self:__calculateMaxDiffMove(gameBoardLogic, moveCount, staticMoveCount, nil, adJustRawData).maxDiffMove
659    fixedMaxDiffMove = maxDiffMove or 0
660
661    if _debug then printx( 1 , "[U5] UBFuuuDiffLogic:calculateDeviation  fixedMaxDiffMove:", fixedMaxDiffMove, "maxDiffMove:", maxDiffMove, "preAddSteps:", preAddSteps ) end
662

664    local _preAddStepIntervalFix = nil
665    local _postAddStepIntervalFix = nil

675    local _postBaseContribution = calculatePostBaseContribution(_preAddStepIntervalFix, _preAddStepCount)

677    if _debug then
678        UBFLogD( "[U5] UBFuuuDiffLogic:calculateDeviation  preAddFixMode:", _preAddStepIntervalFix and _preAddStepIntervalFix.addStepAdjustMode or "nil",
679                    "preAddAvailableSteps:", _preAddStepIntervalFix and _preAddStepIntervalFix.availableSteps or "nil",
680                    "postAddFixMode:", _postAddStepIntervalFix and _postAddStepIntervalFix.addStepAdjustMode or "nil",
681                    "postAddAvailableSteps:", _postAddStepIntervalFix and _postAddStepIntervalFix.availableSteps or "nil",
682                    "postBaseContribution:", _postBaseContribution )
683    end
684
685    if moveCount >= staticMoveCount then
686        local _activeFixType = nil
687        local _addStepIndexAfterStaticMove = moveCount - staticMoveCount
688
689        -- 静态步数后的前 preAddStepCount 步属于前置加步区间。
690        if _preAddStepCount > 0 and _addStepIndexAfterStaticMove >= 0 and _addStepIndexAfterStaticMove < _preAddStepCount then
691            _activeFixType = UBFuuuFixType.kPreAddStepIntervalFix
692            if isAddStepIntervalFixActive(_preAddStepIntervalFix, _addStepIndexAfterStaticMove) then
693                local _preAddStepAdjustMode = _preAddStepIntervalFix.addStepAdjustMode
694                if _preAddStepAdjustMode == UBFuuuAddStepAdjustMode.kDoNothing then
695                    -- 配置明确要求前置加步区间不干预。
696                    fixedMaxDiffMove = 0
697                    adJustRawData.enableAdjust = false
698                elseif _preAddStepAdjustMode == UBFuuuAddStepAdjustMode.kLockLastDs then
699                    -- 沿用旧活动语义：本步计算出的 deviation 作为后续 lockedDeviation。
700                    _needLockDeviation = true
701                elseif _preAddStepAdjustMode == UBFuuuAddStepAdjustMode.kLockMaxDiffMove then
702                    -- 锁定到固定 MaxDiffMove 目标值。
703                    if type(_preAddStepIntervalFix.targetMaxdiffMove) == "number" then
704                        fixedMaxDiffMove = _preAddStepIntervalFix.targetMaxdiffMove
705                    end
706                elseif _preAddStepAdjustMode == UBFuuuAddStepAdjustMode.kAddMaxDiffMoveByFixedValue then
707                    -- step_item_adjust 前置弱化使用这个模式，把前置加步每一步都固定贡献同一个值。
708                    if type(_preAddStepIntervalFix.targetMaxdiffMove) == "number" then
709                        fixedMaxDiffMove = fixedMaxDiffMove + _preAddStepIntervalFix.targetMaxdiffMove
710                    end
711                elseif _preAddStepAdjustMode == UBFuuuAddStepAdjustMode.kAddMaxDiffMoveByStep then
712                    -- 按前置区间内第几步递增贡献；如配置要求，也允许叠加 postBaseContribution。
713                    local _stepContribution = _addStepIndexAfterStaticMove + 1
714                    if _preAddStepIntervalFix.stackingPreValue == true then
715                        _stepContribution = _stepContribution + _postBaseContribution
716                    end
717                    fixedMaxDiffMove = fixedMaxDiffMove + _stepContribution
718                end
719            else
720                -- 前置区间配置不存在或未覆盖当前步时，保持旧逻辑：第 1/2/3 步分别贡献 +1/+2/+3。
721                fixedMaxDiffMove = fixedMaxDiffMove + _addStepIndexAfterStaticMove + 1
722            end
723        else
724            -- 前置区间结束后进入后置加步区间；无后置配置时沿用旧逻辑关闭调节。
725            local _postIntervalStepIndex = moveCount - staticMoveCount - _preAddStepCount
726            _activeFixType = UBFuuuFixType.kPostAddStepIntervalFix
727            if isAddStepIntervalFixActive(_postAddStepIntervalFix, _postIntervalStepIndex) then
728                local _postAddStepAdjustMode = _postAddStepIntervalFix.addStepAdjustMode
729                if _postAddStepAdjustMode == UBFuuuAddStepAdjustMode.kDoNothing then
730                    -- 配置明确要求后置加步区间不干预。
731                    fixedMaxDiffMove = 0
732                    adJustRawData.enableAdjust = false
733                elseif _postAddStepAdjustMode == UBFuuuAddStepAdjustMode.kLockLastDs then
734                    -- 沿用旧活动语义：本步计算出的 deviation 作为后续 lockedDeviation。
735                    _needLockDeviation = true
736                elseif _postAddStepAdjustMode == UBFuuuAddStepAdjustMode.kLockMaxDiffMove then
737                    if type(_postAddStepIntervalFix.targetMaxdiffMove) == "number" then
738                        fixedMaxDiffMove = _postAddStepIntervalFix.targetMaxdiffMove
739                    end
740                elseif _postAddStepAdjustMode == UBFuuuAddStepAdjustMode.kAddMaxDiffMoveByFixedValue then
741                    if type(_postAddStepIntervalFix.targetMaxdiffMove) == "number" then
742                        fixedMaxDiffMove = fixedMaxDiffMove + _postAddStepIntervalFix.targetMaxdiffMove
743                    end
744                elseif _postAddStepAdjustMode == UBFuuuAddStepAdjustMode.kAddMaxDiffMoveByStep then
745                    -- 后置区间按步递增；stackingPreValue=true 时接着前置区间贡献继续累加。
746                    local _stepContribution = _postIntervalStepIndex + 1
747                    if _postAddStepIntervalFix.stackingPreValue == true then
748                        _stepContribution = _stepContribution + _postBaseContribution
749                    end
750                    fixedMaxDiffMove = fixedMaxDiffMove + _stepContribution
751                end
752            else
753                -- 后置区间配置不存在或未覆盖当前步时，保持旧逻辑：关闭 fuuu 调节。
754                fixedMaxDiffMove = 0
755                adJustRawData.enableAdjust = false
756            end
757        end
758
759        if _activeFixType == UBFuuuFixType.kPreAddStepIntervalFix then
760            if _debug then printx( 1 , "[U5] UBFuuuDiffLogic:calculateDeviation  61.1 fixedMaxDiffMove:", fixedMaxDiffMove, "moveCount:", moveCount, "staticMoveCount:", staticMoveCount ) end
761            if _debug then printx( 1 , "[U5] UBFuuuDiffLogic:calculateDeviation  61.1.2 fixedMaxDiffMove:", fixedMaxDiffMove ) end
762        elseif _activeFixType == UBFuuuFixType.kPostAddStepIntervalFix then
763            if _preAddStepCount > 0 then
764                if _debug then printx( 1 , "[U5] UBFuuuDiffLogic:calculateDeviation  61.2 fixedMaxDiffMove:", fixedMaxDiffMove ) end
765            else
766                if _debug then printx( 1 , "[U5] UBFuuuDiffLogic:calculateDeviation  61.3 fixedMaxDiffMove:", fixedMaxDiffMove ) end
767            end
768        end
769    end
770
771    resultData.fixedDiffMove1 = fixedMaxDiffMove
772
773    local levelContext = gameBoardLogic:getLevelContext()
774    if levelContext then
775        local _dcDataRaw = levelContext:getValue( UBFLevelContext.dcDataKey )
776        if not _dcDataRaw.UBFuuu then
777            _dcDataRaw.UBFuuu = {}
778        end
779
780        if moveCount == staticMoveCount then
781            _dcDataRaw.UBFuuu.maxDiffMoveOnStepN0 = maxDiffMove
782            _dcDataRaw.UBFuuu.fixedMaxDiffMoveOnStepN0 = fixedMaxDiffMove
783        elseif _preAddStepCount > 0 and moveCount == staticMoveCount + _preAddStepCount then
784            _dcDataRaw.UBFuuu.maxDiffMoveOnStepN3 = maxDiffMove
785            _dcDataRaw.UBFuuu.fixedMaxDiffMoveOnStepN3 = fixedMaxDiffMove
786        end
787    end
788
789    if _debug then printx( 1 , "[U5] UBFuuuDiffLogic:calculateDeviation  S1 fixedMaxDiffMove:", fixedMaxDiffMove ) end
790    fixedMaxDiffMove = fixedMaxDiffMove + fixedLineFix
791    if _debug then printx( 1 , "[U5] UBFuuuDiffLogic:calculateDeviation  S2 fixedMaxDiffMove:", fixedMaxDiffMove, "lineFix:", fixedLineFix ) end
792    
793
794    resultData.originDiffMove = maxDiffMove
795    resultData.lineFix = fixedLineFix
796    resultData.diffMove = fixedMaxDiffMove
797    resultData.currExpect = _fixedMoveCount / staticMoveCount
798    resultData.curProgress = (_fixedMoveCount - fixedMaxDiffMove) / staticMoveCount
799
800    local _diffmove = math.abs(fixedMaxDiffMove) 
801    if _diffmove > staticMoveCount then
802        _diffmove = staticMoveCount
803    end
804
805    if adJustRawData.lockedDeviation then
806        resultData.deviation = adJustRawData.lockedDeviation
807    else
808        if fixedMaxDiffMove > 0 then
809            resultData.deviation = _diffmove / staticMoveCount
810        else
811            resultData.deviation = _diffmove / staticMoveCount * -1
812        end
813    end
814
815    if _needLockDeviation then
816        adJustRawData.lockedDeviation = resultData.deviation
817    else
818        adJustRawData.lockedDeviation = nil
819    end
820
821    if _debug then printx( 1 , "[U5] UBFuuuDiffLogic:calculateDeviation  ===========  staticMoveCount:" , staticMoveCount , "moveCount:" , moveCount ,"preAddSteps:",preAddSteps, 
822                                "resultData =" , tts(resultData) ) end
823    
824    if _debug then
825        local mf = math.floor(resultData.diffMove * 100 + 0.5)/100
826        table.insert( _G.UBFuuuDebugCore.showPrintInfo, "_" .. tostring(mf)  )
827    end
828    
829    return resultData
830end
