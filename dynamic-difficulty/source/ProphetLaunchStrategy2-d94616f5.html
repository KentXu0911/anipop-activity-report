406local function safeCall(target, methodName, ...)
407    local method = type(target) == "table" and target[methodName] or nil
408    if type(method) ~= "function" then
409        return false, nil
410    end
411    local args = { ... }
412    local ok, result = pcall(function()
413        return method(target, unpackValues(args))
414    end)
415    return ok, result
416end

418local function compareVersion(left, right)
419    local function toParts(value)
420        local parts = {}
421        for part in tostring(value or ""):gmatch("[^.]+") do
422            parts[#parts + 1] = tonumber(part) or 0
423        end
424        return parts
425    end
426
427    local leftParts = toParts(left)
428    local rightParts = toParts(right)
429    local count = math.max(#leftParts, #rightParts)
430    for index = 1, count do
431        local leftValue = leftParts[index] or 0
432        local rightValue = rightParts[index] or 0
433        if leftValue ~= rightValue then
434            return leftValue < rightValue and -1 or 1
435        end
436    end
437    return 0
438end

598local function valueMatches(actual, condition)
599    local op = condition.op
600    local expected = condition.value
601
602    if op == "eq" then
603        return actual == expected
604    end
605    if op == "not_eq" then
606        return actual ~= expected
607    end
608    if op == "in_set" then
609        return type(expected) == "table" and expected[actual] == true
610    end
611    if op == "version_min" then
612        return compareVersion(actual, expected) >= 0
613    end
614    if op == "number_min" or op == "number_max" then
615        local numericActual = tonumber(actual)
616        local numericExpected = tonumber(expected)
617        if numericActual == nil or numericExpected == nil then
618            return false
619        end
620        local epsilon = tonumber(condition.epsilon) or 0
621        if op == "number_min" then
622            if condition.inclusive == false then
623                return numericActual > numericExpected - epsilon
624            end
625            return numericActual >= numericExpected - epsilon
626        end
627        if condition.inclusive == false then
628            return numericActual < numericExpected + epsilon
629        end
630        return numericActual <= numericExpected + epsilon
631    end
632    return false
633end

644local function evaluateConditions(context, conditions)
645    for _, condition in ipairs(conditions or {}) do
646        local actual = context[condition.field]
647        if not valueMatches(actual, condition) then
648            local missing = actual == nil
649            return false, {
650                stage = "condition",
651                ruleId = condition.id,
652                field = condition.field,
653                actual = actual,
654                expected = expectedDescription(condition),
655                reason = missing and condition.missingReason or condition.reason,
656            }
657        end
658    end
659    return true
660end

809local function evaluateNewUser(experiment, context, activation, report)
810    local passed, rejection = evaluateConditions(context, experiment.conditions)
811    if not passed then
812        appendAttempt(report, experiment.id, "rejected", rejection)
813        return nil, experiment.rejectPolicy == "continue"
814    end
815
816    local route = selectRoute(context, experiment.route)
817    if not route then
818        local decision = newUserDecision(context, activation, {
819            routeId = nil,
820            configKey = nil,
821            experienceType = "new_user",
822            context = {},
823        })
824        decision.reason = "new_user_route_not_matched"
825        appendAttempt(report, experiment.id, "selected_but_disabled", {
826            stage = "route",
827            ruleId = "no_route_matched",
828            reason = decision.reason,
829        })
830        return decision, false
831    end
832    local decision = newUserDecision(context, activation, route)
833    report.selectedExperiment = experiment.id
834    report.selectedRoute = route.routeId
835
836    if not context.adjustDataPresent then
837        decision.reason = "adjust_data_missing"
838        appendAttempt(report, experiment.id, "selected_but_disabled", {
839            stage = "pre_action",
840            ruleId = "adjust_data_present",
841            field = "adjustDataPresent",
842            actual = false,
843            expected = { op = "eq", value = true },
844            reason = decision.reason,
845        })
846        return decision, false
847    end
848
849    if not isProphetConfigAvailable(decision.configKey) then
850        decision.reason = "new_user_config_missing"
851        appendAttempt(report, experiment.id, "selected_but_disabled", {
852            stage = "route",
853            ruleId = route.routeId,
854            field = "configKey",
855            actual = decision.configKey,
856            expected = { op = "registered_config" },
857            reason = decision.reason,
858        })
859        return decision, false
860    end
861
862    decision.enabled = true
863    decision.reason = "enabled"
864    appendAttempt(report, experiment.id, "selected", {
865        stage = "route",
866        ruleId = route.routeId,
867        reason = decision.reason,
868    })
869    return decision, false
870end

872local function evaluateExperienceMatrix(experiment, context, activation, report)
873    local decision = experienceDecision(context, activation)
874    report.selectedExperiment = experiment.id
875
876    if not activation.groupConfig then
877        decision.reason = "experience_config_missing"
878        appendAttempt(report, experiment.id, "selected_but_disabled", {
879            stage = "activation",
880            ruleId = activation.group,
881            field = "groupConfig",
882            actual = nil,
883            expected = { op = "defined" },
884            reason = decision.reason,
885        })
886        return decision
887    end
888
889    local stages = experiment.conditionStages or {}
890    local passed, rejection = evaluateConditions(
891        context,
892        stages.beforeAvailability
893    )
894    if not passed then
895        decision.reason = rejection.reason
896        appendAttempt(report, experiment.id, "rejected", rejection)
897        return decision
898    end
899
900    if not decision.available then
901        decision.reason = "control_group"
902        appendAttempt(report, experiment.id, "selected_but_disabled", {
903            stage = "group",
904            ruleId = activation.group,
905            field = "enableProphet",
906            actual = false,
907            expected = { op = "eq", value = true },
908            reason = decision.reason,
909        })
910        return decision
911    end
912
913    decision.userTopLevel = context.userTopLevel
914    passed, rejection = evaluateConditions(
915        context,
916        stages.afterAvailability
917    )
918    if not passed then
919        decision.reason = rejection.reason
920        appendAttempt(report, experiment.id, "rejected", rejection)
921        return decision
922    end
923
924    if decision.requireToplevelBaoDi and not context.toplevelBaoDi then
925        decision.reason = "toplevel_baodi_not_enabled"
926        appendAttempt(report, experiment.id, "rejected", {
927            stage = "condition",
928            ruleId = "toplevel_baodi",
929            field = "toplevelBaoDi",
930            actual = context.toplevelBaoDi,
931            expected = { op = "eq", value = true },
932            reason = decision.reason,
933        })
934        return decision
935    end
936
937    decision.experienceContext = {
938        levelId = context.levelId,
939        levelScene = context.levelScene,
940        difficultyFlag = context.difficultyFlag,
941        difficultyClass = context.difficultyClass,
942        superBirdActive = context.superBirdActive,
943        toplevelBaoDi = context.toplevelBaoDi,
944        userTopLevel = context.userTopLevel,
945        fixedDesiredDiffAvg = context.fixedDesiredDiffAvg,
946        desiredDiffAvg = context.desiredDiffAvg,
947        diffIndex = context.diffIndex,
948    }
949
950    local route = selectRoute(
951        decision.experienceContext,
952        experiment.route,
953        activation.groupConfig.routeMode
954    )
955    if not route then
956        decision.experienceRouteReason = experiment.route.noMatchReason
957        decision.reason = experiment.route.noMatchReason
958        appendAttempt(report, experiment.id, "rejected", {
959            stage = "route",
960            ruleId = "no_route_matched",
961            reason = decision.reason,
962        })
963        return decision
964    end
965
966    decision.experienceRouteReason = "experience_route_matched"
967    decision.experienceType = route.experienceType
968    decision.configKey = route.configKey
969    report.selectedRoute = route.routeId
970
971    if decision.removeDiffFix and not context.adjustDataPresent then
972        decision.reason = "adjust_data_missing"
973        appendAttempt(report, experiment.id, "selected_but_disabled", {
974            stage = "pre_action",
975            ruleId = "adjust_data_present",
976            field = "adjustDataPresent",
977            actual = false,
978            expected = { op = "eq", value = true },
979            reason = decision.reason,
980        })
981        return decision
982    end
983
984    if not isProphetConfigAvailable(decision.configKey) then
985        decision.reason = "experience_config_missing"
986        appendAttempt(report, experiment.id, "selected_but_disabled", {
987            stage = "route",
988            ruleId = route.routeId,
989            field = "configKey",
990            actual = decision.configKey,
991            expected = { op = "registered_config" },
992            reason = decision.reason,
993        })
994        return decision
995    end
996
997    decision.enabled = true
998    decision.reason = "enabled"
999    appendAttempt(report, experiment.id, "selected", {
1000        stage = "route",
1001        ruleId = route.routeId,
1002        reason = decision.reason,
1003    })
1004    return decision
1005end

1007function ProphetLaunchStrategy2:evaluateDetailed(runtime, resultDatas)
1008    local report = {
1009        selectedExperiment = nil,
1010        selectedRoute = nil,
1011        attempts = {},
1012    }
1013    if not CONFIG.enabled then
1014        report.disabled = true
1015        return nil, report
1016    end
1017
1018    local context = buildContext(runtime, resultDatas)
1019    for _, experiment in ipairs(CONFIG.experiments) do
1020        local activation = activateExperiment(experiment, context)
1021        if not activation then
1022            appendAttempt(report, experiment.id, "not_activated", {
1023                stage = "activation",
1024                ruleId = experiment.activation.type,
1025                reason = "experiment_not_activated",
1026            })
1027        elseif experiment.id == "new_user" then
1028            local decision, shouldContinue =
1029                evaluateNewUser(experiment, context, activation, report)
1030            if decision or not shouldContinue then
1031                return decision, report
1032            end
1033        elseif experiment.id == "experience_matrix" then
1034            return evaluateExperienceMatrix(
1035                experiment,
1036                context,
1037                activation,
1038                report
1039            ), report
1040        end
1041    end
1042    return nil, report
1043end

1046    local decision = self:evaluateDetailed(runtime, resultDatas)
1047    return decision
1048end

1116function ProphetLaunchStrategy2:apply(runtime, resultDatas)
1117    local decision = self:evaluate(runtime, resultDatas)
1118    if not decision then
1119        return nil
1120    end
1121
1122    for _, actionName in ipairs(CONFIG.actions) do
1123        if actionName ~= "record_decision" and not decision.enabled then
1124            break
1125        end
1126        local handler = ACTION_HANDLERS[actionName]
1127        if handler and handler(runtime, resultDatas, decision) == false then
1128            break
1129        end
1130    end
1131    return decision
1132end
