5local function cloneComposableConfig(value, seen)
6    if type(value) ~= "table" then return value end
7    seen = seen or {}
8    if seen[value] then return seen[value] end
9    local result = {}
10    seen[value] = result
11    for key, item in pairs(value) do
12        result[cloneComposableConfig(key, seen)] = cloneComposableConfig(item, seen)
13    end
14    return result
15end

17local function omitEmptySequence(parent, key)
18    local value = parent and parent[key]
19    if type(value) == "table" and next(value) == nil then
20        parent[key] = nil
21    end
22end

24local function normalizeDesignatedBoardAliases(normalized)
25    local legacy = normalized.exciting_board_adjustment
26    local current = normalized.designated_board_adjustment
27    if legacy ~= nil and current ~= nil then
28        error("behavior_config must not contain both exciting_board_adjustment and designated_board_adjustment")
29    end
30    if current == nil and legacy ~= nil then
31        normalized.designated_board_adjustment = legacy
32        normalized.exciting_board_adjustment = nil
33    end
34    local policy = normalized.conflict_policy
35    for index, owner in ipairs((policy and policy.priority) or {}) do
36        if owner == "ExcitingBoard" then
37            policy.priority[index] = "DesignatedBoard"
38        end
39    end
40end

42local function fillDefaults(target, defaults)
43    if type(target) ~= "table" or type(defaults) ~= "table" then return target end
44    for key, value in pairs(defaults) do
45        if target[key] == nil then
46            target[key] = cloneComposableConfig(value)
47        elseif type(target[key]) == "table" and type(value) == "table" then
48            fillDefaults(target[key], value)
49        end
50    end
51    return target
52end

74local function normalizedComposableConfig(behaviorConfig)
75    assert(type(behaviorConfig) == "table", "behaviorConfig must be a table")
76    local normalized = cloneComposableConfig(behaviorConfig)
77    normalizeDesignatedBoardAliases(normalized)
78    omitEmptySequence(normalized.tracking, "strength_rules")
79    omitEmptySequence(normalized.fixed_adjustment, "rules")
80    omitEmptySequence(normalized.exciting_operation_adjustment, "ranges")
81    omitEmptySequence(normalized.designated_board_adjustment, "ranges")
82    if normalized.fixed_adjustment then
83        normalized.fixed_adjustment.on_budget_exhausted = nil
84    end
85    return normalized
86end

183local function checkStrength(path, value)
184    if type(value) ~= "number" or value < 0 or value > 10 or value % 1 ~= 0 then
185        return false, path .. " must be an integer between 0 and 10"
186    end
187    return true

190local function checkStepBudget(path, value)
191    if type(value) ~= "table" or value.enabled ~= true then return true end
192    local hasLimit = false
193    for _, key in ipairs({ "max_positive_steps", "max_negative_steps", "max_absolute_steps" }) do
194        local limit = value[key]
195        if limit ~= nil then
196            hasLimit = true
197            if type(limit) ~= "number" or limit < 0 or limit > 10000 then
198                return false, path .. "." .. key .. " must be between 0 and 10000"
199            end
200        end
201    end
202    if not hasLimit then return false, path .. " must define at least one limit when enabled" end
203    return true
204end

206local function checkComboBudget(path, value)
207    if type(value) ~= "table" or value.enabled ~= true then return true end
208    local limit = value.max_combo_count
209    if type(limit) ~= "number" or limit < 0 or limit > 100000 or limit % 1 ~= 0 then
210        return false, path .. ".max_combo_count must be an integer between 0 and 100000"
211    end
212    return true

215local function checkScopedBudget(path, value)
216    value = value or {}
217    local level = value.level or {}
218    local operation = value.per_operation or {}
219    for _, item in ipairs({
220        { path .. ".level.step", level.step, checkStepBudget },
221        { path .. ".level.combo", level.combo, checkComboBudget },
222        { path .. ".per_operation.step", operation.step, checkStepBudget },
223        { path .. ".per_operation.combo", operation.combo, checkComboBudget },
224    }) do
225        local ok, err = item[3](item[1], item[2])
226        if not ok then return false, err end
227    end
228    return true
229end

231local function checkRange(path, range)
232    if type(range) ~= "table" then return false, path .. " must be a table" end
233    if range.type == "Single" then
234        if type(range.move) ~= "number" or range.move % 1 ~= 0
235            or range.move < 1 or range.move > 10000 then
236            return false, path .. ".move must be an integer between 1 and 10000"
237        end
238    elseif range.type == "Interval" then
239        if type(range.start) ~= "number" or type(range["end"]) ~= "number"
240            or range.start % 1 ~= 0 or range["end"] % 1 ~= 0
241            or range.start < 1 or range["end"] > 10000 or range.start > range["end"] then
242            return false, path .. " must be a valid absolute interval"
243        end
244    elseif range.type == "RelativeInterval" then
245        if type(range.start) ~= "number" or type(range["end"]) ~= "number"
246            or range.start % 1 ~= 0 or range["end"] % 1 ~= 0
247            or range.start < -10000 or range["end"] > 10000 or range.start > range["end"] then
248            return false, path .. " must be a valid relative interval"
249        end
250    elseif range.type == "Periodic" then
251        if type(range.start) ~= "number" or type(range.gap) ~= "number"
252            or range.start % 1 ~= 0 or range.gap % 1 ~= 0
253            or range.start < 1 or range.start > 10000 or range.gap < 0 or range.gap > 9999 then
254            return false, path .. " must define start 1..10000 and gap 0..9999"
255        end
256    else
257        return false, path .. ".type is unsupported"
258    end
259    return true
260end

262local function validateCommon(config)
263    local tracking = config.tracking or {}
264    if type(tracking.endpoint_shift_moves) ~= "number" or tracking.endpoint_shift_moves < -9999 or tracking.endpoint_shift_moves > 9999 then
265        return false, "tracking.endpoint_shift_moves must be between -9999 and 9999"
266    end
267    for _, item in ipairs({
268        { "tracking.default_positive_strength", tracking.default_positive_strength },
269        { "tracking.default_negative_strength", tracking.default_negative_strength },
270        { "prop_adjustment.strength", config.prop_adjustment and config.prop_adjustment.strength },
271    }) do
272        local ok, err = checkStrength(item[1], item[2])
273        if not ok then return false, err end
274    end
275    for index, rule in ipairs(tracking.strength_rules or {}) do
276        for _, key in ipairs({ "positive_strength", "negative_strength" }) do
277            local ok, err = checkStrength("tracking.strength_rules[" .. index .. "]." .. key, rule[key])
278            if not ok then return false, err end
279        end
280    end
281    for index, rule in ipairs((config.fixed_adjustment and config.fixed_adjustment.rules) or {}) do
282        local ok, err = checkStrength("fixed_adjustment.rules[" .. index .. "].strength", rule.strength)
283        if not ok then return false, err end
284    end
285    for key, operation in pairs((config.exciting_operation_adjustment and config.exciting_operation_adjustment.operations) or {}) do
286        local ok, err = checkStrength("exciting_operation_adjustment.operations." .. key .. ".strength", operation.strength)
287        if not ok then return false, err end
288    end
289    local board = config.designated_board_adjustment
290    if config.exciting_board_adjustment ~= nil and board ~= nil then
291        return false, "behavior_config must not contain both exciting_board_adjustment and designated_board_adjustment"
292    end
293    if board == nil then board = config.exciting_board_adjustment end
294    local minWeight = config.schema_version == 2 and -99 or 0
295    for key, weight in pairs((board and board.weights) or {}) do
296        if type(weight) ~= "number"
297            or weight < minWeight or weight > 99 or weight % 1 ~= 0
298        then
299            return false, "designated_board_adjustment.weights." .. key
300                .. " must be an integer between " .. minWeight .. " and 99"
301        end
302    end
303    return true
304end

306function RustProphet.validateComposableBehaviorConfigV1(config)
307    if type(config) ~= "table" then return false, "behavior_config must be a table" end
308    if config.schema_version ~= 1 then return false, "schema_version must be 1" end
309    local checks = {
310        { "global_budget", config.global_budget },
311        { "tracking.budget", config.tracking and config.tracking.budget },
312        { "fixed_adjustment.budget", config.fixed_adjustment and config.fixed_adjustment.budget },
313        { "prop_adjustment.budget", config.prop_adjustment and config.prop_adjustment.budget },
314        { "exciting_operation_adjustment.budget", config.exciting_operation_adjustment and config.exciting_operation_adjustment.budget },
315        { "designated_board_adjustment.budget", (config.designated_board_adjustment or config.exciting_board_adjustment)
316            and (config.designated_board_adjustment or config.exciting_board_adjustment).budget },
317    }
318    for _, item in ipairs(checks) do
319        local ok, err = checkStepBudget(item[1], item[2])
320        if not ok then return false, err end
321    end
322    return validateCommon(config)
323end

325function RustProphet.validateComposableBehaviorConfigV2(config)
326    if type(config) ~= "table" then return false, "behavior_config must be a table" end
327    if config.schema_version ~= 2 then return false, "schema_version must be 2" end
328    for _, item in ipairs({
329        { "global_budget", config.global_budget },
330        { "tracking.budget", config.tracking and config.tracking.budget },
331        { "fixed_adjustment.budget", config.fixed_adjustment and config.fixed_adjustment.budget },
332        { "prop_adjustment.budget", config.prop_adjustment and config.prop_adjustment.budget },
333        { "exciting_operation_adjustment.budget", config.exciting_operation_adjustment and config.exciting_operation_adjustment.budget },
334        { "designated_board_adjustment.budget", (config.designated_board_adjustment or config.exciting_board_adjustment)
335            and (config.designated_board_adjustment or config.exciting_board_adjustment).budget },
336    }) do
337        local ok, err = checkScopedBudget(item[1], item[2])
338        if not ok then return false, err end
339    end
340    local exciting = config.exciting_operation_adjustment or {}
341    for index, range in ipairs(exciting.ranges or {}) do
342        local ok, err = checkRange("exciting_operation_adjustment.ranges[" .. index .. "]", range)
343        if not ok then return false, err end
344    end
345    for key, operation in pairs(exciting.operations or {}) do
346        for _, field in ipairs({
347            { "per_operation_step_budget", 10000 },
348            { "per_operation_combo_budget", 100000 },
349            { "max_trigger_count", 1000 },
350        }) do
351            local value = operation[field[1]]
352            if value ~= nil and (type(value) ~= "number" or value < 0
353                or value > field[2] or value % 1 ~= 0) then
354                return false, "exciting_operation_adjustment.operations." .. key .. "."
355                    .. field[1] .. " must be an integer between 0 and " .. field[2]
356            end
357        end
358    end
359    return validateCommon(config)
360end

606local function makeExperienceBehaviorConfig(
607    endpointShiftMoves,
608    trackingPositiveSteps,
609    trackingNegativeSteps,
610    fixedRange,
611    excitingOperationSpecs,
612    excitingBoardRange
613)
614    local config = experienceBaseBehaviorConfig()
615    enableExperienceTracking(
616        config,
617        endpointShiftMoves,
618        trackingPositiveSteps,
619        trackingNegativeSteps
620    )
621    if fixedRange then
622        addFixedExperienceRule(
623            config,
624            fixedRange[1],
625            fixedRange[2],
626            fixedRange[3],
627            fixedRange[4]
628        )
629    end
630    if excitingOperationSpecs then
631        enableExcitingOperations(config, excitingOperationSpecs)
632    end
633    if excitingBoardRange then
634        enableExcitingBoard(
635            config,
636            excitingBoardRange[1],
637            excitingBoardRange[2],
638            excitingBoardRange[3]
639        )
640    end
641    return config
642end
