16function UBFProcessModManager:loadAllWillingMod(defaultWillingList)
17    for sourceId, modClazz in pairs(self.modClazzMap) do
18        local isEnabled = defaultWillingList and defaultWillingList[sourceId]
19        if not isEnabled then
20            isEnabled = modClazz:isWillingEnabled(self.runtime, sourceId)
21        end
22        if isEnabled then
23            table.insert(self.willingModList, sourceId)
24            self.modContextData[sourceId] = {}
25        end
26    end
27
28    -- 活动可在意愿检查时声明隔离策略，因此统一在检查结束后应用白名单。
29    local willingModAllowList = self.runtime:getExtra("willingModAllowList")
30    if type(willingModAllowList) == "table" then
31        for index = #self.willingModList, 1, -1 do
32            local sourceId = self.willingModList[index]
33            if willingModAllowList[tostring(sourceId)] ~= true then
34                table.remove(self.willingModList, index)
35                self.modContextData[sourceId] = nil
36            end
37        end
38    end
39    table.sort(self.willingModList, function(a, b)
40        local na = UBFModHelper:getModPriority(a)
41        local nb = UBFModHelper:getModPriority(b)
42        return na < nb
43    end)

104function UBFProcessModManager:_notifyModEvent(eventName, data, protectedMode)
105    local noError = true
106    if _G.editorMode then
107        if self:isEditorIgnoredEvent(eventName) then
108            return noError
109        end
110    end
111
112    local getNotifyEventModDataList = UBFModHelper:getNotifyEventModDataList(self.willingModList, eventName)
113    local event = UBFSceneModEventMap[eventName]:create(data)
114    local oldSourceId = self.runtime.sourceId
115    for k, v in ipairs(getNotifyEventModDataList) do
116        if event:getBlock() then
117           break
118        end
119        local isGroupBlock = v.groupId and event:getGroupBlock(v.groupId)
120        if not isGroupBlock then
121            self.runtime.sourceId = v.sourceId
122            local strSourceId = tostring(v.sourceId)
123            event:setCurrGroupId(v.groupId)
124            event:setRuntime(self.runtime)
125            event:setEntryTag(v.entryTag)
126
127            event:onPreNotifyMod(v.sourceId)
128
129            local target, funcName
130            --只向mod实例派发
131            if v.useInst then
132                target = UBFGameModManagerOrganizer:getRunningModManager(strSourceId)
133                funcName = v.funcName and v.funcName or "notifyEvent"
134            else
135                target = self.modClazzMap[strSourceId]
136                funcName = v.funcName and v.funcName or "notifyEventStatic"
137            end
138            if protectedMode then
139                if target == nil then
140                    if not v.useInst then
141                        -- 可能是没有注册到活动框架里只是纯加分数buff的活动(签到、加分道具、长尾步数升级、贴纸赢好礼等)
142                        if UBFLogW then UBFLogW(string.format("[UBFProcessModManager] notifyModEvent %s, mod %s, target is nil", eventName, v.sourceId), UBFTraceback()) end
143                        noError = false
144                    end
145                else
146                    local errHandler = string.format("[UBFProcessModManager] notifyModEvent %s, mod %s", eventName, v.sourceId)
147                    local strSrcId = tostring(v.sourceId)
148                    if eventName == UBFGameModEventEnum.LOAD_MOD and (strSrcId == "guide" or strSrcId == "FeedAutoStart") then
149                        local fn = target[funcName]
150                        local ok3, err3 = pcall(fn, target, event)
151                        -- printx(0, "[MODMGR] " .. strSrcId .. " pcall ok=", tostring(ok3), " err=", tostring(err3))
152                        -- printx(0, "[MODMGR] " .. strSrcId .. " enabled after=", tostring(self.modOpenStatusMap[strSrcId]))
153                    else
154                        local ok = UBFHelper:invokeProtected(target, funcName, errHandler, event)
155                        noError = noError and ok
156                    end
157                end
158            else
159                UBFHelper:invoke(target, funcName, event)
160            end
161
162            event:onPostNotifyMod(v.sourceId)
163
164        end
165    end
166    self.runtime.sourceId = oldSourceId
167
168    return noError
169end

207function UBFProcessModManager:_loadModResource(modId, modList)
208    local modClazz = self.modClazzMap[tostring(modId)]
209    if not modClazz then
210        -- 没有注册到活动框架里只是纯加分数buff的活动(长尾步数升级、贴纸赢好礼等)，不需要加载资源
211        -- he_log_error("modClazz a nil value, modId ==>>".. tostring(modId))
212        -- self._last_load_mod_err_ = true
213        return 
214    end
215    local secondaryPrepare = UBFResourceLoadManager:prepare():relax():bestEffort()
216    local primaryPrepare = modClazz:createModResPrepare(self.runtime, secondaryPrepare, modList)
217
218    local modFeature = UBFGameModManagerOrganizer:getModFeatureByModList(modList)
219    if modFeature and not UBFHelper:isTableEmpty(modFeature) then 
220        local UBFResUtils = require 'ubf.UBFPlay.logics.launch.UBFResUtils'
221        primaryPrepare = primaryPrepare or UBFResourceLoadManager:prepare()
222        local resKeys, scatterList = UBFResUtils:getUnloadResByFeatureMap(modFeature)
223        for id, _ in pairs(resKeys) do UBFLVResourceManager:prepare(id, primaryPrepare) end
224        for _, res in ipairs(scatterList) do primaryPrepare:add(res) end
225        local featureManager = self.runtime.featureManager
226        if featureManager then 
227            for itemType, _ in pairs(modFeature) do
228                featureManager:refreshFeatureByItemType(itemType)
229            end
230        end
231    end
232
233    local sceneReource = self.runtime:getSceneResourceMgr()
234    if primaryPrepare then
235        sceneReource:setModPrepare(modId, primaryPrepare)
236    end
237    sceneReource:setModSecondaryPrepare(modId, secondaryPrepare)
238end

251-- 获取 mod 的条件检测数据（缓存命中返回，否则现场收集并写入缓存）
252function UBFProcessModManager:getModConditionData(modId)
253    local modContextData = self.modContextData[tostring(modId)]
254    if not modContextData then return nil end
255    if modContextData.conditionData then
256        return modContextData.conditionData
257    end
258    local modClazz = self.modClazzMap[tostring(modId)]
259    if not modClazz then return nil end
260    local data = modClazz:collectModConditionData(self.runtime)
261    if data then
262        modContextData.conditionData = data
263    end
264    return data
265end

330function UBFProcessModManager:resumeGamePlayModList(modList)
331    self.modOpenStatusMap = {}
332    self.modDataList = {}
333    self.willingModList = {}
334    local willingModMap = {}
335    for _, data in ipairs(modList) do 
336        if data.sourceId then 
337            self.modOpenStatusMap[data.sourceId] = true
338            if not willingModMap[data.sourceId] then
339                willingModMap[data.sourceId] = true
340                table.insert(self.willingModList, data.sourceId)
341            end
342            if not self.modDataList[data.sourceId] then
343                self.modDataList[data.sourceId] = {}
344            end
345            table.insert(self.modDataList[data.sourceId], data)
346        end
347    end
348    table.sort(self.willingModList, function(a, b)
349        local na = UBFModHelper:getModPriority(a)
350        local nb = UBFModHelper:getModPriority(b)
351        return na < nb
352    end)
353    for modId, list in pairs(self.modDataList) do 
354        self:_loadModResource(modId, list)
355        if self._last_load_mod_err_ then
356            return false
357        end
358    end
359end
