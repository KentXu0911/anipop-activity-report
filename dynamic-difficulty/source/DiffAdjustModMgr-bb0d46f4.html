523    local aiParamKey = actData and actData.actKey
524    if not aiParamKey or aiParamKey ~= DiffAdjustModMgr.ACT_KEY  then return end
525
526    actData.enableUBFuuu = true
527
528    local diffFix = tonumber(actData.adjustData.diffFix)
529
530    local adjustData = {
531        enableUBFuuu = true,
532        mode = ProductItemDiffChangeMode.kAIAddColor,
533        ds = 5,
534        reason = "UBFuuu2025ByAI",
535        lastReason = "UBFuuu2025ByAI",
536
537        diffFix = diffFix or 0,
538    }
539
540    if tostring(diffFix) ~= tostring(actData.adjustData.diffFix) then
541
542        local _arr = string.split(actData.adjustData.diffFix, "_")
543
544        actData.ubfuuCheckResult = {
545            stepFixEnable = true,
546            stepFix = tonumber(_arr[2]),
547            desiredStep = tonumber(_arr[2]),
548        }
549    end
550
551    actData.adjustData = table.merge(actData.adjustData, adjustData)
552    ubfSLRuntime.adjustData = actData.adjustData
553    actData.ubfuuCheckResult = {
554        calculateDiffIndexResult = {
555            modGroup = {
556                enableFixScore = ubfSLRuntime.levelId < levelFlag3,
557                enableFixIceCombo = ubfSLRuntime.levelId >= levelFlag3
558            }
559        }
560    }
561
562    if actData.scoreTest then
563        actData.ubfuuCheckResult.calculateDiffIndexResult.modGroup.enableFixScore = actData.scoreTest.fixA
564        actData.ubfuuCheckResult.calculateDiffIndexResult.modGroup.enableFixIceCombo = actData.scoreTest.fixC
565    end
566
567    return actData
568end
