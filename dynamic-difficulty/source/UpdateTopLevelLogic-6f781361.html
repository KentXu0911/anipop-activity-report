3function UpdateTopLevelLogic:execute(passLevelContext)
4    local levelId = passLevelContext.levelId
5    local levelScene = passLevelContext.levelScene
6    local newStar = passLevelContext.star
7    local logicalWin = passLevelContext.logicalWin
8    local logicalFail = passLevelContext.logicalFail
9    local isJumpLevel = passLevelContext.isJumpLevel
10    local isBeHelped = passLevelContext.isBeHelped
11
12    local levelAreaManager = _G.amfBean("LevelAreaManager")
13    local topLevelId = levelAreaManager:getTopLevelId()
14    
15    if levelId == topLevelId and UBFLevelSceneFunc:isMainLevel(levelScene) then
16        -- 代打跳关也算胜利
17        if newStar > 0 or isJumpLevel or isBeHelped then
18            -- 胜利
19            levelAreaManager:setTopLevelFailCount(0)
20            levelAreaManager:setTopLevelLogicalFailCount(0)
21            UserTagManager:onTopLevelChanged()
22            
23            if levelAreaManager:isTopNoPassLevel(levelId) then
24                levelAreaManager:addTopLevelWinCount(1)
25                if logicalWin then
26                    levelAreaManager:addTopLevelLogicalWinCount(1)
27                else
28                    levelAreaManager:setTopLevelLogicalWinCount(0)
29                end
30            end
31
32            local lastScore = levelAreaManager:getUserScore(levelId, levelScene)
33            if not lastScore or lastScore.star <= 0 then
34                --区域最后一关交给区域解锁去更新topLevelId
35                if levelId % 15 ~= 0 then
36                    levelAreaManager:setTopLevelId(topLevelId + 1)
37                    levelAreaManager:updateLastTopLevelUpdateTime()
38                end
39            end
40        else
41            -- 失败
42            if levelAreaManager:isTopNoPassLevel(levelId) then
43                levelAreaManager:setTopLevelWinCount(0)
44                levelAreaManager:setTopLevelLogicalWinCount(0)
45            end
46
47            levelAreaManager:setTopLevelContinueWinCount(0)
48
49            -- 只加逻辑失败，是因为普通失败已经在进关时加过了
50            if logicalFail then
51                levelAreaManager:addTopLevelLogicalFailCount()
52            end
53        end
54    end
55end
