30function UBFFallingItemLogicV2_1:hasFallingBlockBetweenTwoVerticalGrids(gridRawDataUp , gridRawDataDown )

32	local upUsed = gridRawDataUp.isUsed
33	local downUsed = gridRawDataDown.isUsed
34
35	if upUsed then
36		if self:hasBottomBorderFallingBlocker(gridRawDataUp) then return true end
37		-- if self:hasTopBorderFallingBlocker( gridContextDown ) then return true end --新棋盘里，不需要检测另一个格子了
38	else
39		if downUsed then
40			if self:hasTopBorderFallingBlocker(gridRawDataDown) then return true end 
41		end
42	end
43	
44	local upGravity = gridRawDataUp.gravity
45	local downGravity = gridRawDataDown.gravity
46
47	--todo move to opMap
48
49	if upGravity == GravityDirection.kDown then
50		if upUsed and downUsed and downGravity == GravityDirection.kUp then
51			return true
52		end
53		if self:gridHasEnterPortal(gridRawDataUp) then return true end
54	elseif upGravity == GravityDirection.kUp then
55		if self:gridHasExitPortal(gridRawDataUp) then return true end
56	end
57
58	if downGravity == GravityDirection.kDown then
59		if self:gridHasExitPortal(gridRawDataDown) then return true end
60	elseif downGravity == GravityDirection.kUp then
61		if self:gridHasEnterPortal(gridRawDataDown) then return true end
62	end

64	return false

703function UBFFallingItemLogicV2_1:doRecursionVFFalling( boardContext , r , c , viewActlist , needPortalCheckOnFirstGrid , backStop )
704    -- r,c为一个空格子，
705    -- 本方法将从（r,c）开始，沿重力方向的反方向递归寻找每一格，看看能否沿重力方向掉落，如果能掉落就执行掉落
706    -- 如果重力方向的反方向不能掉落，那么会尝试从另外两个90度方向掉落
707    -- needPortalCheckOnFirstGrid 的含义是，如果它为true，（r,c）位置的上一格如果不能掉落，就要检测（r,c）位置是否有传送门传送
708    
709    -- printx( 1 , "doRecursionVFFalling  _checkAndFallingPrevGrid  ---- " , r , c )
710
711    local _checkResult , _prevPos = self:_checkAndFallingPrevGrid( boardContext , r , c , viewActlist, backStop )
712    -- printx( 1 , "doRecursionVFFalling  _checkAndFallingPrevGrid  check result : " , _checkResult )
713
714    if _checkResult then
715        --开始递归
716
717        _checkResult = false
718
719        while(true) do 
720
721            local _pr = _prevPos.r
722            local _pc = _prevPos.c
723            -- printx( 1 , "doRecursionVFFalling  While  _checkAndFallingPrevGrid  ---- " , _pr , _pc )
724            _checkResult , _prevPos = self:_checkAndFallingPrevGrid( boardContext , _pr , _pc , viewActlist , backStop )
725
726            if not _prevPos then
727                -- printx( 1 , "doRecursionVFFalling  While  break 1 !!!"  )
728                break --prev格子不存在，算法结束
729            end
730
731            -- printx( 1 , "doRecursionVFFalling  While  _checkAndFallingPrevGrid  check result :" , _checkResult , _pr , _pc )
732            if not _checkResult then
733                -- 二级递归
734                -- prev格子存在，但无法直线掉落，继续检测prev格子上的传送门
735                -- printx( 1 , "doRecursionVFFalling  While checkGridIsPortalExitAndJustEmpty ---- " , _pr , _pc )
736                if not self:checkGridIsPortalExitAndJustEmpty( boardContext , _pr , _pc , viewActlist , backStop ) then
737					local gridRawData = boardContext:getGridRawDataByPos( _pr , _pc )
738					if gridRawData then
739						local producers = gridRawData[UBFGridContext.itemTypeMapKey][UBFGameItemType.kProducer]
740						local producer = producers and producers[1]
741						-- printx( 1 , "doRecursionVFFalling  While producer ---- " , _pr , _pc )
742						if producer then
743							-- printx( 1 , "doRecursionVFFalling  While checkFallingByCannon " , _pr, _pc)
744							if self:checkFallingByCannon( boardContext , _pr, _pc ) then 
745								-- printx( 1 , "doRecursionVFFalling  While doFallingByCannon " , _pr, _pc)
746								self:doFallingByCannon( boardContext , _pr , _pc , viewActlist )
747							end
748						end
749					end
750				end
751                -- printx( 1 , "doRecursionVFFalling  While  break 2 !!!"  )
752                break
753            end
754        end
755
756		return true
757    else
758        if needPortalCheckOnFirstGrid then
759            self:checkGridIsPortalExitAndJustEmpty( boardContext , r , c , viewActlist , backStop )
760        end
761
762		return false
763    end
764end
