84function UBFSLProcessFlowController:init(createData) 
85	UBFProcessFlowController.init(self, createData)
86    self:setResult("runtime", createData.runtime) -- 不一定开始就有
87    --代码迁移听众
88    local migrator = createData.migrator
89    if migrator then 
90        for key, func in pairs(migrator.after or {}) do
91            self:after(key):next(function() func(self:getResult("runtime")) end)
92        end
93        for key, func in pairs(migrator.before or {}) do
94            self:before(key):next(function() func(self:getResult("runtime")) end)
95        end
96    end
97end

115function UBFSLProcessFlowController:_abort(node)
