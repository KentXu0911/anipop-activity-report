393-- 配置解析、消息构建和发送分开，后续直接重推时不需要借用一次性 pending 字段。
394function UBFGameSceneManager:pushProphetConfig(configSource, thermalState)
395	local config = self:_resolve_prophet_config(configSource)
396	if not config then
397		return false, ProphetReplayResumeLaunchPolicy.AttemptOutcome.InvalidConfig
398	end
399	local data = self:_build_prophet_launch_message(config, thermalState)
400	if not self:_send_prophet_message(data) then
401		return false, ProphetReplayResumeLaunchPolicy.AttemptOutcome.MessageClientUnavailable
402	end
403	return true, ProphetReplayResumeLaunchPolicy.AttemptOutcome.Sent
404end

406function UBFGameSceneManager:_try_push_prophet()

433function UBFGameSceneManager:_sample_thermal_state()
434	if self.isDisposed or self.endLevelData then
435		return false
436	end
437
438	local thermalState = ThermalStateMonitor:getThermalState()
439	if thermalState < 0 then
440		return true
441	end
442
443	local stats = getThermalStateStats(self:getSceneData())
444	local previousMax = tonumber(stats.max) or 0
445	stats.total = stats.total + thermalState
446	stats.count = stats.count + 1
447	if thermalState > previousMax then
448		stats.max = thermalState
449		self:_on_prophet_thermal_state_max_rise(previousMax, thermalState)
450	end
451	return true
452end

454function UBFGameSceneManager:_on_prophet_thermal_state_max_rise(previousMax, currentMax)
455	if not ENABLE_PROPHET_THERMAL_PERFORMANCE_REPUSH
456		or self._prophet_thermal_performance_repush_done
457		or currentMax <= previousMax
458	then
459		return
460	end
461
462	local configSource = self._prophet_config_source_
463	if not configSource then
464		return
465	end
466
467	local success = self:pushProphetConfig(configSource, currentMax)
468	if success then
469		self._prophet_thermal_performance_repush_done = true
470	end
471end

511	self._in_resume_repalying = nil

530function UBFGameSceneManager:finalizeReplayResumeProphetLaunch(onFinalized)
531	if self._resume_prophet_launch_released then
532		return self._resume_prophet_launch_state
533	end
534	if onFinalized and not self._resume_prophet_launch_on_finalized then
535		self._resume_prophet_launch_on_finalized = onFinalized
536	end
537
538	if self.isDisposed then
539		self:disableReplayResumeProphetLaunchOnDispose()
540		return ProphetReplayResumeLaunchPolicy.State.Disabled
541	end
542
543	local state = self._resume_prophet_launch_state
544	if state ~= ProphetReplayResumeLaunchPolicy.State.Pending then
545		if not state then
546			self._resume_prophet_launch_state = ProphetReplayResumeLaunchPolicy.State.Disabled
547		end
548		self:_releaseReplayResumeAfterProphetLaunch()
549		return self._resume_prophet_launch_state
550	end
551
552	local success, outcome = self:_try_push_prophet()
553	local nextState, attemptCount, shouldRetry =
554		ProphetReplayResumeLaunchPolicy:resolveAttempt(
555			state,
556			self._resume_prophet_launch_attempt_count,
557			success and ProphetReplayResumeLaunchPolicy.AttemptOutcome.Sent or outcome
558		)
559	self._resume_prophet_launch_state = nextState
560	self._resume_prophet_launch_attempt_count = attemptCount
561
562	if shouldRetry then
563		if not self._resume_prophet_launch_retry_scheduled then
564			self._resume_prophet_launch_retry_scheduled = true
565			self:addDelayFunc(1, function()
566				self._resume_prophet_launch_retry_scheduled = false
567				self:finalizeReplayResumeProphetLaunch()
568			end)
569		end
570		return nextState
571	end
572
573	self._resume_prophet_launch_retry_scheduled = false
574	if nextState == ProphetReplayResumeLaunchPolicy.State.Disabled then
575		if outcome == ProphetReplayResumeLaunchPolicy.AttemptOutcome.MessageClientUnavailable then
576			he_log_error("ProphetResumeLaunchUnavailable")
577		elseif outcome == ProphetReplayResumeLaunchPolicy.AttemptOutcome.InvalidConfig then
578			he_log_error("ProphetResumeLaunchConfigMissing")
579		end
580		self._prophet_config_ = nil
581	end
582	self:_releaseReplayResumeAfterProphetLaunch()
583	return nextState
584end

731	if self._msgLogic then UBFThrow "不允许重复设置消息逻辑" end
732	if msgLogic.rustMagicNumber then 
733		self._levelInfo.rustMagicNumber = msgLogic.rustMagicNumber 
734	end
735	self._msgLogic = msgLogic
736	self._msgLogic._crashCatcher = function( msg )
737		local boardViewLogic = self:getBoardViewLogic()
738		if boardViewLogic then
739			boardViewLogic:onException(UBFBoardExceptionLevel.kFatal, "消息崩溃了" .. tostring(msg), "msg")
740		else
741			self:onGameOver({
742				isCrash = true,
743				crashMsg = tostring( msg ) ,
744				source = "msg"
745			})
746		end
747	end
748	self._msgLogic.hook:addHook("operation", function(message)
749		if self:shouldGateReplayResumeProphetOperation(message) then
750			message.abandon = true
751			return
752		end
753		if message.opType == UBFOperationType.kSwap or message.opType == UBFOperationType.kProp then
754			local boardViewLogic = self:getBoardViewLogic()
755			if boardViewLogic then
756				boardViewLogic:getBoardStateMap()[UBFBoardStateKey.kCanSwap] = false
757				boardViewLogic:resetInteractor()
758			end
759		end
760	end)
761	self._msgLogic:registerListener(UBFGamePlaySceneEvents.kClosure , function() self:onLogicClosure() end)
762	self._msgLogic:registerListener(UBFGamePlaySceneEvents.kGameOver , function( evt , endLevelData )
763		self:onEndLevelDataRecv( endLevelData )
764		-- AI 直接回收场景，不启动依赖完整用户出关链路的 process。
765		if not self.isAIEnv then
766			self:launchEndLevelProcess(endLevelData)
767		end
768		if self._replayResumeLevelRuntime then
769			-- 回放恢复时，如果关卡收到 endLevelData，直接结束，不再需要进bonus
770			local boardViewLogic = self:getBoardViewLogic()
771			if boardViewLogic then
772				boardViewLogic.isReplay = false
773			end
774			local endFrameIdx = endLevelData.endFrame
775			if boardViewLogic and endFrameIdx then
776				boardViewLogic:addRelativeTickCallback( endFrameIdx + 30 , function() self:onGameOverViewCome() end, true)
777			else
778				self:onGameOverViewCome()
779			end
780			return
781		end
782
783		if not endLevelData.win and endLevelData.failReason == UBFGameFailReason.kDropout
784			or endLevelData.failReason == UBFGameFailReason.kRetry
785		then
786			local scene = self:getScene()
787			if scene and endLevelData.failReason == UBFGameFailReason.kRetry then scene.game_retry_flag = true end
788			self:onGameOverViewCome()
789			return
790		end
791		local boardViewLogic = self:getBoardViewLogic()
792		local endFrameIdx = endLevelData.endFrame
793		if boardViewLogic and endFrameIdx then
794			if not endLevelData.win then
795				boardViewLogic:addRelativeTickCallback( endFrameIdx + 30 , function() self:onGameOverViewCome() end, true)
796			else
797				boardViewLogic:addRelativeTickCallback( endFrameIdx + 5 , function() self:onGameOverViewCome() end, true)
798			end
799		else
800			self:onGameOverViewCome()
801		end
802	end)
803	self._msgLogic:registerListener(UBFGamePlaySceneEvents.kReplayAction , function( evt , dat )
804		local scene = self:getScene()
805
806		if scene and scene._replay_nice_time_line_ then
807			local ret = scene:_debugTrusteeshipReplayAction(dat)
808			if ret then return end
809		end
810
811		local action = dat.action
812		if action == UBFGameReplayActionEnum.kError then
813			if self.resumeLevel then
814				-- resumelevel模式中，后面还会又end
815				if dat.msg then
816					local Error = UError or UBFError
817					Error("replay resume error:" .. dat.msg)
818				end
819				if self:isInRust() then self:onGameReplayEnd(dat)end
820			else
821				if self:isInRust() and EnvUtil:isWindows() then 
822					CommonTip:showTip("Rust回放失败了，不会自动退关哟~")
823				else
824					self:onGameOver({ 
825						replay = true,
826						action = "error",
827					})
828				end
829			end
830		elseif action == UBFGameReplayActionEnum.kEnd then
831			self:onGameReplayEnd(dat)
832		elseif action == UBFGameReplayActionEnum.kBegin then
833			self:onGameReplayBegin(dat)
834		end
835	end )
836	self._msgLogic:registerListener(UBFGamePlaySceneEvents.kAddTempProp, function( evt , propList , info )
837		self:onAddTempProp( propList , info )
838	end )
839	self._msgLogic:registerListener(UBFGamePlaySceneEvents.kAskSection , function( evt , datas )
840		local mode = datas.mode
841		if mode == "take" then
842
843			local removeIds = datas.removeIds
844			local id = datas.id
845			local stepCount = datas.stepCount
846
847			for k , _id in ipairs(removeIds or {}) do

856				UBFGamePlaySceneDebug:static_refreshSectionDebugView()
857			end
858

860
861			local resumeType = datas.resumeType
862			local id = datas.id
863
864			self:onReumeSection( id )
865
866			if self.isDebugMode then
867				UBFGamePlaySceneDebug:debugSaySomething(string.format("进行了%s的恢复，断面ID:%s" , tostring(resumeType) , tostring(id)))
868				UBFGamePlaySceneDebug:static_refreshSectionDebugView()
869			end
870
871		end
872	end )
873	self._msgLogic:registerListener(UBFGamePlaySceneEvents.kGameStateChange, function( evt , datas )

890			local promise = self:onBeforeWonderfulReplayOperation(datas.opData)

1322--[[
1323	场景级延迟面板门控接口。
1324	SceneManager 同一时刻只持有一个活跃 guard，具体拦截哪个面板由 scenePanelType 决定。
1325	业务流程持有 begin 返回的 guard token，并用同一个 token 执行 commit、cancel 或 clear；
1326	这样即使旧 process 的异步回调晚到，也无法操作后续流程新建的 guard。
1327]]
1328
1329-- 开始拦截指定类型的场景面板，并绑定当前 playId。
1330-- 场景已销毁或已有活跃 guard 时返回 nil，调用方不应继续启动依赖该 guard 的流程。
1331function UBFGameSceneManager:beginDeferredScenePanelGuard(scenePanelType)
1332	if self.isDisposed or self._deferredScenePanelGuard then
1333		logDeferredScenePanelGuard("begin_rejected", self._deferredScenePanelGuard, "panelType", scenePanelType)
1334		return nil
1335	end
1336
1337	-- 返回 token 供生命周期回调校验，避免旧 process 误操作后续新建的 guard。
1338	local guard = UBFDeferredScenePanelGuard:create(scenePanelType, self:getPlayId())
1339	self._deferredScenePanelGuard = guard
1340	logDeferredScenePanelGuard("begin", guard, "panelType", scenePanelType)
1341	return guard
1342end
1343
1344-- 尝试由当前 guard 消费一次面板弹出请求。
1345-- 返回 true 表示请求已被 guard 处理：pending 状态会暂存最新请求，committed 状态会直接丢弃；
1346-- 返回 false 表示没有匹配的 guard，调用方应继续执行原有弹出逻辑。
1347function UBFGameSceneManager:tryDeferScenePanel(scenePanelType, params)
1348	local guard = self._deferredScenePanelGuard
1349	if not guard or not guard:tryDefer(scenePanelType, params) then return false end
1350
1351	local action = guard:getState() == UBFDeferredScenePanelGuard.State.PENDING and "defer" or "drop"
1352	logDeferredScenePanelGuard(action, guard, "panelType", scenePanelType)
1353	return true
1354end
1355
1356-- 提交高优先级流程，表示玩家已经确认离开旧流程。
1357-- 只有传入 token 仍是当前活跃 guard 时才生效；提交后继续拦截匹配面板，但不再保存和恢复请求。
1358function UBFGameSceneManager:commitDeferredScenePanelGuard(guard)
1359	if not guard or self._deferredScenePanelGuard ~= guard then return false end
1360	local committed = guard:commit()
1361	logDeferredScenePanelGuard("commit", guard, "committed", committed)
1362	return committed
1363end
1364
1365-- 取消高优先级流程，并按业务条件尝试恢复 pending 阶段暂存的面板请求。
1366-- shouldRestore(sceneMgr, request) 只负责业务有效性判断；playId 和 token 一致性由通用层保证。
1367-- 返回值仅表示本次是否实际调用 popoutScenePanel，不表示 guard 是否曾经存在。
1368function UBFGameSceneManager:cancelDeferredScenePanelGuard(guard, shouldRestore)
1369	if not guard or self._deferredScenePanelGuard ~= guard then return false end
1370
1371	-- 先解除 guard，再恢复面板，避免 popoutScenePanel 被当前 guard 再次拦截。
1372	self._deferredScenePanelGuard = nil
1373	local request = guard:cancel(self:getPlayId())
1374	local restored = false
1375	if request and not self.isDisposed and (not shouldRestore or shouldRestore(self, request)) then
1376		self:popoutScenePanel(request.panelType, request.params)
1377		restored = true
1378	end
1379	logDeferredScenePanelGuard("cancel", guard, "restored", restored)
1380	return restored
1381end
1382
1383-- 终态清理，不恢复任何暂存请求，可安全重复调用。
1384-- 传入 token 时只清理对应 guard；不传时用于场景销毁等确定性终态，直接清理当前活跃 guard。
1385function UBFGameSceneManager:clearDeferredScenePanelGuard(guard)
1386	local activeGuard = self._deferredScenePanelGuard
1387	if not activeGuard or (guard and activeGuard ~= guard) then return false end
1388
1389	self._deferredScenePanelGuard = nil
1390	activeGuard:clear()
1391	logDeferredScenePanelGuard("clear", activeGuard)
1392	return true
1393end
