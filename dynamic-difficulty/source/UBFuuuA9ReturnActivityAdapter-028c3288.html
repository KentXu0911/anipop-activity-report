108function UBFuuuA9ReturnActivityAdapter:calcReturnAvgPlayCount(adjustContext, returnDayForA9)
109    local dayCount = math.max((tonumber(returnDayForA9) or 0) + 1, 1)
110    local dailyPlayCountList = getDailyPlayCountList(adjustContext)
111    local playCount = getTodayPlayCount(adjustContext)
112
113    if dayCount > 1 then
114        for i = 2, dayCount do
115            playCount = playCount + (tonumber(dailyPlayCountList[i]) or 0)
116        end
117    end
118
119    return playCount / dayCount, playCount, dayCount

122function UBFuuuA9ReturnActivityAdapter:findA9ActivityIndex(configMap, group, returnDayForA9, avgPlayCount)
123    if type(configMap) ~= "table" then
124        return nil, "config_nil"
125    end
126
127    local dayKey = "day" .. tostring(returnDayForA9 or 0)
128    local selectedRow = nil
129    for _, row in ipairs(configMap) do
130        if tonumber(row.group) == tonumber(group) and (tonumber(avgPlayCount) or 0) <= (tonumber(row.playcount) or 0) then
131            selectedRow = row
132            break
133        end
134    end
135
136    if not selectedRow then
137        return nil, "row_not_found"
138    end
139
140    local value = selectedRow[dayKey]
141    if value == nil or value == "" then
142        return nil, "empty"
143    end
144
145    local activityIndex = tonumber(value)
146    if not activityIndex then
147        return nil, "invalid_value"
148    end
149
150    return activityIndex, nil, selectedRow
151end

163function UBFuuuA9ReturnActivityAdapter:normalizeContext(adjustContext)
164    if type(adjustContext) ~= "table" then
165        return adjustContext
166    end
167
168    adjustContext.modGroup = adjustContext.modGroup or {}
169    adjustContext.CAGroup = adjustContext.CAGroup or {}
170    adjustContext.reportModGroup = adjustContext.reportModGroup or cloneMap(adjustContext.modGroup)
171
172    local info = getA9Info(adjustContext)
173    local enabled = info.enabled
174    if enabled == nil then
175        if adjustContext.modGroup.A9 ~= nil then
176            enabled = boolValue(adjustContext.modGroup.A9)
177        elseif adjustContext.switchName then
178            local maintenanceManager = MaintenanceManager and MaintenanceManager.getInstance and MaintenanceManager:getInstance()
179            if maintenanceManager and maintenanceManager.isEnabledInGroup then
180                enabled = maintenanceManager:isEnabledInGroup(adjustContext.switchName, "A9") == true
181            else
182                enabled = false
183            end
184        else
185            enabled = false
186        end
187    else
188        enabled = boolValue(enabled)
189    end
190
191    info.enabled = enabled
192    info.a8InjectedByA9 = nil
193    adjustContext.a8InjectedByA9 = nil
194    adjustContext.modGroup.A9 = enabled or nil
195    adjustContext.reportModGroup.A9 = enabled or nil
196
197    if enabled then
198        -- A9 只在外层追加 activityIndex1 fix；基础活跃档位仍复用现有 A8 逻辑。
199        if not adjustContext.reportModGroup.A8 then
200            info.a8InjectedByA9 = true
201            adjustContext.a8InjectedByA9 = true
202        end
203        adjustContext.modGroup.A8 = true
204    end
205
206    info.group = tonumber(info.group or adjustContext.CAGroup.CA10) or 1
207    info.returnKey = info.returnKey or adjustContext.returnKey or adjustContext.returnTime or self:_readReturnKeyFromTag()
208    if info.returnKey == nil and adjustContext.returnState then
209        info.returnKey = tostring(adjustContext.lostDay or "lost") .. "_" .. tostring(adjustContext.returnDayIndex or "day")
210    end
211
212    return adjustContext
213end

220function UBFuuuA9ReturnActivityAdapter:_getServerSmoothMaxValue(adjustContext, info)
221    local statisticInfo = getStatisticInfo(adjustContext)
222    local latestActivityIndex1 = tonumber(statisticInfo.latestActivityIndex1 or adjustContext.latestActivityIndex1)
223    local latestActivityIndex1Day = tonumber(statisticInfo.latestActivityIndex1Day or adjustContext.latestActivityIndex1Day)
224    local currentDay = getCurrentNaturalDay()
225
226    info.latestActivityIndex1 = latestActivityIndex1
227    info.latestActivityIndex1Day = latestActivityIndex1Day
228    info.currentDay = currentDay
229
230    if not latestActivityIndex1 or latestActivityIndex1 < ACTIVITY_INDEX_MIN then
231        return nil, "latest_activity_index_invalid"
232    end
233    if not latestActivityIndex1Day or not currentDay then
234        return nil, "latest_activity_day_invalid"
235    end
236    if latestActivityIndex1Day > currentDay then
237        return nil, "latest_activity_day_after_current_day"
238    end
239
240    return math.min(latestActivityIndex1 + 2, ACTIVITY_INDEX_MAX)
241end

265function UBFuuuA9ReturnActivityAdapter:beforeCheck(ubfSLRuntime, adjustContext, modEnvData)
266    self:normalizeContext(adjustContext)
267    local info = getA9Info(adjustContext)
268    modEnvData = modEnvData or {}
269
270    if not info.enabled then
271        info.source = "Disabled"
272        return
273    end
274
275    if isTopLevel(adjustContext) then
276        info.source = "TopLevel"
277        info.a9ActivityIndex = ACTIVITY_INDEX_MAX
278        self:_appendActivityIndex1Fix(adjustContext, ACT_KEY .. "_top_level", ACTIVITY_INDEX_MAX, UBFuuuFixMode.kSetVaule, {
279            minValue = ACTIVITY_INDEX_MIN,
280            maxValue = ACTIVITY_INDEX_MAX,
281        })
282        return
283    end
284
285    if not adjustContext.returnState then
286        info.source = "NotReturn"
287        info.fallbackReason = "not_return"
288        return
289    end
290
291    local configMap, configSource = self:loadReturnDayConfig()
292    info.configSource = configSource
293    if not configMap then
294        info.source = "A8Fallback"
295        info.fallbackReason = "config_nil"
296        return
297    end
298
299    local returnDayForA9 = self:getReturnDayForA9(adjustContext)
300    local avgPlayCount, totalPlayCount, dayCount = self:calcReturnAvgPlayCount(adjustContext, returnDayForA9)
301    local group = tonumber(info.group or adjustContext.CAGroup.CA10) or 1
302    local activityIndex, fallbackReason, selectedRow = self:findA9ActivityIndex(configMap, group, returnDayForA9, avgPlayCount)
303
304    info.group = group
305    info.returnDayForA9 = returnDayForA9
306    info.returnAvgPlayCount = avgPlayCount
307    info.returnTotalPlayCount = totalPlayCount
308    info.returnDayCount = dayCount
309    info.selectedConfigId = selectedRow and selectedRow.id or nil
310
311    if activityIndex then
312        info.source = "A9"
313        info.a9ActivityIndex = activityIndex
314        self:_appendActivityIndex1Fix(adjustContext, ACT_KEY, activityIndex, UBFuuuFixMode.kSetVaule, {
315            minValue = ACTIVITY_INDEX_MIN,
316            maxValue = ACTIVITY_INDEX_MAX,
317        })
318        return
319    end
320
321    info.source = "A8Fallback"
322    info.fallbackReason = fallbackReason or "empty"
323
324    local maxValue, smoothReason = self:_getServerSmoothMaxValue(adjustContext, info)
325    info.maxValue = maxValue
326
327    if maxValue then
328        self:_appendActivityIndex1Fix(adjustContext, ACT_KEY .. "_smooth", 0, UBFuuuFixMode.kaddVaule, {
329            maxValue = maxValue,
330        })
331    else
332        info.fallbackReason = info.fallbackReason .. "_" .. tostring(smoothReason or "missing_latest_activity_index")
333    end
334end
